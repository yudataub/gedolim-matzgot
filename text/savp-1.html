שׁוֹמְרֵי הַגְּמָרָא הַגִּבּוֹרִים

לִפְנֵי שָׁנִים רַבּוֹת, אַחֲרֵי שֶׁהַחֲכָמִים רַב אָשִׁי וְרָבִינָא סִיְּמוּ לַעֲרֹךְ אֶת הַתַּלְמוּד הַבַּבְלִי, הִגִּיעָה תְּקוּפָה מְיֻחֶדֶת בְּבָבֶל שֶׁנִּקְרֵאת תְּקוּפַת הַסָּבוֹרָאִים. הַחֲכָמִים שֶׁחָיוּ בַּתְּקוּפָה הַזּוֹ נִקְרְאוּ "רַבָּנַן סָבוֹרָאִים". הַשֵּׁם סָבוֹרָאִים בָּא מֵהַמִּלָּה "סְבָרָא". חֲכָמִים אֵלֶּה לֹא חִדְּשׁוּ הֲלָכוֹת חֲדָשׁוֹת מִשֶּׁלָּהֶם, אֶלָּא הִסְבִּירוּ אֶת דִּבְרֵי הָאָמוֹרָאִים, הִכְרִיעוּ בִּשְׁאֵלוֹת שֶׁנִּשְׁאֲרוּ פְּתוּחוֹת, וְסִדְּרוּ אֶת נֻסַּח הַגְּמָרָא כְּדֵי שֶׁהִיא תִּהְיֶה בְּרוּרָה וּשְׁלֵמָה. הַסָּבוֹרָאִים הָיוּ חֲכָמִים עֲנָוִים מְאֹד. הֵם לֹא כָּתְבוּ אֶת שְׁמוֹתֵיהֶם עַל הַהוֹסָפוֹת שֶׁהוֹסִיפוּ לַגְּמָרָא, וְלָכֵן הֵם נֶחְשָׁבִים כְּמוֹ גִּבּוֹרִים שְׁקֶטִים שֶׁשָּׁמְרוּ עַל הַתּוֹרָה בִּשְׁבִילֵנוּ.

בְּתוֹךְ דַּפֵּי הַגְּמָרָא שֶׁאָנוּ לוֹמְדִים עַד הַיּוֹם, יֵשׁ קְטָעִים רַבִּים שֶׁסֻּדְּרוּ עַל יְדֵי הַסָּבוֹרָאִים. לְדֻגְמָה, חַכְמֵי הַסָּבוֹרָאִים רַב עֵינָא וְרַב סִימוֹנָא סִדְּרוּ אֶת הַסֻּגְיָה שֶׁפּוֹתַחַת אֶת מַסֶּכֶת קִדּוּשִׁין. דֻּגְמָה נוֹסֶפֶת מְצוּיָה בְּמַסֶּכֶת סַנְהֶדְרִין, שָׁם הַגְּמָרָא שׁוֹאֶלֶת מִי צָרִיךְ לְשַׁלֵּם עַל הַמַּשְׁקֶה שֶׁנָּתְנוּ נָשִׁים יְקָרוֹת מִירוּשָׁלַיִם לְמִי שֶׁיָּצָא לְהֵהָרֵג. הַחָכָם רַב רַבָּאי מֵרוֹב, שֶׁהָיָה מֵהַסָּבוֹרָאִים, הִכְרִיעַ שֶׁהַמַּשְׁקֶה יְשֻׁלַּם מִכַּסְפֵּי הַצִּבּוּר, כִּי כָּל הַצִּבּוּר אַחֲרַאי לִדְאֹג לְמִי שֶׁנִּמְצָא בְּצַעַר.

תְּקוּפַת הַסָּבוֹרָאִים נִמְשְׁכָה חֲמִשָּׁה דּוֹרוֹת. בַּדּוֹר הָרִאשׁוֹן כִּהֵן רַב יוֹסֵי כְּרֹאשׁ יְשִׁיבַת פּוּמְבְּדִיתָא, וּבְיָמָיו נִסְתַּיֵּם הַתַּלְמוּד. בַּדּוֹר הַשֵּׁנִי חָיוּ חֲכָמִים כְּמוֹ רַב אֲחַאי, רַב שְׁמוּאֵל, רַב סִימוֹנָא וְרַב עֵינָא. בְּסוֹף דּוֹר זֶה נִסְגְּרוּ הַיְּשִׁיבוֹת בְּבָבֶל לְכַחֲמִשִּׁים שָׁנָה בִּגְלַל גְּזֵרוֹת וּשְׁמָדוֹת שֶׁל מַלְכֵי פָּרַס. בַּדּוֹר הַשְּׁלִישִׁי פָּתַח רַב חָנָן מֵאִישְׁקִיָּא מֵחָדָשׁ אֶת יְשִׁיבַת פּוּמְבְּדִיתָא וְהֶחֱזִיר אֶת הַלִּמּוּד. בַּדּוֹר הָרְבִיעִי חָיָה רַב יִצְחָק, רֹאשׁ יְשִׁיבַת פִּירוֹז שָׁבוּר, שֶׁיָּצָא לְקַבֵּל אֶת פְּנֵי הַשַּׁלִּיט עָלִי בְּסֵבֶר פָּנִים יָפוֹת. בַּדּוֹר הַחֲמִישִׁי, שֶׁהָיָה הַדּוֹר הָאַחֲרוֹן, חָיוּ מַר רַבָּא, מַר הוּנָא וְרַב שִׁישְׁנָא, וּבְכָךְ הִסְתַּיְּמָה תְּקוּפַת הַסָּבוֹרָאִים.

בִּתְקוּפָה זוֹ הָיוּ גַּם מַנְהִיגִים שֶׁנִּקְרְאוּ "רָאשֵׁי הַגּוֹלָה". בִּתְחִלַּת הַתְּקוּפָה כִּהֵן מַר זֻטְרָא רֹאשׁ הַגּוֹלָה. מָסֻפָּר שֶׁנִּרְאָה לוֹ עַמּוּד אֵשׁ, וְהוּא יָצָא עִם אַרְבַּע מֵאוֹת אִישׁ וְנִלְחַם בַּפָּרְסִים וְשָׁלַט שִׁבְעָה שָׁנִים. לְאַחַר מִכֵּן תָּפְסוּ אוֹתוֹ הַפָּרְסִים וְהָרְגוּ אוֹתוֹ עַל הַגֶּשֶׁר שֶׁל מָחוֹזָא. בְּאוֹתוֹ יוֹם נוֹלַד בְּנוֹ, שֶׁנִּקְרָא גַּם הוּא מַר זֻטְרָא, וּכְשֶׁגָּדַל עָלָה לְאֶרֶץ יִשְׂרָאֵל וְעָמַד בְּרֹאשׁ הַחֲכָמִים בִּטְבֶרְיָה.

לִקְרַאת סוֹף הַתְּקוּפָה עָמַד בְּרֹאשׁ הַגּוֹלָה מַר בּוּסְתְּנַאי. מָסֻפָּר שֶׁמֶּלֶךְ פָּרַס חָלַם חֲלוֹם שֶׁבּוֹ הוּא עוֹקֵר בֻּסְתָּן שָׁלֵם שֶׁל עֵצִים, וּכְשֶׁבָּא לַעֲקֹר אֶת הַשְּׁתִיל הָאַחֲרוֹן הִגִּיעַ דָּוִד הַמֶּלֶךְ וְעָצַר אוֹתוֹ. הַחָכָם הַיְּהוּדִי פָּתַר לַמֶּלֶךְ שֶׁהַשְּׁתִיל הוּא תִּינוֹק שֶׁעָתִיד לְהִוָּלֵד מִזֶּרַע דָּוִד, וְלָכֵן קָרְאוּ לַיֶּלֶד בּוּסְתְּנַאי. עוֹד מָסֻפָּר שֶׁכַּאֲשֶׁר עָמַד בּוּסְתְּנַאי לִפְנֵי הַמֶּלֶךְ, בָּא זְבוּב וְעָמַד עַל מִצְחוֹ וְעָקַץ אוֹתוֹ עַד שֶׁיָּרַד דָּם, אַךְ בּוּסְתְּנַאי לֹא הִבְרִיחַ אֶת הַזְּבוּב כִּי לָמַד מֵאֲבוֹתָיו שֶׁלֹּא מְרִימִים יָד לִפְנֵי מֶלֶךְ. לָכֵן הָיְתָה צוּרַת זְבוּב עַל הַחוֹתָם שֶׁל רֹאשׁ הַגּוֹלָה.

מִכָּל מַה שֶּׁעָשׂוּ חַכְמֵי הַסָּבוֹרָאִים אָנוּ לוֹמְדִים עַל עֲנָוָה גְּדוֹלָה, עַל מְסִירוּת נֶפֶשׁ לְלִמּוּד הַתּוֹרָה גַּם בִּזְמַנִּים קָשִׁים, וְעַל אַחֲרָיוּת רַבָּה לִשְׁמֹר עַל הַגְּמָרָא מְסֻדֶּרֶת וּבְרוּרָה עֲבוּר כָּל הַדּוֹרוֹת. הֵם הָיוּ חֻלְיָה חֲשׁוּבָה מְאֹד בַּשַּׁלְשֶׁלֶת שֶׁל הַתּוֹרָה שֶׁעוֹבֶרֶת מִדּוֹר לְדוֹר עַד אֵלֵינוּ.

💡 מֵהַסָּבוֹרָאִים אָנוּ לוֹמְדִים עֲנָוָה, מְסִירוּת נֶפֶשׁ לַתּוֹרָה וְאַחֲרָיוּת לִשְׁמֹר עַל שַׁלְשֶׁלֶת הַדּוֹרוֹת שֶׁל הָעַם שֶׁלָּנוּ.
