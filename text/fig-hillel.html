הַגּוֹי שֶׁרָצָה לִהְיוֹת כֹּהֵן גָּדוֹל

גּוֹי אֶחָד חָשׁוּב, שֶׁיָּדַע אֶת שְׂפַת הַקֹּדֶשׁ, הָיָה מִתְהַלֵּךְ לַהֲנָאָתוֹ בָּרְחוֹב. כַּאֲשֶׁר עָבַר לְיַד בֵּית הַמִּדְרָשׁ, הִקְשִׁיב לְקוֹלָם שֶׁל הַיְלָדִים שֶׁלָּמְדוּ תּוֹרָה. הַיְלָדִים לָמְדוּ אֶת פָּרָשַׁת 'תְּצַוֶּה', וְסִפְּרוּ עַל בִּגְדֵי הַכֹּהֵן. הַגּוֹי שָׁמַע אֶת הַדְּבָרִים, וְלִבּוֹ הִתְמַלֵּא הִתְפַּעֲלוּת גְּדוֹלָה מֵהַתֵּאוּר שֶׁל בִּגְדֵי הַכֹּהֵן. הוּא אָמַר לְעַצְמוֹ: "אֵיזֶה בְּגָדִים מְיֻחָדִים יֵשׁ לַכֹּהֵן הַגָּדוֹל, אֵלּוּ בְּגָדִים חֲשׁוּבִים וְיָפִים! גַּם אֲנִי רוֹצֶה מְאֹד לִהְיוֹת כֹּהֵן גָּדוֹל! אֲנִי רוֹצֶה לְהִתְגַּיֵּר וְלִהְיוֹת יְהוּדִי, אֲבָל אֲנִי רוֹצֶה לַעֲשׂוֹת זֹאת רַק בִּתְנַאי אֶחָד: רַק אִם אֶהְיֶה כֹּהֵן גָּדוֹל".

הַגּוֹי הָלַךְ אֶל שַׁמַּאי וְהִצִּיג לְפָנָיו אֶת הַתְּנַאי שֶׁלּוֹ. שַׁמַּאי שָׁמַע אֶת דְּבָרָיו וְכָעַס עָלָיו מְאֹד. שַׁמַּאי דָּחָה אוֹתוֹ מִיָּד וְאָמַר לוֹ: "מָה? מָה זֶה? אַתָּה רוֹצֶה לְהִתְגַּיֵּר רַק בִּגְלַל שֶׁאַתָּה רוֹצֶה אֶת הַבְּגָדִים הַיָּפִים וְהַחֲשׁוּבִים? אַתָּה לֹא מֵבִין בִּכְלָל מָה זוֹ הַתּוֹרָה! לֵךְ מִפֹּה, אֵיךְ אַתָּה יָכוֹל כָּכָה לְזַלְזֵל בַּבְּגָדִים וּבַדְּבָרִים שֶׁל הַכֹּהֵן? אַתָּה לֹא יָכוֹל לִהְיוֹת יְהוּדִי!". שַׁמַּאי לֹא הִסְכִּים לְגַיֵּר אוֹתוֹ וְשָׁלַח אוֹתוֹ לְשָׁלוֹם.

הַגּוֹי לֹא וִתֵּר עַל רְצוֹנוֹ. הוּא שָׁמַע עַל הִלֵּל וְיָדַע שֶׁהִלֵּל הוּא אָדָם סַבְלָן מְאֹד, וְעַל כֵּן הָלַךְ אֵלָיו. הַגּוֹי בָּא אֶל הִלֵּל וּבִקֵּשׁ מִמֶּנּוֹ לְהִתְגַּיֵּר בְּאוֹתוֹ הַתְּנַאי. הִלֵּל שָׁמַע אֶת הַבַּקָּשָׁה, אֲבָל הוּא לֹא דָּחָה אוֹתוֹ. הִלֵּל חָשַׁב שֶׁאוּלַי מַשֶּׁהוּ עָמֹק בִּפְנִים מְנִיעַ אֶת הַגּוֹי וּמֹשֵׁךְ אוֹתוֹ לְהִתְקָרֵב אֶל הַקֹּדֶשׁ וְאֶל הַתּוֹרָה. אָמַר הִלֵּל לַגּוֹי: "אִם אַתָּה רוֹצֶה לִהְיוֹת כֹּהֵן גָּדוֹל, אָז אַתָּה צָרִיךְ לִלְמֹד אֶת הִלְכוֹת הַכְּהֻנָּה וְלָדַעַת מָה הַכֹּהֵן הַגָּדוֹל עוֹשֶׂה. הֲרֵי כְּמוֹ שֶׁאִם אַתָּה רוֹצֶה לִהְיוֹת מֶלֶךְ חָשׁוּב אַתָּה צָרִיךְ לִלְמֹד מָה מֶלֶךְ עוֹשֶׂה, כָּךְ גַּם עָלֶיךָ לִלְמֹד מָה עוֹשֶׂה הַכֹּהֵן הַגָּדוֹל". הַגּוֹי הִסְכִּים לַדְּבָרִים וְהֵחֵל לִלְמֹד תּוֹרָה אֶצְלוֹ.

הַגּוֹי לָמַד אֶצֶל הִלֵּל בִּשְׁקִידָה, וְתוֹךְ כְּדֵי הַלִּמּוּד נִתְקַל בַּפָּסוּק הַכָּתוּב בַּתּוֹרָה: "הַזָּר הַקָּרֵב יוּמָת". הַגּוֹי שָׁאַל אֶת הִלֵּל לְפֵרוּשׁ הַפָּסוּק, וְהִלֵּל הִסְבִּיר לוֹ שֶׁמִּי שֶׁאֵינוֹ כֹּהֵן אוֹ בֵּן שֶׁל כֹּהֵן אֵינוֹ יָכוֹל לִהְיוֹת כֹּהֵן גָּדוֹל. הַגּוֹי הֵבִין כָּעֵת בְּבֵרוּר שֶׁהוּא עַצְמוֹ לֹא יוּכַל לְכַהֵן כְּכֹהֵן גָּדוֹל בְּשׁוּם אֹפֶן.

לַמְרוֹת זֹאת, הַלִּמּוּד עַצְמוֹ הִשְׁפִּיעַ עָלָיו עֲמֻקּוֹת. בְּעִקְבוֹת הַלִּמּוּד אֵצֶל הִלֵּל, חָשַׁב הַגּוֹי לְעַצְמוֹ: "אֲנִי לָמַדְתִּי דְּבָרִים מְיֻחָדִים בַּתּוֹרָה, וְלָמַדְתִּי מִצְווֹת וְלָמַדְתִּי עַל הַשַּׁבָּת, וְהֶחְלַטְתִּי שֶׁבְּכָל זֹאת אֲנִי רוֹצֶה לִהְיוֹת יְהוּדִי, לַמְרוֹת שֶׁאֲנִי לֹא יָכוֹל לִהְיוֹת כֹּהֵן גָּדוֹל". הַגּוֹי הֶחְלִיט לְהִתְגַּיֵּר מִתּוֹךְ אַהֲבָה גְּדוֹלָה לַתּוֹרָה וְלַמִּצְווֹת שֶׁלָּמַד.

💡 לִמּוּד הַתּוֹרָה בְּסַבְלָנוּת וּבְנֹעַם פּוֹתֵחַ אֶת הַלֵּב וּמַרְאֶה לָנוּ אֶת הַיֹּפִי שֶׁל הַמִּצְווֹת.
