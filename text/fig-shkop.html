רַבִּי שִׁמְעוֹן שְׁקוֹפּ – רֹאשׁ הַיְשִׁיבָה שֶׁנִּשְׁאַר עִם תַּלְמִידָיו

רַבִּי שִׁמְעוֹן יְהוּדָה הַכֹּהֵן שְׁקוֹפּ נוֹלַד בִּשְׁנַת ה'תַּרְ"כָּ (1859) בָּעֲיָרָה טוּרֶץ שֶׁבְּרוּסְיָה הַלְּבָנָה. כְּשֶׁהָיָה נַעַר צָעִיר בֶּן שְׁתֵּים עֶשְׂרֵה שָׁנִים, הִתְחִיל לִלְמֹד בִּישִׁיבַת מִיר. כְּשֶׁהִגִּיעַ לְגִיל אַרְבַּע עֶשְׂרֵה שָׁנִים, עָבַר לִלְמֹד בִּישִׁיבַת וּוֹלוֹזִ'ין, וְשָׁם לָמַד בִּקְבוּצָה קְטַנָּה שֶׁל תַּלְמִידִים יַחַד עִם רַבִּי חַיִּים מִבְּרִיסְק.

לְאַחַר מִכֵּן לִמֵּד רַבִּי שִׁמְעוֹן תּוֹרָה בְּמֶשֶׁךְ שְׁמוֹנֶה עֶשְׂרֵה שָׁנִים בִּישִׁיבַת טֶלְז. בְּהֶמְשֵׁךְ חַיָּיו הָיָה רַב וְרֹאשׁ יְשִׁיבָה בָּעֲיָרוֹת מָאלְטְשׁ וּבְרַיְנְסְק, וּכְשֶׁהִגִּיעַ לְגִיל שִׁשִּׁים שָׁנָה, נִקְרָא לַעֲמֹד בְּרֹאשׁ יְשִׁיבַת "שַׁעַר הַתּוֹרָה" בָּעִיר גְּרוֹדְנָה. רַבִּי שִׁמְעוֹן כָּתַב אֶת הַסֵּפֶר הַחָשׁוּב "שַׁעֲרֵי יוֹשֶׁר", שֶׁלּוֹמְדִים בּוֹ עַד הַיּוֹם בְּכָל הַיְּשִׁיבוֹת.

בְּהַקְדָמַת הַסֵּפֶר "שַׁעֲרֵי יוֹשֶׁר" (שֶׁנִּכְתְּבָה בִּשְׁנַת ה'תַּרְפָּ"ח) כָּתַב רַבִּי שִׁמְעוֹן יְסוֹד חָשׁוּב מְאֹד: הָאָדָם צָרִיךְ לִחְיוֹת לֹא רַק בִּשְׁבִיל עַצְמוֹ, אֶלָּא בִּשְׁבִיל אֲחֵרִים – לְהַרְבּוֹת טוֹב לַזּוּלָת, וְלַעֲשׂוֹת אֶת כָּל כֹּחוֹתָיו לְטוֹבַת הַכְּלָל. כָּךְ חַי רַבִּי שִׁמְעוֹן בְּעַצְמוֹ בְּכָל יְמֵי חַיָּיו.

פַּעַם אַחַת נָסַע לְאָמֶרִיקָה כְּדֵי לֶאֱסֹף כֶּסֶף לַיְּשִׁיבָה שֶׁלּוֹ בִּגְרוֹדְנָה. כְּשֶׁהָיָה שָׁם, הִצִּיעוּ לוֹ לִהְיוֹת רֹאשׁ יְשִׁיבָה גְּדוֹלָה בְּנִיוּ יוֹרְק. זֹאת הָיְתָה הַצָּעָה מְכֻבֶּדֶת מְאֹד, אַךְ גְּדוֹלֵי הַדּוֹר – הֶחָפֵץ חַיִּים וְרַבִּי חַיִּים עוֹזֵר גְּרוֹדְזֶנְסְקִי – בִּקְשׁוּ מִמֶּנּוּ לַחְזֹר לְאֵירוֹפָה. רַבִּי שִׁמְעוֹן שָׁמַע לָהֶם וְחָזַר אֶל תַּלְמִידָיו בִּגְרוֹדְנָה.

בִּשְׁנַת ה'תַּרְצָ"ט (1939) פָּרְצָה מִלְחֶמֶת הָעוֹלָם הַשְּׁנִיָּה. רֹב הַתַּלְמִידִים בָּרְחוּ לָעִיר וִילְנָה כְּדֵי לְהִנָּצֵל. רַבִּי שִׁמְעוֹן, שֶׁהָיָה כְּבָר זָקֵן וְחוֹלֶה, לֹא עָזַב אֶת הַיְּשִׁיבָה. הוּא נִשְׁאַר בִּגְרוֹדְנָה עִם קְבוּצָה קְטַנָּה שֶׁל תַּלְמִידִים, וְהִמְשִׁיךְ לְלַמֵּד אוֹתָם תּוֹרָה.

בְּיוֹם ט' בְּחֶשְׁוָן ה'תָּ"שׁ הִתְאַסְפוּ בְּבֵיתוֹ רַבַּנֵּי הַסְּבִיבָה, כְּדֵי לִשְׁאֹל אוֹתוֹ עֵצָה כֵּיצַד לְחַנֵּךְ אֶת הַיְּלָדִים בַּיָּמִים הַקָּשִׁים הָאֵלֶּה. רַבִּי שִׁמְעוֹן דִּבֵּר אִתָּם וְנָתַן לָהֶם עֵצָה טוֹבָה. מִיָּד לְאַחַר מִכֵּן עָמַד לְהִתְפַּלֵּל תְּפִלַּת מִנְחָה. בְּאֶמְצַע הַתְּפִלָּה נִפְטַר רַבִּי שִׁמְעוֹן, וְהוּא בֶּן שְׁמוֹנִים שָׁנָה.

אֶת חִדּוּשָׁיו הַנִּפְלָאִים עַל הַגְּמָרָא הִבְרִיחַ אֶחָד מִתַּלְמִידָיו, תוֹךְ סַכָּנָה גְּדוֹלָה, מִתּוֹךְ בְּרִית הַמּוֹעָצוֹת. בְּנוֹ שֶׁל רַבִּי שִׁמְעוֹן הִדְפִיס אֶת הַסְּפָרִים הָאֵלֶּה, וְעַד הַיּוֹם הַזֶּה לוֹמְדִים בָּהֶם תַּלְמִידִים בְּכָל הַיְּשִׁיבוֹת.
