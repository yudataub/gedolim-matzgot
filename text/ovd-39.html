הַמַּתָּנָה שֶׁהֶחֱזִירָה אֶת הַחַיִּים

בִּימֵי הַשִּׁבְעָה עַל מָרָן הָרַב עֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק לִבְרָכָה, הוּקַם אֹהֶל מְנַחֲמִים עָנָק לְיַד בֵּיתוֹ שֶׁבִּרְחוֹב הַקַּבְּלָן בִּירוּשָׁלַיִם. הָאֹהֶל הָיָה מָלֵא בִּבְנֵי אָדָם בְּכָל שְׁעוֹת הַיּוֹם. אַלְפֵי אֲנָשִׁים הִגִּיעוּ מִכָּל מָקוֹם כְּדֵי לְנַחֵם אֶת בָּנָיו שֶׁל הָרַב. בָּאוּ לְשָׁם אֲנָשִׁים מִכָּל הַחוּגִים וּמִכָּל הָעֵדוֹת: סְפָרַדִּים וְאַשְׁכְּנַזִּים, חֲרֵדִים, דָּתִיִּים, מָסָרְתִּיִּים וְגַם חִלּוֹנִיִּים. כֻּלָּם הִרְגִּישׁוּ עֶצֶב גָּדוֹל עַל לְכָתוֹ שֶׁל הַמַּנְהִיג הָאַהוּב.

בְּכָל יוֹם הִגִּיעוּ לָאֹהֶל עֲשָׂרוֹת רַבָּנִים, דַּיָּנִים וְרָאשֵׁי יְשִׁיבוֹת. הֵם יָשְׁבוּ מוּל בָּנָיו שֶׁל מָרָן וְסִפְּרוּ מַעֲשִׂים מֻפְלָאִים עַל עֵצוֹת טוֹבוֹת שֶׁקִּבְּלוּ מִמֶּנּוּ, עַל בְּרָכוֹת שֶׁהִתְקַיְּמוּ וְעַל פִּסְקֵי הֲלָכָה חֲשׁוּבִים שֶׁסִּיְּעוּ לְרַבִּים.

בְּאֶחָד מִימֵי הַשִּׁבְעָה, הִבְחִין נֶכְדּוֹ שֶׁל מָרָן בְּאִישׁ אֶחָד פָּשׁוּט שֶׁעָמַד בְּפִנַּת הָאֹהֶל. הָאִישׁ בָּכָה מְאֹד וּבָכָה מִכָּל הַלֵּב, מַמָּשׁ כְּמוֹ יֶלֶד קָטָן. הַנֶּכֶד רָאָה אֶת הַדְּמָעוֹת הָרַבּוֹת וְנִגַּשׁ אֵלָיו בְּעַדְנוּת. הוּא שָׁאַל אוֹתוֹ בְּדַאֲגָה: "מָה קָרָה לְךָ? לָמָּה אַתָּה בּוֹכֶה כָּל כָּךְ?".

הָאִישׁ הָיָה נִרְגָּשׁ מְאֹד וְלֹא יָכֹל לַעֲנוֹת מִיָּד. גְּרוֹנוֹ נִשְׁנַק מִגֹּדֶל הַבְּכִי. אֲבָל כְּשֶׁהִצְלִיחַ לְהוֹצִיא מִילִים מִפִּיו, הוּא אָמַר שׁוּב וָשׁוּב: "לִפְנֵי שָׁנָה וְחֹדֶשׁ נוֹלַדְתִּי! לִפְנֵי שָׁנָה וְחֹדֶשׁ נוֹלַדְתִּי!".

הַנֶּכֶד תָּמַהּ מְאֹד עַל הַמִּילִים הַמּוּזָרוֹת הָאֵלֶּה. הוּא הֵבִין כִּי מֵאֲחוֹרֵי הַבְּכִי מִסְתַּתֵּר סִפּוּר מְיֻחָד וּמַרְגִּישׁ. הוּא אָחַז בִּזְרוֹעוֹ שֶׁל הָאִישׁ וְהוֹבִיל אוֹתוֹ בְּנַחַת אֶל הָרַב יִצְחָק יוֹסֵף, בְּנוֹ שֶׁל מָרָן. הָרַב יִצְחָק הִבִּיט בּוֹ בְּרַחֲמִים וְאָמַר לוֹ בְּרֹךְ: "סַפֵּר לִי, בְּבַקָּשָׁה, מָה מְבִיאוֹת אֶת הַדְּמָעוֹת הָאֵלֶּה?".

הָאִישׁ הִתְאַמֵּץ לְהֵרָגַע, נָשַׁם עָמֹק וְהִתְחִיל לְסַפֵּר אֶת קוֹרוֹתָיו: "אֲנִי בֶּן אַרְבָּעִים וְאַחַת. כְּשֶׁהָיִיתִי בֶּן שְׁמוֹנֶה עֶשְׂרֵה, יָצְאָה עָלַי שְׁמוּעָה רָעָה שֶׁאֵינִי יְהוּדִי כָּשֵׁר וְשֶׁאֲנִי פָּסוּל לְחַתְנִי. הַשְּׁמוּעָה הַזֹּאת הָיְתָה קָשָׁה וְנוֹרָאָה. בְּמֶשֶׁךְ עֶשְׂרִים וּשְׁתַּיִם שָׁנִים חָיִיתִי כְּמוֹ אָדָם שֶׁאֵינוֹ חַי. אֲנָשִׁים הִתְרַחֲקוּ מִמֶּנִּי, אַף אֶחָד לֹא רָצָה לְהִתְקָרֵב אֵלַי, וְלֹא יָכֹלְתִּי לְהָקִים מִשְׁפָּחָה. הָלַכְתִּי מֵרַב לְרַב וּמִדַּיָּן לְדַיָּן, הִתְחַנַּנְתִּי שֶׁיִּבְדְּקוּ וְיִמְצְאוּ פַתְרוֹן כְּדֵי לְהַכְשִׁיר אוֹתִי, אֲבָל אַף אֶחָד לֹא הִצְלִיחַ לַעֲזֹר לִי".

הָאִישׁ עָצַר, וּדְמָעוֹת חֲדָשׁוֹת זָלְגוּ מֵעֵינָיו. הוּא הִמְשִׁיךְ וְסִפֵּר: "אֲבָל לִפְנֵי שָׁנָה וְחֹדֶשׁ הִגִּיעַ הָעִנְיָן שֶׁלִּי אֶל מָרָן הָרַב עֹבַדְיָה. מָרָן יָשַׁב, לָמַד אֶת הַנּוֹשֵׂא לְעֹמֶק, וּבְכֹחַ תּוֹרָתוֹ הַגְּדוֹלָה כָּתַב עָלַי פְּסָק הֲלָכָה מַקִּיף וּבָרוּר – וְהִכְשִׁיר אוֹתִי לַחֲלוּטִין! בְּאוֹתוֹ יוֹם נוֹלַדְתִּי מֵחָדָשׁ! מָרָן נָתַן לִי אֶת חַיַּי בְּמַתָּנָה!".

כְּשֶׁסִּיֵּם הָאִישׁ אֶת דְּבָרָיו, פָּרְצוּ כָּל הַנִּמְצָאִים בָּאֹהֶל בִּבְכִי נִרְגָּשׁ יַחַד עִמּוֹ, מִתּוֹךְ הוֹדָיָה עַל הַחֶסֶד הַגָּדוֹל וְהָאַהֲבָה שֶׁהִרְעִיף הָרַב עַל כָּל אָדָם.
