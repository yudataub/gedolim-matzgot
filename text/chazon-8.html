הַחַזּוֹן אִישׁ וּמְשַׁל הַחַיָּט

פַּעַם אַחַת הִגִּיעַ יְהוּדִי אֶחָד אֶל הַחֲזוֹן אִישׁ, כְּדֵי לִשְׁאֹל אוֹתוֹ שְׁאֵלָה שֶׁהָיְתָה קָשָׁה לוֹ מְאֹד. הַיְהוּדִי רָאָה שֶׁעוֹבְרִים עַל עַם יִשְׂרָאֵל קְשָׁאִים וְסֵבֶל, וְלָכֵן הוּא פָּנָה אֶל הַחֲזוֹן אִישׁ וְשָׁאַל אוֹתוֹ בִּכְאֵב: "רַבִּי, לָמָּה עַל עַם יִשְׂרָאֵל עוֹבְרִים כָּל כָּךְ הַרְבֵּה צָרוֹת?".

הַחֲזוֹן אִישׁ הִקְשִׁיב לַשְּׁאֵלָה בְּסַבְלָנוּת וּבְאַהֲבָה רַבָּה, וְהֶחְלִיט לַעֲנוֹת לוֹ עַל יְדֵי מָשָׁל קָצָר, יָפֶה וּבָהִיר.

אָמַר לוֹ הַחֲזוֹן אִישׁ: "פַּעַם אַחַת קָנָה אָדָם אֶחָד בַּד מְאֹד יָקָר, יָפֶה וּמְיֻחָד. הָאָדָם רָצָה מְאֹד שֶׁיִּהְיֶה לוֹ מֵהַבַּד הַזֶּה בֶּגֶד נָאֶה וּמְכֻבָּד, וְלָכֵן הוּא לָקַח אֶת הַבַּד הַיָּקָר וֶהֱבִיאוֹ אֶל הַחַיָּט, כְּדֵי שֶׁהַחַיָּט יִתְפּוֹר לוֹ מִזֶּה חֲלִיפָה מְכֻבֶּדֶת.

כַּאֲשֶׁר הִגִּיעַ הָאָדָם אֶל הַחַיָּט, נָטַל הַחַיָּט מִסְפָּרַיִם וְהֵחֵל לִגְזֹר אֶת הַבַּד הַיָּקָר. הַחַיָּט חָתַךְ וְגָזַר אֶת הַבַּד לִחֲלָקִים וְלִגְזָרִים שׁוֹנִים. כַּאֲשֶׁר רָאָה הָאָדָם מָה הַחַיָּט עוֹשֶׂה לַבַּד הַיָּקָר שֶׁלּוֹ, הוּא מְאֹד נִבְהַל וְהִתְפַּלֵּא עַל מַעֲשֵׂה הַחַיָּט. הָאָדָם לֹא הֵבִין לָמָּה הַחַיָּט חוֹתֵךְ וְגוֹזֵר בַּד כָּל כָּךְ יָקָר וּמְפֹאָר, וְחָשַׁב בִּדְאָגָה שֶׁהַחַיָּט הוֹרֵס אֶת הַבַּד הַמְּיֻחָד שֶׁלּוֹ.

אֲבָל הַחַיָּט יָדַע הֵיטֵב אֶת מְלַאכְתּוֹ. הַחַיָּט לָקַח אֶת כָּל חַלְקֵי הַבַּד הַגְּזוּרִים, סִדֵּר אוֹתָם כָּרָאוּי, וְהֵחֵל לִתְפּוֹר אוֹתָם זֶה לָזֶה בִּמְיוּמָנוּת וּבְסַבְלָנוּת רַבָּה. הוּא תָּפַר חֵלֶק אֶל חֵלֶק בְּדִיּוֹק רַב, עַד שֶׁכָּל הַחֲלָקִים הַגְּזוּרִים הִתְחַבְּרוּ יַחְדָּו לְבֶגֶד אֶחָד מוּכָן.

רַק לְאַחַר שֶׁהַחַיָּט סִיֵּם לִתְפּוֹר אֶת כָּל הַחֲלִיפָה הַמְּכֻבֶּדֶת, וְהֶרְאָה אוֹתָהּ לָאָדָם, הֵבִין הָאָדָם אֶת מַעֲשֵׂי הַחַיָּט. פִּתְאוֹם הֵבִין הָאָדָם שֶׁהַחַיָּט לֹא הָרַס אֶת הַבַּד, אֶלָּא כָּל גְּזִירָה וְכָל חִתּוּךְ הָיוּ נְחוּצִים וַחֲשׁוּבִים מְאֹד כְּדֵי לִתְפּוֹר אֶת הַחֲלִיפָה הַיָּפָה וְהַמֻּשְׁלֶמֶת.

כָּךְ גַּם אֲנַחְנוּ, הִסְבִּיר הַחֲזוֹן אִישׁ: עַכְשָׁיו אֲנַחְנוּ עוֹבְרִים קְשָׁאִים וְצָרוֹת, וְלֹא תָּמִיד מְבִינִים מָה הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה וְלָמָּה דְּבָרִים קוֹרִים. אֲבָל כְּמוֹ שֶׁהָאָדָם הֵבִין אֶת מַעֲשֵׂי הַחַיָּט רַק כְּשֶׁהַבֶּגֶד הָיָה מוּכָן, כָּךְ גַּם אֲנַחְנוּ נָבִין אֶת מַעֲשֵׂי הַשֵּׁם רַק לְאַחַר בִּיאַת מְשִׁיחַ צִדְקֵנוּ בִּמְהֵרָה בְיָמֵינוּ. אָז נִרְאֶה וְנָבִין אֶת כָּל הַתַּכְנִית הַשְּׁלֵמָה, הַטּוֹבָה וְהַנִּפְלָאָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא עַל כָּל פְּרָטֶיהָ.".

💡 גַּם כַּאֲשֶׁר אֲנַחְנוּ לֹא מְבִינִים אֶת מָה שֶׁקּוֹרֶה, אֲנַחְנוּ מַאֲמִינִים שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה הַכֹּל לְטוֹבָתֵנוּ וּלְפִי תַּכְנִית מֻשְׁלֶמֶת.
