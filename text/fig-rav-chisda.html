רַב חִסְדָּא וְהַלִּמּוּד שֶׁלֹּא נִפְסַק

חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה מְסַפְּרִים לָנוּ עַל **רַב חִסְדָּא**, שֶׁהָיָה מִגְּדוֹלֵי הַחֲכָמִים הַחֲשׁוּבִים בְּבָבֶל וְרֹאשׁ יְשִׁיבָה גְדוֹלָה בָּעִיר סוּרָא. רַב חִסְדָּא אָהַב אֶת הַתּוֹרָה הַקְּדוֹשָׁה בְּכָל לִבּוֹ. הוּא הָיָה יוֹשֵׁב בְּבֵית הַמִּדְרָשׁ שָׁעוֹת רַבּוֹת, הוֹגֶה בַּתּוֹרָה וְלוֹמֵד בִּשְׁקִידָה נִפְלָאָה. פִּיו שֶׁל רַב חִסְדָּא לֹא הָיָה שׁוֹתֵק מִלִּמּוּדוֹ אֲפִילוּ לְרֶגַע אֶחָד, וְהוּא הָיָה מְשַׁנֶּה וְלוֹמֵד אֶת דִּבְרֵי הַתּוֹרָה בִּרְצִיפוּת.

מַאֲמַר חֲכָמֵינוּ מְלַמֵּד אוֹתָנוּ כִּי הַתּוֹרָה הַקְּדוֹשָׁה מְגִנָּה עַל מִי שֶׁעוֹסֵק בָּהּ. בִּזְכוּת לִמּוּד הַתּוֹרָה הָרָצוּף וְהַמַּתְמִיד שֶׁל רַב חִסְדָּא, שׁוּם דָּבָר רַע לֹא יָכוֹל הָיָה לִפְגֹּעַ בּוֹ. אֲפִילוּ **שְׁלִיחָא דִּמְלַאֲכָא דְּמוֹתָא**, שְׁלִיחוֹ שֶׁל מַלְאַךְ הַמָּוֶת, לֹא יָכֹל הָיָה לְהִתְקָרֵב אֵלָיו בְּשׁוּם אֹפֶן כָּל עוֹד הוּא הָיָה יוֹשֵׁב וְלוֹמֵד תּוֹרָה. מַלְאַךְ הַמָּוֶת יָדַע כִּי כָּל עוֹד קוֹל הַתּוֹרָה בְּפִיו שֶׁל רַב חִסְדָּא אֵינוֹ נִפְסָק, הַתּוֹרָה מְגִנָּה עָלָיו כְּמוֹ חוֹמָה חֲזָקָה וּמַצִּילָה אוֹתוֹ מִכָּל פְּגִיעָה.

שְׁלִיחוֹ שֶׁל מַלְאַךְ הַמָּוֶת רָצָה לְהִתְקָרֵב אֶל רַב חִסְדָּא, אַךְ הוּא לֹא מָצָא שׁוּם דֶּרֶךְ לַעֲשׂוֹת זֹאת, מִפְּנֵי שֶׁרַב חִסְדָּא לֹא הִפְסִיק אֶת גִּירְסָתוֹ וְלֹא שָׁתַק מִלִּמּוּדוֹ מֵעוֹלָם. הַשָּׁלִיחַ הֵבִין כִּי רַק אִם רַב חִסְדָּא יַפְסִיק אֶת הַלִּמּוּד וְיִשְׁתֹּק לְרֶגַע אֶחָד, הוּא יוּכַל לְהִתְקָרֵב אֵלָיו. מָה עָשָׂה שְׁלִיחוֹ שֶׁל מַלְאַךְ הַמָּוֶת? הוּא עָלָה לְמַעְלָה וְיָשַׁב עַל **עֵץ אֶרֶז** גָּבוֹהַּ וְחָזָק שֶׁעָמַד בְּבֵית הַמִּדְרָשׁ.

לְפֶתַע פִּתְאֹם, כַּאֲשֶׁר יָשַׁב הַשָּׁלִיחַ עַל הָעֵץ, נִשְׁבַּר עֵץ הָאֶרֶז וְנִפְקַע בְּקוֹל רַעַשׁ גָּדוֹל וְחָזָק מְאֹד. הָרַעַשׁ הַפִּתְאוֹמִי וְהַחָזָק שֶׁל עֵץ הָאֶרֶז הַנִּשְׁבָּר הִפְתִיעַ אֶת רַב חִסְדָּא. בִּגְלַל קוֹל הַשְּׁבִירָה הַגָּדוֹל, נִבְהַל רַב חִסְדָּא לְרֶגַע אֶחָד קָטָן, הִפְסִיק אֶת לִמּוּדוֹ וְשָׁתַק מִגִּירְסָתוֹ לְכַמָּה שְׁנִיּוֹת.

בְּאוֹתוֹ רֶגַע מַמָּשׁ, כַּאֲשֶׁר רַב חִסְדָּא שָׁתַק מִלִּמּוּד הַתּוֹרָה וְלֹא הָגָה בָּהּ, הִצְלִיחַ שְׁלִיחוֹ שֶׁל מַלְאַךְ הַמָּוֶת לְהִתְקָרֵב אֵלָיו וּלְהִתְגַּבֵּר עָלָיו. הַסִּפּוּר מַרְאֶה לָנוּ אֶת הַכֹּחַ הָעֲצוּם וְהַנִּפְלָא שֶׁל לִמּוּד הַתּוֹרָה, שֶׁשּׁוֹמֵר וּמֵגֵן עַל הָאָדָם כָּל עוֹד הוּא עָסוּק בָּהּ בִּשְׁקִידָה וּבְאַהֲבָה.
