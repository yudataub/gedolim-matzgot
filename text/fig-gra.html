הַבְּכִי עַל הַצִּיצִית

הַגָּאוֹן רַבִּי אֵלִיָּהוּ מִוִּילְנָא, שֶׁכֻּנָּה גַּם בְּשֵׁם הַגְּרָא, הַגָּאוֹן מִוִּילְנָא, אוֹ "הֶחָסִיד", יָשַׁב בָּעִיר וִילְנָא כַּאֲשֶׁר הָיָה סָמוּךְ מַמָּשׁ לִפְטִירָתוֹ. סְבִיבוֹ שֶׁל הָרַב הַגָּדוֹל עָמְדוּ תַּלְמִידָיו, אֲשֶׁר הָיוּ לְצִדּוֹ בְּאוֹתָן דַּקּוֹת מְרַגְּשׁוֹת וַחֲשׁוּבוֹת. פִּתְאֹם, מַמָּשׁ לִפְנֵי פְּטִירָתוֹ, אָחַז הַגָּאוֹן מִוִּילְנָא אֶת הַצִּיצִיּוֹת שֶׁהָיוּ עָלָיו בִּבְגָדוֹ. הוּא הִבִּיט בַּצִּיצִיּוֹת הָאֵלֶּה שֶׁהֶחֱזִיק בְּיָדָיו, וּפִתְאֹם הִתְחִיל לִבְכּוֹת בְּכִי תַּמְרוּרִים. הַבְּכִי שֶׁל הָרַב הָיָה חָזָק מְאֹד, וּדְמָעוֹת רַבּוֹת יָרְדוּ מֵעֵינָיו בְּהִתְרַגְּשׁוּת גְּדוֹלָה.

תַּלְמִידָיו אֲשֶׁר הָיוּ עִמּוֹ וְלְצִדּוֹ בַּחֶדֶר רָאוּ אֶת הַבְּכִי הַכּוֹאֵב וְהַגָּדוֹל שֶׁל רַבָּם. הַתַּלְמִידִים תָּמְהוּ מְאֹד עַל כָּךְ וְלֹא הֵבִינוּ מָה מְבִיא אֶת הָרַב הַגָּדוֹל לִבְכּוֹת כָּכָה. פָּנוּ תַּלְמִידָיו אֵלָיו וְשָׁאֲלוּ אוֹתוֹ בִּדְאָגָה רַבָּה: "הָרַב, לָמָּה אַתָּה בּוֹכֶה? מָה קָרָה?". הַתַּלְמִידִים רָצוּ מְאֹד לְהָבִין מָה הַסִּבָּה לַבְּכִי הַגָּדוֹל וּמָה מְצַעֵר אֶת לִבּוֹ שֶׁל הַגָּאוֹן מִוִּילְנָא בְּרֶגַע זֶה.

עָנָה לָהֶם הַגָּאוֹן מִוִּילְנָא וְהִסְבִּיר לְתַלְמִידָיו אֶת הַסִּבָּה לִבְכִיָּתוֹ הַנִּרְגֶּשֶׁת. הוּא אָמַר לָהֶם: "תִּרְאוּ, כָּאן בָּעוֹלָם הַזֶּה, בְּכַמָּה פְּרוּטוֹת, בְּכַמָּה שְׁקָלִים, אֲנִי יָכוֹל לִקְנוֹת צִיצִית, וּבְכָל רֶגַע וָרֶגַע לְקַיֵּם מִצְווֹת שֶׁעֲלֵיהֶן אֲנִי אֶקַבֵּל שָׂכָר לְאֵין עֲרֹךְ". הַגָּאוֹן מִוִּילְנָא הִסְבִּיר לָהֶם עַד כַּמָּה קַל וּפָשׁוּט לְקַיֵּם אֶת מִצְוַת הַצִּיצִית כְּשֶׁאֲנַחְנוּ חַיִּים בָּעוֹלָם הַזֶּה. כָּל מָה שֶׁצָּרִיךְ הוּא בְּסַךְ הַכֹּל סְכוּם קָטָן שֶׁל כַּמָּה פְּרוּטוֹת אוֹ כַּמָּה שְׁקָלִים כְּדֵי לִקְנוֹת בֶּגֶד עִם צִיצִית, וּבְכָל רֶגַע שֶׁלּוֹבְשִׁים אוֹתוֹ זוֹכִים לְקַיֵּם מִצְוָה שֶׁהַשָּׂכָר עָלֶיהָ הוּא לְאֵין עֲרוּךְ.

אָז הִמְשִׁיךְ הַגָּאוֹן מִוִּילְנָא וְאָמַר לְתַלְמִידָיו אֶת הַדְּבָרִים הָעֲמֻקִּים שֶׁנָּגְעוּ יָשָׁר לְלִבָּם. הוּא אָמַר לָהֶם: "אֲבָל לְאַחַר מֵאָה וְעֶשְׂרִים כְּבָר לֹא אוּכַל לְקַיֵּם אֶת הַמִּצְווֹת הָאֵלֶּה בְּכָזוֹ קַלּוּת". מִכֵּיוָן שֶׁהָיָה סָמוּךְ מַמָּשׁ לִפְטִירָתוֹ, הָרַב הֵבִין כִּי הוּא עוֹמֵד לַעֲזֹב אֶת הָעוֹלָם הַזֶּה. הוּא יָדַע שֶׁבָּעוֹלָם הַבָּא כְּבָר אִי אֶפְשָׁר לֶאֱסֹף עוֹד מִצְווֹת חֲדָשׁוֹת כְּמוֹ שֶׁאֶפְשָׁר לַעֲשׂוֹת בְּקַלּוּת כָּזֹאת בָּעוֹלָם הַזֶּה.

בְּמַעֲשֶׂה זֶה וּבַבְּכִי הַגָּדוֹל שֶׁלּוֹ, רָצָה הַגָּאוֹן מִוִּילְנָא לְהַעֲבִיר לְתַלְמִידָיו וּלְכָל אֶחָד מֵאִתָּנוּ מַסָּר חָשׁוּב מְאֹד. הָרַב רָצָה לְהַרְאוֹת לָנוּ עַד כַּמָּה חֲשׁוּבוֹת וִיקָרוֹת הַמִּצְווֹת שֶׁאָנוּ יְכוֹלִים לְקַיֵּם בְּכָל יוֹם וָיוֹם בְּחַיֵּינוּ. הוּא רָצָה לְהַדְגִּישׁ בִּמְיֻחָד עַד כַּמָּה חֲשׁוּבָה מִצְוַת הַצִּיצִית, שֶׁנּוֹתֶנֶת לָנוּ אֶת הַזְּכוּת וְאֶת הַהִזְדַּמְנוּת לֶאֱסֹף שָׂכָר עָצוּם בְּקַלּוּת רַבָּה.

💡 עָלֵינוּ לִשְׂמֹחַ בְּכָל מִצְוָה שֶׁאָנוּ יְכוֹלִים לְקַיֵּם בָּעוֹלָם הַזֶּה, וּבִמְיֻחָד בְּמִצְוַת הַצִּיצִית הַיְקָרָה.
