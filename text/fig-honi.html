חוֹנִי הַמְּעַגֵּל וְעֵץ הַחֲרוּבִים

הָיָה הָיָה אָדָם חָכָם וְדָגוּל, וּשְׁמוֹ חוֹנִי הַמְּעַגֵּל. יוֹם אֶחָד, חוֹנִי הַמְּעַגֵּל הָיָה מִתְהַלֵּךְ לוֹ בַּדֶּרֶךְ וְהָלַךְ בְּנַחַת. כְּשֶׁהָיָה בַּדֶּרֶךְ, הוּא הִבִּיט סְבִיבוֹ וְרָאָה מַרְאֶה מְעַנְיֵן וּמְעוֹרֵר מַחְשָׁבָה. וְהִנֵּה, הוּא רוֹאֶה אִישׁ מְבוּגָר אֶחָד, שֶׁעוֹמֵד בַּדֶּרֶךְ, עוֹדֵר בָּאֲדָמָה וְשׁוֹתֵל שְׁתִיל. הָאִישׁ הַמְּבוּגָר הָיָה עָסוּק בַּעֲבוֹדָתוֹ, עָדַר בָּאֲדָמָה בִּשְׁקִידָה, וְשָׁתַל אֶת הַשְּׁתִיל הַקָּטָן בְּזֶהִירוּת וּבְאַהֲבָה.

חוֹנִי הַמְּעַגֵּל הִתְקָרֵב אֶל הָאִישׁ הַמְּבוּגָר, נִגַּשׁ אֵלָיו בְּכָבוֹד רַב, וּפָנָה אֵלָיו בִּשְׁאֵלָה. אָמַר לוֹ חוֹנִי הַמְּעַגֵּל: "סְלִיחָה אֲדֹנִי, אַתָּה כָּל כָּךְ מְבוּגָר, מָה אַתָּה שׁוֹתֵל?".

הָאִישׁ הַמְּבוּגָר הִפְסִיק אֶת עֲבוֹדָתוֹ בָּאֲדָמָה, הִבִּיט בְּחוֹנִי הַמְּעַגֵּל, וְעָנָה לוֹ בְּסַבְלָנוּת וּבְנַחַת. הָאִישׁ עָנָה: "אֲנִי שׁוֹתֵל עֵץ חֲרוּבִים".

כְּשֶׁשָּׁמַע חוֹנִי הַמְּעַגֵּל אֶת דִּבְרֵי הָאִישׁ, הוּא תָּמַהּ עַל כָּךְ מְאֹד וְהִתְפַּלֵּא עַל מַעֲשֵׂהוּ. חוֹנִי הַמְּעַגֵּל יָדַע כִּי לְעֵץ הַחֲרוּבִים לוֹקֵחַ זְמַן רַב מְאֹד עַד שֶׁהוּא גוֹדֵל וּמוֹצִיא פֵּרוֹת. הוּא יָדַע כִּי מִכֵּיוָן שֶׁלּוֹקֵחַ לְעֵץ הַחֲרוּבִים שִׁבְעִים שָׁנָה עַד שֶׁהוּא מוֹצִיא פֵּרוֹת, יַעַבְרוּ שָׁנִים רַבּוֹת כָּל כָּךְ עַד שֶׁיִּהְיוּ עַל הָעֵץ פֵּרוֹת מְתוּקִים.

לָכֵן, פָּנָה חוֹנִי הַמְּעַגֵּל שׁוּב אֶל הָאִישׁ הַמְּבוּגָר וְאָמַר לוֹ: "סְלִיחָה שֶׁאֲנִי אוֹמֵר, אַתָּה כְּבָר מְבוּגָר, אַתָּה שׁוֹתֵל עֵץ שֶׁיִּצְמַח בְּעוֹד שִׁבְעִים שָׁנָה, וּבְעוֹד שִׁבְעִים שָׁנָה אַתָּה לֹא תִּהְיֶה כָּאן!". חוֹנִי הַמְּעַגֵּל תָּמַהּ מַדּוּעַ אָדָם שֶׁהוּא כְּבָר מְבוּגָר מְאֹד מַשְׁקִיעַ מַאֲמָץ וְעוֹבֵד קָשֶׁה כָּל כָּךְ בִּשְׁתִילַת עֵץ שֶׁיִּגְדַּל רַק בְּעוֹד שִׁבְעִים שָׁנָה, כַּאֲשֶׁר הוּא עַצְמוֹ לֹא יִזְכֶּה לֶאֱכֹל מִפֵּרוֹתָיו.

אָמַר לוֹ אוֹתוֹ מְבוּגָר בְּחִיּוּךְ וּבְנַחַת: "אַתָּה צוֹדֵק, בְּעוֹד שִׁבְעִים שָׁנָה אֲנִי לֹא אֶהְיֶה כָּאן. אֲבָל כְּשֶׁיֵּצְאוּ הַפֵּרוֹת, הַיְּלָדִים שֶׁלִּי יִזְכּוּ לֶאֱכֹל מֵהֶם, וְהַנְּכָדִים שֶׁלִּי יִזְכּוּ לֶאֱכֹל מֵהֶם. לֹא כָּל מָה שֶׁבֶּן אָדָם עוֹשֶׂה בָּעוֹלָם הוּא עוֹשֶׂה רַק לְעַצְמוֹ, הוּא עוֹשֶׂה גַּם לַאֲחֵרִים".

חוֹנִי הַמְּעַגֵּל שָׁמַע אֶת הַדְּבָרִים הַחֲכָמִים שֶׁל הָאִישׁ הַמְּבוּגָר, שִׂמַּח אוֹתוֹ הַמַּעֲנֶה הַנִּפְלָא, וְהוּא הֵבִין אֶת הַמַּסָּר הֶחָשׁוּב. הוּא רָאָה כַּמָּה זֶה יָפֶה וּמַשְׁמָעוּתִי לִדְאֹג לַדּוֹרוֹת הַבָּאִים, וְלַעֲשׂוֹת מַעֲשִׂים טוֹבִים שֶׁיַּעַזְרוּ לַאֲחֵרִים גַּם בֶּעָתִיד.
