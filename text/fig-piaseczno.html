הַשִּׂמְחָה שֶׁנִּצְחָה אֶת הַחֹשֶׁךְ

בִּזְמַן הַשּׁוֹאָה הַנּוֹרָאָה, בְּתוֹךְ מַחֲנֵה הַשְׁמָדָה גֶּרְמָנִי, חָיוּ יְהוּדִים רַבִּים בְּתַנְאִים קָשִׁים מְאֹד. הֵם סָבְלוּ מֵעֶצֶב גָּדוֹל, מִכְּאֵב עָמֹק וּמִדַּאֲגָה רַבָּה. אַךְ גַּם בַּמָּקוֹם הַקָּשֶׁה הַזֶּה, הָיוּ אֲנָשִׁים שֶׁשָּׁמְרוּ בְּלִבָּם עַל אֱמוּנָה חֲזָקָה בַּה' יִתְבָּרַךְ וְעַל אַהֲבָה גְּדוֹלָה לַתּוֹרָה וְלַמִּצְוֹת.

אֶחָד מֵהָאֲנָשִׁים הָאֵלֶּה הָיָה יְהוּדִי יָקָר בְּשֵׁם דָּוִד וַיְס. יוֹם אֶחָד, בְּתוֹךְ הַמַּחֲנֶה הַגֶּרְמָנִי, רָאָה דָּוִד אֵצֶל חַיָּל גֶּרְמָנִי דַּף אֶחָד מִתּוֹךְ סֵפֶר "שֻׁלְחָן עָרוּךְ". דָּוִד רָצָה מְאֹד לִלְמֹד תּוֹרָה מִתּוֹךְ סֵפֶר כָּתוּב, וְהִרְגִּישׁ רָצוֹן עַז לִזְכּוֹת בַּדַּף הַקָּדוֹשׁ. הוּא נָשָׂא אֶת עֵינָיו וְהִתְפַּלֵּל בְּלִבּוֹ לַה', וְאָז פָּנָה אֶל הַחַיָּל הַגֶּרְמָנִי וּבִקֵּשׁ מִמֶּנּוּ לָתֵת לוֹ אֶת הַדַּף כְּדֵי שֶׁיּוּכַל לִלְמֹד תּוֹרָה. הַחַיָּל הַגֶּרְמָנִי כָּעַס וְעָנָה לוֹ: "אַתָּה יָכוֹל לָתֵת לִי שֶׁקֶט לֶאֱכֹל?! קַח אֶת זֶה וְלֵךְ מִפֹּה!" וּמָסַר לוֹ אֶת הַדַּף. דָּוִד שָׂמַח מְאֹד, הוֹדָה לַה' עַל הַזְּכוּת הַגְּדוֹלָה, וּמֵאוֹתוֹ יוֹם הָיָה לוֹמֵד בְּכָל יוֹם מִתּוֹךְ הַדַּף וּמְלַמֵּד אֶת כָּל אַנְשֵׁי הַמַּחֲנֶה תּוֹרָה.

כַּאֲשֶׁר הִגִּיעַ חַג שִׂמְחַת תּוֹרָה, הָיוּ כָּל הַיְּהוּדִים בַּמַּחֲנֶה עֲצוּבִים וְכוֹאֲבִים מְאֹד. הֵם סָבְלוּ מִקְּשָׁיִם רַבִּים, וְלֹא יָדְעוּ כֵּיצַד יוּכְלוּ לַחְגֹּג וְלִשְׂמֹחַ בַּחַג. בַּמַּחֲנֶה שָׁהָה גַּם הָאַדְמוֹ"ר מִפִּיאַסֶצְנָה. הָאַדְמוֹ"ר הָיָה מַנְהִיג רוּחָנִי שֶׁתָּמִיד הָיָה רָגִיל לְרוֹמֵם וּלְחַזֵּק אֶת כָּל הַצִּבּוּר. אַךְ מַצָּבוֹ הָיָה קָשֶׁה וְכוֹאֵב מְאֹד: הָאַדְמוֹ"ר שִׁכֵּל אֶת שְׁנֵי יְלָדָיו שֶׁנֶּהֶרְגוּ מוּל עֵינָיו, וְרַק כַּמָּה שָׁבוֹעוֹת לִפְנֵי כֵן קָם מִיְּמֵי הַשִּׁבְעָה. כָּל הַחֲסִידִים וְהַנִּמְצָאִים בַּמַּחֲנֶה תָּהוּ וְדָאֲגוּ בְּלִבָּם: הַאִם יִהְיֶה לָאַדְמוֹ"ר כֹּחַ לִהְיוֹת הַחַזָּן וּלְהוֹבִיל אֶת שִׂמְחַת הַחַג?

וְהִנֵּה, כַּאֲשֶׁר הִגִּיעַ זְמַן הַתְּפִלָּה בְּחַג שִׂמְחַת תּוֹרָה, הִגִּיעַ הָאַדְמוֹ"ר מִפִּיאַסֶצְנָה אֶל הָאוּלָם שֶׁבּוֹ הִתְכַּנְּסוּ הַיְּהוּדִים. הוּא הִבִּיט בַּחֲסִידִים הַדּוֹאֲגִים, פָּנָה אֲלֵיהֶם בְּאַהֲבָה רַבָּה וְאָמַר: "יְהוּדִים יְקָרִים, חַג הַיּוֹם, חַג שִׂמְחַת תּוֹרָה! בּוֹאוּ וְנִגַּשׁ לַתְּפִלָּה...". אָז עָמַד הָאַדְמוֹ"ר, וּבֶאֱמוּנָה עֲמֻקָּה וּבְעֹצְמָה גְּדוֹלָה קָרָא אֶת מִלּוֹת הַתְּפִלָּה: "בָּרְכוּ אֶת ה' הַמְּבֹרָךְ!"

בְּאוֹתוֹ רֶגַע מַמָּשׁ, חָשׁוּ כָּל הַנִּמְצָאִים בָּאוּלָם זֶרֶם חָזָק מְאֹד שֶׁעָבַר בְּכָל הָאוּלָם. הַקּוֹל הַחַזָּק, מִלּוֹת הַתְּפִלָּה וְהָאֱמוּנָה הָעַזָּה שֶׁל הָאַדְמוֹ"ר מִפִּיאַסֶצְנָה נִכְנְסוּ יָשָׁר לְלִבּוֹ שֶׁל כָּל אֶחָד וְאֶחָד. בְּאוֹתָהּ שָׁעָה, הָעֶצֶב וְהַכְּאֵב הַגָּדוֹל פִּנּוּ אֶת מְקוֹמָם, וְכָל הַיְּהוּדִים קִבְּלוּ כֹּחַ מְחֻדָּשׁ לְהִתְמוֹדֵד עִם כָּל הַקְּשָׁיִם. הֵם הִרְגִּישׁוּ מְלֵאֵי אֱמוּנָה, וְהָיוּ מוּכָנִים לִרְקֹד וְלִשְׂמֹחַ בְּכָל הַכֹּחַ לִכְבוֹד שִׂמְחַת תּוֹרָה.

💡 גַּם בָּרְגָעִים הַקָּשִׁים בְּיוֹתֵר, הָאֱמוּנָה בַּה' וְשִׂמְחַת הַתּוֹרָה נֹתְנוֹת לָנוּ כֹּחַ וְאוֹר.
