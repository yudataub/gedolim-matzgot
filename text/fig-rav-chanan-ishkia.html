הַלַּפִּיד שֶׁלֹּא כָבָה

בְּאֶרֶץ בָּבֶל הָעַתִּיקָה, בִּתְקוּפָה מְיֻחֶדֶת שֶׁנִּקְרֵאת תְּקוּפַת רַבָּנַן סָבוֹרָאֵי, פָּעֲלוּ שְׁתֵּי יְשִׁיבוֹת תּוֹרָה גְּדוֹלוֹת וּמְפֻרְסָמוֹת: יְשִׁיבַת סוּרָא וִישִׁיבַת פּוּמְבְּדִיתָא. הַחֲכָמִים בְּאוֹתָם יָמִים נִקְרְאוּ "סָבוֹרָאִים" מִשּׁוּם שֶׁהֵם לֹא חִדְּשׁוּ הֲלָכוֹת חֲדָשׁוֹת לְאַחַר חֲתִימַת הַתַּלְמוּד, אֶלָּא הִסְבִּירוּ, בֵּאֲרוּ וְסִדְּרוּ אֶת דִּבְרֵי הַגְּמָרָא בְּהֶסְבֵּרִים בְּהִירִים כְּדֵי שֶׁכָּל יְהוּדִי יוּכַל לַהֲבִינָם. בְּבָתֵּי הַמִּדְרָשׁ לָמְדוּ תַּלְמִידִים רַבִּים בְּאַהֲבַת הַתּוֹרָה וּבְשִׂמְחָה עֲצוּמָה, וְקוֹל הַלִּמּוּד נִשְׁמַע בְּכָל מָקוֹם. אַךְ לְאַחַר שֶׁנִּפְטְרוּ מַנְהִיגֵי הַדּוֹר, רַב עֵינָא רֹאשׁ יְשִׁיבַת סוּרָא וְרַב סִימוֹנָא רֹאשׁ יְשִׁיבַת פּוּמְבְּדִיתָא, הִגִּיעָה שָׁעָה קָשָׁה מְאֹד לְעַם יִשְׂרָאֵל. מַלְכֵי פָּרַס הַסַּאסָאנִיִּים הָיוּ מְלֵאֵי שִׂנְאָה, וְהֵם גָּזְרוּ גְּזֵרוֹת קָשׁוֹת וּשְׁמָדוֹת נוֹרָאִים כְּדֵי לְהַפְסִיק אֶת לִמּוּד הַתּוֹרָה בָּרַבִּים. בִּגְלַל הָרְדִיפוֹת הַקָּשׁוֹת וְהַסַּכָּנָה הַגְּדוֹלָה, נֶאֶלְצוּ הַחֲכָמִים לִסְגֹּר אֶת הַיְשִׁיבוֹת, וְהַדְּלָתוֹת הַגְּדוֹלוֹת שֶׁל בָּתֵּי הַמִּדְרָשׁ נִנְעֲלוּ בְּמַנְעוּלֵי בַּרְזֶל כְּבֵדִים.

בְּמֶשֶׁךְ כְּחֲמִשִּׁים שָׁנָה עָמְדוּ הַיְשִׁיבוֹת הַגְּדוֹלוֹת שׁוֹמְמוֹת, וְהַדְּלָתוֹת נִשְׁאֲרוּ סְגוּרוֹת וַחֲתוּמוֹת. יְלָדִים, נְעָרִים וּמְבֻגָּרִים הִתְגַּעְגְּעוּ מְאֹד לְקוֹל הַתּוֹרָה הַמָּתוֹק, אַךְ בָּרְחוֹבוֹת שָׂרַר שֶׁקֶט עָצוּב וְאִישׁ לֹא הֵעֵז לִפְתֹּחַ אֶת הַשְּׁעָרִים. וְאוּלָם, מְסֻפָּר שֶׁהַלַּפִּיד הַקָּדוֹשׁ שֶׁל הַתּוֹרָה לֹא כָּבָה לְעוֹלָם. מְסֻפָּר כִּי בְּכָל אוֹתָן שָׁנִים אֲפֵלוֹת, הַחֲכָמִים וְתַלְמִידֵיהֶם, יַחַד עִם יְלָדִים צְעִירִים, לֹא אִבְּדוּ אֶת תִּקְוָתָם. הֵם הִמְשִׁיכוּ לִלְמֹד תּוֹרָה בַּסֵּתֶר, בְּמַחֲבוֹאִים וּבַחֲדָרִים פְּנִימִיִּים, בִּמְסִירוּת נֶפֶשׁ אֲמִתִּית וּבְאַהֲבַת הַשֵּׁם עֲמֻקָּה, כְּדֵי לִשְׁמֹר עַל הַגַּחֶלֶת שֶׁלֹּא תִּכְבֶּה. תַּלְמִידֵי הַסָּבוֹרָאִים הֶעֱבִירוּ אֶת מָסֹרֶת הַגְּמָרָא מֵאִישׁ לְרֵעֵהוּ בִּצְנִיעוּת וּבְעֹז רוּחַ.

כַּאֲשֶׁר חָלְפוּ כַּחֲמִשִּׁים שָׁנָה וְהַגְּזֵרוֹת נִרְגְּעוּ מְעַט, קָם מַנְהִיג אַמִּיץ וְקָדוֹשׁ שֶׁשְּׁמוֹ רַב חָנָן מֵאִישְׁקִיָא, שֶׁקִּבֵּל תּוֹרָה מִתַּלְמִידֵי רַב סִימוֹנָא וְרַב עֵינָא. רַב חָנָן הֵבִין שֶׁעַם יִשְׂרָאֵל אֵינוֹ יָכוֹל לְהַמְשִׁיךְ לִלְמֹד רַק בַּסֵּתֶר, וְשֶׁהִגִּיעַ הַזְּמַן לְהַחֲזִיר אֶת קוֹל הַתּוֹרָה לָרַבִּים. בִּשְׁנַת ד' אֲלָפִים שְׁלוֹשׁ מֵאוֹת אַרְבָּעִים וָתֵשַׁע לִבְרִיאַת הָעוֹלָם (ד'שְׁמַ"ט), יָצָא רַב חָנָן בְּאֹמֶץ לֵב וּבִטָּחוֹן גָּמוּר בַּה' בְּרֹאשׁ סִיעַת חֲכָמִים גְּדוֹלָה אֶל הָעִיר פּוּמְבְּדִיתָא. הוּא צָעַד יָשָׁר אֶל בֵּית הַמִּדְרָשׁ הַגָּדוֹל, נִגַּשׁ אֶל הַשְּׁעָרִים הַנְּעוּלִים, פָּתַח אֶת דַּלְתוֹת הַיְשִׁיבָה לִרְוָחָה, וְקָרָא בְּקוֹל גָּדוֹל לְכָל הַתַּלְמִידִים לָשׁוּב אֶל סַפְסְלֵי הַלִּמּוּד וְאֶל הַסְּפָרִים.

בְּאוֹתוֹ יוֹם מְרַגֵּשׁ חָזַר סֵדֶר הַלִּמּוּדִים הַקָּבוּעַ כְּבֶעָבָר, וְקוֹל הַתּוֹרָה הַפֻּמְבִּי חָזַר לְהַדְהֵד בְּעָצְמָה רַבָּה בְּכָל רַחֲבֵי הָעִיר וּבְכָל אַרְץ בָּבֶל. הַשִּׂמְחָה בְּקֶרֶב הַיְּהוּדִים הָיְתָה עֲצוּמָה, וּמַעֲשֵׂהוּ שֶׁל רַב חָנָן מֵאִישְׁקִיָא הֵאִיר כְּמוֹ מַשּׂוּאָה עֲנָקִית בַּחֹשֶׁךְ. אִגֶּרֶת רַב שְׁרִירָא גָּאוֹן מְסַפֶּרֶת כִּי פְּתִיחַת הַיְשִׁיבָה עַל יְדֵי רַב חָנָן סִמְּנָה אֶת תְּחִלָּתָהּ שֶׁל תְּקוּפַת הַגְּאוֹנִים, וּמֵאָז נִמְשַׁךְ רֶצֶף לִמּוּד הַתּוֹרָה עַד שֶׁהִתְחַדְּשָׁה לְאַחַר מִכֵּן גַּם יְשִׁיבַת סוּרָא בְּרָאשׁוּת רַב מָר בַּר רַב הוּנָא. כָּךְ נִשְׁמְרָה שַׁרְשֶׁרֶת הַדּוֹרוֹת הַקְּדוֹשָׁה, וְאוֹר הַתּוֹרָה לֹא כָּבָה לְעוֹלָם.

💡 גַּם כְּשֶׁיֵּשׁ קְשָׁיִים וְחֹשֶׁךְ, אוֹר הַתּוֹרָה תָּמִיד חוֹזֵר לִזְרֹחַ בִּזְכוּת מְסִירוּת נֶפֶשׁ, אֹמֶץ לֵב וְאַהֲבַת הַשֵּׁם.
