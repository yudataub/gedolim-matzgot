הַדֶּלֶת הַפְּתוּחָה שֶׁל הַמַּהַרְשָׁ"א

לִפְנֵי שָׁנִים רַבּוֹת, בְּעִיר קָזִ'ימְיֶז' שֶׁלְּיַד קְרָקוֹב שֶׁבְּאֶרֶץ פּוֹלִין, נוֹלַד יֶלֶד מְיֻחָד מְאֹד בְּשֵׁם שְׁמוּאֵל אֱלִיעֶזֶר הַלֵּוִי, אֲשֶׁר הָיָה מִמִּשְׁפַּחַת הַמַּהֲרַ"ל מִפְּרָאג. כְּבָר מִיְּמֵי יַלְדוּתוֹ הַצְּעִירָה, נוֹדַע הַיֶּלֶד כְּעִלּוּי חָכָם וּכְשְׁרוֹנִי יוֹצֵא דֹּפֶן, הַלּוֹמֵד תּוֹרָה בִּשְׁקִידָה גְּדוֹלָה וּבְאַהֲבָה רַבָּה. כַּאֲשֶׁר גָּדַל, הָפַךְ לְאֶחָד מִגְּדוֹלֵי חַכְמֵי יִשְׂרָאֵל, הַמֻּכָּר בְּשֵׁם הַמַּהַרְשָׁ"א. הוּא כִּהֵן כְּרַב מְפֻרְסָם בְּשָׁלֹשׁ עָרִים חֲשׁוּבוֹת: בְּחֶלְם, בְּטִיקְטִין וּבְאוֹסְטְרָהָא, וְשָׁם גַּם עָמַד בְּרֹאשׁ יְשִׁיבָה גְּדוֹלָה וְלִמֵּד תַּלְמִידִים רַבִּים מִכָּל הָאֵזוֹר.

הַמַּהַרְשָׁ"א חִבֵּר שְׁנֵי סְפָרִים חֲשׁוּבוֹת וִיסוֹדִיִּים מְאֹד עַל הַגְּמָרָא. הַסֵּפֶר הָרִאשׁוֹן נִקְרָא "חִדּוּשֵׁי הֲלָכוֹת", וּבוֹ מְפֹרָשִׁים דִּבְרֵי הַגְּמָרָא, פֵּרוּשׁ רַשְׁאִ"י וּדְבָרֵי הַתּוֹסָפוֹת. הַסֵּפֶר הַשֵּׁנִי נִקְרָא "חִדּוּשֵׁי אַגָּדוֹת", וּבוֹ מְפֹרָשִׁים דִּבְרֵי הָאַגָּדָה וְהַסִּפּוּרִים שֶׁל חֲזַ"ל בַּגְּמָרָא. עַד הַיּוֹם הַזֶּה, הַפֵּרוּשׁ הַמְּפֹאָר שֶׁל הַמַּהַרְשָׁ"א מֻדְפָּס כִּמְעַט בְּכָל סֵפֶר גְּמָרָא, וְתַלְמִידֵי חֲכָמִים וּבְנֵי יְשִׁיבוֹת לוֹמְדִים בּוֹ בְּכָל יَوْם.

אַךְ הַמַּהַרְשָׁ"א לֹא הָיָה רַק גָּאוֹן עָצוּם בַּתּוֹרָה, אֶלָּא הָיָה מְפֻרְסָם גַּם בְּמַעֲשֵׂי חֶסֶד רַבִּים מְאֹד. עַל דֶּלֶת בֵּיתוֹ הָיָה כָּתוּב פָּסוּק מְיֻחָד מִסֵּפֶר אִיּוֹב: "דַּלְתַי לָאֹרַח אֶפְתָּח". כָּל אָדָם שֶׁעָבַר לְיַד בֵּיתוֹ יָדַע שֶׁהַדֶּלֶת שֶׁל הָרַב פְּתוּחָה תָּמִיד עֲבוּר כָּל אוֹרֵחַ. כָּל עָנִי, כָּל עוֹבֵר אֹרַח וְכָל תַּלְמִיד שֶׁהִגִּיעַ לָעִיר יָדַע שֶׁבְּבֵית הַמַּהַרְשָׁ"א יֵשׁ לוֹ מָקוֹם בָּטוּחַ וְחַם. בְּבֵית הָרַב קִבְּלוּ הָאוֹרְחִים אֹכֶל, שְׁתִיָּה וּמָקוֹם נָעִים לִישֹׁן, וְהַכֹּל בְּסֵבֶר פָּנִים יָפוֹת וּבְפָנִים מְאִירוֹת.

דָּבָר מְיֻחָד נוֹסָף שֶׁלָּמַדְנוּ עַל הַמַּהַרְשָׁ"א הוּא מִדַּת הַכָּרַת הַתּוֹדָה הָעֲמֻקָּה שֶׁלּוֹ. כְּשֶׁהָיָה צָעִיר, מִשְׁפָּחָה נְדִיבָה פִּרְנְסָה אוֹתוֹ וּמִמְּנָה אֶת כָּל הוֹצָאוֹת הַיְּשִׁיבָה שֶׁלּוֹ, כְּדֵי שֶׁיּוּכַל לִלְמֹד וּלְלַמֵּד תּוֹרָה בְּשַׁלְוָה וּבְלִי דְּאָגוֹת. הַמַּהַרְשָׁ"א זָכַר אֶת הַטּוֹבָה הַזֹּאת כָּל יְמֵי חַיָּיו וְלֹא שָׁכַח אוֹתָהּ לְעוֹלָם. הוּא אֲפִלּוּ הוֹסִיף לִשְׁמוֹ אֶת הַשֵּׁם "אַיְדֶלְס" עַל שֵׁם הָאִשָּׁה הַנְּדִיבָה שֶׁתָּמְכָה בּוֹ, וְתָמִיד הִזְכִּיר אוֹתָהּ בְּתוֹדָה רַבָּה.

כְּמוֹ כֵן, הַמַּהַרְשָׁ"א דָּאַג מְאֹד לִכְבוֹד הַתּוֹרָה. הוּא הָיָה מֵרָאשֵׁי "וַעַד אַרְבַּע אֲרָצוֹת", וְחָתַם עַל תַּקָּנָה מְיֻחֶדֶת שֶׁאֲסָרָה לִקְנוֹת מִשְׁרוֹת רַבָּנוּת בְּכֶסֶף, כְּדֵי שֶׁרַק תַּלְמִידֵי חֲכָמִים רְאוּיִם יִהְיוּ רַבָּנִים בְּיִשְׂרָאֵל.
