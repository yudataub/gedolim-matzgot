מְסִירוּת הַתּוֹרָה שֶׁל הָרַב עוֹבַדְיָה

כְּאַרְבַּע עֶשְׂרֵה שָׁנִים לִפְנֵי שֶׁנִּפְטַר מָרָן הָרַב עוֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק וְקָדוֹשׁ לִבְרָכָה, כַּאֲשֶׁר הָיָה כִּבְן שְׁמוֹנִים שָׁנִים, הוּא חָשׁ כְּאֵבִים חֲזָקִים בְּלִבּוֹ. הָרַב הֻבְהַל לְבֵית הַחוֹלִים, וְהָרוֹפְאִים הַמֻּמְחִים שֶׁבָּדְקוּ אוֹתוֹ קָבְעוּ כִּי הָרַב חַיָּב לַעֲבֹר נִתּוּחַ וְצִנְתּוּר בִּלְבָבוֹ. בְּנֵי מִשְׁפַּחְתּוֹ הָאֲהוּבִים שֶׁל הָרַב דָּאֲגוּ מְאֹד וְחָרְדוּ לִשְׁלוֹמוֹ, אַךְ הָרוֹפְאִים הַמּוֹסוּרִים הִרְגִּיעוּ אוֹתָם וְהִבְטִיחוּ לָהֶם כִּי כַּעֲבֹר כַּמָּה יְמֵי אִשְׁפּוּז הָרַב יַחֲזֹר לְאֵיתָנוֹ, יַבְרִיא לְגַמְרֵי וִישֻׁחְרַר בְּשָׁלוֹם לְבֵיתוֹ. הַנִּתּוּחַ הַחָשׁוּב נִקְבַּע לְיוֹם מְסֻיָּם, וּבִבְקָרוֹ שֶׁל אוֹתוֹ יוֹם הִגִּיעוּ הָרוֹפְאִים לְחַדְרוֹ שֶׁל הָרַב וּבִקְּשׁוּ לְהָכִין אוֹתוֹ לִקְרַאת הַטִּפּוּל הָרְפוּאִי.

אוּלָם מָרָן הָרַב עוֹבַדְיָה יוֹסֵף פָּנָה אֶל הָרוֹפְאִים וּבִקֵּשׁ מֵהֶם בַּקָּשָׁה מְיֻחֶדֶת וְיוֹצֵאת דֹּפֶן שֶׁהִפְתִיעָה אוֹתָם מְאֹד. הָרַב אָמַר לָהֶם בְּנַחַת: "אָנָּא, תְּנוּ לִי חֻפְשָׁה שֶׁל שָׁלוֹשׁ שָׁעוֹת בִּלְבַד מִבֵּית הַחוֹלִים, וּבְעוֹד שָׁלוֹשׁ שָׁעוֹת אֶחֱזֹר לְכָאן שׁוּב עַל מְנָת לַעֲבֹר אֶת הַנִּתּוּחַ". הָרוֹפְאִים הַנִּדְהָמִים תָּמְהוּ בֵּינָם לְבֵין עַצְמָם וְשָׁאֲלוּ בִּפְלִיאָה: "מַדּוּעַ הָרַב זָקוּק לְחֻפְשָׁה בַּת שָׁלוֹשׁ שָׁעוֹת מִבֵּית הַחוֹלִים מַמָּשׁ לִפְנֵי הַנִּתּוּחַ?". אַךְ הֵם הֶעֱרִיכוּ מְאֹד אֶת הָרַב וְלֹא הֵעֵזּוּ לְהַבִּיעַ אֶת מַחְשְׁבוֹתֵיהֶם הַפְּרָטִיּוֹת בְּפָנָיו. הֵם אִשְּׁרוּ אֶת הַבַּקָּשָׁה וְהִסְכִּימוּ לִדְחוֹת אֶת הַנִּתּוּחַ בְּשָׁלוֹשׁ שָׁעוֹת, וְהָרַב עוֹבַדְיָה יָצָא מִיָּד לְבֵיתוֹ כְּשֶׁהוּא מְלֻוֶּה בִּבְנֵי מִשְׁפַּחְתּוֹ הַקְּרוֹבִים.

כַּאֲשֶׁר הִגִּיעַ הָרַב לְבֵיתוֹ, הוּא לֹא נָח וְלֹא שָׁכַב לִישֹׁן, אֶלָּא מִהֵר יָשָׁר אֶל שֻׁלְחַן הַלִּמּוּד שֶׁלּוֹ. בְּמֶשֶׁךְ שָׁלוֹשׁ הַשָּׁעוֹת הָאֵלֶּה עָבַד הָרַב בִּשְׁקִידָה עֲמֻקָּה וּבְרִכּוּז רַב כְּדֵי לְסַיֵּם כְּתִיבַת פְּסַק הֲלָכָה חָשׁוּב וּמֻרְכָּב מְאֹד בְּעִנְיָנָהּ שֶׁל אִשָּׁה עֲגוּנָה שֶׁהָיְתָה זְקוּקָה לְעֶזְרָתוֹ הַדְּחוּפָה. הָרַב חָשַׁב רַק עַל אוֹתָהּ אִשָּׁה וְרָצָה לַעֲזֹר לָהּ בְּכָל לִבּוֹ. בְּסִיַּעְתָּא דִּשְׁמַיָּא וּבְבָרוּךְ הַשֵּׁם, הָרַב מָצָא אֶת הַפֶּתַח הַהֲלָכָתִי הַנָּכוֹן לְהַתִּיר אֶת הָעֲגוּנָה וּלְסַיֵּם אֶת הַתְּשׁוּבָה הַהֲלָכָתִית.

לְאַחַר שֶׁסִּיֵּם אֶת כְּתִיבַת הַפְּסָק, חָזַר הָרַב עוֹבַדְיָה יֵשָׁר לְבֵית הַחוֹלִים. הוּא הִסְבִּיר בְּנַחַת כִּי בִּקֵּשׁ לִדְחוֹת אֶת הַנִּתּוּחַ בְּשָׁלוֹשׁ שָׁעוֹת כְּדֵי לְסַיֵּם אֶת הַתְּשׁוּבָה הַהֲלָכָתִית וּלְהַתִּיר אֶת הָעֲגוּנָה מִסִּבְלָהּ. רַק אָז, כַּאֲשֶׁר יָדַע שֶׁעָזַר לָאִשָּׁה וְסִיֵּם אֶת מְלַאכְתּוֹ, נִגַּשׁ הָרַב לַנִּתּוּחַ בְּלֵב רָגוּעַ וּבְשִׂמְחָה.
