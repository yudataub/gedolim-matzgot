פַּת לֶחֶם בַּסֵּתֶר

בַּגְּמָרָא בְּמַסֶּכֶת שַׁבָּת מְסֻפָּר עַל שְׁמוּאֵל הָאָמוֹרָא וְעַל חָכָם גּוֹי אֶחָד שֶׁנִּקְרָא אֲבְלָט. אֲבְלָט הָיָה אָדָם שֶׁקּוֹרֵא בַּכּוֹכָבִים, וְהוּא יָדַע לְפִי הַכּוֹכָבִים מָה יִהְיֶה הַגּוֹרָל שֶׁל כָּל בֶּן אָדָם. יוֹם אֶחָד עָמְדוּ שְׁמוּאֵל וַאֲבְלָט יַחְדָּו, וְרָאוּ שְׁנֵי אֲנָשִׁים שֶׁעוֹבְרִים בַּדֶּרֶךְ וְהוֹלְכִים לְדַרְכָּם. הִבִּיט אֲבְלָט בַּכּוֹכָבִים, פָּנָה אֶל שְׁמוּאֵל וְאָמַר לוֹ כִּי אֶחָד מִשְּׁנֵי הָאֲנָשִׁים הָאֵלֶּה שֶׁהוֹלְכִים בַּדֶּרֶךְ לֹא יַחְזֹר מֵהַדֶּרֶךְ, חַס וְשָׁלוֹם, וְהוּא לֹא יִשָּׁאֵר חַי. שְׁמוּאֵל שָׁמַע אֶת דִּבְרֵי אֲבְלָט וְאָמַר לוֹ שֶׁזֶּה אֵינוֹ נָכוֹן. שְׁמוּאֵל הִסְבִּיר כִּי הָאִישׁ הַזֶּה כֵּן יַחְזֹר מֵהַדֶּרֶךְ וְיִהְיֶה חַי, מִפְּנֵי שֶׁלְּכָל אָדָם יֵשׁ בְּחִירָה חָפְשִׁית לַעֲשׂוֹת מַעֲשִׂים טוֹבִים.

וְאָכֵן, כָּךְ בְּדִיּוֹק קָרָה! אוֹתוֹ אָדָם שֶׁאֲבְלָט אָמַר עָלָיו שֶׁלֹּא יַחְזֹר – חָזַר מֵהַדֶּרֶךְ כְּשֶׁהוּא חַי וּבָרִיא. שְׁמוּאֵל פָּנָה אֶל הָאָדָם הַזֶּה וְשָׁאַל אוֹתוֹ אֵיךְ זֶה יָכוֹל לִהְיוֹת? הֲרֵי אֲבְלָט הֶחָכָם הַגּוֹי יָדַע הֵיטֵב עַל מָה הוּא מְדַבֵּר. כְּשֶׁאֲבְלָט אָמַר אֶת הַדְּבָרִים הָאֵלֶּה, הוּא אָמַר זֹאת מִתּוֹךְ כָּךְ שֶׁהוּא יָדַע מָה יִהְיֶה הַגּוֹרָל שֶׁל הָאָדָם לְפִי הַכּוֹכָבִים. אִם הוּא אָמַר שֶׁהוּא לֹא יִחְיֶה, הוּא יָדַע שֶׁזֶּהוּ הַגּוֹרָל שֶׁלּוֹ. אִם כָּךְ, הֵבִין שְׁמוּאֵל, כְּנִרְאֶה שֶׁהָאָדָם עָשָׂה מַשֶּׁהוּ טוֹב מְאֹד שֶׁהִצִּיל אוֹתוֹ מִגְּזַר דִּין שֶׁל מָוֶת, חַס וְשָׁלוֹם. אָמַר לוֹ שְׁמוּאֵל: "בּוֹא, תַּסְבִּיר לִי מָה עָשִׂיתָ!"

סִפֵּר הָאָדָם לִשְׁמוּאֵל עַל מִנְהָג יָפֶה שֶׁהוּא וַחֲבֵרָיו נוֹהֲגִים: כָּל אֶחָד מֵהֶם לוֹקֵחַ פַּת לֶחֶם, שָׂם אוֹתָהּ יַחַד בְּמַאֲגָר מְשֻׁתָּף, וְכָל הַחֲבֵרִים אוֹכְלִים אֶת הַלֶּחֶם יַחְדָּו. בְּאוֹתוֹ יוֹם הָיָה בֵּינֵיהֶם אָדָם אֶחָד שֶׁלֹּא הָיָה לוֹ לֶחֶם כְּלָל, וְהָיָה חָסֵר לוֹ לֶחֶם. אוֹתוֹ בֶּן אָדָם שֶׁחָזַר חַי מֵהַדֶּרֶךְ רָאָה אֶת הַחָבֵר הַזֶּה, וְלֹא רָצָה שֶׁהוּא יִתְבַּיֵּשׁ. הוּא חָשַׁב שֶׁכַּאֲשֶׁר יֶאֶסְפוּ אֶת הַלֶּחֶם מִכָּל אֶחָד, הַחָבֵר יִרְאֶה פִּתְאוֹם שֶׁאֵין לוֹ כְּלוֹם וְיִהְיֶה נָבוֹךְ מְאֹד מוּל כֻּלָּם. מָה עָשָׂה? הוּא הוֹסִיף פַּת לֶחֶם נוֹסֶפֶת מִשֶּׁלּוֹ, וְשָׂם לָאִישׁ הַזֶּה בַּסֵּתֶר בְּתוֹךְ הַיָּד. כְּשֶׁהִגִּיעַ הַזְּמַן לֶאֱסֹף אֶת הַלֶּחֶם מִכָּל אֶחָד, הָאִישׁ שֶׁהָיָה חָסֵר לוֹ לֶחֶם לֹא הִתְבַּיֵּשׁ, כִּי גַּם לוֹ הָיָה לֶחֶם לָתֵת. וְלֹא רַק זֹאת, אֶלָּא כַּאֲשֶׁר אוֹתוֹ אָדָם בָּא לֶאֱסֹף אֶת הַלֶּחֶם מֵאוֹתוֹ חָבֵר, הוּא עָשָׂה אֶת עַצְמוֹ כְּאִלּוּ הוּא אוֹסֵף לֶחֶם שֶׁשַּׁיָּךְ מַמָּשׁ לְאוֹתוֹ בֶּן אָדָם, כְּדֵי שֶׁהוּא לֹא יִרְגִּישׁ נָבוֹךְ כְּלָל.

כַּאֲשֶׁר שָׁמַע שְׁמוּאֵל שֶׁזֶּה מָה שֶׁהָאָדָם עָשָׂה, הוּא אָמַר לוֹ: "עָשִׂיתָ מִצְוָה גְּדוֹלָה, עָשִׂיתָ צְדָקָה, וְלָכֵן נִצַּלְתָּ וְלָכֵן זָכִיתָ בְּרַחֲמִים!" וּמִכָּאן אָמַר שְׁמוּאֵל אֶת הַמַּאֲמָר הַיָּדוּעַ: "וּצְדָקָה תַּצִּיל מִמָּוֶת". הוּא הִסְבִּיר כִּי מִפְּנֵי שֶׁהוּא עָשָׂה צְדָקָה וּמַעֲשֶׂה טוֹב, הוּא זָכָה בְּרַחֲמִים גְּדוֹלִים וְזָכָה לְחַיִּים.

💡 מַעֲשֵׂה צְדָקָה וְחֶסֶד בַּסֵּתֶר מוֹנֵעַ בּוּשָׁה מֵהַחָבֵר, מֵבִיא רַחֲמִים מִשָּׁמַיִם וּמַצִּיל מִכָּל רַע.
