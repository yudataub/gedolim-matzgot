תּוֹלָע בֶּן פּוּאָה – הַשּׁוֹפֵט שֶׁהֶחֱזִיר אֶת הַשֶּׁקֶט

לִפְנֵי שָׁנִים רַבּוֹת עָבְרָה עַל עַם יִשְׂרָאֵל תְּקוּפָה קָשָׁה וּמֻרְכֶּבֶת מְאֹד. בְּאוֹתָם יָמִים קָם אָדָם רַע שֶׁנִּקְרָא אֲבִימֶלֶךְ בֶּן גִּדְעוֹן, אֲשֶׁר רָצָה לִמְלֹךְ עַל הָעָם בְּכֹחַ, בְּגַאֲוָה וּבְאַכְזָרִיּוֹת רַבָּה. אֲבִימֶלֶךְ עָשָׂה מַעֲשֶׂה נוֹרָא וַאֲיֹם מְאֹד, כַּאֲשֶׁר הָרַג אֶת שִׁבְעִים אֶחָיו בְּנֵי גִּדְעוֹן. גַּם אַנְשֵׁי הָעִיר שְׁכֶם עָזְרוּ לוֹ בַּמַּעֲשִׂים הָרָעִים הָאֵלֶּה וְתָמְכוּ בּוֹ. אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרוֹאֶה אֶת כָּל מַעֲשֵׂי בְּנֵי הָאָדָם וְיוֹדֵעַ אֶת כָּל מַה שֶּׁקּוֹרֶה, לֹא וִתֵּר עַל הָרָעָה הַזֹּאת. הַקָּדוֹשׁ בָּרוּךְ הוּא הֵשִׁיב אֶת הָרָעָה שֶׁל אֲבִימֶלֶךְ עַל רֹאשׁוֹ, וְגַם אֶת כָּל הָרָעָה שֶׁעָשׂוּ אַנְשֵׁי שְׁכֶם הֵשִׁיב אֱלֹהִים בְּרֹאשָׁם. כָּךְ הִגִּיעָה עֲלֵיהֶם קִלְלַת יוֹתָם בֶּן יְרֻבַּעַל, וּבַסּוֹף אֲבִימֶלֶךְ נֶעֱנַשׁ כְּפִי שֶׁמַּגִּיעַ לוֹ וּמֵת, וְהָרָעָה הַגְּדוֹלָה שֶׁלּוֹ הֻפְסְקָה.

לְאַחַר שֶׁהִסְתַּיְמָה הַתְּקוּפָה הַקָּשָׁה וְהַסּוֹעֶרֶת שֶׁל אֲבִימֶלֶךְ, עַם יִשְׂרָאֵל הָיָה זָקוּק מְאֹד לְשֶׁקֶט, לְשָׁלוֹם, לִנְחָמָה וּלְהַנְהָגָה טוֹבָה וִישָׁרָה. בַּזְּמַן הַזֶּה קָם מַנְהִיג חָדָשׁ כְּדֵי לְהוֹשִׁיעַ אֶת עַם יִשְׂרָאֵל מֵהַמַּצָּב הַקָּשֶׁה. לַמַּנְהִיג הַחָדָשׁ הַזֶּה קָרְאוּ תּוֹלָע בֶּן פּוּאָה. הוּא הָיָה בֶּן דּוֹדוֹ, וְכָךְ הָיָה שְׁמוֹ הַמְּפֹרָשׁ כְּפִי שֶׁמַּסְבִּיר רַשִׁ"י. תּוֹלָע בֶּן פּוּאָה הָיָה אִישׁ מִשֵּׁבֶט יִשָּׂשכָר. הוּא בָּחַר לָשֶׁבֶת וְלָגוּר בִּמְקוֹם מְיֻחָד שֶׁנִּקְרָא שָׁמִיר, שֶׁנִּמְצָא בְּאֵזוֹר הַר אֶפְרַיִם. מִתּוֹךְ הָעִיר שָׁמִיר הוּא הִנְהִיג אֶת הָעָם, כִּוֵּן אוֹתָם וְשָׁפַט אֶת יִשְׂרָאֵל.

תּוֹלָע בֶּן פּוּאָה שָׁפַט אֶת יִשְׂרָאֵל בֶּאֱמֶת וּבְתָמִים עֶשְׂרִים וָשָׁלֹשׁ שָׁנִים. פֵּרוּשׁ מְצֻדַּת דָּוִד מַסְבִּיר שֶׁהוּא נִפְטַר בְּתוֹךְ שְׁנַת הָעֶשְׂרִים וְשָׁלֹשׁ לִשְׁפִיטָתוֹ, וְהַשָּׁנָה הַזֹּאת נֶחְשְׁבָה גַּם לוֹ וְגַם לַשּׁוֹפֵט יָאִיר שֶׁשָּׁפַט אֶת הָעָם אַחֲרָיו, כְּפִי שֶׁכָּתוּב בְּסֵפֶר סֵדֶר עוֹלָם. בְּכָל שְׁנוֹת הַנְּהָגָתוֹ שֶׁל תּוֹלָע הָיָה שֶׁקֶט גָּדוֹל וּמְבֹרָךְ בְּכָל הָאָרֶץ. הַכָּתוּב בַּתָּנָ"ךְ אֵינוֹ מְסַפֵּר עַל מִלְחָמוֹת כְּלָל בְּכָל יָמָיו, כִּי הוּא הֶחֱזִיר אֶת הַשָּׁלוֹם, אֶת הָאַחֲוָה וְאֶת הַמְּנוּחָה לְעַם יִשְׂרָאֵל. בִּזְכּוּת שְׁפִיטָתוֹ הַטּוֹבָה הָעָם יָכֹל לִחְיוֹת בְּבִטָּחוֹן, לִלְמֹד תּוֹרָה וְלַעֲבֹד אֶת הַשֵּׁם בְּנַחַת, בְּשִׂמְחָה וּבְשַׁלְוָה.

כַּאֲשֶׁר מָלְאוּ יְמֵי שְׁפִיטָתוֹ הַבְּרוּכָה שֶׁל תּוֹלָע בֶּן פּוּאָה, הוּא נִפְטַר וְנִקְבַּר בִּשְׁמִיר, בְּאוֹתוֹ הַמָּקוֹם שֶׁבּוֹ יָשַׁב וְשָׁפַט אֶת כָּל עַם יִשְׂרָאֵל. תּוֹלָע הָיָה שׁוֹפֵט מְיֻחָד שֶׁהֵבִיא עִמּוֹ יְשׁוּעָה, שָׁלוֹם וְשֶׁקֶט רַב לְאַחַר הַסְּעָרָה וְהָרָעָה הַגְּדוֹלָה שֶׁהָיוּ בִּימֵי אֲבִימֶלֶךְ. זְכוּתוֹ הַגְּדוֹלָה וּמַעֲשָׂיו הַטּוֹבִים עָזְרוּ לְעַם יִשְׂרָאֵל לְהִשְׁתַּקֵם, לְהִתְאַחֵד וְלַחֲזֹר לְחַיִּים שְׁלֵוִים, טוֹבִים וּמְבֹרָכִים בְּאֶרֶץ יִשְׂרָאֵל.
