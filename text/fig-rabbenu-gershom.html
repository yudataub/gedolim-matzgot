הַגְּזָרָה שֶׁל רַבֵּנוּ גֵּרְשֹׁם

לִפְנֵי כְּאֶלֶף שָׁנִים חַי הַחָכָם הַגָּדוֹל רַבֵּנוּ גֵּרְשֹׁם בָּעִיר מַגֶּנְצָא שֶׁבְּאַשְׁכְּנַז. רַבֵּנוּ גֵּרְשֹׁם עָמַד בְּרֹאשׁ הַיְּשִׁיבָה, וְכָל חַכְמֵי אַשְׁכְּנַז הָיוּ תַּלְמִידָיו, עַד שֶׁרַשִׁ"י כָּתַב עָלָיו שֶׁמִּפִּיו אָנוּ חַיִּים כֻּלָּנוּ. רַבֵּנוּ גֵּרְשֹׁם הִקְפִּיד מְאֹד עַל כְּבוֹדָם שֶׁל יְהוּדִים, וְלָכֵן תִּקֵּן תַּקָּנוֹת חֲשׁוּבוֹת וְגָזַר גְּזָרָה קָשָׁה וְנִדּוּי עַל כָּל מִי שֶׁיַּזְכִּיר לִיהוּדִי שֶׁחָזַר בִּתְשׁוּבָה אֶת מַעֲשָׂיו הָרָעִים מֵהָעָבָר, מֵהַיָּמִים שֶׁבָּהֶם הִכְרִיחוּ אוֹתוֹ הַגּוֹיִים בִּימֵי הַשְּׁמָד לַעֲזֹב אֶת דַּת יִשְׂרָאֵל.

כַּמָּה שָׁנִים אַחֲרֵי כֵן, אֵרַע מַעֲשֶׂה בְּעִיר אַחַת. שְׁתֵּי מִשְׁפָּחוֹת הֵחֵלּוּ לָרִיב זוֹ עִם זוֹ וּלְהַעֲלִיב אַחַת אֶת הַשְּׁנִיָּה בְּמִלִּים קָשׁוֹת וּבִגְנַאי. אַנְשֵׁי הַקְּהִילָה שָׁמְעוּ אֶת הַמְּרִיבָה הַנּוֹרָאָה, וְהֶחְלִיטוּ לִגְזֹר עַל שְׁתֵּי הַמִּשְׁפָּחוֹת לְהַפְסִיק מִיָּד אֶת הַחֵרוּפִים וְאֶת הַהַעֲלָבוֹת. אַךְ אִשָּׁה אַחַת מֵהַמִּשְׁפָּחוֹת קָפְצָה וְנִשְׁבְּעָה עוֹד לִפְנֵי גְּזֵרַת הַקָּהָל שֶׁהֵם לֹא יְקַבְּלוּ עֲלֵיהֶם אֶת הַגְּזֵרָה, וְהֵם הִמְשִׁיכוּ לְהַעֲלִיב וּלְחָרֵף.

בְּתוֹךְ הַמְּרִיבָה, קָפַץ אָדָם אֶחָד וְהִזְכִּיר לְאֶחָד מִבְּנֵי הַמִּשְׁפָּחָה הַשְּׁנִיָּה דָּבָר מֵבִישׁ: הוּא הִזְכִּיר לוֹ שֶׁבִּימֵי הַשְּׁמָד הוּא נִטְמַע בֵּין הַגּוֹיִים וְעָזַב אֶת הַדַּת. מִיָּד עָמַד אָדָם אַחֵר וְאָמַר לוֹ: הַס! אָסוּר לְךָ לְהַזְכִּיר זֹאת, שֶׁהֲרֵי נִגְזְרָה גְּזָרָה עַל כָּךְ! אַךְ הוּא לֹא אָמַר לוֹ מִי הָרַב שֶׁגָּזַר אֶת הַגְּזָרָה. רַק מְעַט מְאֹד אַחַר כָּךְ נוֹדַע לְכֻלָּם שֶׁרַבֵּנוּ גֵּרְשֹׁם עַצְמוֹ גָּזַר שֶׁכָּל מִי שֶׁיַּזְכִּיר גְּנַאי כָּזֶה יִהְיֶה בְּנִדּוּי.

תַּלְמִיד אֶחָד מֵאֹיְבֵיהֶם שֶׁל אוֹתָם אֲנָשִׁים הוֹרָה וְאָמַר שֶׁהָאָדָם שֶׁהִזְכִּיר אֶת הָעָבָר נִמְצָא כָּעֵת בְּנִדּוּי חָמוּר שֶׁל רַבֵּנוּ גֵּרְשֹׁם, וְאֵין שׁוּם חָכָם בַּדּוֹר שֶׁיָּכוֹל לְהַתִּיר לוֹ אֶת הַנִּדּוּי. הוּא נִשְׁאָר בְּנִדּוּי לְעוֹלָם. אֲבָל אֲנָשִׁים אֲחֵרִים טָעֲנוּ שֶׁאוֹתוֹ אָדָם כְּלָל לֹא יָדַע מִי הָרַב שֶׁגָּזַר וְלֹא שָׁמַע עַל הַגְּזָרָה מֵעוֹלָם, וְגַם שֶׁהוּא מֵעִיר אַחֶרֶת. כְּמוֹ כֵן אָמְרוּ שֶׁאִם הוּא מְקַבֵּל עָלָיו אֶת הַדִּין, מֻתָּר לְחַכְמֵי הַדּוֹר לְהַתִּיר לוֹ אֶת הַנִּדּוּי.

חַכְמֵי הַדּוֹר פָּנוּ אֶל הַגָּאוֹן הַגָּדוֹל רַשִׁ"י וְשָׁאַלּוּ אוֹתוֹ מַהִי הַהֲלָכָה. רַשִׁ"י עָנָה לָהֶם בִּתְשׁוּבָה מְפֹרָשֶׁת. רַשִׁ"י הִסְבִּיר שֶׁאִם יִתְבָּרֵר בְּעֵדִים נֶאֱמָנִים שֶׁרַבֵּנוּ גֵּרְשֹׁם גָּזַר אֶת הַגְּזָרָה בְּלָשׁוֹן קָשָׁה וְחֲמוּרָה, וְקָבַע שֶׁאֲסוּר לְהַזְכִּיר אֶת הַגְּנַאי הַזֶּה לֹא לְבַעֲלֵי הַתְּשׁוּבָה עַצְמָם וְלֹא לְדוֹרֵי דּוֹרוֹתֵיהֶם, אָז אָסוּר לָנוּ לִנְהֹג קַלּוּת רֹאשׁ בַּנִּדּוּי שֶׁלּוֹ. אֵין בְּדוֹרֵנוּ חָכָם גָּדוֹל כְּרַבֵּנוּ גֵּרְשֹׁם שֶׁיּוּכַל לְהַתִּיר אֶת הַגְּזָרָה שֶׁגָּזַר.

רַשִׁ"י הוֹסִיף וּבֵאֵר שֶׁהַגְּזָרָה שֶׁל רַבֵּנוּ גֵּרְשֹׁם נַעֲשְׂתָה לִכְבוֹד שָׁמַיִם, שֶׁהֲרֵי עוֹד מֵהַר סִינַי הִזְהִירָה אוֹתָנוּ הַתּוֹרָה עַל אוֹנָאַת דְּבָרִים וְאָמְרָה "לֹא תוֹנוּ". הַתּוֹרָה צִוְּתָה שֶׁאִם אָדָם הָיָה בַּעַל תְּשׁוּבָה, אָסוּר לִפְגֹעַ בּוֹ וְלוֹמַר לוֹ "זְכֹר מַעֲשֶׂיךָ הָרִאשׁוֹנִים". רַשִׁ"י הִדְגִּישׁ שֶׁהַתְּשׁוּבָה מַגַּעַת עַד כִּסֵּא הַכָּבוֹד, וַאֲפִלּוּ צַדִּיקִים גְּמוּרִים אֵינָם עוֹמְדִים בִּמְקוֹם שֶׁבַּעֲלֵי תְּשׁוּבָה עוֹמְדִים. לָכֵן אָסוּר בְּאִסּוּר חָמוּר לְהַעֲלִיב אָדָם שֶׁחָזַר בִּתְשׁוּבָה אוֹ לְהַזְכִּיר לוֹ אֶת חֲטָאָיו מֵהָעָבָר.

💡 אָסוּר לָנוּ לְהַזְכִּיר לְשׁוּם אָדָם אֶת מַעֲשָׂיו הָרָעִים מֵהָעָבָר אִם הוּא חָזַר בִּתְשׁוּבָה, כִּי הַתְּשׁוּבָה יְקָרָה מְאֹד לִפְנֵי הַשֵּׁם וּמַגַּעַת עַד כִּסֵּא הַכָּבוֹד.
