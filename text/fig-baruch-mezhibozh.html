הַנֶּכֶד שֶׁהִתְחַבֵּא וְהַסַּבָּא שֶׁבָּכָה

לִפְנֵי הַרְבֵּה שָׁנִים חַי רַב גָּדוֹל וְצַדִּיק גָּדוֹל מְאֹד שֶׁקָּרְאוּ לוֹ רַבִּי בָּרוּךְ מִמֶּזְ'יבּוּז'. רַבִּי בָּרוּךְ גָּר בָּעִיר מֶזְ'יבּוּז' וְהָיָה אָדָם חָכָם, קָדוֹשׁ וְאוֹהֵב אֲנָשִׁים. יוֹם אֶחָד הִגִּיעַ הַנֶּכֶד שֶׁל רַבִּי בָּרוּךְ אֶל הַבַּיִת שֶׁל סַבָּא שֶׁלּוֹ. הַנֶּכֶד הַקָּטָן נִכְנַס אֶל הַחֶדֶר כְּשֶׁהוּא בּוֹכֶה, בּוֹכֶה, בּוֹכֶה מְאֹד וְדְמָעוֹת רַבּוֹת יוֹרְדוֹת עַל לְחָיָיו. הַנֶּכֶד הָיָה עָצוּב כָּל כָּךְ, וְהוּא קָרָא מִתּוֹךְ הַדְּמָעוֹת אֶל סַבָּא שֶׁלּוֹ: "סַבָּא, סַבָּא, הֵם לֹא חִפְּשׂוּ אוֹתִי!".

רַבִּי בָּרוּךְ, הַסַּבָּא הַטּוֹב וְהָאוֹהֵב, רָאָה אֶת נֶכְדּוֹ הַקָּטָן בּוֹכֶה בְּצַעַר גָּדוֹל. הוּא פָּנָה אֵלָיו בְּרַכּוּת וּבְאַהֲבָה רַבָּה, וְאָמַר לוֹ: "מֶה קָרָה חָמוּד? בּוֹא תֵּשֵׁב, בּוֹא תְּסַפֵּר לִי מֶה קָרָה". הַנֶּכֶד הַחָמוּד יָשַׁב עַל הַכִּסֵּא לְיַד סַבָּא שֶׁלּוֹ, נִגֵּב אֶת הַדְּמָעוֹת וְהִתְחִיל לְסַפֵּר אֶת כָּל מָה שֶׁקָּרָה לוֹ בַּחוּץ מָה שֶׁגָּרַם לוֹ לִבְכּוֹת.

הוּא אָמַר לְסַבָּא: "סַבָּא, שִׂחַקְנוּ מַחְבּוֹאִים וְהַחֲבֵרִים הִפְסִיקוּ לְחַפֵּשׂ אוֹתִי כִּי הִתְחַבֵּאתִי כָּל-כָּךְ, כָּל-כָּךְ טוֹב. וּבְאֵיזֶהשֶׁהוּ שָׁלָב הֵם הִתְיָאֲשׁוּ מִמֶּנִּי וְהִפְסִיקוּ לְחַפֵּשׂ אוֹתִי, וְזֶה עָצוּב לִי!". הַנֶּכֶד הִסְבִּיר לְסַבָּא שֶׁהוּא הִתְחַבֵּא בִּמְקוֹם מַחְבּוֹא נִפְלָא וְחִכָּה בְּשֶׁקֶט שֶׁהַחֲבֵרִים יָבֹאוּ לִמְצֹא אוֹתוֹ. אֲבָל בִּגְלַל שֶׁהוּא הִתְחַבֵּא כָּל כָּךְ טוֹב, הַחֲבֵרִים פְּשׁוּט הִתְיָאֲשׁוּ מִלִּמְצֹא אוֹתוֹ, הִפְסִיקוּ לְחַפֵּשׂ וְהָלְכוּ מִשָּׁם. הַדָּבָר הַזֶּה גָּרַם לַנֶּכֶד לְהַרְגִּישׁ עָצוּב מְאֹד, כִּי הוּא רָצָה שֶׁיַּמְשִׁיכוּ לְחַפֵּשׂ אוֹתוֹ.

כְּשֶׁשָּׁמַע רַבִּי בָּרוּךְ אֶת הַדְּבָרִים שֶׁל נֶכְדּוֹ, פִּתְאֹם גַּם הַסַּבָּא מַתְחִיל לִבְכּוֹת. הַסַּבָּא רַבִּי בָּרוּךְ יָשַׁב מְרֻגָּשׁ, וּדְמָעוֹת חַמּוֹת יָרְדוּ מֵעֵינָיו. הַנֶּכֶד הַקָּטָן הֻפְתַּע מְאֹד לִרְאוֹת אֶת סַבָּא שֶׁלּוֹ הַצַּדִּיק בּוֹכֶה, וְשָׁאַל אוֹתוֹ בִּפְלִיאָה: "סַבָּא, מֶה קָרָה? לָמָּה אַתָּה בּוֹכֶה?".

אָז עָנָה לוֹ הַסַּבָּא, רַבִּי בָּרוּךְ מִמֶּזְ'יבּוּז', וְאָמַר לוֹ בְּקוֹל מִתְרַגֵּשׁ: "אוֹי נֶכֶד יָקָר, אַתָּה יוֹדֵעַ, זֶה בְּדִיּוּק מָה שֶׁקָּרָה לַקָּדוֹשׁ בָּרוּךְ הוּא בָּעוֹלָם. הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת הָעוֹלָם, בָּרָא עוֹלָם כָּל-כָּךְ נִפְלָא, וּבְתוֹךְ הָעוֹלָם הַזֶּה הַקָּדוֹשׁ בָּרוּךְ הוּא מִתְחַבֵּא. וְהוּא מִתְחַבֵּא כָּל-כָּךְ טוֹב, כִּי הַקָּדוֹשׁ בָּרוּךְ הוּא יוֹדֵעַ לְהִתְחַבֵּא טוֹב, שֶׁאֲנַחְנוּ לֹא מוֹצְאִים אוֹתוֹ. וִיכוֹלִים הַרְבֵּה אֲנָשִׁים לָלֶכֶת, לְהִסְתּוֹבֵב בָּעוֹלָם, וְלֹא לִרְאוֹת אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא וְלֹא לְהַרְגִּישׁ אוֹתוֹ".

רַבִּי בָּרוּךְ בָּכָה כִּי הוּא הֵבִין שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת כָּל הָעוֹלָם הַיָּפֶה וְהוּא מִתְחַבֵּא בּוֹ, אֲבָל בְּנֵי הָאָדָם לִפְעָמִים מִתְיָאֲשִׁים וּמַפְסִיקִים לְחַפֵּשׂ אוֹתוֹ, וְזֶה דָּבָר עָצוּב מְאֹד.

💡 אֲנַחְנוּ צְרִיכִים לְחַפֵּשׂ אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא תָּמִיד בְּכָל מָקוֹם בָּעוֹלָם וְלֹא לְהִתְיָאֵשׁ לְעוֹלָם.
