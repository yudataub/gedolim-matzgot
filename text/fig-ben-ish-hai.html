הַיֶּלֶד יוֹסֵף חַיִּים וּבוֹר הַמַּיִם

לִפְנֵי יוֹתֵר מִמֵּאָה וַחֲמִשִּׁים שָׁנִים, בָּעִיר הַגְּדוֹלָה בַּגְדָּד שֶׁבְּאֶרֶץ עִירָאק, חַי יֶלֶד קָטָן, מָתוֹק וּמְיוּחָד מְאֹד שֶׁקָּרְאוּ לוֹ יוֹסֵף. יוֹסֵף הַקָּטָן הָיָה יֶלֶד חָכָם וְטוֹב לֵב, שֶׁאָהַב מְאֹד אֶת הַתּוֹרָה הַקְּדוֹשָׁה וְאֶת הַקָּדוֹשׁ בָּרוּךְ הוּא. אָבִיו שֶׁל יוֹסֵף הָיָה רַב גָּדוֹל וּמַנְהִיג רוּחָנִי חָשׁוּב מְאֹד עַבּוּר כָּל יְהוּדֵי בָּבֶל.

יוֹם אֶחָד, כַּאֲשֶׁר הָיָה יוֹסֵף הַקָּטָן בַּדֶּרֶךְ, הוּא עָבַר לְיַד בּוֹר מַיִם עָמֹק וְגָדוֹל. פִּתְאֹם, בְּלִי שֶׁשָּׂם לֵב, נָפַל יוֹסֵף הַקָּטָן לְתוֹךְ בּוֹר הַמַּיִם, וְהוּא מָצָא אֶת עַצְמוֹ מַמָּשׁ בְּסַכָּנַת חַיִּים גְּדוֹלָה וּמַפְחִידָה מְאֹד. אֲנָשִׁים רַבִּים שֶׁהָיוּ בַּמָּקוֹם רָאוּ אֶת הָאָסוֹן הַנּוֹרָא, וּמִהֲרוּ בִּמְהִירוּת רַבָּה לְהַצִּיל אוֹתוֹ מִתּוֹךְ הַבּוֹר. לְאַחַר שֶׁהֶעֱלוּ אוֹתוֹ מִן הַמַּיִם, הָיָה הַמַּצָּב שֶׁל יוֹסֵף הַקָּטָן מַמָּשׁ מַמָּשׁ לֹא פָּשׁוּט, וְהוּא הָיָה חוֹלֶה מְאֹד וְשָׁכַב לְלֹא תְּנוּעָה.

כַּאֲשֶׁר שָׁמְעוּ עַל כָּךְ אֲנָשֵׁי הַקְּהִלָּה הַיְּהוּדִית בְּבַגְדָּד, הֵם דָּאֲגוּ לוֹ מְאֹד. מִיָּד הִתְכַּנְסוּ אֲנָשִׁים רַבִּים בְּבָתֵּי הַכְּנֶסֶת בְּכָל הָעִיר, פָּתְחוּ אֶת הַסְּפָרִים, הִתְפַּלְלוּ בְּכַוָּנָה גְּדוֹלָה מִכָּל הַלֵּב וּבִקְשׁוּ רַחֲמִים מֵהַשֵּׁם יִתְבָּרַךְ שֶׁיִּשְׁלַח רְפוּאָה שְׁלֵמָה לַיֶּלֶד יוֹסֵף הַקָּטָן. כָּל אֶחָד וְאֶחָד בִּקֵּשׁ שֶׁהַשֵּׁם יַעֲשֶׂה נֵס וִישַׁחְרֵר אֶת הַיֶּלֶד מֵהַסַּכָּנָה הַגְּדוֹלָה.

בִּזְכוּת הַתְּפִלּוֹת הָרַבּוֹת וְהַדְּמָעוֹת שֶׁל אֲנָשֵׁי הַקְּהִלָּה, וּבִזְכוּת הָרַחֲמִים הַגְּדוֹלִים שֶׁל הַשֵּׁם יִתְבָּרַךְ, אָכֵן קָרָה נֵס נִפְלָא. לְאַט לְאַט פָּקַח יוֹסֵף הַקָּטָן אֶת עֵינָיו. וּבָרֶגַע הָרִאשׁוֹן מַמָּשׁ שֶׁבּוֹ פָּקַח אֶת עֵינָיו וְחָזַר לְהַכָּרָתוֹ, הוּא נָשָׂא אֶת עֵינָיו לַשָּׁמַיִם בֶּאֱמוּנָה גְּדוֹלָה, וְאָמַר מִכָּל לִבּוֹ: "רִבּוֹנוֹ שֶׁל עוֹלָם! בְּבַקָּשָׁה מִמְּךָ, אֲנִי רוֹצֶה לִהְיוֹת בָּרִיא! וְאִם אֲנִי אֶבְרָא, אֲנִי אֲקַדֵּשׁ אֶת כָּל חַיַּי לְלִמּוּד הַתּוֹרָה הַקְּדוֹשָׁה!".

הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַע אֶת תְּפִלָּתוֹ הַזַּכָּה וְהַיְּפָה שֶׁל הַיֶּלֶד, וּבֶאֱמֶת לְאַט לְאַט הִבְרִיא יוֹסֵף הַקָּטָן וְחָזַר לְכֹחוֹ. בַּזְּמַן שֶׁהָיָה חוֹלֶה, הוֹסִיפוּ לוֹ אֲנָשֵׁי הַקְּהִלָּה אֶת הַשֵּׁם 'חַיִּים' כִּסְגֻלָּה נִפְלָאָה לִרְפוּאָה וּלְחַיִּים, וּמֵאָז נִקְרָא שְׁמוֹ יוֹסֵף חַיִּים הַקָּטָן. יוֹסֵף חַיִּים הֶחֱלִים לְגַמְרֵי, הִתְחִיל לְהִתְחַזֵּק, וּכְפִי שֶׁהִבְטִיחַ לַהַשֵּׁם יִתְבָּרַךְ, הוּא הִתְחִיל לִלְמֹד בִּשְׁקִיקָה גְּדוֹלָה, בְּשִׂמְחָה וּבְאַהֲבָה רַבָּה אֶת הַתּוֹרָה הַקְּדוֹשָׁה.

יוֹסֵף חַיִּים הַקָּטָן לָמַד וְלָמַד בִּשְׁקִידָה נִפְלָאָה יָמִים וְלֵילוֹת, וְהִתְעַלָּה מְאֹד בְּחָכְמַת הַתּוֹרָה. כַּאֲשֶׁר הָיָה בֶּן עֶשְׂרִים וָשֵׁשׁ שָׁנִים, נִפְטַר אָבִיו הָרַב, וְיוֹסֵף חַיִּים יָרַשׁ אֶת מְקוֹמוֹ שֶׁל אָבִיו. הוּא הָפַךְ לִגְדוֹל יִשְׂרָאֵל וּלְרַב גָּדוֹל וּמְפֻרְסָם מְאֹד בְּכָל בָּבֶל. אֲנָשִׁים רַבִּים קָרְאוּ לוֹ 'רַבִּי יוֹסֵף חַיִּים מִבַּגְדָּד', וְגַם קָרְאוּ לוֹ 'הַבֶּן אִישׁ חַי' עַל שֵׁם הַסֵּפֶר הַחָשׁוּב וְהַמְּפֻרְסָם שֶׁכָּתַב.

💡 כַּאֲשֶׁר אָנוּ מִתְפַּלְלִים לַשֵּׁם מִכָּל הַלֵּב בְּעֵת צָרָה וּמַבְטִיחִים לְהַקְדִּישׁ אֶת חַיֵּינוּ לַתּוֹרָה, הַשֵּׁם שׁוֹמֵעַ אֶת תְּפִלָּתֵנוּ, עוֹשֶׂה עִמָּנוּ נִסִּים וּמְבָרֵךְ אוֹתָנוּ.
