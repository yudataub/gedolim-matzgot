אֵילּוֹן הַזְּבוּלֹנִי – הַשּׁוֹפֵט מֵחוֹף הַיָּם

בְּסֵפֶר שׁוֹפְטִים מְסֻפָּר עַל תְּקוּפָה חֲשׁוּבָה בְּתוֹלְדוֹת עַם יִשְׂרָאֵל, שֶׁבָּהּ הִנְהִיגוּ אֶת הָעָם שׁוֹפְטִים שׁוֹנִים מִשְּׁבָטִים שׁוֹנִים. לִפְנֵי שֶׁקָּם הַשּׁוֹפֵט אֵילוֹן הַזְּבוּלֹנִי, שָׁפַט אֶת יִשְׂרָאֵל שׁוֹפֵט שֶׁקָּרְאוּ לוֹ אִבְצָן מִבֵּית לֶחֶם. אִבְצָן הִנְהִיג אֶת עַם יִשְׂרָאֵל בֶּאֱמֶת וּבְמִשְׁפָּט בַּמֶּשֶׁךְ שֶׁבַע שָׁנִים תְּמִימוֹת. בַּמָּקוֹר מְסֻפָּר כִּי לְאִבְצָן הָיוּ שְׁלֹשִׁים בָּנִים וְגַם שְׁלֹשִׁים בָּנוֹת. אֶת שְׁלֹשִׁים בְּנוֹתָיו שִׁלַּח אִבְצָן לַחוּץ, כְּלוֹמַר הֵן הִתְחַתְּנוּ מִחוּץ לְמִשְׁפַּחְתּוֹ, וּשְׁלֹשִׁים בָּנוֹת הֵבִיא לִבְנָיו מִן הַחוּץ. כַּאֲשֶׁר עָבְרוּ שֶׁבַע הַשָּׁנִים שֶׁל מַנְהִיגֻתּוֹ, נִפְטַר אִבְצָן וְנִקְבַּר בָּעִיר בֵּית לֶחֶם.

אַחֲרֵי מוֹתוֹ שֶׁל אִבְצָן מִבֵּית לֶחֶם, קָם מַנְהִיג חָדָשׁ וְשׁוֹפֵט חָדָשׁ לְעַם יִשְׂרָאֵל. לַשּׁוֹפֵט הַזֶּה קָרְאוּ אֵילוֹן הַזְּבוּלֹנִי. אֵילוֹן הָיָה דְּמוּת מְיֻחֶדֶת מְאֹד בְּתוֹךְ סֵפֶר שׁוֹפְטִים, מִפְּנֵי שֶׁהוּא הָיָה הַשּׁוֹפֵט הַיָּחִיד בְּכָל הַהִיסְטוֹרְיָה שֶׁבָּא מִשֵּׁבֶט זְבוּלֻן.

כְּדֵי לְהָבִין אֶת יִחוּדוֹ שֶׁל אֵילוֹן הַזְּבוּלֹנִי, כְּדַאי לְהַכִּיר אֶת שֵׁבֶט זְבוּלֻן. שֵׁבֶט זְבוּלֻן יָשַׁב בְּאֶרֶץ יִשְׂרָאֵל לְחוֹף הַיָּם וְעָסַק בַּמִּסְחָר. שֵׁבֶט זֶה זָכָה לִבְרָכוֹת מְיֻחָדוֹת עוֹד מִימֵי אֲבוֹתֵינוּ. יַעֲקֹב אָבִינוּ בֵּרֵךְ אֶת זְבוּלֻן לִפְנֵי מוֹתוֹ בְּסֵפֶר בְּרֵאשִׁית, כְּפִי שֶׁנֶּאֱמַר: "זְבוּלֻן לְחוֹף יַמִּים יִשְׁכֹּן, וְהוּא לְחוֹף אֳנִיּוֹת, וְיַרְכָתוֹ עַל צִידֹן". הַבְּרָכָה הַזֹּאת תֵּאֲרָה כִּיצַד שֵׁבֶט זְבוּלֻן יִהְיֶה שֵׁבֶט שֶׁשּׁוֹכֵן לְחוֹף הַיָּם וּמִתְפַּרְנֵס מִמִּסְחָר עִם אֳנִיּוֹת בַּיָּם.

גַּם מֹשֶׁה רַבֵּנוּ בֵּרֵךְ אֶת שֵׁבֶט זְבוּלֻן בְּסֵפֶר דְּבָרִים לִפְנֵי פְטִירָתוֹ. מֹשֶׁה אָמַר: "שְׂמַח זְבוּלֻן בְּצֵאתֶךָ, וְיִשָּׂשכָר בְּאָהֳלֶיךָ. עַמִּים הַר יִקְרָאוּ, שָׁם יִזְבְּחוּ זִבְחֵי צֶדֶק, כִּי שֶׁפַע יַמִּים יִינָקוּ וּשְׂפֻנֵי טְמוּנֵי חוֹל". מֹשֶׁה רַבֵּנוּ בֵּרֵךְ אֶת זְבוּלֻן שֶׁיִּשְׂמַח בְּצֵאתוֹ לַמִּסְחָר, וְשֶׁיֵּהָנֶה מֵהַשֶּׁפַע שֶׁהַיָּם וְהַחוֹל מְבִיאִים עִמָּהֶם.

מִתּוֹךְ הַשֵּׁבֶט הַזֶּה, שֶׁזָּכָה לִבְרָכוֹת מִיַעֲקֹב וּמִמֹּשֶׁה, יָצָא הַשּׁוֹפֵט אֵילוֹן הַזְּבוּלֹנִי. אֵילוֹן שָׁפַט אֶת עַם יִשְׂרָאֵל בֶּאֱמָנָה וְהוֹבִיל אֶת הָעָם בַּמֶּשֶׁךְ עֶשֶׂר שָׁנִים. כָּל עַם יִשְׂרָאֵל כִּבֵּד אֶת אֵילוֹן וְהִקְשִׁיב לִדְבָרָיו הַחֲכָמִים. הוּא הָיָה הַשּׁוֹפֵט הַיָּחִיד שֶׁיָּצָא מִשֵּׁבֶט זְבוּלֻן כְּדֵי לְהַנְהִיג אֶת כָּל הָעָם.

כַּאֲשֶׁר תַּמּוּ עֶשֶׂר הַשָּׁנִים שֶׁבָּהֶן שָׁפַט אֶת יִשְׂרָאֵל, נִפְטַר אֵילוֹן הַזְּבוּלֹנִי. הוּא הוּבָא לִקְבוּרָה בִּמְקוֹם מוֹשָׁבוֹ בְּנַחֲלַת שִׁבְטוֹ, בָּעִיר אַיָּלוֹן אֲשֶׁר בְּאֶרֶץ זְבוּלֻן. כָּךְ הוֹתִיר אֵילוֹן הַזְּבוּלֹנִי זִכָּרוֹן שֶׁל מַנְהִיגֻת מְבוֹרֶכֶת שֶׁל שׁוֹפֵט יָחִיד מִשֵּׁבֶט זְבוּלֻן.

💡 כָּל אָדָם, מִכָּל שֵׁבֶט וּמִכָּל מָקוֹם, יָכוֹל לִהְיוֹת מַנְהִיג חָשׁוּב וּלְסַיֵּעַ לְכָל עַם יִשְׂרָאֵל.
