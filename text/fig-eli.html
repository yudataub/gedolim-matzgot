הַקּוֹל שֶׁקָּרָא לִשְׁמוּאֵל בַּלַּיְלָה

עֵלִי הַכֹּהֵן שָׁפַט אֶת עַם יִשְׂרָאֵל בְּשִׁילֹה אַרְבָּעִים שָׁנָה. בְּאוֹתָם יָמִים הַנַּעַר שְׁמוּאֵל גָּדַל אֶצְלוֹ וְשֵׁרֵת אֶת הַשֵּׁם לִפְנֵי עֵלִי. עֵלִי הָיָה אִישׁ זָקֵן מְאֹד, וְעֵינָיו הֵחֵלּוּ לִכְהוֹת כָּךְ שֶׁלֹּא יָכֹל לִרְאוֹת הֵיטֵב. בְּאַחַד הַלֵּילוֹת, כַּאֲשֶׁר נֵר אֱלֹהִים טֶרֶם כָּבָה וּשְׁמוּאֵל שָׁכַב בְּהֵיכַל הַשֵּׁם אֲשֶׁר שָׁם אֲרוֹן הָאֱלֹהִים, קָרָא הַשֵּׁם אֶל שְׁמוּאֵל. שְׁמוּאֵל עָנָה מִיָּד "הִנֵּנִי", וְרָץ אֶל עֵלִי הַכֹּהֵן וְאָמַר לוֹ: "הִנְנִי כִּי קָרָאתָ לִי". אַךְ עֵלִי עָנָה לוֹ: "לֹא קָרָאתִי, שׁוּב שְׁכַב". שְׁמוּאֵל חָזַר לִשְׁכַּב, וְהַשֵּׁם קָרָא לוֹ שֵׁנִית: "שְׁמוּאֵל!". שׁוּב קָם שְׁמוּאֵל וְרָץ אֶל עֵלִי, וְשׁוּב עָנָה עֵלִי שֶׁלֹּא קָרָא לוֹ וּבִקֵּשׁ מִמֶּנּוּ לַחְזֹר לִישֹׁן, כִּי שְׁמוּאֵל עֲדַיִן לֹא יָדַע כִּי דְּבַר הַשֵּׁם נִגְלָה אֵלָיו.

כַּאֲשֶׁר קָרָא הַשֵּׁם לִשְׁמוּאֵל בַּפַּעַם הַשְּׁלִישִׁית, הֵבִין עֵלִי הַכֹּהֵן כִּי הַשֵּׁם הוּא זֶה שֶׁקּוֹרֵא לַנַּעַר. עֵלִי אָמַר לִשְׁמוּאֵל לַחְזֹר לִשְׁכַּב, וְאִם יִשְׁמַע שׁוּב אֶת הַקּוֹל, יַעֲנֶה: "דַּבֵּר הַשֵּׁם כִּי שֹׁמֵעַ עַבְדֶּךָ". שְׁמוּאֵל חָזַר לִמְקוֹמוֹ, וְהַשֵּׁם בָּא וְנִתְיַצֵּב וְקָרָא: "שְׁמוּאֵל שְׁמוּאֵל!". שְׁמוּאֵל עָנָה כְּפִי שֶׁלִּמְּדוֹ עֵלִי: "דַּבֵּר כִּי שֹׁמֵעַ עַבְדֶּךָ". אָז גִּלָּה הַשֵּׁם לִשְׁמוּאֵל כִּי הוּא עוֹמֵד לַעֲשׂוֹת דָּבָר גָּדוֹל בְּיִשְׂרָאֵל, וְכִי הוּא יַעֲנִישׁ אֶת בֵּית עֵלִי עַל עֲוֹן בָּנָיו, חָפְנִי וּפִינְחָס, אֲשֶׁר חָטְאוּ וְעֵלִי לֹא מָנַע אוֹתָם מִכָּךְ.

שְׁמוּאֵל שָׁכַב עַד הַבֹּקֶר, וְאָז פָּתַח אֶת דַּלְתוֹת בֵּית הַשֵּׁם. הוּא יָרֵא מְאֹד לְסַפֵּר לְעֵלִי אֶת הַנְּבוּאָה הַקָּשָׁה. אַךְ עֵלִי קָרָא לוֹ וּבִקֵּשׁ מִמֶּנּוּ שֶׁלֹּא יַסְתִּיר מִמֶּנּוּ דָּבָר. שְׁמוּאֵל סִפֵּר לְעֵלִי אֶת כָּל הַדְּבָרִים, וְעֵלִי קִבֵּל אֶת הַדְּבָרִים בִּכְנִיעָה וְאָמַר: "הַשֵּׁם הוּא, הַטּוֹב בְּעֵינָיו יַעֲשֶׂה". שְׁמוּאֵל הָלַךְ וְגָדַל, וְהַשֵּׁם הָיָה עִמּוֹ וְלֹא הִפִּיל מִכָּל דְּבָרָיו אָרְצָה. כָּל עַם יִשְׂרָאֵל, מִדָּן וְעַד בְּאֵר שֶׁבַע, יָדַע כִּי שְׁמוּאֵל הוּא נָבִיא נֶאֱמָן לַה'.

כַּעֲבֹר זְמַן, כַּאֲשֶׁר עֵלִי הָיָה בֶּן תִּשְׁעִים וּשְׁמֹנֶה שָׁנָה, יָצְאוּ יִשְׂרָאֵל לַמִּלְחָמָה נֶגֶד הַפְּלִשְׁתִּים. עֵלִי יָשַׁב עַל הַכִּסֵּא לְיַד הַדֶּרֶךְ וְצִפָּה בַּחֲרָדָה, כִּי לִבּוֹ חָרַד עַל אֲרוֹן הָאֱלֹהִים. וְהִנֵּה בָּא אִישׁ מִשֵּׁבֶט בִּנְיָמִין מֵהַמַּעֲרָכָה, כְּשֶׁבְּגָדָיו קְרֻעִים וַאֲדָמָה עַל רֹאשׁוֹ, וְסִפֵּר אֶת הַבְּשׂוֹרָה הַקָּשָׁה: עַם יִשְׂרָאֵל נָס לִפְנֵי הַפְּלִשְׁתִּים, הָיְתָה מַגֵּפָה גְּדוֹלָה, שְׁנֵי בָּנָיו שֶׁל עֵלִי מֵתוּ, וַאֲרוֹן הָאֱלֹהִים נִלְקַח. כְּשֶׁשָּׁמַע עֵלִי אֶת הַזְכָּרַת אֲרוֹן הָאֱלֹהִים, הוּא נָפַל מֵעַל הַכִּסֵּא אֲחוֹרַנִּית, מִפְרַקְתּוֹ נִשְׁבְּרָה וָמֵת.

💡 כַּאֲשֶׁר הַשֵּׁם קוֹרֵא לָנוּ אוֹ מְבַקֵּשׁ מֵאִתָּנוּ דָּבָר, עָלֵינוּ לְהַקְשִׁיב בִּתְשׂוּמַת לֵב, בְּעֲנָוָה וּבִנְכוֹנוּת לְמַלֵּא אֶת דְּבָרוֹ.
