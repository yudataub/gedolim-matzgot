הָאֱמוּנָה בְּלֵב כָּל יְהוּדִי

הָרַב יוֹסֵף מַשָׁאשׁ נָלַד בִּמְדִינַת מָרוֹקוֹ. לְאַחַר מִכֵּן, עָבַר הָרַב מַשָׁאשׁ לָגוּר בָּעִיר תִּלְמְסָאן, וְשָׁם הוּא כִּהֵן כְּרַב הָעִיר בַּמֶּשֶׁךְ שָׁנִים רַבּוֹת. כַּאֲשֶׁר הִגִּיעַ הָרַב מַשָׁאשׁ לָעִיר תִּלְמְסָאן, הָיְתָה שָׁם קְהִילָה יְהוּדִית שֶׁאֲנָשֶׁיהָ הָיוּ פָּחוֹת שׁוֹמְרֵי תּוֹרָה וּמִצְווֹת. בַּקְּהִילָה הָיוּ אֲנָשִׁים רַבִּים שֶׁשָּׁמְרוּ עַל הַמָּסֹרֶת הַיְהוּדִית, אַךְ הֵם לֹא מַמָּשׁ הִקְפִּידוּ עַל כָּל הַמִּצְווֹת. נְשִׂיא הַקְּהִילָה עַצְמוֹ הָיָה אָדָם שֶׁלֹּא שָׁמַר כַּשְׁרוּת בְּבֵיתוֹ, וְהוּא גַּם לֹא הָיָה בָּא לְהִתְפַּלֵּל בְּבֵית הַכְּנֶסֶת.

בַּשַּׁבָּת הָרִאשׁוֹנָה שֶׁבָּהּ הִתְאָרֵחַ הָרַב מַשָׁאשׁ בָּעִיר תִּלְמְסָאן, נְשִׂיא הַקְּהִילָה יָדַע שֶׁאֵין לוֹ שׁוּם סִכּוּי לְאָרֵחַ אֶת הָרַב הַחָדָשׁ בְּבֵיתוֹ הַפְּרָטִי, מִפְּנֵי שֶׁאֶצְלוֹ בַּבַּיִת לֹא שָׁמְרוּ כַּשְׁרוּת. לָכֵן, הִתְאָרֵחַ הָרַב מַשָׁאשׁ בִּסְעוּדַת הַשַּׁבָּת אֵצֶל אִישׁ אַחֵר שֶׁהָיָה אָדָם חָשׁוּב בַּקְּהִילָה. נְשִׂיא הַקְּהִילָה בָּא גַּם הוּא לְהִתְאָרֵחַ יַחַד אִתָּם בְּבֵיתוֹ שֶׁל אוֹתוֹ אִישׁ חָשׁוּב.

בְּמַהֲלַךְ הַמִּפְגָּשׁ וְהַשְּׁהוּת הַמְּשׁוּתֶפֶת עִם הָרַב, הִתְקָרְבוּ אַנְשֵׁי הַקְּהִילָה אֶל הָרַב מַשָׁאשׁ וְאָמְרוּ לוֹ בַּלָּחַשׁ: "הָרַב מַשָׁאשׁ, תֵּדַע לְךָ שֶׁאִם נְשִׂיא הַקְּהִילָה יַזְמִין אוֹתְךָ אֵלָיו לַבַּיִת לֶאֱכֹל, תֵּדַע שֶׁאֶצְלוֹ בַּבַּיִת לֹא כָּשֵׁר, אָז מַמָּשׁ מַמָּשׁ תִּתְחַמֵּק מִזֶּה וְאַל תַּסְכִּים לֶאֱכֹל אֶצְלוֹ".

וְהִנֵּה, הָרַב מַשָׁאשׁ דִּבֵּר עִם נְשִׂיא הַקְּהִילָה, וְתוֹךְ כְּדֵי שֶׁהֵם מְדַבְּרִים בִּידִידוּת, אָמַר לוֹ הָרַב מַשָׁאשׁ מִיזְמָתוֹ: "אֲבָל אֲנִי, בַּשַּׁבָּת הַבָּאָה, אֲנִי רוֹצֶה לֶאֱכֹל אֶצְלְךָ בַּבַּיִת". כְּשֶׁשָּׁמַע זֹאת נְשִׂיא הַקְּהִילָה הוּא מְאֹד הִתְרַגֵּשׁ לִשְׁמֹעַ שֶׁהָרַב הַחָשׁוּב מוֹכָן לֶאֱכֹל אֶצְלוֹ בַּבַּיִת, וּלְאַחַר מִכֵּן נִפְרְדוּ כֻּלָּם בְּשִׂמְחָה וְהָלְכוּ לְבָתֵּיהֶם.

כַּאֲשֶׁר שָׁמְעוּ זֹאת הָאֲנָשִׁים בַּקְּהִילָה, הֵם בָּאוּ אֶל הָרַב מַשָׁאשׁ וְשָׁאַלוּ אוֹתוֹ בִּפְלִיאָה רַבָּה: "אֵיךְ אַתָּה אוֹמֵר לוֹ שֶׁתֹּאכַל אֶצְלוֹ? הֲרֵי אָמַרְנוּ לְךָ שֶׁאֶצְלוֹ בַּבַּיִת לֹא כָּשֵׁר?!". אָמַר לָהֶם הָרַב מַשָׁאשׁ בְּחִיּוּךְ וּבְשַׁלְוָה: "חַכּוּ וְתִרְאוּ".

וּבֶאֱמָת, בְּאוֹתוֹ שָׁבוּעַ שֶׁלְּאַחַר מִכֵּן, נְשִׂיא הַקְּהִילָה הָלַךְ וְהִכְשִׁיר אֶת כָּל הַמִּטְבָּח שֶׁלּוֹ, הָלַךְ לִקְנוֹת אוֹכֵל כָּשֵׁר, וְדָאַג שֶׁיִּהְיֶה הֲכִי כָּשֵׁר שֶׁבָּעוֹלָם בַּבַּיִת שֶׁלּוֹ. וְלֹא רַק זֶה קָרָה, אֶלָּא שֶׁהוּא גַּם הִתְקָרֵב מְאֹד אֶל הָרַב, וְהִתְחִיל בְּכָל שַׁבָּת לָבוֹא לְבֵית הַכְּנֶסֶת.

אָמַר הָרַב מַשָׁאשׁ לְאַנְשֵׁי הַקְּהִילָה שֶׁתָּמְהוּ עַל הַשִּׁנּוּי הַנִּפְלָא: "אַתֶּם רוֹאִים, צְרִיכִים לְהַאֲמִין בַּלֵּב שֶׁל כָּל יְהוּדִי, הוּא רוֹצֶה לִשְׁמֹר אֶת כָּל הַמִּצְווֹת".

💡 כְּשֶׁאָנוּ מַאֲמִינִים בַּטּוֹב שֶׁבְּלֵב כָּל יְהוּדִי, אָנוּ עוֹזְרִים לוֹ לְהִתְקָרֵב לַתּוֹרָה וְלַמִּצְווֹת.
