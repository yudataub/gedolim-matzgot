הַתְּקִיעָה מִן הַלֵּב שֶׁל מוֹיְשְׁלֶה

בְּאֶמְצַע חֹדֶשׁ אֱלוּל, הִכְרִיז רַבִּי לֵוִי יִצְחָק מִבֶּרְדִּיצְ'בְֿ, שֶׁמְּכֻנֶּה גַּם "סַנְגּוֹרָם שֶׁל יִשְׂרָאֵל", כִּי הוּא מְחַפֵּשׂ בַּעַל תּוֹקֵעַ לְרֹאשׁ הַשָּׁנָה. רַבִּי לֵוִי יִצְחָק הִצִּיב תְּנַאי מְיֻחָד לְכָל מִי שֶׁרָצָה לִהְיוֹת הַבַּעַל תּוֹקֵעַ שֶׁלּוֹ: הָאִישׁ יִצְטָרֵךְ לֹא רַק לַעֲבֹר בְּחִינָה עַל תְּקִיעוֹת וְלִתְקֹעַ הֵיטֵב, אֶלָּא גַּם לְגַלּוֹת לוֹ מָה הִיא הַכַּוָּנָה שֶׁהוּא מְכַוֵּן בִּזְמַן הַתְּקִיעָה.

לְאַחַר הַהַכְרָזָה הִגִּיעוּ אֶל הָרַב הַצַּדִּיקִים הַקְּדוֹשִׁים וְהַמְּקֻבָּלִים שֶׁל הָעֲיָרָה. הֵם נָטְלוּ אֶת הַשּׁוֹפָר וְתָקְעוּ תְּקִיעוֹת מְצֻיָּנוֹת מְאֹד. שָׁאַל אוֹתָם רַבִּי לֵוִי יִצְחָק: מָה אַתֶּם מְכַוְּנִים כְּשֶׁאַתֶּם תּוֹקְעִים? הִתְחִילוּ הַמְּקֻבָּלִים לְהַסְבִּיר לוֹ שֶׁהֵם מְכַוְּנִים עַל שֵׁמוֹת קְדוֹשִׁים, עַל יִחוּדִים וְעַל עוֹלָמוֹת רוּחָנִיִּים גְּבוֹהִים מְאֹד. שָׁמַע רַבִּי לֵוִי יִצְחָק אֶת דִּבְרֵיהֶם, עָנָה לָהֶם תּוֹדָה רַבָּה, וְאָמַר שֶׁאִם יִצְטָרֵךְ אוֹתָם הוּא יִתְקַשֵּׁר אֲלֵיהֶם.

כָּךְ עָבְרוּ כָּל הָאֲנָשִׁים שֶׁעָמְדוּ בַּתּוֹר, וְהַתּוֹר הִסְתַּיֵּם לַלְּחָלוּטִין, אַךְ רַבִּי לֵוִי יִצְחָק עֲדַיִן לֹא מָצָא אֶת הַבַּעַל תּוֹקֵעַ שֶׁחִפֵּשׂ. בִּקֵּשׁ הָרַב מֵאַנְשֵׁי הָעֲיָרָה: תְּחַפְּשׂוּ לִי עוֹד מִישֶׁהוּ בָּעֲיָרָה שֶׁיּוֹדֵעַ לִתְקֹעַ. אֲנָשִׁים יָצְאוּ וְחִפְּשׂוּ, וּלְבַסּוֹף מָצְאוּ עוֹד אֶחָד – אֶת מוֹיְשְׁלֶה שׁוֹאֵב הַמַּיִם.

כְּשֶׁשָּׁמַע מוֹיְשְׁלֶה שֶׁמְּחַפֵּשִׂים אוֹתוֹ, הוּא הָיָה בְּהַפְתָּעָה גְּדוֹלָה. הוּא אָמַר לְעַצְמוֹ: אֲנִי? מוֹיְשְׁלֶה הַפָּשׁוּט אֶהְיֶה הַבַּעַל תּוֹקֵעַ שֶׁל רַבִּי לֵוִי יִצְחָק? הֲרֵי יֵשׁ פֹּה הַרְבֵּה קְדוֹשִׁים וְצַדִּיקִים, מִי אֲנִי בִּכְלָל? אַךְ מִכֵּוָן שֶׁאַף אֶחָד לֹא מָצָא חֵן בְּעֵינֵי רַבִּי לֵוִי יִצְחָק, נִכְנַס מוֹיְשְׁלֶה אֶל הָרַב. מוֹיְשְׁלֶה לָקַח אֶת הַשּׁוֹפָר וּבָא לִתְקֹעַ. הוּא תָּקַע בַּשּׁוֹפָר מְצֻיָּן מְאֹד.

כָּעֵת הִגִּיעַ הַחֵלֶק הַשֵּׁנִי שֶׁל הַבְּחִינָה. רַבִּי לֵוִי יִצְחָק שָׁאַל אֶת מוֹיְשְׁלֶה: תַּגִּיד לִי, מָה אַתָּה מְכַוֵּן כְּשֶׁאַתָּה תּוֹקֵעַ? מוֹיְשְׁלֶה שׁוֹאֵב הַמַּיִם הִתְבַּיֵּשׁ מְאֹד וְאָמַר: רַבֶּה, זֶה מַמָּשׁ דְּבָרִים אִישִׁיִּים, לֹא נָעִים לִי לְסַפֵּר. אַךְ רַבִּי לֵוִי יִצְחָק אָמַר לוֹ: הֶסְכֵּם זֶה הֶסְכֵּם! אֲנִי צָרִיךְ לָדַעַת מָה אַתָּה מְכַוֵּן.

מוֹיְשְׁלֶה נֶאֱנַח מֵעֹמֶק הַלֵּב וְאָז אָמַר: רַבֶּה, אַתָּה יוֹדֵעַ שֶׁיֵּשׁ לִי בַּבַּיִת שֶׁבַע בָּנוֹת וַאֲנִי צָרִיךְ לְחַתֵּן אוֹתָן, וְאֵין לִי מֵאֵיפֹה לְחַתֵּן אוֹתָן. כְּשֶׁאֲנִי מִתְפַּלֵּל לַקָּדוֹשׁ בָּרוּךְ הוּא, כְּשֶׁאֲנִי תּוֹקֵעַ, הַלֵּב שֶׁלִּי פָּצוּעַ, הַלֵּב שֶׁלִּי כּוֹאֵב, וְהַלֵּב שֶׁלִּי מִתְפַּלֵּל לַה': רִבּוֹנוֹ שֶׁל עוֹלָם, תַּעֲזֹר לִי לְחַתֵּן אֶת הַבָּנוֹת שֶׁלִּי!

אָמַר לוֹ רַבִּי לֵוִי יִצְחָק: אַתָּה תִּהְיֶה הַבַּעַל תּוֹקֵעַ שֶׁלִּי!

💡 הַקָּדוֹשׁ בָּרוּךְ הוּא מְחַפֵּשׂ אֶת עֹמֶק הַלֵּב וְאֶת הַתְּפִלָּה הַכְּנֵה שֶׁלָּנוּ, וְלֹא מִלִּים וְכַוָּנוֹת מְסֻבָּכוֹת.
