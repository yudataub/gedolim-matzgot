נִצְחוֹנוֹ שֶׁל דָּוִד בִּזְכוּת הָאֱמוּנָה

לִפְנֵי כְּאַלְפַּיִם וּתְשַׁע מֵאוֹת שָׁנִים, הִכְרִיזוּ הַפְּלִשְׁתִּים מֵעַזָּה מִלְחָמָה עַל עַם יִשְׂרָאֵל. הֵם בָּחֲרוּ לוֹחֵם עֲנַק מְמַדִּים, גָּלְיָת שְׁמוֹ, שֶׁהָיָה לָבוּשׁ בְּשִׁרְיוֹן כָּבֵד וְאָחַז בְּחֶרֶב עֲנָקִית. בְּמֶשֶׁךְ אַרְבָּעִים יָמִים עָמַד גָּלְיָת מוּל מַחֲנֵה יִשְׂרָאֵל, בִּזָּה אוֹתָם וְקָרָא לָהֶם: "בְּשָׁבִיל מָה שֶׁנַּעֲשֶׂה מִלְחָמָה גְּדוֹלָה, בּוֹאוּ תִּבְחֲרוּ אֶת הָאִישׁ הֲכִי חָזָק שֶׁלָּכֶם, וּמִי שֶׁיְּנַצֵּחַ בַּקְּרָב – הוּא יוֹכְרַז הַמְּנַצֵּחַ בְּכָל הַמִּלְחָמָה". אַף חַיָּל מִלּוֹחֲמֵי יִשְׂרָאֵל לֹא הָיָה מְסֻגָּל לָצֵאת נֶגְדּוֹ מֵרֹב פַּחַד.

דָּוִד הַצָּעִיר הִגִּיעַ לַמַּחֲנֶה כְּדֵי לִפְגֹּשׁ אֶת שְׁלֹשֶׁת אֶחָיו הַלּוֹחֲמִים. הוּא שָׁמַע מֵהֶם עַל הַפְּלִשְׁתִּי מֵעַזָּה, וְכֵן עַל הַבְטָחַת הַמֶּלֶךְ שָׁאוּל שֶׁמִּי שֶׁיְּנַצֵּחַ אוֹתוֹ יִזְכֶּה בְּבִתּוֹ מִיכַל וּבְכֶסֶף רָב. דָּוִד לֹא נִבְהַל וְאָמַר לָהֶם: "מָה, מָה אַתֶּם מְפַחֲדִים? תִּבְטְחוּ בַּקָּדוֹשׁ בָּרוּךְ הוּא! אֲנִי לֹא מְפַחֵד מִמֶּנּוּ, אֲנִי מוּכָן לְהִלָּחֵם נֶגְדּוֹ!".

דָּוִד נִגַּשׁ אֶל הַמֶּלֶךְ שָׁאוּל וְהִצִּיעַ לָצֵאת לַקְּרָב. הַמֶּלֶךְ שָׁאוּל תָּמַהּ כֵּיצַד נַעַר צָעִיר יָכוֹל לְהִתְמוֹדֵד מוּל גָּלְיָת, שֶׁאֲפִלּוּ מֵיטַב הַלּוֹחֲמִים מְפַחֲדִים מִמֶּנּוּ. דָּוִד הֵשִׁיב לַמֶּלֶךְ שָׁאוּל: "כְּשֶׁהָיִיתִי פַּעַם רוֹעֵה צֹאן, בָּאוּ אַרְיֵה וְדֹב גָּדוֹל וְחָטְפוּ כֶּבֶשׂ מֵהָעֵדֶר, וַאֲנִי לֹא וִתַּרְתִּי וְרַצְתִּי אַחֲרֵיהֶם, וְנִלְחַמְתִּי בָּהֶם וְנִצַּחְתִּי אוֹתָם. וְהַכֹּל בִּזְכוּת הַקָּדוֹשׁ בָּרוּךְ הוּא, בִּזְכוּת הַתְּפִלּוֹת שֶׁהִתְפַּלַּלְתִּי אֵלָיו. וְגַם עַכְשָׁו, בְּעֶזְרַת הַתְּפִלּוֹת וְהָאֱמוּנָה הַחֲזָקָה שֶׁיֵּשׁ לִי, אֲנִי אַצְלִיחַ לִגְבֹּר עַל אוֹתוֹ גָּלְיָת רָשָׁע מְרֻשָּׁע".

דָּוִד לָקַח חֲמִשָּׁה חַלּוּקֵי נַחַל, אֲבָנִים חֲלָקוֹת מֵהַנַּחַל, וּמַקְלֵעַ בְּיָדוֹ, וְנִצַּב לְהִלָּחֵם. גָּלְיָת זִלְזֵל בְּדָּוִד וְאָמַר לוֹ: "מִי אַתָּה? אַתָּה יֶלֶד, אֲנִי תּוֹךְ שְׁנִיָּה מְנַצֵּחַ אוֹתְךָ!". דָּוִד עָנָה לוֹ בְּאֹמֶץ: "אַתָּה בָּא אֵלַי בְּחֶרֶב, בַּחֲנִית וּבְכִידוֹן, וְאָנֹכִי בָא אֵלֶיךָ בְּשֵׁם ה' צְבָאוֹת אֱלֹהֵי מַעַרְכוֹת יִשְׂרָאֵל אֲשֶׁר חֵרַפְתָּ. הַYּוֹם הַזֶּה יְסַגֶּרְךָ ה' בְּיָדִי וְהִכִּיתִךָ וַהֲסִרֹתִי אֶת רֹאשְׁךָ מֵעָלֶיךָ, וְיֵדְעוּ כָּל הַקָּהָל הַזֶּה כִּי לֹא בְּחֶרֶב וּבַחֲנִית יְהוֹשִׁיעַ ה', כִּי לַה' הַמִּלְחָמָה וְנָתַן אֶתְכֶם בְּיָדֵנוּ!".

דָּוִד הִתְפַּלֵּל מִכָּל הַלֵּב, לָקַח חֲלוּק נַחַל אֶחָד, שָׂם בַּמַּקְלֵעַ, כִּוֵּן וְהֵעִיף לְעֵבֶר גָּלְיָת. הָאֶבֶן פָּגְעָה לְגָּלְיָת בְּדִיּוּק בַּמֵּצַח, בַּמָּקוֹם שֶׁבּוֹ לֹא הָיָה שִׁרְיוֹן, וְהוּא נָפַל וּמֵת. בְּעִקְבוֹת זֹאת קָמוּ חַיָּלֵי יִשְׂרָאֵל, רָדְפוּ אַחֲרֵי הַפְּלִשְׁתִּים מֵעַזָּה וְנִצְחוּ אוֹתָם.

💡 כַּאֲשֶׁר מִתְפַּלְלִים לַה' וּבֹטְחִים בּוֹ בְּאֱמוּנָה חֲזָקָה, אֶפְשָׁר לְנַצֵּחַ גַּם אֶת הַקְּשָׁיִים הַגְּדוֹלִים בְּיוֹתֵר.
