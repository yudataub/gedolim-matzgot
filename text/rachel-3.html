הַתְּפִלָּה הַמְּרַגֶּשֶׁת עַל קֶבֶר רָחֵל אִמֵּנוּ

בִּימֵי מִלְחֶמֶת הָעוֹלָם הַשְּׁנִיָּה, חָיוּ יְהוּדִים רַבִּים בְּאֵירוֹפָה וְסָבְלוּ סֵבֶל נוֹרָא וְקָשֶׁה מְאֹד תַּחַת שִׁלְטוֹן הַנָּאצִים. בְּאוֹתָם יָמִים קָשִׁים וּמֻרְכָּבִים, תַּלְמִידֵי יְשִׁיבַת מִיר וּבְרָאשָׁם רֹאשׁ הַיְּשִׁיבָה, הָרַב חַיִּים שְׁמוּאֶלֵבִיץ זֵכֶר צַדִּיק לִבְרָכָה, נֶאֶלְצוּ לִבְרֹחַ מֵאַרְצָם לִיטָא כְּדֵי לְהִנָּצֵל מֵהַסַּכָּנָה הַגְּדוֹלָה שֶׁאִיְּמָה עֲלֵיהֶם. הֵם עָבְרוּ דֶּרֶךְ אֲרֻכָּה וְקָשָׁה מְאֹד עַד הָעִיר שַׁנְהַאי שֶׁבְּסִין, וּמִשָּׁם הִמְשִׁיכוּ בְּמַסָּעָם וְעָלוּ בְּשָׁלוֹם לְאֶרֶץ יִשְׂרָאֵל.

בְּאוֹתָהּ תְּקוּפָה, הַיְּהוּדִים שֶׁכְּבָר חָיוּ בְּאֶרֶץ יִשְׂרָאֵל כִּמְעַט וְלֹא יָדְעוּ כְּלָל עַל הַסֵּבֶל הַנּוֹרָא וְעַל הַצָּרוֹת הַקָּשׁוֹת שֶׁסּוֹבְלִים אֲחֵיהֶם הַיְּהוּדִים בְּאֵירוֹפָה. רַק כַּאֲשֶׁר הִגִּיעוּ נִצּוֹלִים רִאשׁוֹנִים לָאָרֶץ, הֵם הֵחֵלוּ לְסַפֵּר לְאַנְשֵׁי הָאָרֶץ עַל הַמַּצָּב הַקָּשֶׁה, עַל הָרְדִיפוֹת וְעַל הַסֵּבֶל הָרַב שֶׁעוֹבְרִים הַיְּהוּדִים שָׁם.

כַּאֲשֶׁר הִגִּיעַ הָרַב חַיִּים שְׁמוּאֶלֵבִיץ לְאֶרֶץ יִשְׂרָאֵל, הוּא מִיָּד הֵחֵל לְעוֹרֵר אֶת כָּל יוֹשְׁבֵי הָאָרֶץ לְהִתְפַּלֵּל, לִבְכּוֹת וּלְהִתְחַנֵּן בְּכָל הַלֵּב לַהַצָּלָה שֶׁל עַם יִשְׂרָאֵל. הָרַב חַיִּים עַצְמוֹ הָיָה מִסְתּוֹבֵב בַּמְּקוֹמוֹת הַקְּדוֹשִׁים בָּאָרֶץ, כְּגוֹן בַּכֹּתֶל הַמַּעֲרָבִי, בִּמְעָרַת הַמַּכְפֵּלָה וּבְעוֹד מְקוֹמוֹת קְדוֹשִׁים נוֹסָפִים. בְּכָל מָקוֹם שֶׁבּוֹ הִגִּיעַ, הִתְפַּלֵּל הָרַב בִּבְכִי רַב וּבִכְאֵב גָּדוֹל עַל הַמַּצָּב הַקָּשֶׁה שֶׁל עַמּוֹ.

יוֹם אֶחָד הֶחְלִיט הָרַב חַיִּים לָלֶכֶת גַּם לְקֶבֶר רָחֵל אִמֵּנוּ כְּדֵי לְהִתְפַּלֵּל שָׁם. הָרַב חַיִּים וְתַלְמִידָיו הִצְטָרְפוּ לְשַׁיָּרָה שֶׁנָּסְעָה לְכִוּוּן הָעִיר בֵּית לֶחֶם, הַסְּמוּכָה לִירוּשָׁלַיִם, שָׁם נִמְצָא קֶבֶר רָחֵל.

כַּאֲשֶׁר הִגִּיעוּ הָרַב וְתַלְמִידָיו אֶל מְקוֹם קְבוּרַת רָחֵל אִמֵּנוּ, הֵחֵל הָרַב חַיִּים לְהִתְפַּלֵּל מִקִּירות לִבּוֹ. הַתַּלְמִידִים שֶׁעָמְדוּ סְבִיבוֹ רָאוּ אֶת רַבָּם בִּתְפִלָּתוֹ הַנִּרְגֶּשֶׁת וּבְהִתְרַגְּשׁוּתוֹ הַגְּדוֹלָה, וְהֵם הִצְטָרְפוּ גַּם הֵם לְתַחֲנוּנָיו הָרַבִּים וְלִתְפִלָּתוֹ הַזַּכָּה.

הָרַב חַיִּים נָטַף דְּמָעוֹת רַבּוֹת, בָּכָה בְּקוֹל גָּדוֹל וְקָרָא מֵעוֹמֶק הַלֵּב: "אִמָּא רָחֵל! אִמָּא רָחֵל! אֲנַחְנוּ יוֹדְעִים שֶׁכָּתוּב בַּנָּבִיא יִרְמְיָהוּ אֶת דִּבְרֵי הַשֵּׁם: 'מִנְעִי קוֹלֵךְ מִבֶּכִי וְעֵינַיִךְ מִדִּמְעָה'. אֲבָל אֲנִי, חַיִּים, הִגַּעְתִּי עַכְשָׁו מֵהַמִּלְחָמָה הַנּוֹרָאָה וְרָאִיתִי בְּעֵינַי אֶת כָּל הַצָּרוֹת הַקָּשׁוֹת שֶׁעַם יִשְׂרָאֵל סוֹבֵל. מִתְחַנֵּן אֲנִי וּמְבַקֵּשׁ מִמֵּךְ שֶׁלֹּא תַּפְסִיקִי לִבְכּוֹת! תִּבְכִּי, רָחֵל, תִּבְכִּי! תִּבְכִּי בִּשְׁבִיל עַם יִשְׂרָאֵל! תְּבַקְשִׁי רַחֲמִים מֵרִבּוֹנוֹ שֶׁל עוֹלָם שֶׁיֹּאמַר דַּי לְצָרוֹתֵינוּ. בְּעֶזְרַת הַשֵּׁם תִּתְקַבֵּלְנָה תְּפִלּוֹתַיִךְ וְעַם יִשְׂרָאֵל יָשׁוּב לִגְבוּלוֹ וּלְאַרְצוֹ בִּמְהֵרָה בְּיָמֵינוּ, אָמֵן!"
