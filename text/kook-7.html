הַתְּפִלָּה שֶׁשִּׁנְתָה אֶת הַלֵּב

אֶחָד הָאֲנָשִׁים הַחֲשׁוּבִים וְהַיְדוּעִים שֶׁמְּעִידִים עַל גְּדֻלָּתוֹ שֶׁל הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק הוּא תַּלְמִידוֹ הַמֻּבְהָק, הָרַב הַנָּזִיר – הָרַב דָּוִד הַכֹּהֵן. בִּתְחִלַּת דַּרְכּוֹ הָיָה הָרַב דָּוִד הַכֹּהֵן אִישׁ מַשְׂכִּיל. הָיוּ לוֹ שְׁאֵלוֹת מְאֹד מְאֹד עֲמֻקּוֹת עַל הַשְׁגָחָתוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בָּעוֹלָם וְעַל עִנְיְנֵי אֱמוּנָה. הוּא חִפֵּשׂ בְּכָל לִבּוֹ תְּשׁוּבָה אֲמִתִּית וּבְרוּרָה לַשְּׁאֵלוֹת שֶׁהִטְרִידוּ אֶת נַפְשׁוֹ, וַאֲנָשִׁים שֶׁהִכִּירוּ אוֹתוֹ אָמְרוּ לוֹ: "לֵךְ אֶל הָרַב קוּק". בְּאוֹתָם יָמִים שָׁהָה הָרַב קוּק בִּשְׁוַיְץ.

הָרַב הַנָּזִיר הֶחְלִיט לִנְסֹעַ וְלָלֶכֶת אֶל הָרַב קוּק כְּדֵי לִמְצֹא תְּשׁוּבוֹת לִשְׁאֵלוֹתָיו. הוּא הִגִּיעַ לְבֵיתוֹ שֶׁל הָרַב קוּק וְהִתְאَرֵחַ שָׁם בַּלַּיְלָה עַד אֶמְצַע הַלַּיְלָה. בְּמַהֲלַךְ הַשָּׁעוֹת הָאֲרֻכּוֹת שֶׁל הַלַּיְלָה, שָׁאַל הָרַב הַנָּזִיר אֶת כָּל הַשְּׁאֵלוֹת הָעֲמֻקּוֹת שֶׁהָיוּ לוֹ, וְהָרַב קוּק עָנָה לוֹ בְּסַבְלָנוּת וְנָתַן לוֹ תְּשׁוּבוֹת. אֲבָל אַף עַל פִּי שֶׁקִּבֵּל תְּשׁוּבוֹת שִׂכְלִיּוֹת, הַתְּשׁוּבוֹת הָאֵלֶּה עֲדַיִן לֹא נָתְנוּ מָנוֹחַ לְנַפְשׁוֹ. הָרַב הַנָּזִיר כָּתַב מְאֻחָר יוֹתֵר עַל אוֹתוֹ לַיְלָה מְתוּחַ וְקָשֶׁה: "עַל מִשְׁכָּבִי לֹא שָׁכַב לִבִּי, גּוֹרַל חַיַּי הָיוּ עַל כַּפּוֹת הַמֹּאזְנַיִם". הוּא הִרְגִּישׁ שֶׁכָּל עֲתִידוֹ, אֱמוּנָתוֹ וְכָל דַּרְכּוֹ בַּחַיִּים תְּלוּיִים בַּתְּשׁוּבָה שֶׁהוּא מְחַפֵּשׂ.

וְהִנֵּה הִגִּיעַ הַבֹּקֶר. הָרַב הַנָּזִיר שָׁמַע אֶת צַעֲדָיו שֶׁל הָרַב קוּק הַמִּתְהַלֵּךְ בַּבַּיִת. מִיָּד אַחַר כָּךְ שָׁמַע אֶת הָרַב קוּק מְבָרֵךְ בְּכַוָּנָה רַבָּה אֶת בִּרְכוֹת הַשַּׁחַר, מִתְפַּלֵּל אֶת תְּפִלַּת הָעֲקֵדָה, וּמִתְפַּלֵּל בְּשִׁיר וּבְנִגּוּן עֶלְיוֹן וְקָדוֹשׁ. הָרַב הַנָּזִיר עָמַד וְהִקְשִׁיב בְּהִתְרַגְּשׁוּת לַתְּפִלָּה הַנִּפְלָאָה הַזֹּאת. הוּא חָשׁ כֵּיצַד הַתְּפִלָּה הַזֹּאת חוֹדֶרֶת אֶל תּוֹךְ עֹמֶק נִשְׁמָתוֹ, וְכָךְ כָּתַב מְאֻחָר יוֹתֵר: "וְהִנֵּה נֶהֶפַכְתִּי וְהָיִיתִי לְאִישׁ אַחֵר". תְּפִלָּתוֹ הַזַּכָּה שֶׁל הָרַב קוּק שִׁנְתָה אֶת כָּל הַרְגָּשָׁתוֹ וְנָתְנָה לוֹ אֶת הַתְּשׁוּבָה הָעֲמֻקָּה שֶׁחִפֵּשׂ.

מִיָּד אַחֲרֵי שֶׁהִסְתַּיְמָה הַתְּפִלָּה, מִהֵר הָרַב הַנָּזִיר לְבַשֵּׂר אֶת הַבְּשׂוֹרָה הַטּוֹבָה בְּמִכְתָּב. הוּא כָּתַב בְּהִתְרַגְּשׁוּת רַבָּה: "יוֹתֵר מֵאֲשֶׁר פִּלַּלְתִּי, מָצָאתִי – מָצָאתִי לִי רַב". כַּאֲשֶׁר הָרַב דָּוִד הַכֹּהֵן רָאָה בְּעֵינָיו אֶת הָרַב קוּק מִתְפַּלֵּל, הוּא לֹא רָאָה סְתָם אָדָם שֶׁמִּתְפַּלֵּל, אֶלָּא הוּא רָאָה נְשָׁמָה אֱלֹהִית שֶׁבּוֹעֶרֶת דֶּרֶךְ הַכְּלִי שֶׁנִּקְרָא הַגּוֹף שֶׁלָּהּ. מֵאוֹתוֹ רֶגַע מֻפְלָא הֵבִין הָרַב הַנָּזִיר שֶׁהוּא מָצָא אֶת הָרַב וְאֶת הַמּוֹרֶה הָרוּחָנִי שֶׁחִפֵּשׂ בְּכָל לִבּוֹ.

💡 תְּפִלָּה זַכָּה וֶאֱמֶת שֶׁיּוֹצֵאת מֵעֹמֶק הַלֵּב יְכוֹלָה לְהָאִיר אֶת הָעוֹלָם וּלְשַׁנּוֹת אֶת הַלְּבָבוֹת לְטוֹבָה.
