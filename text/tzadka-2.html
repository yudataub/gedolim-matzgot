הַתַּכְשִׁיט וְהָעֲנָוָה שֶׁל הַיֶּלֶד יְהוּדָה

זֶהוּ סִפּוּר מְיֻחָד שֶׁסִּפֵּר מָרָן הָרַב יְהוּדָה צָדְקָה זֵכֶר צַדִּיק לִבְרָכָה עַל יְמֵי יַלְדוּתוֹ.

כַּאֲשֶׁר הָיָה יְהוּדָה יֶלֶד צָעִיר, קָרוֹב לְגִיל עֶשֶׂר שָׁנִים, הִגִּיעַ עֶרֶב חַג הַפֶּסַח.

בְּאוֹתוֹ הַיּוֹם, מְעַט לִפְנֵי שֶׁנִּכְנְסָה וְהִתְקַדְּשָׁה קְדֻשַּׁת הַחַג, רָצְתָה אִמּוֹ לְהָכִין אוֹתוֹ לִכְבוֹד יוֹם הַטּוֹב.

הִיא הִלְבִּישָׁה אֶת יְהוּדָה הַקָּטָן בַּחֲלִיפַת בְּגָדִים חֲדָשָׁה, יָפָה וַחֲגִיגִית שֶׁהוּכְנָה מְיֻחָד לִכְבוֹד הֶחָג.

כְּדֵי לְהוֹסִיף יֹפִי וְכָבוֹד לַחֲלִיפָה הַחֲדָשָׁה, הִצְמִידָה לוֹ אִמּוֹ תַּכְשִׁיט נָאֶה עַל גַּבֵּי הַבֶּגֶד, מִמַּעַל לַכִּיס.

לְאַחַר שֶׁהִלְבִּישָׁה אוֹתוֹ בַּחֲלִיפָה הַחֲדָשָׁה עִם הַתַּכְשִׁיט הַנָּאֶה, נָתְנָה אִמּוֹ בְּיָדוֹ מַצּוֹת אֲחָדוֹת.

מַצּוֹת אֵלּוּ הָיוּ מַצּוֹת מְיֻחָדוֹת מְאֹד, שֶׁנֶּאֱפוּ בְּאוֹתוֹ הַיּוֹם מַמָּשׁ בִּשְׁעוֹת אַחַר הַצָּהֳרַיִם, בַּאֲפִיָּה מְיֻחֶדֶת שֶׁנַּעֲשְׂתָה לְשֵׁם מִצְוָה.

אִמּוֹ בִּקְשָׁה מִמֶּנּוּ לָקַחַת אֶת הַמַּצּוֹת הָאֵלֶּה וּלְהוֹלִיכָן מַתָּנָה וְדוֹרוֹן אֶל אִישׁ קָדוֹשׁ וְצַדִּיק, הֲלוֹא הוּא הַמְּקֻבָּל רַבִּי אַבְרָהָם עַדֵס זֵכֶר לִבְרָכָה.

יְהוּדָה הַקָּטָן לָקַח אֶת הַמַּצּוֹת הָאֲפוּיוֹת לְשֵׁם מִצְוָה, וְהָלַךְ לְהוֹלִיכָן אֶל בֵּיתוֹ שֶׁל הַצַּדִּיק רַבִּי אַבְרָהָם עַדֵס.

כֵּיוָן שֶׁהִגִּיעַ יְהוּדָה סָמוּךְ לִמְעוֹנוֹ וּלְבֵיתוֹ שֶׁל הַצַּדִּיק, עָצַר פִּתְאֹם מַמָּשׁ עַל סַף פֶּתַח הַכְּנִיסָה לַבַּיִת.

בְּאוֹתוֹ רֶגַע שֶׁעָמַד עַל הַסַּף, חָרַד לִבּוֹ שֶׁל יְהוּדָה מְאֹד.

הוּא הִרְגִּישׁ בְּתוֹךְ תּוֹכוֹ כִּי אֵינוֹ רָאוּי וְאֵינוֹ נָכוֹן לְהוֹפִיעַ עִם תַּכְשִׁיט מְפֹאָר וּלְהִתְגָּאוֹת בִּלְבוּשׁוֹ לִפְנֵי אִישׁ קָדוֹשׁ וְרוֹמָם כָּזֶה.

הוּא הֵבִין כִּי כַּאֲשֶׁר נִכְנָסִים אֶל אִישׁ קָדוֹשׁ וְצַדִּיק, יֵשׁ לָלֶכֶת בְּדֶרֶךְ שֶׁל עֲנָוָה, וְלֹא לְהִתְגָּאוֹת בְּתַכְשִׁיטִים וּבִבְגָדִים נָאִים.

מִיָּד, כְּשֶׁהִרְגִּישׁ זֹאת, שָׁלַח יְהוּדָה אֶת יָדוֹ וְתָלַשׁ מֵעַל הַחֲלִיפָה הַחֲדָשָׁה אֶת הַתַּכְשִׁיט הַנָּאֶה שֶׁהִצְמִידָה לוֹ אִמּוֹ מִמַּעַל לַכִּיס.

הוּא לֹא רָצָה שֶׁיִּהְיֶה בּוֹ שׁוּם שֶׁמֶץ שֶׁל גַּאֲוָה, וְלָכֵן הֵסִיר אֶת הַתַּכְשִׁיט בַּמְּהִירוּת מֵעַל בִּגְדוֹ.

רַק לְאַחַר שֶׁתָּלַשׁ אֶת הַתַּכְשִׁיט, הִרְגִּישׁ יְהוּדָה כִּי כָּעֵת הוּא נִכְנָס אֶל הָרַב בַּדֶּרֶךְ הַנְּכוֹנָה וְהָרְאוּיָה.

כָּךְ נִכְנַס יְהוּדָה אֶל מְעוֹנוֹ שֶׁל הַצַּדִּיק רַבִּי אַבְרָהָם עַדֵס בְּצוּרָה נְאוֹתָה וּרְאוּיָה לְפִי הַרְגָּשָׁתוֹ הַפְּנִימִית, בְּדֶרֶךְ שֶׁל עֲנָוָה, צְנִיעוּת וּשְׁפְלוּת רוּחַ.

💡 מִמַּעֲשֵׂהוּ שֶׁל הַיֶּלֶד יְהוּדָה אָנוּ לוֹמְדִים לִבְרֹחַ מִן הַגַּאֲוָה וְלִנְהֹג תָּמִיד בַּעֲנָוָה וּבִצְנִיעוּת.
