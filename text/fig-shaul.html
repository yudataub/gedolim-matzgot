שָׁאוּל הַמֶּלֶךְ – מֵחִפּוּשׂ אֲתוֹנוֹת לִמְלוּכָה

פַּעַם אַחַת אָבַד לְאָבִיו שֶׁל שָׁאוּל עֵדֶר שֶׁל אֲתוֹנוֹת. פָּנָה הָאָב אֶל שָׁאוּל בְּנוֹ וְאָמַר לוֹ: 'תַּחְפֵּשׂ אֶת הָאֲתוֹנוֹת'. שָׁאוּל קִבֵּל מִיָּד אֶת הַצִּוּוּי מֵאָבִיו בְּאַהֲבָה וּבְכָבוֹד רַב, וְלֹא הִסְהֵס כְּלָל. הוּא יָצָא מִיָּד לַדֶּרֶךְ כְּדֵי לְמַלֵּאת אֶת בַּקָּשַׁת אָבִיו. שָׁאוּל לֹא הָיָה מֵהָאֲנָשִׁים שֶׁמִּתְיָאֲשִׁים מַהֵר, וְהוּא לֹא וִתֵּר עַל הַצִּוּוּי הַזֶּה שֶׁקִּבֵּל. הוּא לֹא חִפֵּשׂ רַק קְצָת אֶת הָאֲתוֹנוֹת, וְהוּא בְּשׁוּם אֹפֶן לֹא חָזַר כָּךְ סְתָם אֶל אָבִיו וְאָמַר לוֹ: 'אַבָּא, מִצְטַעֵר, לֹא מָצָאתִי'. מַמָּשׁ לֹא! שָׁאוּל יָצָא לְמַסָּע אָרֹךְ וּמַאֲמִיץ, וְהוּא הָלַךְ וְעָבַר מֵאֶרֶץ לְאֶרֶץ, וּמִמָּקוֹם לְמָקוֹם, כְּדֵי לַחְפּוֹשׂ אֵיפֹה נִמְצָאוֹת הָאֲתוֹנוֹת שֶׁל אָבִיו. הוּא הִמְשִׁיךְ לַחְפּוֹשׂ וְלַחְפּוֹשׂ בִּמְסִירוּת רַבָּה, כִּי הָיָה חָשׁוּב לוֹ מְאֹד לְקַיֵּם אֶת דִּבְרֵי אָבִיו.

הַחִפּוּשׂ הָאָרֹךְ נִמְשַׁךְ זְמַן רַב, עַד כְּדֵי כָּךְ שֶׁשָּׁאוּל הֵחֵל לִדְאֹג לְאָבִיו. הוּא בָּא וְאָמַר: 'אוּלַי אַבָּא שֶׁלִּי יַרְגִּישׁ רַע, יִתְמַלֵּא בְּצַעַר שֶׁהוּא כְּבָר לֹא יוֹדֵעַ מָה קוֹרֶה אִתִּי'. שָׁאוּל לֹא דָּאַג רַק לַמְּשִׂימָה, אֶלָּא הָיָה מָלֵא בְּרַחֲמִים וּבִדְאָגָה עֲמֻקָּה לָרְגָשׁוֹת שֶׁל אָבִיו שֶׁמַּמְתִּין לוֹ בַּבַּיִת.

יַחַד עִם שָׁאוּל יָצָא לַמַּסָּע גַּם הַנַּעַר שֶׁלּוֹ, הַנַּעַר הַמְּשָׁרֵת. כַּאֲשֶׁר רָאָה הַנַּעַר שֶׁהֵם לֹא מוֹצְאִים אֶת הָאֲתוֹנוֹת, הוּא אָמַר לְשָׁאוּל: 'בּוֹא נֵלֵךְ וְנִשְׁאַל אֶת הָרוֹאֶה, אוּלַי הוּא יוֹדֵעַ מָה קָרָה לָאֲתוֹנוֹת'. שָׁאוּל יָדַע שֶׁצָּרִיךְ לָתֵת כָּבוֹד גָּדוֹל לַ'רוֹאֶה'. אֲבָל כַּאֲשֶׁר בָּדְקוּ מָה יֵשׁ אִתָּם, הִתְבָּרֵר שֶׁלֹּא נִשְׁאַר לְשָׁאוּל כֶּסֶף כְּלָל. וּלְמִי כֵּן נִשְׁאַר כֶּסֶף? לַנַּעַר שֶׁלּוֹ! אֵיךְ קָרָה שֶׁלַּנַּעַר הַמְּשָׁרֵת הָיָה כֶּסֶף וּלְשָׁאוּל לֹא? הַסִּבָּה הִיא בִּגְלַל שֶׁשָּׁאוּל הִתְיַחֵס אֶל הַנַּעַר בְּאַהֲבָה גְּדוֹלָה וּבְכָבוֹד גָּדוֹל. כְּשֶׁשָּׁאוּל הָיָה קוֹנֶה אֹכֶל בַּדֶּרֶךְ, הוּא לֹא שָׁמַר אֶת הַכֶּסֶף רַק לְעַצְמוֹ, אֶלָּא הָיָה נֹתֵן גַּם לַנַּעַר שֶׁלּוֹ מֵהָאֹכֶל וּמֵהַכֶּסֶף. שָׁאוּל נָהַג בַּנַּעַר הַמְּשָׁרֵת מַמָּשׁ בְּאַהֲבָה, בְּכָבוֹד וּבִדְאָגָה כֵּנָה.

בְּסוֹפוֹ שֶׁל דָּבָר, הוּא מַגִּיעַ אֶל הָרוֹאֶה, שֶׁהוּא שְׁמוּאֵל הַנָּבִיא. כַּאֲשֶׁר פָּגַשׁ שָׁאוּל אֶת שְׁמוּאֵל הַנָּבִיא, קָרָה דָּבָר מֻפְלָא מְאֹד. שְׁמוּאֵל הַנָּבִיא בִּשֵּׂר לוֹ: 'אַתָּה תִּהְיֶה מֶלֶךְ יִשְׂרָאֵל!'. שָׁאוּל יָצָא מֵהַבַּיִת שֶׁל אָבִיו רַק כְּדֵי לַחְפּוֹשׂ אֲתוֹנוֹת, אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הַמִּדּוֹת הַטּוֹבוֹת שֶׁלּוֹ. כְּשֶׁאָבִיו אָמַר לוֹ מַשֶּׁהוּ, הוּא מִיָּד רָץ לַעֲשׂוֹת וְלֹא וִתֵּר. גַּם הַהִתְיַחֲסוּת שֶׁלּוֹ אֶל הַנַּעַר הַמְּשָׁרֵת הָיְתָה מְלֵאַת כָּבוֹד. בִּגְלָל הַמִּדּוֹת הָנִפְלָאוֹת הָאֵלֶּה, בִּגְלַל כְּבוֹד הָאָב וּכְבוֹד הַזּוּלָת, הֵבִיא אוֹתוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא אֶל שְׁמוּאֵל הַנָּבִיא, וְכָךְ הוּא זָכָה בַּמְּלוּכָה עַל כָּל עַם יִשְׂרָאֵל.

💡 כַּאֲשֶׁר אָנוּ מְקַיְּמִים אֶת דִּבְרֵי הַהוֹרִים בִּמְסִירוּת וּמִתְנַהֲגִים בְּכָבוֹד וּבְאַהֲבָה לְכָל אָדָם, אָנוּ זוֹכִים לִבְרָכָה גְּדוֹלָה מֵהַקָּדוֹשׁ בָּרוּךְ הוּא.
