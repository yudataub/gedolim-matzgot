הָאוֹתִיּוֹת שֶׁפָּרְחוּ מֵהַתְּפִלִּין

לִפְנֵי כִּשְׁלֹשׁ מֵאוֹת שָׁנִים חַי בְּמָרוֹקוֹ רַב גָּדוֹל וְקָדוֹשׁ מְאֹד. שְׁמוֹ הָיָה רַבִּי חַיִּים בֶּן עַטָּר, וּבִגְלַל הַקְּדֻשָּׁה וְהַטָּהֳרָה הָרַבָּה שֶׁלּוֹ כִּנּוּ אוֹתוֹ בְּשֵׁם מְיֻחָד: "אוֹר הַחַיִּים הַקָּדוֹשׁ". לְרַבִּי חַיִּים הָיוּ תְּפִלִּין מְיֻחָדוֹת שֶׁאוֹתָן הָיָה מַנִּיחַ בְּכָל יוֹם וָיוֹם. בְּכָל פַּעַם שֶׁהִנִּיחַ אֶת הַתְּפִלִּין, הָיָה מִתְפַּלֵּל בִּדְבֵקוּת גְּדוֹלָה, בְּשִׂמְחָה וּבְאַהֲבָה רַבָּה אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא.

יוֹם אֶחָד הִגִּיעָה אֵלָיו אִשְׁתּוֹ כְּשֶׁהִיא בּוֹכָה מְאֹד. הִיא אָמְרָה לוֹ בְּדַאֲגָה: "רַבִּי חַיִּים, מָה אֶעֱשֶׂה אַחֲרֵי שֶׁתַּעֲזֹב אֶת הָעוֹלָם? לֹא יִהְיֶה מִי שֶׁיִּתֵּן לִי כֶּסֶף, לֹא יִהְיֶה מִי שֶׁיְּפַרְנֵס אוֹתִי, וַאֲנִי אֶהְיֶה עֲנִיָּה מְאֹד!" רַבִּי חַיִּים הִרְגִּיעַ אוֹתָהּ וְאָמַר לָהּ: "מִנְעִי קוֹלֵךְ מִבֶּכִי וְעֵינַיִךְ מִדִּמְעָה, אֵין לָךְ מָה לִהְיוֹת בִּדְאָגָה. אַחֲרֵי שֶׁאֶפְטַר מִן הָעוֹלָם, יָבֹא אִישׁ עָשִׁיר מְאֹד וְיִרְצֶה לִקְנוֹת אֶת הַתְּפִלִּין שֶׁלִּי. אַתְּ תִּמְכְּרִי לוֹ אוֹתָן וְתַקַּבְלִי עֲלֵיהֶן כֶּסֶף רַב שֶׁיַּסְפִּיק לָךְ. אֲבָל חָשׁוּב מְאֹד שֶׁתַּגִּידִי לוֹ שֶׁלֹּא יִשְׁכַּח: בְּכָל פַּעַם שֶׁהוּא מַנִּיחַ אֶת הַתְּפִלִּין, אָסוּר לוֹ לַחְשֹׁב מַחֲשָׁבוֹת שֶׁאֵינָן קְשׁוּרוֹת לַתְּפִלָּה וְלַקְּדֻשָּׁה. עָלָיו לַחְשֹׁב רַק מַחֲשָׁבוֹת קְדוֹשׁוֹת עַל הַתְּפִלָּה, וְשֶׁשּׁוּם דָּבָר לֹא יַסִּיחַ אֶת דַּעְתּוֹ".

לְאַחַר שֶׁאוֹר הַחַיִּים הַקָּדוֹשׁ נִפְטַר, הִגִּיעַ אֶל בֵּיתוֹ אִישׁ עָשִׁיר מְאֹד וְקָנָה אֶת הַתְּפִלִּין מֵאִשְׁתּוֹ שֶׁל הָרַב. מִיָּד כְּשֶׁהִגִּיעַ הָאִישׁ הָעָשִׁיר אֶל בֵּיתוֹ, הוּא הִתְיַשֵּׁב וְהִנִּיחַ אֶת הַתְּפִלִּין. כְּבָר בָּרֶגַע הָרִאשׁוֹן הוּא הִרְגִּישׁ אֶת הַקְּדֻשָּׁה וְאֶת הַטָּהֳרָה הַגְּדוֹלָה שֶׁל אוֹר הַחַיִּים הַקָּדוֹשׁ. הוּא הִרְגִּישׁ כֵּיצַד הוּא מִתְקָרֵב וְדָבֵק בַּקָּדוֹשׁ בָּרוּךְ הוּא, וּמִתְפַּלֵּל בְּשִׂמְחָה וּבְאַהֲבָה רַבָּה. בְּכָל יוֹם שֶׁהִנִּיחַ אֶת הַתְּפִלִּין, הוּא הִרְגִּישׁ עוֹד וְעוֹד הִתְעַלּוּת רוּחָנִית גְּדוֹלָה.

יוֹם אֶחָד, בְּאֶמְצַע הַיּוֹם, כְּשֶׁהָאִישׁ הָעָשִׁיר עוֹד הָיָה עָטוּר בַּתְּפִלִּין עַל רֹאשׁוֹ וְעַל יָדוֹ, הִגִּיעַ אֵלָיו אֶחָד הַסּוֹחֲרִים. הַסּוֹחֵר שָׁאַל אוֹתוֹ שְׁאֵלָה, וְהָאִישׁ שֶׁקָּנָה אֶת הַתְּפִלִּין לֹא שָׂם לֵב וְהִתְחִיל לְדַבֵּר אִתּוֹ עַל דִּבְרֵי חוֹל וְעַל דִּבְרֵי עֲבוֹדָה. אַחֲרֵי שֶׁסִּיֵּם לְדַבֵּר אִתּוֹ, הוּא הִרְגִּישׁ פִּתְאֹם מַשֶּׁהוּ מוּזָר מְאֹד. הוּא כְּבָר לֹא הִרְגִּישׁ אֶת אוֹתָהּ קְדֻשָּׁה וְטָהֳרָה שֶׁהִרְגִּישׁ בְּכָל יוֹם.

הוּא קָרָא מִיָּד לְסוֹפֵר סְתָ"ם וּבִקֵּשׁ מִמֶּנּוּ: "בֹּא בְּבַקָּשָׁה, בְּדֹק אֶת הַתְּפִלִּין שֶׁלִּי וּרְאֵה מָה קָרָה לָהֵן". הַסּוֹפֵר פָּתַח אֶת הַתְּפִלִּין וְנִדְהַם לִרְאוֹת: לֹא נִשְׁאֲרָה אַף אוֹת אַחַת עַל הַקְּלָף! כָּל הָאוֹתִיּוֹת פָּרְחוּ וְנֶעֶלְמוּ לַלְּחָלוּצִין.

אָז הֵבִין הָאִישׁ הָעָשִׁיר אֶת הַלֶּקַח הַגָּדוֹל. הוּא הֵבִין שֶׁבְּכָל פַּעַם שֶׁהוּא חוֹשֵׁב אוֹ מְדַבֵּר דְּבָרִים שֶׁאֵינָם שַׁיָּכִים לַקְּדֻשָּׁה וְלַתְּפִלָּה בִּזְמַן הֲנָחַת הַתְּפִלִּין, הַקְּדֻשָּׁה הַגְּדוֹלָה מִסְתַּלֶּקֶת מֵהֶן.
