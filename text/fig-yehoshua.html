רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה וּפִדְיוֹן הַיֶּלֶד מִבֵּית הַכֶּלֶא

הָרַב אִיסֵר יְהוּדָה אוּנְטֶרְמַן סִפֵּר סִפּוּר מְרַגֵּשׁ וּמֻפְלָא שֶׁמּוֹפִיעַ בַּגְּמָרָא. בַּסִּפּוּר הַזֶּה מְסֻפָּר עַל חֲכַם יִשְׂרָאֵל הַגָּדוֹל, רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה.

מַעֲשֶׂה בְּרַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה, שֶׁהִגִּיעַ בְּמַהֲלַךְ מַסָּעוֹ אֶל כְּרַךְ גָּדוֹל, כְּלוֹמַר אֶל עִיר אַחַת בְּרוֹמִי. כַּאֲשֶׁר הִגִּיעַ רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה אֶל תּוֹךְ הָעִיר הַזֹּאת בְּרוֹמִי, סִפְּרוּ לוֹ הָאֲנָשִׁים שָׁם מַעֲשֶׂה מְצַעֵר וּמְעוֹרֵר רַחֲמִים. הֵם סִפְּרוּ לוֹ שֶׁבְּתוֹךְ בֵּית הַכֶּלֶא הַמְּקוֹמִי כָּלוּא יֶלֶד יְהוּדִי יְפֵה תֹּאַר, שֶׁתַּלְתַּלָּיו וּפֵאוֹתָיו מְסֻדָּרוֹת יָפֶה מְאֹד כִּתְלַתְלִים. הַיֶּלֶד הַקָּטָן וְהַיָּפֶה הָיָה כָּלוּא שָׁם בְּתוֹךְ בֵּית הַכֶּלֶא בְּרוֹמִי, וְהַדָּבָר הַזֶּה נָגַע עֲמֹק בְּלִבּוֹ שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה.

כַּאֲשֶׁר שָׁמַע רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה אֶת הַדְּבָרִים הָאֵלֶּה, הוּא רָצָה מְאֹד לְבָרֵר מִי הוּא הַיֶּלֶד הַזֶּה שֶׁנִּמְצָא בְּתוֹךְ בֵּית הַכֶּלֶא, וּמָה טִיבוֹ. לָכֵן, הָלַךְ רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה אֶל בֵּית הַכֶּלֶא בָּעִיר, וְעָמַד מִחוּץ לַקִּיר הַפְּנִימִי שֶׁל בֵּית הַכֶּלֶא, בַּמָּקוֹם שֶׁבּוֹ הָיָה הַיֶּלֶד כָּלוּא. רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה רָצָה לָדַעַת כֵּיצַד יָגִיב הַיֶּלֶד הַיְּהוּדִי שֶׁנִּמְצָא בִּפְנִים, בְּתוֹךְ הַקִּיר הַפְּנִימִי שֶׁל בֵּית הַכֶּלֶא.

כְּדֵי לִבְדֹּק אֶת תְּגוּבָתוֹ וְאֶת חָכְמָתוֹ שֶׁל הַיֶּלֶד בְּתוֹךְ בֵּית הַכֶּלֶא, עָמַד רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה עַל יַד הַקִּיר, וְהִתְחִיל לְצַטֵּט בְּקוֹל פָּסוּק מִתּוֹךְ סֵפֶר יְשַׁעְיָהוּ. רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה אָמַר אֶת הַחֵלֶק הָרִאשׁוֹן שֶׁל הַפָּסוּק: "מִי נָתַן לִמְשִׁסָּה יַעֲקֹב וְיִשְׂרָאֵל לְבֹזְזִים?".

וְהִנֵּה, מִיָּד מִתּוֹךְ הַקִּיר הַפְּנִימִי שֶׁל בֵּית הַכֶּלֶא, עָנָה הַיֶּלֶד הַיְּהוּדִי וְהִשְׁלִים אֶת הַפָּסוּק מִסֵּפֶר יְשַׁעְיָהוּ בְּדִיּוּק רַב. הַיֶּלֶד אָמַר: "הֲלוֹא ה' זוּ חָטָאנוּ לוֹ וְלֹא אָבוּ בִדְרָכָיו הָלוֹךְ וְלֹא שָׁמְעוּ בְּתוֹרָתוֹ".

כַּאֲשֶׁר שָׁמַע רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה אֶת תְּשׁוּבָתוֹ הַחֲכָמָה שֶׁל הַיֶּלֶד, הוּא הִתְרַגֵּשׁ מְאֹד וְאָמַר בְּבִטָּחוֹן רַב: אֲנִי בָּטוּחַ שֶׁהַיֶּלֶד הַזֶּה יִהְיֶה מוֹרֵה הוֹרָאָה בְּיִשְׂרָאֵל!

רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה לֹא הִמְתִּין וְלֹא הִסְהֵס. הוּא לָקַח הַרְבֵּה מְאֹד כֶּסֶף, שִׁלֵּם אוֹתוֹ וּפָדָה אֶת הַיֶּלֶד מִבֵּית הַכֶּלֶא, כְּדֵי שֶׁיֵּצֵא לַחָפְשִׁי.

וּבֶאֱמַת, כְּפִי שֶׁחָזָה רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה, הַיֶּלֶד הַזֶּה גָּדַל וְהָפַךְ לִהְיוֹת הַתַּנָּא הַקָּדוֹשׁ רַבִּי יִשְׁמָעֵאל בֶּן אֱלִישָׁע, שֶׁהָיָה מוֹרֵה הוֹרָאָה גָּדוֹל, חָכָם וּמְפֻרְסָם בְּכָל עַם יִשְׂרָאֵל.

💡 כַּאֲשֶׁר אָנוּ לוֹמְדִים תּוֹרָה וְזוֹכְרִים אֶת פְּסוּקֶיהָ, הַחָכְמָה וְהָאֱמוּנָה נִשְׁאָרוֹת אִתָּנוּ בְּכָל מָקוֹם וּמְבִיאוֹת לִישׁוּעָה.
