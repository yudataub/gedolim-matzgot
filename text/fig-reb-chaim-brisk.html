רַבִּי חַיִּים מִבְּרִיסְק וּמִצְוַת פִּדְיוֹן שְׁבוּיִים

רַבִּי חַיִּים הַלֵּוִי סוֹלוֹבֵיְצִ'יק, הַמּُכָּר בְּשֵׁם "רַבִּי חַיִּים מִבְּרִיסְק", נוֹלַד בָּעִיר וָאלוֹזִ'ין. כְּבָר כְּשֶׁהָיָה צָעִיר מְאֹד, בֶּן עֶשְׂרִים וָשֶׁבַע בִּלְבַד, הוּא הָיָה מֵרָאשֵׁי יְשִׁיבַת וָאלוֹזִ'ין הַגְּדוֹלָה שֶׁבְּלִיטָא. בְּהֶמְשֵׁךְ חַיָּיו, הוּא נַעֲשָׂה הָרַב שֶׁל הָעִיר בְּרִיסְק. רַבִּי חַיִּים הָיָה גָּאוֹן עָצוּם בַּתּוֹרָה, וְהוּא הִמְצִיא דֶּרֶךְ לִמּוּד מְיֻחֶדֶת, עֲמֻקָּה וּמְדֻיֶּקֶת שֶׁל הַגְּמָרָא וְשֶׁל דִּבְרֵי הָרַמְבַּ"ם. דֶּרֶךְ זוֹ נִקְרֵאת "דֶּרֶךְ הַלִּמּוּד הַבְּרִיסְקָאִית", וְעַד הַיּוֹם תַּלְמִידִים רַבִּים בַּיְּשִׁיבוֹת לוֹמְדִים לְפִיהָ. בֵּין תַּלְמִידָיו הָיוּ חֲכָמִים גְּדוֹלִים כְּמוֹ רַבִּי בָּרוּךְ בֶּר לֵיבּוֹבִיץ, רַבִּי אִיסֶר זַלְמָן מֶלְצֶר, רַבִּי שִׁמְעוֹן שְׁקוֹפּ וְרַבִּי אֶלְחָנָן וַסֶרְמַן.

לְצַד הַגְּדוּלָה הָעֲצוּמָה שֶׁלּוֹ בַּתּוֹרָה, רַבִּי חַיִּים הָיָה גַּם אִישׁ חֶסֶד גָּדוֹל מְאֹד. בֵּיתוֹ הָיָה פָּתוּחַ לִרְוָחָה בְּכָל עֵת, וְכָל אָדָם עָנִי, חַלָּשׁ אוֹ מִסְכֵּן יָכֹל הָיָה לִכְנֹס אֵלָיו וּלְקַבֵּל עֶזְרָה, חֹם וְאַהֲבָה.

פַּעַם אַחַת, בְּלֵיל יוֹם הַכִּפּוּרִים, הִגִּיעָה הָעֵת לְהַתְחִיל אֶת הַתְּפִלָּה הַקְּדוֹשָׁה. כָּל אַנְשֵׁי הַקְּהִלָּה כְּבָר עָמְדוּ בְּבֵית הַכְּנֶסֶת, עֲטוּפִים בְּטַלִּיתוֹת לְבָנוֹת, וְחִכּוּ בְּרַעַד וּבְיִרְאָה לִתְפִלַּת "כָּל נִדְרֵי". וְהִנֵּה, בְּדִיּוּק בָּאוֹתוֹ רֶגַע קָדוֹשׁ, נוֹדְעָה בְּשׂוֹרָה מְבַהֶלֶת: יְהוּדִים נִתְפְּסוּ וְנִלְקְחוּ לַשְּׁבִי עַל יְדֵי הַשִּׁלְטוֹן שֶׁל הַצָּאר. כְּדֵי לְהַצִּיל אֶת הַשְּׁבוּיִים הָאֵלֶּה וּלְפַדּוֹת אוֹתָם, הָיָה צָרִיךְ לֶאֱסֹף מִיָּד סְכוּם כֶּסֶף גָּדוֹל מְאֹד.

מָה עָשָׂה רַבִּי חַיִּים? הוּא לֹא הִסֵּס לְרֶגַע. הוּא הוֹרָה לְכָל הַצִּבּוּר שֶׁלֹּא לְהַתְחִיל אֶת תְּפִלַּת "כָּל נִדְרֵי", עַד שֶׁיֶּאֶסְפוּ אֶת כָּל הַכֶּסֶף הַנִּדְרָשׁ מִן הַקָּהָל – דַּוְקָא עַכְשָׁו, בְּלֵיל יוֹם הַכִּפּוּרִים עַצְמוֹ! הָאֲנָשִׁים בְּבֵית הַכְּנֶסֶת הֵבִינוּ אֶת חֲשִׁיבוּת הַמִּצְוָה, מִהֲרוּ וְהֵבִיאוּ אֶת הַכֶּסֶף, וְהַיְּהוּדִים הַשְּׁבוּיִים נִפְדּוּ וְנִצְּלוּ. רַק לְאַחַר שֶׁהַשְּׁבוּיִים נִפְדּוּ בְּהַצְלָחָה, נָתַן רַבִּי חַיִּים אֶת הָאוֹת וְהַקָּהָל הִתְחִיל בִּתְפִלַּת "כָּל נִדְרֵי".

עוֹד מְסֻפָּר עַל רַבִּי חַיִּים, שֶׁהוּא הָיָה מֵקֵל מְאֹד בְּדִינֵי הַצּוֹם עֲבוּר אֲנָשִׁים חוֹלִים. כַּאֲשֶׁר אֲנָשִׁים תָּמְהוּ עַל כָּךְ וְאָמְרוּ לוֹ: "רַבִּי, כֵּיצַד אַתָּה מֵקֵל כָּל כָּךְ בְּדִינֵי הַצּוֹם שֶׁל יוֹם הַכִּפּוּרִים?", הֱשִׁיב לָהֶם רַבִּי חַיִּים בְּחָכְמָה: "אֲנִי כְּלָל אֵינִי מֵקֵל בְּדִינֵי יוֹם הַכִּפּוּרִים. אֲנִי פָּשׁוּט מַחְמִיר מְאֹד בְּמִצְוַת פִּקּוּחַ נֶפֶשׁ!".

רַבִּי חַיִּים לִמֵּד אוֹתָנוּ שֶׁמִּצְווֹת שֶׁל חֶסֶד, הַצָּלַת יְהוּדִים וּשְׁמִירָה עַל חַיִּים שֶׁל חוֹלִים, הֵן חֵלֶק בִּלְתִּי נִפְרָד מֵעֲבוֹדַת הַשֵּׁם, אֲפִלּוּ בַּיּוֹם הַקָּדוֹשׁ בְּיוֹתֵר בַּשָּׁנָה. מִי שֶׁהוּא גָּדוֹל אֲמִתִּי בַּתּוֹרָה, הוּא גַּם גָּדוֹל מְאֹד בְּחֶסֶד.
