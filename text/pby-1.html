הָאֶבֶן הַיְקָרָה וְהָעַכְבָּר

לִפְנֵי שָׁנִים רַבּוֹת, בְּמַמְלֶכֶת סִירְקְיָה הָרְחוֹקָה, אִי שֶׁהָיָה קַיָּם בַּחֲצִי הָאִי עֲרָב, חַי מֶלֶךְ עָשִׁיר מְאֹד. לַמֶּלֶךְ הַזֶּה הָיְתָה תֵּבָה מְיֻחֶדֶת בָּאַרְמוֹן, וּבְתוֹכָהּ הָיוּ מַטְבְּעוֹת רַבִּים, שַׁרְשְׁרוֹת נוֹצְצוֹת וְאַבְנִים מְיֻחָדוֹת מְאֹד. יוֹם אֶחָד, יָשַׁב הַמֶּלֶךְ בְּאַרְמוֹנוֹ, פָּתַח אֶת הַתֵּבָה וְהִסְתַּכֵּל בְּאֶבֶן מְיֻחֶדֶת מְאֹד שֶׁהָיְתָה לוֹ. בְּעוֹד הַמֶּלֶךְ מַבִּיט בָּאֶבֶן הַיְקָרָה, פִּתְאֹם נָפְלָה הָאֶבֶן מִיָּדוֹ, הִתְגַּלְגְּלָה עַל הָרִצְפָּה, וּבְאוֹתוֹ רֶגַע מַמָּשׁ בָּא עַכְבָּר, בָּלַע אֶת הָאֶבֶן הַזֹּאת וּבָרַח מְהֵרָה.

הַמֶּלֶךְ הִתְבַּלְבֵּל וְנִבְהַל מְאֹד, וּמִיָּד קָרָא לְכָל אַנְשֵׁי הָאַרְמוֹן וּבִקֵּשׁ מֵהֶם לְחַפֵּשׁ אֶת הָעַכְבָּר בְּכָל מָקוֹם. אַנְשֵׁי הָאַרְמוֹן חִפְּשׂוּ בְּכָל הַפִּנּוֹת, אַךְ הֵם לֹא מָצְאוּ אֶת הָעַכְבָּר וְלֹא מָצְאוּ אֶת הָאֶבֶן. הַמֶּלֶךְ הָיָה עָצוּב מְאֹד עַל אָבְדַן הָאֶבֶן הַיְקָרָה, הוּא יָשַׁב וּבָכָה מְרָרֹת, וּמֵרֹב צַעַר לֹא הִסְכִּים לֶאֱכֹל וְלֹא הִסְכִּים לִשְׁתּוֹת.

כְּשֶׁרָאָה זֹאת הַיּוֹעֵץ שֶׁל הַמֶּלֶךְ, נִגַּשׁ אֵלָיו וְאָמַר לוֹ שֶׁשָּׁמַע שֶׁלַּיְּהוּדִים יֵשׁ אֵיזֶה רַב גָּדוֹל שֶׁקּוֹרְאִים לוֹ רַבִּי פִּנְחָס בֶּן יָאִיר. הַיּוֹעֵץ הִצִּיעַ שֶׁיֵּלְכוּ אֵלָיו כְּדֵי שֶׁיְּדַבֵּר עִם הַשֵּׁם שֶׁלּוֹ, וְאוּלַי יַחְזִיר לוֹ אֶת הָאֶבֶן הַיְקָרָה. הַמֶּלֶךְ הִסְכִּים, וְהֵם הָלְכוּ אֶל רַבִּי פִּנְחָס בֶּן יָאִיר, אֲשֶׁר קִבֵּל אוֹתָם בְּסֵבֶר פָּנִים יָפוֹת וּבְשִׂמְחָה.

רַבִּי פִּנְחָס בֶּן יָאִיר שָׁאַל אֶת הַמֶּלֶךְ מָה הוּא עוֹשֶׂה שָׁם וְלָמָה בָּא. הַמֶּלֶךְ סִפֵּר לוֹ אֶת כָּל מָה שֶׁקָּרָה, וְשֶׁהָעַכְבָּר בָּלַע אֶת הָאֶבֶן הַיְקָרָה שֶׁלּוֹ. כְּשֶׁשָּׁמַע זאת רַבִּי פִּנְחָס בֶּן יָאִיר, שָׁאַל אוֹתוֹ בִּפְלִיאָה הַאִם הוּא חוֹשֵׁב שֶׁהוּא מְכַשֵּׁף שֶׁיָּכוֹל לַעֲשׂוֹת כִּשּׁוּף וְלָמָה בָּאתָ אֵלָיו. עָנָה הַמֶּלֶךְ וְאָמַר כִּי שָׁמַע שֶׁהוּא מִתְפַּלֵּל לֵאלֹהִים שֶׁלּוֹ, וְהוּא אָדָם טוֹב שֶׁעוֹזֵר לַאֲחֵרִים, וְהָאֱלֹהִים שֶׁלּוֹ שׁוֹמֵעַ אֶת תְּפִלּוֹתָיו.

כְּשֶׁשָּׁמַע זֹאת הַצַּדִּיק, רַבִּי פִּנְחָס בֶּן יָאִיר עָמַד בִּתְפִלָּה לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא וּבִקֵּשׁ מִמֶּנּוּ עֶזְרָה. לְאַחַר הַתְּפִלָּה, רַבִּי פִּנְחָס בֶּן יָאִיר צִוָּה עַל כָּל הָעַכְבָּרִים שֶׁנִּמְצָאִים בְּאוֹתוֹ אֵזוֹר לָבוֹא וְלַעֲבוֹר לְפָנָיו. וְהִנֵּה קָרָה פֶּלֶא, וְעַכְבָּרִים רַבִּים הֵחֵלּוּ לָצֵאת מֵהַחַלּוֹנוֹת, מֵהַדְּלָתוֹת וּמֵהַשְּׁבִילִים, וְעָבְרוּ כֻּלָּם לְפָנָיו.

רַבִּי פִּנְחָס בֶּן יָאִיר הִבִּיט בָּעַכְבָּרִים הָעוֹבְרִים, וּכְשֶׁעָבְרָה לְפָנָיו עַכְבָּרָה אַחַת שֶׁהָיְתָה קְצָת מְלֵאָה וּגְדוֹלָה, הוּא בִּקֵּשׁ מִמֶּנָּה לַעֲצוֹר וְלִפְלֹט אֶת הָאֶבֶן. הָעַכְבָּרָה עָמְדָה, פָּלְטָה אֶת הָאֶבֶן הַיְקָרָה מִפִּיהָ, וְהָאֶבֶן הֻחְזְרָה אֶל הַמֶּלֶךְ. הַמֶּלֶךְ שָׂמַח שִׂמְחָה גְּדוֹלָה, בֵּרַךְ אֶת רַבִּי פִּנְחָס בֶּן יָאִיר וְהוֹדָה לוֹ מְאֹד.
