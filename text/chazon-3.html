הַכָּרַת הַטּוֹב שֶׁל הַחֲזוֹן אִישׁ

הָיָה הָיָה מַעֲשֶׂה יָפֶה עַל הַחֲזוֹן אִישׁ, עַל אִשְׁתּוֹ, וְעַל אַחַת מִשְּׁכֵנוֹתָיו הַטּוֹבוֹת. הַחֲזוֹן אִישׁ הָיָה לוֹמֵד תּוֹרָה וְאִישׁ קָדוֹשׁ וְחָכָם. בְּבֵיתוֹ שֶׁל הַחֲזוֹן אִישׁ הָיְתָה אִשְׁתּוֹ, וּבַשְּׁכֵנוּת חָיְתָה אַחַת מִשְּׁכֵנוֹתָיו.

הַשְּׁכֵנָה הַזֹּאת הָיְתָה אִשָּׁה טוֹבָה וּמְסוּרָה מְאֹד. הִיא הָיְתָה דּוֹאֶגֶת וּמְסַיַּעַת לְאִשְׁתּוֹ שֶׁל הַחֲזוֹן אִישׁ מְאֹד מְאֹד. בְּכָל יוֹם וָיוֹם, כַּאֲשֶׁר אִשְׁתּוֹ שֶׁל הַחֲזוֹן אִישׁ הָיְתָה זְקוּקָה לְעֶזְרָה וּלְתְמִיכָה, אוֹתָהּ אִשָּׁה הָיְתָה בָּאָה, דּוֹאֶגֶת לָהּ, עֹזֶרֶת לָהּ וּמְסַיַּעַת לָהּ בְּכָל לִבָּהּ. הַדְּאָגָה וְהַסִּיּוּעַ שֶׁל הַשְּׁכֵנָה הָיוּ גְּדוֹלִים מְאֹד וְעָזְרוּ מְאֹד לְאִשְׁתּוֹ שֶׁל הַחֲזוֹן אִישׁ.

הַחֲזוֹן אִישׁ רָאָה אֶת כָּל הַדְּאָגָה וְאֶת כָּל הַסִּיּוּעַ שֶׁהַשְּׁכֵנָה מְסַיַּעַת לְאִשְׁתּוֹ. הַחֲזוֹן אִישׁ לֹא שָׁכַח אֶת הָעֶזְרָה הַזֹּאת. הוּא הִכִּיר לָהּ טוֹבָה גְּדוֹלָה וַעֲמֻקָּה מְאֹד עַל כָּל מַה שֶּׁעָשְׂתָה עֲבוּר אִשְׁתּוֹ. הַחֲזוֹן אִישׁ יָדַע כִּי כַּאֲשֶׁר אָדָם מְסַיֵּעַ וְדוֹאֵג לָנוּ אוֹ לִבְנֵי מִשְׁפַּחְתֵּנוּ, מְחוֹבָתֵנוּ לְהַכִּיר לוֹ טוֹבָה וּלְהוֹדוֹת לוֹ מִכָּל הַלֵּב.

הַחֲזוֹן אִישׁ הִרְהֵר בְּלִבּוֹ וְשָׁאַל אֶת עַצְמוֹ: כֵּיצַד יוּכַל לְהָשִׁיב לַשְּׁכֵנָה טוֹבָה עַל כָּל הַסִּיּוּעַ וְהַדְּאָגָה? הוּא רָצָה לְהֵיטִיב עִם אוֹתָהּ אִשָּׁה בַּדֶּרֶךְ הַטּוֹבָה בְּיוֹתֵר. הַחֲזוֹן אִישׁ הֵבִין כִּי הַדֶּרֶךְ הַנִּפְלָאָה בְּיוֹתֵר לְהֵיטִיב עִם אֵם שֶׁעָזְרָה וְסִיְּעָה, הִיא לְהֵיטִיב לָהּ עִם בְּנֵיהָ. כַּאֲשֶׁר עוֹשִׂים טוֹב עִם הַבָּנִים, הָאֵם מִתְמַלֵּאת שִׂמְחָה וְהַכָּרַת תּוֹדָה.

לְפִיכָךְ, הֶחְלִיט הַחֲזוֹן אִישׁ לַעֲשׂוֹת מַעֲשֶׂה גָּדוֹל. הוּא הֶחְלִיט לִלְמֹד תּוֹרָה עִם בְּנֵיהָ שֶׁל אוֹתָהּ אִשָּׁה. הַחֲזוֹן אִישׁ פִּנָּה מִזְּמַנּוֹ כְּדֵי לִלְמֹד תּוֹרָה מַמָּשׁ עִם הַבָּנִים שֶׁל הַשְּׁכֵנָה.

וְהַלִּמּוּד הַזֶּה לֹא נַעֲשָׂה פַּעַם אַחַת בִּלְבַד, אֶלָּא הָיָה קָבוּעַ וּמַתְמִיד. הַחֲזוֹן אִישׁ הָיָה לוֹמֵד תּוֹרָה עִם בְּנֵיהָ שֶׁל אוֹתָהּ אִשָּׁה מִדֵּי שַׁבָּת בְּשַׁבַּתּוֹ! בְּכָל יוֹם שַׁבָּת, שָׁבוּעַ אַחַר שָׁבוּעַ, הָיוּ בְּנֵיהָ שֶׁל הַשְּׁכֵנָה בָּאִים אֶל הַחֲזוֹן אִישׁ, וְהוּא הָיָה יוֹשֵׁב עִמָּהֶם וְלוֹמֵד אִתָּם תּוֹרָה בְּאַהֲבָה וּבְסַבְלָנוּת.

וְכַמָּה זְמַן נִמְשַׁךְ הַלִּמּוּד הַזֶּה? הַחֲזוֹן אִישׁ לָמַד תּוֹרָה עִם בְּנֵיהָ שֶׁל הַשְּׁכֵנָה בְּמֶשֶׁךְ שָׁלֹשׁ שָׁנִים תְּמִימֹת! שָׁלֹשׁ שָׁנִים שְׁלֵמוֹת, בְּכָל שַׁבָּת וְשַׁבָּת, לֹא וִתֵּר הַחֲזוֹן אִישׁ וְלִמֵּד אֶת הַבָּנִים תּוֹרָה. בְּכָךְ הֶרְאָה הַחֲזוֹן אִישׁ עַד כַּמָּה גְּדוֹלָה הַכָּרַת הַטּוֹב שֶׁלּוֹ לַשְּׁכֵנָה שֶׁדָּאֲגָה וְסִיְּעָה לְאִשְׁתּוֹ.

מֵהַסִּפּוּר הַזֶּה אָנוּ לוֹמְדִים מַהִי הַכָּרַת טוֹב אֲמִתִּית. כַּאֲשֶׁר מִישֶׁהוּ עוֹזֵר לָנוּ אוֹ מְסַיֵּעַ לִבְנֵי מִשְׁפַּחְתֵּנוּ, אָנוּ צְרִיכִים לִזְכֹּר אֶת הַחֶסֶד וְאֶת הַטּוֹבָה, וּלְחַפֵּשׂ דְּרָכִים לְהֵיטִיב עִמּוֹ וְעִם יְלָדָיו. הַחֲזוֹן אִישׁ הֶרְאָה לָנוּ כֵּיצַד מְשִׁיבִים טוֹבָה עַל דְּאָגָה וְסִיוּעַ בְּאַהֲבָה וּבְלִמּוּד תּוֹרָה.

💡 כַּאֲשֶׁר מִישֶׁהוּ עוֹזֵר לָנוּ, אָנוּ לוֹמְדִים מֵהַחֲזוֹן אִישׁ לְהַכִּיר לוֹ טוֹבָה וּלְהָשִׁיב לוֹ בְּמַעֲשִׂים טוֹבִים.
