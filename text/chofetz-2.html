הֶחָפֵץ חַיִּים – מַעֲשִׂים טוֹבִים מִנְעוּרָיו

בָּעַיָּרָה הָיְתָה אִשָּׁה מוֹכֶרֶת תַּפּוּחִים שֶׁהָיְתָה מִסְתּוֹבֶבֶת בַּשּׁוּק עִם סַל תַּפּוּחִים. בְּאוֹתָהּ עַיָּרָה גָּדַל יִשְׂרָאֵל מֵאִיר הַקָּטָן מִרָאדִין, שֶׁלִּפְעָמִים נִקְרָא גַּם רַבִּי יִשְׂרָאֵל מֵאִיר הַכֹּהֵן מִרָאדִין, וּלְאַחַר מִכֵּן נוֹדַע בְּשֵׁם "הֶחָפֵץ חַיִּים". לְיִשְׂרָאֵל מֵאִיר הַקָּטָן הָיוּ חֲבֵרִים שׁוֹבָבִים. הַחֲבֵרִים הַשּׁוֹבָבִים הָאֵלֶּה הָיוּ גּוֹרְמִים לְכַמָּה תַּפּוּחִים לִפּוֹל מִתּוֹךְ הַסַּל שֶׁל מוֹכֶרֶת הַתַּפּוּחִים, וְאָז מוֹכֶרֶת הַתַּפּוּחִים הָיְתָה רָצָה וְרָצָה אַחֲרֵי הַתַּפּוּחִים הַמִּתְגַּלְגְּלִים. מַרְאֶה זֶה הָיָה מַצְחִיק מְאֹד אֶת הַחֲבֵרִים הַשּׁוֹבָבִים. אֲבָל אֶת 'הֶחָפֵץ חַיִּים', אֶת יִשְׂרָאֵל מֵאִיר הַקָּטָן מִרָאדִין, זֶה לֹא הִצְחִיק בִּכְלָל. הוּא רָאָה אֵיךְ מוֹכֶרֶת הַתַּפּוּחִים רָצָה רָצָה אַחֲרֵי הַתַּפּוּחִים, וְגַם רָאָה שֶׁהַתַּפּוּחִים שֶׁנָּפְלוּ כְּבָר מִתְקַלְקְלִים, וְהִיא כְּבָר לֹא יָכְלָה לִמְכֹּר אוֹתָם לָאֲנָשִׁים. יִשְׂרָאֵל מֵאִיר הַקָּטָן חָשַׁב מָה לַעֲשׂוֹת כְּדֵי לַעֲזֹר לָהּ. מָה עָשָׂה יִשְׂרָאֵל מֵאִיר הַקָּטָן? הוּא בִּקֵּשׁ מֵאַבָּא שֶׁלּוֹ מַטְבֵּעַ, כִּי הוּא הָיָה אוֹמֵר שֶׁהוּא רוֹצֶה מַטְבֵּעַ כְּדֵי לִקְנוֹת תַּפּוּחַ. נָכוֹן, וְאָז הוּא הָיָה קוֹנֶה תַּפּוּחַ מִמּוֹכֶרֶת הַתַּפּוּחִים, אֲבָל הוּא לֹא בָּאֱמֶת הָיָה אוֹכֵל אוֹתוֹ. בִּמְקוֹם לֶאֱכֹל אֶת הַתַּפּוּחַ, הוּא הָיָה מַחֲזִיר אוֹתוֹ חֲזָרָה לְתוֹךְ הַסַּל שֶׁלָּהּ. וְכָכָה הוּא כְּאִלּוּ הָיָה גּוֹרֵם שֶׁלֹּא יִהְיֶה לָהּ הֶפְסֵד, שֶׁגַּם עַל הַתַּפּוּחִים שֶׁהִתְגַּלְגְּלוּ לָהּ – אָז בִּזְכוּת הַמַּטְבֵּעַ שֶׁלּוֹ לֹא יִהְיֶה לָהּ הֶפְסֵד כַּסְפִּי.

פַּעַם אַחַת, כַּאֲשֶׁר רַבִּי יִשְׂרָאֵל מֵאִיר מִרָאדִין הָיָה נַעַר בַּיְשִׁיבָה, אֵרַע מַעֲשֶׂה נוֹסָף בָּעַיָּרָה. בָּא אֵיזֶה מִישֶׁהוּ לַיְשִׁיבָה לְבַקֵּשׁ צְדָקָה עֲבוּר הַכְנָסַת כַּלָּה, וְכָל אֶחָד מֵאַנְשֵׁי הַמָּקוֹם וְהַתַּלְמִידִים תָּרַם מַה שֶּׁהָיָה יָכוֹל. מִישֶׁהוּ תָּרַם שְׁמִיכָה, וּמִישֶׁהוּ תָּרַם אֲרוֹנִית. מִישֶׁהוּ תָּרַם מְנוֹרָה, וּמִישֶׁהוּ תָּרַם סִפְרִיָּה יְשָׁנָה וּסְפָרִים יְשָׁנִים, כִּי זֹאת הָיְתָה עַיָּרָה מְאֹד יְשָׁנָה. 'הֶחָפֵץ חַיִּים' הָיָה נַעַר בַּיְשִׁיבָה, וְגַם הוּא רָצָה מְאֹד לָתֵת צְדָקָה, אֲבָל לֹא הָיָה לוֹ מַה לָּתֵת. אָז אַתֶּם יוֹדְעִים אֶת מָה הוּא נָתַן? הוּא נָתַן אֶת הַכָּרִית שֶׁלּוֹ שֶׁהוּא הָיָה יָשֵׁן עָלֶיהָ. הוּא נָתַן אֶת הַכָּרִית שֶׁלּוֹ לִצְדָקָה עֲבוּר הַכְנָסַת הַכַּלָּה. וּמָה עָשָׂה הַנַּעַר בַּלַּיְלָה כַּאֲשֶׁר הָלַךְ לִישׁוֹן? הוּא פָּשׁוּט גִּלְגֵּל אֶת הַמְּעִיל שֶׁלּוֹ וְיָשַׁן.

💡 תָּמִיד אֶפְשָׁר לִמְצֹא דֶּרֶךְ לַעֲזֹר לַאֲחֵרִים וְלָתֵת מֵהַלֵּב, גַּם כְּשֶׁזֶּה דּוֹרֵשׁ מֵאִתָּנוּ לְוַתֵּר עַל מַשֶּׁהוּ מִשֶּׁלָּנוּ.
