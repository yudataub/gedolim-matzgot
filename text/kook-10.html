מִשְׁקְלוֹת הַזְּכוֹכִית שֶׁל הָרַב קוּק

כַּאֲשֶׁר הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק זֵכֶר צַדִּיק לִבְרָכָה הָיָה עֲדַיִן בְּחוּץ לָאָרֶץ, הוּא נִתְמַנָּה לְכַהֵן כְּרַבָּהּ שֶׁל הַקְּהִילָה הַיְּהוּדִית בָּעִיר זַיְמֶל. בַּיָּמִים הָהֵם, בַּקְּהִילּוֹת הַיְּהוּדִיּוֹת שֶׁבְּאוֹתוֹ אֵזוֹר, הָיָה מִנְהָג מְקֻבָּל וּמוֹעִיל מְאֹד לְעֶזְרַת הָרַבָּנִים. הַקְּהִילָה הָיְתָה מַצִּיעָה לָרַב אֶת הַזִּכָּיוֹן הַבִּלְעָדִי לִמְכִירַת מֶלַח לְכָל תּוֹשָׁבֵי הַמָּקוֹם. בְּדֶרֶךְ זוֹ, מִמְּכִירַת הַמֶּלַח לְאַנְשֵׁי הָעִיר, יָכְלוּ הָרַבָּנִים לְהַרְוִיחַ חֵלֶק מֵהַכֶּסֶף הַדָּרוּשׁ לָהֶם לְהוֹצָאוֹת מִחְיָתָם וּלְכַלְכָּלַת בְּנֵי בֵּיתָם בְּכָבוֹד.

לְפִי הַמִּנְהָג הַזֶּה, הִצִּיעוּ גַּם לָרַב קוּק לְקַבֵּל אֶת הַזִּכָּיוֹן הַמְּיוּחָד לִמְכִירַת הַמֶּלַח בָּעִיר זַיְמֶל. הַצָּעָה זוֹ הָיְתָה יְכוֹלָה לַעֲזֹר לוֹ מְאֹד וּלְסַפֵּק לוֹ פַּרְנָסָה טוֹבָה וּבְטוּחָה. אוּלָם, לְהַפְתָּעַת כֻּלָּם, הָרַב קוּק סֵרֵב לְקַבֵּל אֶת הַזִּכָּיוֹן לִמְכִירַת הַמֶּלַח. הָרַב הִסְבִּיר אֶת סִבַּת סֵרוּבוֹ הַתַּקִיף: הוּא הוֹדִיעַ כִּי הוּא אֵינוֹ מוֹכָן לְקַבֵּל עָלָיו אֶת מְכִירַת הַמֶּלַח, כָּל עוֹד לֹא יַחְלִיפוּ אֶת כָּל מִשְׁקְלוֹת הַבַּרְזֶל וְאֶת מִשְׁקְלוֹת הָאֶבֶן שֶׁבָּהֶם מִשְׁתַּמְשִׁים לִשְׁקִילַת הַמֶּלַח, וְיַתְקִינוּ בִּמְקוֹמָם מִשְׁקְלוֹת מְיוּחָדוֹת הָעֲשׂוּיוֹת מִזְּכוֹכִית.

הָרַב קוּק הִסְבִּיר לְאַנְשֵׁי הַקְּהִילָה אֶת הַסִּבָּה הַמַּדַּעִית וְהַהֲלָכָתִית לִדְרִישָׁתוֹ הַקְּפֵדָנִית. הַמֶּלַח, בִּגְלַל הַחֲרִיפוּת וְהַחֻמְצִיּוּת הַטִּבְעִית שֶׁלּוֹ, פּוֹגֵעַ בְּמַהֲלַךְ הַזְּמַן בַּבַּרְזֶל וּבָאֶבֶן. הַמֶּלַח שוֹחֵק אֶת מִשְׁקְלוֹת הַבַּרְזֶל וְהָאֶבֶן, מְכַרְסֵם בָּהֶן וּמַפְחִית מִמִּשְׁקָלָן הַמְּקוֹרִי מְעַט מְעַט, בְּלִי שֶׁאָדָם יַרְגִּישׁ בְּכָךְ בְּנָקֵל. כַּאֲשֶׁר הַמִּשְׁקָלוֹת נִשְׁחָקוֹת וּמִתְמַעֲטוֹת, מִשְׁקָלָן הוֹפֵךְ לִהְיוֹת קַל יוֹתֵר מִמַּה שֶׁהוּא אָמוּר לִהְיוֹת, וְכָךְ עֲלוּלָה לְהִוָּצֵר טָעוּת בִּשְׁקִילַת הַמֶּלַח לַלָּקוֹחוֹת.

לְעֻמַּת זֹאת, הַזְּכוֹכִית הִיא חֹמֶר מְיוּחָד שֶׁאֵינוֹ מוּשְׁפָּע מֵהַחֲרִיפוּת שֶׁל הַמֶּלַח. הַזְּכוֹכִית אֵינָהּ נִפְחֶתֶת, אֵינָהּ נִשְׁחֶקֶת וְאֵינָהּ מְאַבֶּדֶת דָּבָר מִמִּשְׁקָלָהּ גַּם לְאַחַר שִׁמּוּשׁ מְמוּשָׁךְ בְּמַגָּע עִם מֶלַח. מִשְׁקְלוֹת הַזְּכוֹכִית נִשְׁאָרוֹת מְדֻיָּקוֹת לַחֲלוּטִין וּשְׁלֵמוֹת תָּמִיד. הָרַב קוּק רָצָה לִהְיוֹת בָּטוּחַ בְּמֵאָה אֲחוּזִים כִּי כָּל מְכִירָה וּמְכִירָה שֶׁל מֶלַח תִּהְיֶה יְשָׁרָה, הֲגוּנָה וּמְדוּיֶקֶת לְלֹא שׁוּם חֲשָׁשׁ שֶׁל הַפְחָתָה אוֹ טָעוּת בַּמִּשְׁקָל. הוּא הִתְעַקֵּשׁ עַל כָּךְ וְלֹא וִיתַּר, עַד שֶׁהוּחְלְפוּ כָּל הַמִּשְׁקְלוֹת לְמִשְׁקְלוֹת זְכוֹכִית מְדוּיָקוֹת, וְרַק אָז הִסְכִּים לְקַבֵּל אֶת הַזִּכָּיוֹן לִמְכִירַת הַמֶּלַח.
