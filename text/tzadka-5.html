הַקְּפִיצָה מֵהַחַלּוֹן לְמַעַן הַתּוֹרָה

הַרַב יְהוּדָה צְדָקָה זֵכֶר צַדִּיק לִבְרָכָה הָיָה חָכָם גָּדוֹל, מַנְהִיג רוּחָנִי וְרֹאשׁ יְשִׁיבָה אָהוּב. הַרַב יְהוּדָה צְדָקָה אָהַב אֶת הַתּוֹרָה הַקְּדוֹשָׁה בְּכָל מְאֹדוֹ, וְשָׁמַר עַל זְמַנֵּי הַלִּמּוּד שֶׁלּוֹ בְּדַיְקָנוֹת גְּדוֹלָה וּמוּפְלָאָה. בְּסֵפֶר "וְזֹאת לִיהוּדָה", הַמְּסַפֵּר עַל תּוֹלְדוֹת חַיָּיו, קוֹרוֹתָיו וּמִשְׁנָתוֹ שֶׁל מָרָן רֹאשׁ הַיְּשִׁיבָה, מוֹפִיעַ פֶּרֶק מְיֻחָד בְּמִינוֹ שֶׁכּוֹתַרְתּוֹ הִיא "חֹק וְלֹא יַעֲבֹר". פֶּרֶק זֶה מְתָאֵר אֶת הַדַּיְקָנוֹת הַמְּיוּחֶדֶת שֶׁל הַרַב בִּזְמַנֵּי הַשִּׁעוּרִים, וּמַסְבִּיר כֵּיצַד שָׁקַד עַל תַּלְמוּדוֹ וְלֹא וִתֵּר עַל שׁוּם שִׁעוּר תּוֹרָה קָבוּעַ.

כְּשֶׁהִגִּיעַ זְמַנּוֹ שֶׁל הַרַב יְהוּדָה צְדָקָה לָלֶכֶת וְלִלְמֹד אֶת שִׁעוּרָיו הַקְּבוּעִים בַּתּוֹרָה, לֹא מָנַע אוֹתוֹ שׁוּם דָּבָר שֶׁבָּעוֹלָם. הַרַב הֶאֱמִין שֶׁלִּמּוּד הַתּוֹרָה הוּא הַדָּבָר הַחָשׁוּב וְהַיָּקָר בְּיוֹתֵר, וְלָכֵן הִקְפִּיד מְאֹד לְהַגִּיעַ לְכָל שִׁעוּר וְשִׁעוּר בַּזְּמַן הַמְּדֻיָּק, בְּלִי לְאַחֵר אַף לֹא בְּדַקָּה אַחַת. הַדַּיְקָנוֹת שֶׁלּוֹ הָיְתָה יְדוּעָה לַכֹּל, וְכָל מְכִירָיו וְתַלְמִידָיו יָדְעוּ שֶׁכַּאֲשֶׁר מַגִּיעַ מוֹעֵד הַשִּׁעוּר, אֵין שׁוּם סִבָּה אוֹ מִכְשׁוֹל שֶׁיּוּכְלוּ לַעֲצֹר אֶת הַרַב מִלָּלֶכֶת וְלִלְמֹד תּוֹרָה.

פַּעַם אַחַת אֵרַע מִקְרֵה מְיֻחָד וּבִלְתִּי צָפוּי בְּבֵיתוֹ שֶׁל הַרַב יְהוּדָה צְדָקָה. מַנְעוּל דֶּלֶת הַבַּיִת הִתְקַלְקֵל פִּתְאֹם, וּבְנֵי הַבַּיִת גִּלּוּ שֶׁלֹּא הָיוּ יְכוֹלִים לִפְתֹּחַ אֶת הַדֶּלֶת. הֵם נִסּוּ וְנִסּוּ לְסוֹבֵב אֶת הַמַּנְעוּל שׁוּב וָשׁוּב, אַךְ הַמַּנְעוּל נִשְׁאַר תָּקוּעַ, וְהַדֶּלֶת הָיְתָה נְעוּלָה חָזָק. אִי אֶפְשָׁר הָיָה לִפְתֹּחַ אֶת הַדֶּלֶת, וְלֹא הָיְתָה שׁוּם דֶּרֶךְ רְגִילָה לָצֵאת מֵהַבַּיִת הַחוּצָה.

וְהִנֵּה, מַמָּשׁ בְּאוֹתָן דַּקּוֹת, הִגִּיעַ מוֹעֵד הַשִּׁעוּר הַקָּבוּעַ בַּתּוֹרָה שֶׁל הַרַב. הַרַב יְהוּדָה צְדָקָה רָאָה שֶׁהַזְּמַן לָלֶכֶת לַשִּׁעוּר הִגִּיעַ, וְהוּא חַיָּב לָצֵאת מִיָּד כְּדֵי לֹא לְאַחֵר. הַרַב לֹא הִסְתַּפֵּק בַּתֵּרוּץ שֶׁהַדֶּלֶת מְקֻלְקֶלֶת וְשֶׁאִי אֶפְשָׁר לִפְתֹּחַ אוֹתָהּ, וְלֹא הִמְתִין שֶׁמִּישֶׁו יָבֹא לְתַקֵּן אֶת הַמַּנְעוּל.

כְּשֶׁהִגִּיעַ מוֹעֵד הַשִּׁעוּר בַּתּוֹרָה, עָמַד הַרַב יְהוּדָה צְדָקָה בִּזְרִיזוּת גְּדוֹלָה. לַמְרֹאת שֶׁלֹּא הָיָה צָעִיר, הוּא קָפַץ מִן הַחַלּוֹן הַחוּצָה כְּעֵלֶם צָעִיר, מְלֵא מֶרֶץ, זְרִיזוּת וְאַהֲבַת תּוֹרָה. הַרַב קָפַץ מֵהַחַלּוֹן כְּדֵי לָרוּץ אֶל הַשִּׁעוּר וּלְהַגִּיעַ בַּזְּמַן הַמְּדֻיָּק, מִבְּלִי לְאַחֵר. הוּא לֹא נָתַן לַדֶּלֶת הַמְּקֻלְקֶלֶת אוֹ לְשׁוּם מִכְשׁוֹל אַחֵר לִמְנֹעַ מִמֶּנּוּ לָלֶכֶת וְלִלְמֹד תּוֹרָה.

מַעֲשֶׂה מֻפְלָא זֶה מַרְאֶה לָנוּ עַד כַּמָּה הָיְתָה הַתּוֹרָה יְקָרָה וַחֲשׁוּבָה לָרַב יְהוּדָה צְדָקָה, וְכַמָּה שִׂמְחָה הָיְתָה לוֹ בְּכָל רֶגַע שֶׁל לִמּוּד. הַרַב לִמֵּד אוֹתָנוּ שֶׁכַּאֲשֶׁר מַגִּיעַ זְמַן לִמּוּד הַתּוֹרָה, עֹשִׂים כָּל מַאֲמָץ כְּדֵי לֹא לְאַחֵר, בִּזְרִיזוּת, בְּאַהֲבָה וּבְשִׂמְחָה גְּדוֹלָה.

💡 כְּשֶׁאָנוּ אוֹהֲבִים אֶת הַתּוֹרָה וְאֶת הַמִּצְווֹת, אָנוּ עוֹשִׂים כָּל מַאֲמָץ לְהַגִּיעַ בַּזְּמַן וְלֹא נוֹתְנִים לְשׁוּם מִכְשׁוֹל לַעֲצֹר אוֹתָנוּ.
