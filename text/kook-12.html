הָרַב קוּק וְהַמַּפְתֵּחַ שֶׁנִּשְׁכַּח

הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק הָיָה רַב גָּדוֹל וְחָשׁוּב מְאֹד. הוּא הָיָה הָרַב הָרָאשִׁי הָרִאשׁוֹן לְיִשְׂרָאֵל, וְהוּא גַּם זֶה שֶׁהֵקִים אֶת הָרַבָּנוּת הָרָאשִׁית לְיִשְׂרָאֵל. בְּאוֹתָם יָמִים, הָרַב קוּק הָיָה נוֹהֵג לָקוּם מֻקְדָּם מְאֹד בַּבֹּקֶר. בְּכָל בֹּקֶר הוּא הָלַךְ לְבֵית הַכְּנֶסֶת, הִתְפַּלֵּל בְּכַוָּנָה, וְלָמַד תּוֹרָה מִכָּל הַלֵּב. הָרַב קוּק גַּם כָּתַב סְפָרִים רַבִּים, וְהָיוּ לוֹ תַּלְמִידִים רַבִּים שֶׁלָּמְדוּ מִמֶּנּוּ תּוֹרָה בְּאַהֲבָה.

בְּמִסְגֶּרֶת תַּפְקִידוֹ כְּרַב גָּדוֹל, הָרַב קוּק הָיָה רַב מְאֹד עָסוּק. הוּא הָיָה נִפְגָּשׁ עִם אֲנָשִׁים חֲשׁוּבִים, וְגַם הָיָה עוֹבֵר בֵּין כָּל מִינֵי מְקוֹמוֹת שׁוֹנִים בְּאֶרֶץ יִשְׂרָאֵל כְּדֵי לְחַזֵּק אֶת הַמִּתְיַשְּׁבִים. הוּא עָשָׂה מַעֲשִׂים רַבִּים וַחֲשׁוּבִים מִבֹּקֶר עַד עֶרֶב, וְהַיָּמִים שֶׁלּוֹ הָיוּ מְלֵאִים בַּעֲבוֹדָה קָשָׁה וּבַעֲשִׂיָּה רַבָּה לְמַעַן הַכְּלָל.

וְהִנֵּה, יוֹם אֶחָד, אַחֲרֵי כָּל הַיּוֹם הָאֲרֹךְ וְהָעָמוּס הַזֶּה, הָרַב קוּק חָזַר אֶל בֵּיתוֹ. כְּשֶׁהִגִּיעַ לְיַד הַדֶּלֶת, הוּא מִשֵּׁשׁ בַּכִּיסִים שֶׁלּוֹ כְּדֵי לִמְצֹא אֶת הַמַּפְתֵּחַ, וּפִתְאוֹם הוּא רוֹאֶה וּמְגַלֶּה שֶׁהוּא שָׁכַח אֶת הַמַּפְתֵּחַ! הוּא בָּדַק שׁוּב וָשׁוּב בְּכָל הַכִּיסִים שֶׁלּוֹ, אַךְ הַמַּפְתֵּחַ פָּשׁוּט לֹא הָיָה שָׁם.

מָה עוֹשֶׂה בֶּן אָדָם שֶׁשּׁוֹכֵחַ מַפְתֵּחַ? בְּדֶרֶךְ כְּלָל, כְּשֶׁאָדָם שׁוֹכֵחַ אֶת הַמַּפְתֵּחַ שֶׁלּוֹ, הוּא נוֹקֵשׁ בַּדֶּלֶת כְּדֵי שֶׁיִּפְתְּחוּ לוֹ, אוֹ מְנַסֶּה לְהִכָּנֵס הַבַּיְתָה דֶּרֶךְ הַחַלּוֹן. אֲבָל הָרַב קוּק לֹא נָקַשׁ בַּדֶּלֶת וְגַם לֹא נִכְנַס דֶּרֶךְ הַחַלּוֹן.

לָמָה הָרַב קוּק לֹא נָקַשׁ בַּדֶּלֶת וְלֹא נִכְנַס דֶּרֶךְ הַחַלּוֹן? מִפְּנֵי שֶׁהוּא לֹא רָצָה לְהָעִיר אֶת אִשְׁתּוֹ הָרַבָּנִית! אִשְׁתּוֹ הָרַבָּנִית עָבְדָה כָּל הַיּוֹם קָשֶׁה מְאֹד בַּבַּיִת עִם הַיְּלָדִים. כְּשֶׁבָּאוּ אֲנָשִׁים וְאוֹרְחִים אֶל הַבַּיִת, הִיא הָיְתָה צְרִיכָה שֶׁהַבַּיִת יִהְיֶה תָּמִיד נָקִי וּמְסֻדָּר, וְהִיא הָיְתָה מַכְנִיסָה אֶת הָאוֹרְחִים וּמַגִּישָׁה לָהֶם מַשֶׁהוּ לִשְׁתּוֹת. הָרַבָּנִית עָבְדָה קָשֶׁה מְאֹד בְּמֶשֶׁךְ כָּל הַיּוֹם, וְלָכֵן הָרַב קוּק בְּשׁוּם פָּנִים וָאֹפֶן לֹא הִסְכִּים לְהָעִיר אוֹתָהּ.

אָז מָה עָשָׂה הָרַב קוּק? הַאִם הוּא יָשַׁן בַּחוּץ? עַל הַמַּדְרֵגוֹת? כֵּן, בְּדִיּוּק כָּךְ! הָרַב קוּק פָּשׁוּט נִשְׁעַן עַל הַקִּיר וְיָשַׁן עַל הַמַּדְרֵגוֹת, זֶה הַכֹּל. הוּא הֶעֱדִיף לִישֹׁן בַּחוּץ עַל הַמַּדְרֵגוֹת מִחוּץ לַבַּיִת כְּדֵי שֶׁלֹּא לְהַפְרִיעַ לִמְנוּחָתָהּ שֶׁל אִשְׁתּוֹ הָרַבָּנִית.
