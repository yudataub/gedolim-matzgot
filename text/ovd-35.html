עֶזְרָה סוֹדִית מֵהָרַב עוֹבַדְיָה

לִפְנֵי שָׁנִים רַבּוֹת, חָיְתָה בָּעִיר אֶלְעָד אִשָּׁה אַלְמָנָה שֶׁבַּעְלָהּ נִפְטַר. הָאִמָּא נִשְׁאֲרָה לְבַדָּהּ עִם יְלָדֶיהָ הַיְתוֹמִים, וְלֹא הָיָה לָהֶם כֶּסֶף לִקְנוֹת אֹכֶל וְלַכַלְכֵּל אֶת הַבַּיִת. הַיְתוֹמִים הַקְּטַנִּים הָיוּ רְעֵבִים, וְהָאִמָּא הָיְתָה עֲצוּבָה וְדוֹאֶגֶת מְאֹד.

בְּצָרָתָהּ הַגְּדוֹלָה, הֶחְלִיטָה הָאִשָּׁה לִנְסֹעַ לִירוּשָׁלַיִם אֶל בֵּיתוֹ שֶׁל מָרָן הָרַב עוֹבַדְיָה יוֹסֵף זַצַּ"ל כְּדֵי לְבַקֵּשׁ עֶזְרָה. הִיא הִגִּיעָה לְבֵית הָרַב בְּשָׁעָה עֶשֶׂר בַּלַּיְלָה וְנָקְשָׁה עַל הַדֶּלֶת. אֶחָד מִבְּנֵי הַבַּיִת פָּתַח אֶת הַדֶּלֶת וְאָמַר לָהּ שֶׁהָרַב לוֹמֵד עַכְשָׁו תּוֹרָה וְאֵין קַבָּלַת קָהָל בַּלַּיְלָה, וְשָׁאַל אוֹתָהּ לָבוֹא מָחָר. הָאִשָּׁה אָמְרָה לָאִישׁ: "בְּבַקָּשָׁה תַּגִּיד לָרַב שֶׁאֲנִי אִשָּׁה אַלְמָנָה. אִם הָרַב יַגִּיד לִי לָבוֹא מָחָר, אֲנִי אָבוֹא מָחָר". הָאִישׁ נִכְנַס אֶל הָרַב עוֹבַדְיָה וְסִפֵּר לוֹ שֶׁמְּחַכָּה בַּחוּץ אַלְמָנָה מֵאֶלְעָד. כְּשֶׁשָּׁמַע הָרַב מָרָן שֶׁמְּדוּבָּר בְּאַלְמָנָה, אָמַר מִיָּד: "הִיא אַלְמָנָה, תַּכְנִיסוּ אוֹתָהּ מִיָּד!".

הָאַלְמָנָה נִכְנְסָה אֶל הָרַב, פָּרְצָה בִּבְכִי וְסִפֵּרָה עַל בַּעְלָהּ שֶׁנִּפְטַר וְעַל הַיְתוֹמִים הַקְּטַנִּים שֶׁאֵין לָהֶם מָה לֶאֱכֹל. הָרַב עוֹבַדְיָה יָרַד דְּמָעוֹת מֵרַחֲמִים עַל צָרָתָהּ. הָאִשָּׁה הִסְבִּירָה שֶׁהִיא בָּאָה לְבַקֵּשׁ עֶזְרָה מַמָּשִׁית, אוּלַי הָרַב יַמְלִיץ עָלֶיהָ לְאַרְגּוֹנֵי חֶסֶד. אָמַר לָהּ הָרַב עוֹבַדְיָה: "הָסִירִי דְּאָגָה מִלִּבֵּךְ, אֲנִי אֶעֱזֹר לָךְ בְּעַצְמִי בְּכָל חֹדֶשׁ. אֲבָל אֲנִי מְבַקֵּשׁ שֶׁאַף אֶחָד לֹא יֵדַע עַל כָּךְ. תָּבוֹאִי לְפֹה כָּל חֹדֶשׁ וְתִקְבְּלִי מִמֶּנִּי מַעֲטָפָה, וְרַק אֲנִי וְאַתְּ נֵדַע מִזֶּה".

וְכָךְ הָיָה בְּמֶשֶׁךְ שֶׁבַע שָׁנִים! הָאִשָּׁה הָיְתָה מַגִּיעָה פַּעַם בְּחֹדֶשׁ, מְקַבֶּלֶת מֵהָרַב מַעֲטָפָה וּבָהּ כֶּסֶף מֵהַמַּשְׂכֹּרֶת שֶׁלּוֹ, וְאַף אֶחָד בַּבַּיִת לֹא יָדַע לָמָּה הִיא בָּאָה.

כַּעֲבֹר שֶׁבַע שָׁנִים נִפְטַר הָרַב עוֹבַדְיָה זַצַּ"ל, וְהָאַלְמָנָה בָּכְתָה בְּכִי כָּפוּל – עַל אָבְדַן גְּדוֹל הַדּוֹר, וְעַל כָּךְ שֶׁאֵין מִי שֶׁיַּעֲזֹר לָהּ וּלִילָדֶיהָ. כַּעֲבֹר חֲמִשָּׁה חֳדָשִׁים, כְּשֶׁמַּצָּבָהּ הָיָה קָשֶׁה מְאֹד, נָסְעָה הָאִשָּׁה לְקִבְרוֹ שֶׁל הָרַב עוֹבַדְיָה בִּירוּשָׁלַיִם. הִיא בָּכְתָה שָׁם מֵעוֹמֵק הַלֵּב וְקָרְאָה אֶת כָּל סֵפֶר תְּהִילִים בִּדְמָעוֹת. בָּעֶרֶב חָזְרָה לְבֵיתָהּ עֲיֵפָה, וּלְפָנֵי שֶׁהִסְפִּיקָה לִשְׁתּוֹת מַיִם נָפְלָה עָלֶיהָ תַּרְדֵמָה.

בַּחֲלוֹם רָאֲתָה אֶת הָרַב עוֹבַדְיָה אוֹמֵר לָהּ: "בִּתִּי, לָמָּה תִּבְכִּי? כְּפִי שֶׁעָזַרְתִּי לָךְ בְּחַיַּי, אֲנִי אֶעֱזֹר לָךְ גַּם עַכְשָׁו. בְּיוֹם חֲמִישִׁי הַקָּרוֹב תִּתְקַיֵּם בְּרִית מִילָה בְּאוּלָם 'בֵּית יִשְׂרָאֵל' בִּירוּשָׁלַיִם. יִהְיֶה שָׁם יְהוּדִי בְּשֵׁם שָׁלוֹם סַעֲדוֹן. לְכִי אֵלָיו וְתַגִּידִי לוֹ שֶׁהָרַב עוֹבַדְיָה שָׁלַח אוֹתָךְ, וְהוּא יַעֲזֹר לָךְ".

הָאִשָּׁה הִתְעוֹרְרָה וְחָשְׁבָה שֶׁזֶּה רַק חֲלוֹם שָׁוְא. אַךְ מִיָּד נִרְדְּמָה שׁוּב וְחָלְמָה אֶת אוֹתוֹ חֲלוֹם מַמָּשׁ בַּפַּעַם הַשְּׁנִיָּה! בְּיוֹם חֲמִישִׁי נָסְעָה הָאַלְמָנָה לָאוּלָם בִּירוּשָׁלַיִם. הִיא פָּנְתָה לְאַבְרֵךְ שֶׁהָיָה שָׁם וְשָׁאֲלָה אִם נִמְצָא שָׁלוֹם סַעֲדוֹן. הָאַבְרֵךְ קָרָא לוֹ, וְהָאִשָּׁה אָמְרָה לוֹ: "אֲנִי אִשָּׁה אַלְמָנָה, וְהָרַב עוֹבַדְיָה יוֹסֵף שָׁלַח אוֹתִי אֵלֶיךָ כְּדֵי שֶׁתַּעֲזֹר לִי".

כְּשֶׁשָּׁמַע שָׁלוֹם סַעֲדוֹן אֶת דְּבָרֶיהָ, נָפַל וְהִתְעַלֵּף מֵרֹב הִתְרַגְּשׁוּת! כְּשֶׁהִתְעוֹרֵר סִפֵּר שֶׁגַּם אֵלָיו בָּא הָרַב עוֹבַדְיָה בַּחֲלוֹם בְּאוֹתוֹ שָׁבוּעַ, וּבִקֵּשׁ מִמֶּנּוּ לָבוֹא לַבְּרִית וְלַעֲזֹר לָאַלְמָנָה שֶׁתַּגִּיעַ. שָׁלוֹם שָׁאַל אוֹתָהּ כַּמָּה כֶּסֶף נָתַן לָהּ הָרַב מָרָן בְּכָל חֹדֶשׁ, וְכַאֲשֶׁר עָנְתָה שֶׁהָרַב נָתַן לָהּ אַרְבַּעַת אֲלָפִים שְׁקָלִים, הִבְטִיחַ לָהּ שָׁלוֹם: "מֵהַיּוֹם תְּקַבְּלִי מִמֶּנִּי בְּכָל חֹדֶשׁ אַרְבַּעַת אֲלָפִים שְׁקָלִים בְּשֵׁם הָרַב, וְעוֹד אַרְבַּעַת אֲלָפִים שְׁקָלִים מִמֶּנִּי!".
