וְשָׁבוּ בָנִים לִגְבוּלָם

הַסִּפּוּר שֶׁלְּפָנֵינוּ מְבֻסָּס עַל עֵדוּת אִישִׁית מְרַגֶּשֶׁת שֶׁל חָנָן פּוֹרָת, מִתּוֹךְ שִׁעוּר שֶׁנָּתַן עַל דְּמוּתָהּ הָאֲהוּבָה שֶׁל רָחֵל אִמֵּנוּ. כְּדֵי לְהָבִין אֶת גֹּדֶל הָרֶגַע, עָלֵינוּ לַחְזֹר מַעַט בַּזְּמַן אֶל מִלְחֶמֶת הָעַצְמָאוּת: בַּמִּלְחָמָה הַהִיא נָפַל כְּפַר עֶצְיוֹן בִּידֵי הַיַּרְדְּנִים, וַאֲבוֹתֵיהֶם שֶׁל הַמִּתְיַשְּׁבִים נֶהֶרְגוּ בַּקְרָב הָאַחֲרוֹן. בְּמֶשֶׁךְ תְּשַׁע-עֶשְׂרֵה שָׁנִים אֲרֻכּוֹת לֹא יָכְלוּ נִצּוֹלֵי גּוּשׁ עֶצְיוֹן וִילָדֵיהֶם לַחְזֹר אֶל בֵּית אֲבוֹתָם.

לְאַחַר מִלְחֶמֶת שֵׁשֶׁת הַיָּמִים, בִּשְׁנַת תשכ"ז, נִפְתְּחָה הַדֶּרֶךְ וְנִצּוֹלֵי גּוּשׁ עֶצְיוֹן יָכְלוּ לָשׁוּב וּלְהִתְיַשֵּׁב מֵחָדָשׁ בִּכְפַר עֶצְיוֹן. שְׁלֹשָׁה חֳדָשִׁים בִּלְבַד לְאַחַר הַמִּלְחָמָה, יָמִים סְפוּרִים מַמָּשׁ לִפְנֵי רֹאשׁ הַשָּׁנָה תשכ"ח, הִגִּיעַ הַיּוֹם הָרִאשׁוֹן שֶׁל הָעֲלִיָּה הַמְּחֻדֶּשֶׁת הַבַּיְתָה. חָנָן פּוֹרָת, יַחַד עִם קוֹמֶץ קָטָן שֶׁל חֲבֵרִים צְעִירִים, יָצְאוּ לַדֶּרֶךְ בְּהִתְרַגְּשׁוּת רַבָּה.

בַּדֶּרֶךְ אֶל הַכְּפָר, בְּכִבְרַת הָאָרֶץ שֶׁבְּדֶרֶךְ אֶפְרָת וּבֵית לֶחֶם, עָצְרוּ הַחֲבֵרִים הַצְּעִירִים לְיַד קֶבֶר רָחֵל אִמֵּנוּ. הֵם עָמְדוּ שָׁם בְּהִתְרוֹמְמוֹת רוּחַ וּבְיִרְאַת כָּבוֹד. אָז קָם אֶחָד מֵהַחֲבֵרִים, שֶׁאָבִיו נָפַל בַּקְרָב הָאַחֲרוֹן בִּגְוּשׁ עֶצְיוֹן, פָּתַח אֶת סֵפֶר הַתַּנָּךְ וְקָרָא אֶת הַפְּסוּקִים מִסֵּפֶר יִרְמְיָהוּ.

הוּא קָרָא בִּפְשָׁטוּת אֶת מַאֲמַר ה': 'כֹּה אָמַר ה' קוֹל בְּרָמָה נִשְׁמָע נְהִי בְּכִי תַמְרוּרִים, רָחֵל מְבַכָּה עַל בָּנֶיהָ מֵאָנָה לְהִנָּחֵם עַל בָּנֶיהָ כִּי אֵינֶנּוּ'. אֵלֶּה הַפְּסוּקִים הַמְּתָאֲרִים אֶת רָחֵל אִמֵּנוּ הַבּוֹכִיָּה עַל בָּנֶיהָ שֶׁיָּצְאוּ לַגָּלוּת. אָז קָרָא הַחָבֵר אֶת דִּבְרֵי הַהַבְטָחָה וְהַנֶּחָמָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא אֶל הָאֵם: 'כֹּה אָמַר ה' מִנְעִי קוֹלֵךְ מִבֶּכִי וְעֵינַיִךְ מִדִּמְעָה כִּי יֵשׁ שָׂכָר לִפְעֻלָּתֵךְ נְאֻם ה' וְשָׁבוּ מֵאֶרֶץ אוֹיֵב. וְיֵשׁ תִּקְוָה לְאַחֲרִיתֵךְ נְאֻם ה' וְשָׁבוּ בָנִים לִגְבוּלָם'.

חָנָן פּוֹרָת סִפֵּר שֶׁמֵּאָז הוּא לֹא יָכֹל לִשְׁכֹּחַ אֶת הָרֶגַע הַזֶּה בְּחַיָּיו. כְּשֶׁהַחָבֵר קָרָא אֶת הַפְּסוּקִים הָאֵלֶּה, חָשׁ חָנָן מַשֶּׁהוּ מְיֻחָד בְּיוֹתֵר, חֲוָיָה שֶׁאוּלַי מַרְגִּישִׁים רַק פַּעַם אַחַת בַּחַיִּים. הוּא הִרְגִּישׁ שֶׁהַפְּסוּקִים הַקְּדוֹשִׁים הָאֵלֶּה נִכְתָּבִים מַמָּשׁ עֲלֵיהֶם, כָּאן וְעַכְשָׁו! הֵם שָׁמְעוּ כִּבְיָכוֹל אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא כּוֹתֵב עֲלֵיהֶם וּמַלְהִיב אֶת לִבָּם.

חָנָן וְקוֹמֶץ הַחֲבֵרִים הֵבִינוּ שֶׁהֵם, הַחוֹזְרִים לִכְפַר עֶצְיוֹן לְאַחַר תְּשַׁע-עֶשְׂרֵה שָׁנִים, מְקַיְּמִים אֶת דְּבַר ה'. בַּעֲלִיָּתָם מֵחָדָשׁ הֵם מוֹחִים אֶת דִּמְעוֹתֶיהָ שֶׁל רָחֵל אִמֵּנוּ, שֶׁבָּכְתָה כָּל הַשָּׁנִים עַל בָּנֶיהָ, וּמְבַשְּׂרִים בְּשִׂמְחָה לָעוֹלָם כֻּלּוֹ: וְשָׁבוּ בָנִים לִגְבוּלָם.

💡 הַבְטָחוֹתָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בַּתַּנָּךְ תַּמִיד מִתְקַיְּמוֹת, וְהַבָּנִים חוֹזְרִים לְבֵיתָם.
