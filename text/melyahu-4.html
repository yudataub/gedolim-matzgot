הַתְּפִלָּה הַסּוֹדִית בָּאַרְנָק

הָרַב מָרְדְּכַי אֵלִיָּהוּ, זֵכֶר צַדִּיק לִבְרָכָה, הָיָה הָרַב הָרָאשִׁי לְיִשְׂרָאֵל וְהָרִאשׁוֹן לְצִיּוֹן. בְּמֶשֶׁךְ עֲשָׂרוֹת שָׁנִים, לְמַעְלָה מֵאַרְבָּעִים שָׁנָה, הוּא עָבַד כְּדַיָּן בְּבֵית הַדִּין וְשָׁפַט בֵּין אֲנָשִׁים. תַּפְקִידוֹ שֶׁל דַּיָּן הוּא לַעֲזֹר לַאֲנָשִׁים לִפְתֹּר מַחְלֹקוֹת וּלְהוֹרוֹת לָהֶם אֶת הַדֶּרֶךְ הַנְּכוֹנָה עַל פִּי הַתּוֹרָה. בְּכָל שָׁנִים אֵלּוּ, הָרַב מָרְדְּכַי אֵלִיָּהוּ הִתְפַּלֵּל תָּמִיד תָּמִיד אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא. הוּא יָדַע שֶׁלִּהְיוֹת דַּיָּן זוֹ אַחְרָיוּת גְּדוֹלָה מְאֹד, וְשֶׁהוּא זָקוּק לְעֶזְרַת ה' כְּדֵי לֹא לִטְעוֹת.

כַּאֲשֶׁר הָרַב מָרְדְּכַי אֵלִיָּהוּ נִפְטַר, אֲנָשִׁים פָּתְחוּ אֶת הָאַרְנָק הָאִישִׁי שֶׁלּוֹ. בְּתוֹךְ הָאַרְנָק הֵם גִּלּוּ דָּבָר מַפְתִיעַ וּמְרַגֵּשׁ: פֶּתֶק קָטָן וּבוֹ תְּפִלָּה מְיוּחֶדֶת שֶׁהָרַב הָיָה מִתְפַּלֵּל בְּכָל יוֹם. בַּתְּפִלָּה הַזּוֹ כָּתַב הָרַב: רִבּוֹנוֹ שֶׁל עוֹלָם, גָּלוּי וְיָדוּעַ לְפָנֶיךָ שֶׁלֹּא עַל דַּעְתִּי עָמַדְתִּי לְשָׁרֵת עַם קָדְשְׁךָ, לָדִון וּלְהוֹרוֹת. הָרַב הִסְבִּיר בִּדְבָרָיו שֶׁהוּא מַרְגִּישׁ יִרְאָה וּרְעָדָה גְּדוֹלָה עַל הַסַּכָּנָה הַנּוֹרָאָה שֶׁל טָעוּת בַּדִּין, וְהוּא שָׁאַל מִתּוֹךְ עֲנָוָה: אָנָה אֶפְנֶה לְעֶזְרָה, וְאָנָה אֶבְרַח? וְאָז הוּא עָנָה מִתּוֹךְ אֱמוּנָה חֲזָקָה: אֲבָל בָּטַחְתִּי בְּרֹב חַסְדְּךָ, וּמַפִּיל תְּחִנָּתִי לְפָנֶיךָ, כִּי אַתָּה שׁוֹמֵעַ תְּפִלָּה, וְאַתָּה תַּצִילֵנוּ מִכָּל שְׁגִיאוֹת וְטָעוּיוֹת. תְּפִלָּה זוֹ הֶרְאֲתָה אֶת מִדַּת הָעֲנָוָה הָעֲמֻקָּה שֶׁל הָרַב וְאֶת הַבִּטָּחוֹן הַמָּלֵא שֶׁלּוֹ בַּה'.

בְּסִרְטוֹן שֶׁבּוֹ הָרַב מָרְדְּכַי אֵלִיָּהוּ עַצְמוֹ מְסַפֵּר, הוּא נִזְכָּר בְּמִקְרֶה שֶׁאֵרַע לוֹ כְּשֶׁהִגִּיעַ לְבֵית הַדִּין. כַּאֲשֶׁר נִכְנַס אֶל הַחֶדֶר, הוּא רָאָה שֶׁהַדַּיָּנִים הָאַחֲרִים מְדַבְּרִים בֵּינֵיהֶם בְּשָׂפָה אַחֶרֶת – בְּשָׂפַת יִידִישׁ. הָרַב מָרְדְּכַי אֵלִיָּהוּ לֹא יָדַע יִידִישׁ וְלֹא הֵבִין אֶת הַשָּׂפָה הַזּוֹ. הוּא הִרְגִּישׁ קֹשִׁי גָּדוֹל, כִּי כֵּיצַד יוּכַל לָדִין וּלְהָבִין אֶת הַדְּבָרִים? בִּמְקוֹם לְהִתְיָאֵשׁ, הָרַב פָּשׁוּט עָמַד שָׁם וְהִתְפַּלֵּל מֵעֹמֶק הַלֵּב אֶל ה': אָנָה ה', תַּעֲזֹר לִי! הוּא פָּנָה אֶל ה' בְּמִלִּים פְּשׁוּטוֹת וְכֵנוֹת. וּבִזְכוּת הַתְּפִלָּה הַזּוֹ, רוֹאִים מַמָּשׁ אֵיךְ הוּא הִצְלִיחַ לְכַוֵּן לָאֱמֶת, זָכָה לְכָךְ וְהִצְלִיחַ בְּתַפְקִידוֹ.

הַסִּפּוּר מַזְכִּיר גַּם אֶת דִּבְרֵי רַבִּי יִשְׁמָעֵאל בְּנוֹ שֶׁל רַבִּי יוֹסֵי, הַמּוֹפִיעִים בְּמִשְׁנָיוֹת פִּרְקֵי אָבוֹת, הַמְּלַמְּדִים עַל הַנְהָגַת הַדַּיָּנִים וְעַל הַחֲשִׁיבוּת שֶׁל יִרְאַת שָׁמַיִם וְזִהוּר בַּדִּין. הָרַב מָרְדְּכַי אֵלִיָּהוּ שָׁמַר אֶת הַתְּפִלָּה בְּאַרְנָקוֹ בְּכָל עֵת, כְּדֵי לִזְכֹּר תָּמִיד שֶׁהָאֱמֶת וְהַהַצְלָחָה מַגִּיעוֹת מֵאֵאת ה' בִּזְכוּת הַתְּפִלָּה.

💡 כַּאֲשֶׁר אָנוּ עוֹמְדִים בִּפְנֵי קֹשִׁי אוֹ מְשִׁימָה חֲשׁוּבָה, תְּפִלָּה כֵּנָה מֵהַלֵּב אֶל ה' עוֹזֶרֶת לָנוּ לִמְצֹא אֶת הַדֶּרֶךְ הַנְּכוֹנָה וּלְהַצְלִיחַ.
