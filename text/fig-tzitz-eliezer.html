הָרַב שֶׁל בֵּית הַחוֹלִים

הָרַב אֱלִיעֶזֶר יְהוּדָה וַלְדֶנְבֶּרְג נוֹלַד בִּירוּשָׁלַיִם. כְּשֶׁהָיָה יֶלֶד וְנַעַר, הוּא לָמַד תּוֹרָה בִּישִׁיבַת "עֵץ חַיִּים" וּבִישִׁיבַת "חֶבְרוֹן". הוּא אָהַב לִלְמֹד תּוֹרָה בִּשְׁקִידָה רַבָּה, וּכְבָר כְּשֶׁהָיָה בָּחוּר צָעִיר בֶּן תְּשַׁע עֶשְׂרֵה שָׁנִים, הוּא זָכָה לִכְתֹּב וּלְהוֹצִיא לָאוֹר אֶת סִפְרוֹ הָרִאשׁוֹן שֶׁנִּקְרָא "דְּבַר אֱלִיעֶזֶר". בְּמֶשֶׁךְ חַיָּיו הַמְבֹרָכִים הוּא כִּהֵן כְּדַיָּן, וְהָיָה אַב בֵּית דִּין בִּירוּשָׁלַיִם וְגַם חָבֵר בְּבֵית הַדִּין הָרַבָּנִי הַגָּדוֹל. הוּא חִבֵּר עֶשְׂרִים וּשְׁתַּיִם כְּרָכִים שֶׁל שְׁאֵלוֹת וּתְשׁוּבוֹת שֶׁנִּקְרְאוּ "צִיץ אֱלִיעֶזֶר", וּבִזְכוּת חָכְמָתוֹ הַתּוֹרָנִית הָרַבָּה הוּא זָכָה בִּפְרָס יִשְׂרָאֵל לְסִפְרוּת תּוֹרָנִית וְגַם בִּפְרָס הָרַב קוּק.

בְּבֵית הַחוֹלִים "שַׁעֲרֵי צֶדֶק" אֲשֶׁר בִּירוּשָׁלַיִם עָבְדוּ רוֹפְאִים יִרְאֵי שָׁמַיִם. בְּכָל יוֹם וָיוֹם פָּגְשׁוּ הָרוֹפְאִים הָאֵלֶּה שְׁאֵלוֹת חֲדָשׁוֹת וְחֲשׁוּבוֹת מְאֹד בַּעֲבוֹדָתָם. הֵם שָׁאֲלוּ אֶת עַצְמָם: אֵיךְ מְטַפְּלִים בְּחוֹלֶה בְּיוֹם שַׁבָּת? מָה מֻתָּר לַעֲשׂוֹת כְּדֵי לְהַצִּיל חַיִּים שֶׁל בֶּן אָדָם? וְאֵיךְ מִשְׁתַּמְּשִׁים בְּטֶכְנוֹלוֹגִיּוֹת חֲדָשׁוֹת בִּרְפוּאָה – עַל פִּי הַתּוֹרָה הַקְּדוֹשָׁה?

לִשְׁאֵלוֹת מֻרְכָּבוֹת כָּאֵלֶה לֹא תָּמִיד הָיוּ תְּשׁוּבוֹת כְּתוּבוֹת מְפֹרָשׁוֹת בַּסְּפָרִים הַיְשָׁנִים, כִּי חֵלֶק מֵהַמַּכְשִׁירִים הָרְפוּאִיִּים וּמֵהַטִּפּוּלִים הַחֲדָשִׁים כְּלָל לֹא הָיוּ קַיָּמִים בִּימֵי חַכְמֵי הָרִאשׁוֹנִים! אָז לְאָן פָּנוּ הָרוֹפְאִים כְּדֵי לְקַבֵּל הַדְרָכָה וּתְשׁוּבָה? הֵם פָּנוּ אֶל הָרַב אֱלִיעֶזֶר יְהוּדָה וַלְדֶנְבֶּרְג. בְּשֶׁל עֶזְרָתוֹ הָרַבָּה וִידִיעוֹתָיו הָרְחָבוֹת, הוּא הָיָה עֲבוּרָם מַמָּשׁ כְּמוֹ "הָרַב שֶׁל בֵּית הַחוֹלִים".

הָרַב וַלְדֶנְבֶּרְג הָיָה בָּקִי מֻפְלָא בְּכָל חֶלְקֵי הַתּוֹרָה, אַךְ הוּא לֹא הִסְתַּפֵּק רַק בְּכָךְ: הוּא גַּם לָמַד וְהֵבִין בְּעוֹמֶק אֶת חָכְמַת הָרְפוּאָה עַצְמָהּ, שָׁאַל אֶת הָרוֹפְאִים שְׁאֵלוֹת מְפֹרָטוֹת כְּדֵי לְהָבִין אֶת אֹפֶן הַטִּפּוּל, וְחִפֵּשׂ בַּגְּמָרָא, בְּדִבְרֵי הָרִאשׁוֹנִים וּבְדִבְרֵי הָאַחֲרוֹנִים אֶת הַיְסוֹדוֹת שֶׁל כָּל שְׁאֵלָה וּשְׁאֵלָה.

כָּךְ כָּתַב הָרַב תְּשׁוּבוֹת חֲלוּצִיּוֹת וַחֲשׁוּבוֹת בְּעִנְיְנֵי רְפוּאָה וַהֲלָכָה, וְגַם בְּעִנְיְנֵי מְדִינָה, בְּעֶשְׂרִים וּשְׁתַּיִם כְּרָכִים מְפֹאָרִים שֶׁל סִפְרוֹ "צִיץ אֱלִיעֶזֶר". רוֹפְאִים וְרַבָּנִים בְּכָל הָעוֹלָם כֻּלּוֹ לוֹמְדִים מֵהַתְּשׁוּבוֹת הָאֵלֶּה וּמִתְיַעֲצִים עִמָּהֶן עַד הַיּוֹם הַזֶּה.

💡 הַתּוֹרָה נוֹתֶנֶת תְּשׁוּבָה לְכָל דּוֹר וּלְכָל שְׁאֵלָה חֲדָשָׁה, וּמִי שֶׁלּוֹמֵד אוֹתָהּ לְעֹמֶק יָכוֹל לַעֲזֹר לְרוֹפְאִים לְהַצִּיל חַיִּים בְּדֶרֶךְ הַתּוֹרָה.
