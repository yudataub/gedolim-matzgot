הַבָּאבָּא סָאלִי וְהַיָּם הַסּוֹעֵר

הַסִּפּוֹר הַמֻּפְלָא שֶׁאֲנִי רוֹצֶה לְסַפֵּר לָכֶם כַּיּוֹם הוּא עַל הַבָּאבָּא סָאלִי, הֲלֹא הוּא הַצַּדִּיק רַבִּי יִשְׂרָאֵל אֲבוּחַצִירָא. זֶהוּ סִפּוֹר מֻפְלָא מְאֹד שֶׁקָּרָה לוֹ בְּאֶמְצַע הַיָּם הַגָּדוֹל.

הַבָּאבָּא סָאלִי הָיָה גָּר בָּעִיר תָּפִילָאלְתְּ שֶׁבְּמָרוֹקוֹ. הוּא אָהַב מְאֹד אֶת הַתּוֹרָה וְאֶת אֶרֶץ יִשְׂרָאֵל. יוֹם אֶחָד הוּא רָצָה מְאֹד לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, כְּדֵי לִבְדֹּק אִם אֶפְשָׁר לַעֲלוֹת לְשָׁמָּה, וְאִם יֵשׁ בְּאֶרֶץ יִשְׂרָאֵל מַסְפִּיק יְשִׁיבוֹת וּמֶרְכְּזֵי תּוֹרָה עֲבוּר לוֹמְדֵי הַתּוֹרָה.

כְּדֵי לָצֵאת לַדֶּרֶךְ הָאֲרֻכָּה, הַבָּאבָּא סָאלִי לָקַח אִתּוֹ תִּיק קָטָן בִּלְבַד. בְּתוֹךְ הַתִּיק הַקָּטָן הוּא שָׂם אֶת הַחֲפָצִים הַחֲשׁוּבִים לוֹ בְּיוֹתֵר: טַלִּית, סִדּוּר תְּפִלָּה, וְאֶת גְּבִיעַ הַשַּׁבָּת הַמְּיֻחָד שֶׁלּוֹ, שֶׁאִתּוֹ הוּא הָיָה מְקַדֵּשׁ תָּמִיד עַל הַיַּיִן.

רַבִּי יִשְׂרָאֵל עָלָה עַל אוֹנִיָּה שֶׁהִפְלִיגָה בַּיָּם. כְּשֶׁעָלָה לָאוֹנִיָּה, הוּא יָרַד לַחֲדַר הֲכִי הֲכִי תַּחְתּוֹן, לְמַטָּה לְמַטָּה בְּתַחְתִּית הָאוֹנִיָּה. שָׁם בַּחֶדֶר הַתַּחְתּוֹן יָשַׁב רַבִּי יִשְׂרָאֵל בְּשַׁלְוָה וְלָמַד תּוֹרָה בִּשְׁקִידָה רַבָּה מְאֹד.

פִּתְאֹם הִתְחִילָה סְעָרָה גְּדוֹלָה וְחָזָקָה בַּיָּם. הַסְּעָרָה הָיְתָה חֲזָקָה חֲזָקָה, וְהַגַּלִּים הַגְּבוֹהִים עָלוּ וְיָרְדוּ וְעָלוּ וְיָרְדוּ לְלֹא הַפְסָקָה. הָאֲנָשִׁים שֶׁהָיוּ עַל הָאוֹנִיָּה נִבְהֲלוּ מְאֹד מִן הַסְּעָרָה הַקָּשָׁה. הֵם רָצוּ לְחַדְרוֹ שֶׁל הַבָּאבָּא סָאלִי, נָקְשׁוּ בַּדֶּלֶת וְקָרְאוּ לוֹ בְּדַאֲגָה: "רַבִּי יִשְׂרָאֵל, רַבִּי יִשְׂרָאֵל, תַּעֲזֹר לָנוּ, תִּתְפַּלֵּל!". אַךְ הַבָּאבָּא סָאלִי כְּלָל לֹא שָׁמַע אֶת הַנְּקִישׁוֹת עַל הַדֶּלֶת, מֵרֹב שֶׁהָיָה עָסוּק וְשׁוֹקֵעַ כֻּלּוֹ בְּלִמּוּד הַתּוֹרָה הַקְּדוֹשָׁה.

וְאָז, מָה שֶׁקָּרָה הוּא שֶׁהֵם נָקְשׁוּ עַל הַדֶּלֶת עוֹד יוֹתֵר חָזָק וְעוֹד יוֹתֵר חָזָק. מֵרֹב שֶׁהַנְּקִישׁוֹת הָיוּ חֲזָקוֹת מְאֹד, רַבִּי יִשְׂרָאֵל שָׁמַע אוֹתָן וּפָתַח לָהֶם אֶת הַדֶּלֶת. הוּא הֵבִין שֶׁהֵם זְקוּקִים לְעֶזְרָה, וְאָז הוּא עָלָה לְמַעְלָה, אֶל סִיפּוֹן הָאוֹנִיָּה, עִם הַטַּלִּית שֶׁלּוֹ וְעִם גְּבִיעַ הַשַּׁבָּת שֶׁלּוֹ, וּפָשׁוּט עָמַד שָׁם וְהִתְפַּלֵּל.

הַבָּאבָּא סָאלִי הִתְעַטֵּף בַּטַּלִּית שֶׁלּוֹ, לָקַח אֶת גְּבִיעַ הַשַּׁבָּת שֶׁלּוֹ, מִלֵּא אֹתוֹ מִמֵּי הַיָּם הַסּוֹעֲרִים, וְשָׁפַךְ אֶת הַמַּיִם שָׁלֹשׁ פְּעָמִים לְתוֹךְ הַיָּם. וּפִתְאֹם, רַבִּי יִשְׂרָאֵל עָמַד בִּתְפִלָּה וְהִתְפַּלֵּל, וְאָז הַיָּם שָׁקַט, וְהַגַּלִּים הַסּוֹעֲרִים פָּשׁוּט נֶעֶלְמוּ. בִּזְכוּת תְּפִלָּתוֹ הַזַּכָּה שֶׁל הַצַּדִּיק, הַסְּעָרָה פָּשׁוּט שָׁכְכָה.

💡 תְּפִלָּה זַכָּה שֶׁבָּאָה מִתּוֹךְ אֱמוּנָה פְּנוּמִית וְאַהֲבַת תּוֹרָה יְכוֹלָה לְבַטֵּל סְעָרוֹת וּלְהָבִיא שַׁלְוָה וִישׁוּעָה.
