הַנְּסִיעָה הָאִטִּית שֶׁל הֶחָזוֹן אִישׁ

הַגָּאוֹן הָרַב אַבְרָהָם יְשַׁעְיָהוּ קָרֶלִיץ זֵכֶר צַדִּיק לִבְרָכָה, שֶׁכֻּלָּם כִּנּוּ אוֹתוֹ בְּשֵׁם הַ'חָזוֹן אִישׁ', הָיָה מִגְּדוֹלֵי הַפּוֹסְקִים בְּאֶרֶץ יִשְׂרָאֵל לַאֲחַר הַשּׁוֹאָה. שְׁמוֹ נִזְכַּר בְּהַעֲרָצָה רַבָּה בְּפִי כָּל יְהוּדִי שׁוֹמֵר תּוֹרָה וּמִצְוֹת.

בְּאַחַד הַיָּמִים הָיָה הָרַב אָמוּר לָצֵאת מִבֵּיתוֹ כְּדֵי לְהִשְׁתַּתֵּף בַּחֲגִיגַת בְּרִית מִילָה לִבְנוֹ שֶׁל אֶחָד מִמְּקֹרָבָיו. הָרַב כְּבָר לָבַשׁ אֶת מְעִילוֹ, חָבַשׁ אֶת מִגְבַּעְתּוֹ, וְעָמַד מַמָּשׁ בְּפֶתַח הַבַּיִת כְּדֵי לָצֵאת לַדֶּרֶךְ.

וְהִנֵּה, בְּדִיּוּק בְּאוֹתוֹ רֶגַע, הוֹפִיעוּ בַּפֶּתַח מִסְפָּר אֲנָשִׁים. אֵלּוּ הָיוּ אֲנָשִׁים שְׁבוּרֵי לֵב, שֶׁבָּאוּ לְהָשִׂיחַ אֶת מוֹעֲקָתָם וְאֶת צַעֲרָם בְּאָזְנָיו שֶׁל הָרַב. כַּאֲשֶׁר הִבְחִין בָּהֶם הַ'חָזוֹן אִישׁ', הוּא לֹא אָמַר דָּבָר עַל כָּךְ שֶׁהוּא מְמַהֵר אוֹ שֶׁמַּמְתִּינִים לוֹ. בְּלִי לוֹמַר מָלָה, שָׁב מִיָּד אֶל חַדְרוֹ, וּבְסַבְלָנוּת נְדִירָה מְאֹד הִזְמִין אוֹתָם לְהִכָּנֵס.

הָרַב יָשַׁב עִמָּהֶם וְשׂוֹחַח אִתָּם בְּסַבְלָנוּת וּבְנַחַת בְּמֶשֶׁךְ קָרוֹב לַחֲצִי שָׁעָה תְּמִימָה. הוּא הִקְשִׁיב בִּתְשׂוּמַת לֵב לְכָל הֵיבֵט וּלְכָל פְּרָט שֶׁל הַבְּעָיָה הַסְּבוּכָה שֶׁהִצִּיגוּ לְפָנָיו. הַשִּׂיחָה הִתְנַהֲלָה בְּשַׁלְוָה גְּדוֹלָה, כְּאִלּוּ לֹא אֲמוּרָה לְהִתְקַיֵּם כָּעֵת שׁוּם בְּרִית מִילָה. הָרַב הֶעֱנִיק לִבְנֵי שִׂיחוֹ תְּחוּשָׁה נְעִימָה וּרְגוּעָה, כְּאִלּוּ כָּל הַזְּמַן שֶׁלְּפָנָיו מֻקְדָּשׁ כָּעֵת אַךְ וְרַק לָהֶם וּלְפִתְרוֹן בַּעֲיָתָם.

כַּאֲשֶׁר סִיְּמוּ אֶת הַשִּׂיחָה וְיָצְאוּ מֵהַחֶדֶר, קָם הַ'חָזוֹן אִישׁ' וְיָצָא מֵהַבַּיִת אֶל עֵבֶר הָרֶכֶב שֶׁהִמְתִּין לוֹ בַּחוּץ. מִיָּד עִם כְּנִיסָתוֹ לָרֶכֶב, רָאָה הָרַב כִּי הַנַּהָג כְּבָר הִתְנִיעַ אֶת הַמְּכוֹנִית. כְּפִי שֶׁהָיָה נִכָּר בִּתְנוּעוֹתָיו שֶׁל הַנַּהָג, הוּא תִּכְנֵן לִפְתֹּחַ בִּנְסִיעָה מְהִירָה מְאֹד וּבִדְהִירָה לַאֲחַר הָאִחוּר הַגָּדוֹל, כְּדֵי לְצַמְצֵם כַּמָּה שֶׁאֶפְשָׁר אֶת הָאִחוּר לִשְׁמְחַת הַבְּרִית. אַךְ הַ'חָזוֹן אִישׁ' פָּנָה אֵלָיו וְהוֹרָה לוֹ מִיָּד: "סַע בְּאִטִּיּוּת!".

כַּאֲשֶׁר הִתְרַחֲקוּ קְצָת מֵהַמָּקוֹם, הִסְבִּיר הָרַב לַנַּהָג מַדּוּעַ הוֹרָה לוֹ לִנְסֹעַ לְאַט לַמְרוֹת שֶׁהַשָּׁעָה הָיְתָה מְאֻחֶרֶת מְאֹד. הָרַב אָמַר לוֹ: "כְּשֶׁיָּצָאנוּ מֵהַבַּיִת, הָאֲנָשִׁים הָאֵלֶּה הָיוּ עֲדַיִן עֹמְדִים בְּקִרְבַת מָקוֹם. אִלּוּ הָיִיתָ מַתְחִיל לִנְסֹעַ בִּמְהִירוּת גְּדוֹלָה, הָיְתָה עוֹבֶרֶת בָּהֶם מַחֲשָׁבָה שֶׁאוּלַי הֵם עִכְּבוּ אוֹתִי, אוֹ שֶׁאוּלַי לֹא הָיָה לִי בֶּאֱמֶת כֹּחַ וְסַבְלָנוּת עֲבוּרָם. לִרְאָיָה: הֵם הָיוּ אוֹמְרִים שֶׁבָּרֶגַע שֶׁיָּצְאוּ מִמֶּנִּי 'בָּרַחְתִּי'..."

אָז סִכֵּם הַ'חָזוֹן אִישׁ' אֶת דְּבָרָיו וְאָמַר: "מֵאֲנָשִׁים שְׁבוּרֵי לֵב לֹא בּוֹרְחִים...". בְּדֶרֶךְ זוֹ הֶחְדִּיר הָרַב בְּשֹׁמְעָיו רְגִישׁוּת גְּדוֹלָה מְאֹד וְלִמֵּד אוֹתָם לְהִתְחַשֵּׁב תָּמִיד בְּרִגְשׁוֹתָיו שֶׁל כָּל אָדָם.
