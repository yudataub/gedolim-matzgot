הַדְּרָשָׁה שֶׁהַמַּלְאָכִים בָּאוּ לִשְׁמֹעַ

לִפְנֵי מֵאוֹת שָׁנִים, בָּעִיר הַקְּדוֹשָׁה צְפַת, חָיוּ חַכְמֵי תּוֹרָה גְּדוֹלִים וּמְפֻרְסָמִים מְאֹד. בֵּינֵיהֶם הָיוּ רַבִּי יוֹסֵף קָארוֹ, רַבֵּנוּ הָאֲרִי הַקָּדוֹשׁ, וְרַבִּי מֹשֶׁה אַלְשֵׁיךְ, שֶׁכֻּלָּם קָרְאוּ לוֹ "הָאַלְשֵׁיךְ הַקָּדוֹשׁ". רַבִּי מֹשֶׁה אַלְשֵׁיךְ נוֹלַד בְּטוּרְקִיָּה, לָמַד אֵצֶל רַבִּי יוֹסֵף קָארוֹ, וּבָא לָגוּר בְּאֶרֶץ יִשְׂרָאֵל. הוּא הָיָה דַּיָּן חָכָם, עָמַד בְּרֹאשׁ שְׁתֵּי יְשִׁיבוֹת, וְהָיָה רַבָּם שֶׁל תַּלְמִידֵי חֲכָמִים גְּדוֹלִים.

בְּכָל יוֹם שִׁשִּׁי הָיָה רַבִּי מֹשֶׁה אַלְשֵׁיךְ יוֹשֵׁב וְכוֹתֵב אֶת הַדְּרָשָׁה שֶׁלּוֹ לִכְבוֹד שַׁבָּת. וּבְכָל שַׁבָּת קֹדֶשׁ, כְּשֶׁהִגִּיעָה הָעֵת, עָמַד בְּבֵית הַכְּנֶסֶת הַגָּדוֹל בִּצְפַת וְדָרַשׁ לִפְנֵי כָּל הַקָּהָל. דְּבָרָיו הַמְּתוּקִים נִכְנְסוּ יָשָׁר אֶל הַלְּבָבוֹת. הוּאִיל וְדִבֵּר בְּאַהֲבָה רַבָּה, לִמֵּד אֶת הָעָם לֶאֱהֹב אֶת הַשֵּׁם יִתְבָּרַךְ, לְהִתְפַּלֵּל בְּכַוָּנָה, לִירֹא אֶת הַשָּׁמַיִם וְלַחֲזֹר בִּתְשׁוּבָה שְׁלֵמָה. כָּל אַנְשֵׁי הָעִיר צְפַת אָהֲבוּ מְאֹד לָבוֹא וְלִשְׁמֹעַ אֶת דְּרָשׁוֹתָיו.

יוֹם אֶחָד אָמַר רַבֵּנוּ הָאֲרִי הַקָּדוֹשׁ לְתַלְמִידָיו דָּבָר מַפְלִיא וּמַרְהִיב. הוּא סִפֵּר לָהֶם שֶׁבְּכָל פַּעַם שֶׁרַבִּי מֹשֶׁה אַלְשֵׁיךְ עוֹמֵד וְדוֹרֵשׁ אֶת דִּבְרֵי הַתּוֹרָה בְּבֵית הַכְּנֶסֶת, כָּל הַפַּמַלְיָה שֶׁל מַעְלָה – כָּל הַמַּלְאָכִים וְהַנְּשָׁמוֹת הַקְּדוֹשׁוֹת מֵהַשָּׁמַיִם – יוֹרְדִים וּבָאִים לִשְׁמֹעַ אֶת הַדְּרָשָׁה שֶׁלּוֹ. מִפְּנֵי שֶׁרָאָה אֶת הַדָּבָר הַגָּדוֹל הַזֶּה, הָיָה הָאֲרִי הַקָּדוֹשׁ עַצְמוֹ הוֹלֵךְ בְּכָל שַׁבָּת לְבֵית הַכְּנֶסֶת כְּדֵי לִשְׁמֹעַ אֶת דִּבְרֵי הָאַלְשֵׁיךְ.

רַבִּי מֹשֶׁה אַלְשֵׁיךְ יָדַע שֶׁהָאֲרִי הַקָּדוֹשׁ יוֹדֵעַ אֶת כָּל סוֹדוֹת הַתּוֹרָה הָעֲמֻקִּים בְּחָכְמַת הַקַּבָּלָה. בִּקֵּשׁ הָאַלְשֵׁיךְ מֵהָאֲרִי הַקָּדוֹשׁ: "אָנָּא, לַמֵּד גַּם אוֹתִי אֶת חָכְמַת הַסּוֹד, כְּדֵי שֶׁאֵדַע אֶת סוֹדוֹת הַתּוֹרָה הַנִּסְתָּרִים!" אַךְ הָאֲרִי הַקָּדוֹשׁ לֹא הִסְכִּים לְלַמֵּד אוֹתוֹ. הוּא הִסְבִּיר לוֹ בְּעַנְוָה: "הַנְּשָׁמָה שֶׁלְּךָ לֹא יָרְדָה לָעוֹלָם הַזֶּה כְּדֵי לִלְמֹד אֶת חָכְמַת הַסּוֹד. הַתַּפְקִיד הַמְּיֻחָד שֶׁלְּךָ מֵהַשֵּׁם יִתְבָּרַךְ הוּא לִכְתֹּב סְפָרִים נִפְלָאִים עַל פִּי הַפְּשָׁט וְהַדְּרוּשׁ, וּלְבָאֵר אֶת מִלּוֹת הַתּוֹרָה לְכָל אָדָם מִיִּשְׂרָאֵל, כְּדֵי שֶׁכָּל יְהוּדִי יוּכַל לְהָבִין וְלִלְמֹד."

כְּשֶׁשָּׁמַע זֹאת רַבִּי מֹשֶׁה אַלְשֵׁיךְ, הוּא הִצְטַעֵר מְאֹד בְּלִבּוֹ, כִּי נִכְסַף וְהִשְׁתוֹקֵק לִלְמֹד אֶת הַסּוֹד. אֲבָל הוּא קִבֵּל אֶת דִּבְרֵי הָאֲרִי הַקָּדוֹשׁ בְּאַהֲבָה וּבִתְמִימוּת. הוּא הֵבִין שֶׁזֶּהוּ הַתַּפְקִיד שֶׁהָאֵל הִטִּיל עָלָיו, וּבָחַר לַעֲשׂוֹת אוֹתוֹ בְּכָל כֹּחוֹ וּבְכָל לְבָבוֹ. הוּא הִמְשִׁיךְ לִלְמֹד בִּשְׁקִידָה רַבָּה, לִדְרֹשׁ לָעָם בְּכָל שַׁבָּת וְשַׁבָּת, וְלִכְתֹּב אֶת דִּבְרֵי תּוֹרָתוֹ הַמְּתוּקִים.

מִן הַדְּרָשׁוֹת הָאֵלֶּה שֶׁדָּרַשׁ רַבִּי מֹשֶׁה אַלְשֵׁיךְ בִּצְפַת, נוֹלְדוּ הַסְּפָרִים הַקְּדוֹשִׁים שֶׁלּוֹ, וּבְרֹאשָׁם הַסֵּפֶר הַמְּפֻרְסָם "תּוֹרַת מֹשֶׁה" עַל חֲמִשָּׁה חֻמְשֵׁי תּוֹרָה, וְכֵן פֵּרוּשִׁים עַל נְבִיאִים, מְגִלּוֹת וּתְהִלִּים. עַד הַיּוֹם הַזֶּה, בְּכָל דּוֹר וָדוֹר, יוֹשְׁבִים יְהוּדִים בְּכָל הָעוֹלָם וְלוֹמְדִים אֶת פֵּרוּשׁ הָאַלְשֵׁיךְ הַקָּדוֹשׁ עַל פָּרָשַׁת הַשָּׁבוּעַ, וְנֶהֱנִים מֵהָאוֹר הַגָּדוֹל שֶׁהוּא הֵבִיא לָעוֹלָם.

💡 לְכָל אֶחָד מֵאִתָּנוּ יֵשׁ תַּפְקִיד מְיֻחָד שֶׁהַשֵּׁם נָתַן לוֹ, וּכְשֶׁאָנוּ עוֹשִׂים אֶת הַתַּפְקִיד שֶׁלָּנוּ בְּכָל הַלֵּב – אָנוּ מְבִיאִים אוֹר גָּדוֹל לָעוֹלָם.
