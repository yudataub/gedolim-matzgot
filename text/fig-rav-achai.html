רַב אַחַאי גָּאוֹן וְסֵפֶר הַשְּׁאִילְתּוֹת

לִפְנֵי שָׁנִים רַבּוֹת, בִּתְקוּפַת הַגְּאוֹנִים, חַי בְּבָבֶל חָכָם גָּדוֹל וְחָשׁוּב מְאֹד שֶׁנִּקְרָא רַב אַחַאי מִשַּׁבְחָא. רַב אַחַאי הָיָה מִגְּדוֹלֵי חַכְמֵי בָּבֶל. וְהִנֵּה, הִגִּיעַ זְמַן שֶׁבּוֹ צָרִיךְ הָיָה לִבְחֹר רֹאשׁ יְשִׁיבָה חָדָשׁ לִישִׁיבַת פּוּמְבְּדִיתָא הַמְּפֻרְסֶמֶת. רֹאשׁ הַגּוֹלָה, שֶׁהָיָה הַמַּנְהִיג, הֶחְלִיט לֹא לְמַנּוֹת אֶת רַב אַחַאי לְרֹאשׁ הַיְשִׁיבָה, אֶלָּא מִנָּה אֶת מַר רַב נַטְרוֹנַאי, שֶׁהָיָה תַּלְמִידוֹ וְשַׁמָּשׁוֹ שֶׁל רַב אַחַאי. רַב אַחַאי יָכֹל הָיָה לִכְעֹס אוֹ לְהִלָּחֵם עַל הַכָּבוֹד שֶׁלּוֹ, אַךְ הוּא הָיָה חָכָם וְעַנְוְתָן וְלֹא נִלְחַם עַל הַכָּבוֹד כְּלָל. בִּמְקוֹם זֹאת, הוּא הֶחְלִיט לַעֲשׂוֹת מַעֲשֶׂה מְפֹאָר וְעָלָה לְאֶרֶץ יִשְׂרָאֵל.

כַּאֲשֶׁר הִגִּיעַ רַב אַחַאי לְאֶרֶץ יִשְׂרָאֵל, הוּא חִבֵּר אֶת סֵפֶר "הַשְּׁאִילְתּוֹת". זֶהוּ הַסֵּפֶר הָרִאשׁוֹן מַמָּשׁ שֶׁנִּכְתַּב וְחֻבַּר לְאַחַר חֲתִימַת הַתַּלְמוּד. בַּסֵּפֶר הַמְּיֻחָד הַזֶּה כָּתַב רַב אַחַאי דְּרָשׁוֹת חֲשׁוּבוֹת עַל פָּרָשִׁיּוֹת הַשָּׁבוּעַ וְעַל מִצְוֹת הַתּוֹרָה, כְּדֵי לְלַמֵּד אֶת עַם יִשְׂרָאֵל.

הַדְּרָשָׁה הָרִאשׁוֹנָה בַּסֵּפֶר, הַנִּקְרֵאת הַשְּׁאִילְתָּא הָרִאשׁוֹנָה, עוֹסֶקֶת בִּכְבוֹד הַשַּׁבָּת וּבְעֹנֶג הַשַּׁבָּת. רַב אַחַאי מַסְבִּיר שֶׁכָּל בְּנֵי יִשְׂרָאֵל חַיָּבִים לָנוּחַ בְּיוֹם הַשַּׁבָּת. הוּא מַסְבִּיר כִּי כַּאֲשֶׁר בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת עוֹלָמוֹ בְּשִׁשָּׁה יָמִים, הוּא נָח בְּיוֹם הַשַּׁבָּת, בֵּרַךְ אוֹתוֹ וְקִדֵּשׁ אוֹתוֹ. הַדָּבָר דוֹמֶה לְאָדָם שֶׁבּוֹנֶה בַּיִת, מְקַשֵּׁט וּמְעַצֵּב אוֹתוֹ יָפֶה, וּכְשֶׁהוּא מְסַיֵּם אֶת כָּל עֲבוֹדָתוֹ, הוּא עוֹשֶׂה מְסִבָּה וַחֲגִיגָה גְּדוֹלָה לְיוֹם אֶחָד. כָּךְ אָמַר לָנוּ הַשֵּׁם לָנוּחַ בַּשַּׁבָּת כְּפִי שֶׁהוּא נָח, כְּפִי שֶׁכָּתוּב: "וַיְכַל אֱלֹהִים בַּיּוֹם הַשְּׁבִיעִי", וְכֵן אָמַר לָנוּ: "זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ".

רַב אַחַאי מְלַמֵּד שֶׁאָסוּר לְהִתְעַנּוֹת אוֹ לָצוּם בַּשַּׁבָּת. בְּנֵי יִשְׂרָאֵל צְרִיכִים לְהִתְעַנֵּג בַּשַּׁבָּת בְּמַאֲכָל וּבְמַשְׁתֶּה, וּלְכַבֵּד אֶת הַשַּׁבָּת בִּלְבוּשׁ יָפֶה וּבְכִסּוּי נָאֶה, כְּפִי שֶׁכָּתוּב: "וְקָרָאתָ לַשַּׁבָּת עֹנֶג... וְכִבַּדְתּוֹ". הַבְּגָדִים שֶׁל שַׁבָּת לֹא יִהְיוּ כְּבִגְדֵי הַחֹל. רַב אַחַאי מְבִיא כְּדֻגְמָה אֶת רַבִּי יוֹחָנָן, שֶׁהָיָה קוֹרֵא לַבְּגָדִים הַנָּאִים שֶׁלָּבַשׁ בַּשַּׁבָּת "מְכַבְּדוֹתַי", כִּי הֵם מְכַבְּדִים אֶת הָאָדָם וְאֶת הַשַּׁבָּת. וְעוֹד אָמַר רַב הוּנָא שֶׁמִי שֶׁיֵּשׁ לוֹ בְּגָדִים לְהַחְלִיף – יַחְלִיף לִכְבוֹד הַשַּׁבָּת, וּמִי שֶׁאֵין לוֹ בְּגָדִים אֲחֵרִים לְהַחְלִיף – יַשְׁלְשֵׁל בִּבְגָדָיו וִישַׁנֶּה אֶת מְהַלְכָם, כְּדֵי לְהַרְאוֹת כָּבוֹד מְיֻחָד לַשַּׁבָּת הַקְּדוֹשָׁה.

💡 מִמַּעֲשֵׂהוּ שֶׁל רַב אַחַאי אָנוּ לוֹמְדִים לֹא לִלְחֹם עַל כָּבוֹד, אֶלָּא לְהַרְבּוֹת בְּלִמּוּד תּוֹרָה וּבִכְבוֹד הַשַּׁבָּת.
