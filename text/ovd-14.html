הַכֹּחַ שֶׁל הַתּוֹרָה

לִפְנֵי כַּמָּה שָׁנִים חָשׁ מָרָן הָרַב עוֹבַדְיָה יוֹסֵף כְּאֵבִים עַזִּים וְקָשִׁים מְאֹד בְּגוּפוֹ, וְהֻבְהַל בִּמְהִירוּת רַבָּה לְבֵית הַחוֹלִים. עֲנָנָה כְּבֵדָה עָלְתָה עַל פְּנֵי הָרוֹפְאִים שֶׁבָּדְקוּ אוֹתוֹ בְּחַדְרֵי הַבְּדִיקָה. הֵם לָחֲשׁוּ בִּדְאָגָה לִמְקֹרָבָיו הַנֶּאֱמָנִים שֶׁלִּוּוּ אוֹתוֹ, כִּי הֵם חוֹשְׁשִׁים לְמַחֲלָה מְסֻיֶּמֶת בְּגוּפוֹ, וּכְדֵי לִשְׁלֹל זֹאת וְלָדַעַת בְּבֵרוּר מָה מַצָּבוֹ, עֲלֵיהֶם לַעֲשׂוֹת לָרַב בְּדִיקָה קַפְּדָנִית וּמְפֹרֶטֶת מְאֹד. הָרוֹפְאִים הִסְבִּירוּ לִמְקֹרָבָיו: "אֶת הַבְּדִיקָה הַזֹּאת יִצְטָרֵךְ הָרַב לַעֲבֹר תַּחַת הַרְדָּמָה, שֶׁכֵּן מְדֻבָּר בִּבְדִיקָה כּוֹאֶבֶת בִּמְיֻחָד".

אֲבָל מָרָן הָרַב עוֹבַדְיָה, שֶׁשָּׁמַע אֶת הַעֲרָכוֹת הָרוֹפְאִים וְאֶת דִּבְרֵיהֶם, הִתְנַגֵּד לְכָךְ בְּתֹקֶף וְאָמַר לָהֶם נַחֲרָצוֹת: "אֶעֱבֹר אֶת הַבְּדִיקָה הַזֹּאת בְּלֹא הַרְדָּמָה". הָרוֹפְאִים הֻפְתְּעוּ מְאֹד מִדְּבָרָיו, נִעְנְעוּ בְּרָאשֵׁיהֶם וְאָמְרוּ לוֹ בִּדְאָגָה: "זֶה לֹא יִתָּכֵן! מְדֻבָּר בִּבְדִיקָה קָשָׁה כָּל כָּךְ, שֶׁאֲפִלּוּ אָדָם צָעִיר וּבָרִיא שֶׁיַּעֲבֹר אוֹתָהּ לְלֹא הַרְדָּמָה, יְאַבֵּד אֶת הַכָּרָתוֹ מֵרֹב כְּאֵב!".

מָרָן שָׁמַע אֶת דִּבְרֵיהֶם הַקָּשִׁים שֶׁל הָרוֹפְאִים וְשָׁקַע בְּמַחְשָׁבוֹת עֲמֻקּוֹת לְמֶשֶׁךְ כַּמָּה רְגָעִים. לְאַחַר מִכֵּן הִתְנַעֵר מִמַּחְשַׁבְתּוֹ, פָּנָה אֶל מְקֹרָבוֹ הַסּוֹבֵב אוֹתוֹ וְאָמַר לוֹ: "רֵד לְבֵית הַכְּנֶסֶת שֶׁבְּבֵית הַחוֹלִים, וְהָבֵא לִי גְּמָרָא". שָׁאַל הַמְּקֹרָב בְּסַקְרָנוּת: "אֵיזוֹ מַסֶּכֶת לְהָבִיא לְךָ?". הֵשִׁיב לוֹ מָרָן בְּפַשְׁטוּת: "לֹא מְשַׁנֶּה, אֵיזוֹ מַסֶּכֶת שֶׁתִּמְצָא". מִהֵר הַמְּקֹרָב בִּרְצִיפָה לְבֵית הַכְּנֶסֶת שֶׁבְּבֵית הַחוֹלִים, וְשָׁלַף מִן הַסִּפְרִיָּה אֶת הַכֶּרֶךְ הָרִאשׁוֹן שֶׁהִזְדַּמֵּן לְיָדוֹ. הָיְתָה זוֹ מַסֶּכֶת עֲבוֹדָה זָרָה. הוּא מִהֵר וְחָזַר אֶל חַדְרוֹ שֶׁל מָרָן, וְהִגִּישׁ לוֹ אֶת הַגְּמָרָא.

כַּאֲשֶׁר הִגִּישׁ לוֹ אֶת הַסֵּפֶר, עֵינָיו שֶׁל מָרָן אוֹרוּ בְּשִׂמְחָה רַבָּה. הוּא פָּתַח אֶת הַמַּסֶּכֶת, וְשָׁקַע בְּאַחַת הַסּוּגְיוֹת הָעֲמֻקּוֹת בְּמֶשֶׁךְ חָמֵשׁ דַּקּוֹת תְּמִימוֹת. לְאַחַר מִכֵּן הֵרִים אֶת עֵינָיו מִן הַסֵּפֶר, סָגַר אֶת הַגְּמָרָא וְאָמַר בְּרֹגַע לָרוֹפְאִים: "עַכְשָׁו אֶפְשָׁר לְבַצֵּעַ אֶת הַבְּדִיקָה, בְּלִי הַרְדָּמָה".

בְּכִירֵי הָרוֹפְאִים נִעְנְעוּ שׁוּב בְּרָאשֵׁיהֶם לִשְׁלִילָה. לֹא, הֵם לֹא הָיוּ מְסֻגָּלִים אֲפִלּוּ לַחְשֹׁב עַל אֶפְשָׁרוּת כָּזֹאת שֶׁל בְּדִיקָה כּוֹאֶבֶת לְלֹא הַרְדָּמָה. אוּלָם לְבַקָּשׁוֹתָיו הַחוֹזְרוֹת וְנִשְׁנוֹת שֶׁל מָרָן, הִסְכִּימוּ לְבַסּוֹף שְׁנֵי רוֹפְאִים לַעֲשׂוֹת זֹאת.

שֵׁשׁ דַּקּוֹת אָרְכָה הַבְּדִיקָה הַקָּשָׁה. בְּאוֹתָם רְגָעִים מָרָן צָלַל עָמֹק אֶל הַסּוּגְיוֹת הָעֲמֻקּוֹת שֶׁבְּמַסֶּכֶת עֲבוֹדָה זָרָה, כְּשֶׁהוּא אֵינוֹ חָשׁ בִּמְאוּמָה – לֹא בָּרוֹפְאִים הַסּוֹבְבִים אוֹתוֹ, לֹא בַּמְּקֹרָבִים הַמַּבִּיטִים בּוֹ, לֹא בַּבְּדִיקָה הַקָּשָׁה וְלֹא בַּכְּאֵבִים הַנּוֹרָאִים. הַבְּדִיקָה הִסְתַּיְּמָה, וּמָרָן הִתְנַעֵר מֵעֹמֶק עִיּוּנוֹ בַּתּוֹרָה. הָרוֹפְאִים, שֶׁהִקִּיפוּ אֶת מִטָּתוֹ בְּתִמָּהוֹן, הִבִּיטוּ בּוֹ בְּעֵינַיִם דּוֹמְעוֹת מֵהִתְרַגְּשׁוּת וְאָמְרוּ: "מִיָּמֵינוּ לֹא רָאִינוּ מַחֲזֶה מַדְהִים שֶׁכָּזֶה! לֹא רָאִינוּ מְעוֹלָם אָדָם בָּעוֹלָם שֶׁיִּשְׁקַע כָּךְ בְּעִנְיָן מְסֻיָּם, וְיִהְיֶה מְסֻגָּל לִסְבֹּל כְּאֵבִים נוֹרָאִים שֶׁכָּאֵלּוּ!".

💡 כַּאֲשֶׁר שׁוֹקְעִים בְּאַהֲבָה בְּלִמּוּד הַתּוֹרָה הַקְּדוֹשָׁה, כֹּחַ הַתּוֹרָה מְמַלֵּא אֶת הַלֵּב וּמִתְגַּבֵּר עַל כָּל קֹשִׁי וּכְאֵב.
