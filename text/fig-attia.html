תְּפִלַּת הָרְחוֹב שֶׁל הָרַב מָרְדְּכַאי עַטִיָּה

בִּשְׁנַת תַּשְׁ"ח, בִּזְמַן מִלְחֶמֶת הַשִּׁחְרוּר, קְצָת אַחֲרֵי שֶׁזָּכִינוּ לְהָקִים אֶת מְדִינַת יִשְׂרָאֵל, הָיָה פַּחַד גָּדוֹל בְּאֶרֶץ יִשְׂרָאֵל. עַם יִשְׂרָאֵל וְתוֹשְׁבֵי יְרוּשָׁלַיִם הָיוּ מְפוֹחָדִים מְאֹד. אַף אֶחָד לֹא יָדַע מָה יִהְיוּ תּוֹצְאוֹת הַמִּלְחָמָה וּמָה יַעֲשׂוּ הָעַרְבִים. לַיְּהוּדִים לֹא הָיוּ כְּלֵי נֶשֶׁק וְלֹא הָיוּ טַנְקִים. הַפַּחַד הָיָה כָּל כָּךְ גָּדוֹל, עַד שֶׁכְּדֵי לְהַבְרִיחַ אֶת הָעַרְבִים, הָיוּ אֲנָשִׁים שֶׁעָמְדוּ עִם מַקְלוֹת וְדָפְקוּ עַל פַּחִים וְעַל חֲפָצִים שׁוֹנִים כְּדֵי לַעֲשׂוֹת רַעַשׁ גָּדוֹל, כְּדֵי שֶׁהָאוֹיְבִים יַחְשְׁבוּ שֶׁיֵּשׁ שָׁם קוֹלוֹת שֶׁל מִלְחָמָה.

בְּאוֹתָהּ תְּקוּפָה חַי בִּירוּשָׁלַיִם הָרַב מָרְדְּכַאי עַטִיָּה, סַבָּא רַבָּא שֶׁל הָרַב נֹעַם הַלֵּוִי. הָרַב מָרְדְּכַאי עַטִיָּה רָאָה אֶת הַפַּחַד שֶׁל הָאֲנָשִׁים וְרָצָה מְאֹד לְחַזֵּק אֶת תּוֹשְׁבֵי יְרוּשָׁלַיִם. הוּא הֶחְלִיט לְאַרְגֵּן תְּפִילּוֹת מְיֻחָדוֹת מַמָּשׁ בִּרְחוֹבָהּ שֶׁל עִיר, בַּרְחוֹב לְיַד שׁוּק מַחֲנֵה יְהוּדָה. הָרַב הֵבִיא לַשָּׁם סֵפֶר תּוֹרָה וְשֻׁלְחָן, וְחִזֵּק אֶת כָּל הָאֲנָשִׁים בִּתְפִילָה לַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיַּצִּיל אוֹתָנוּ מֵהַצָּרָה וְיִגְאַל אוֹתָנוּ גְּאֻלָּה שְׁלֵמָה.

כְּדֵי לְחַזֵּק אֶת הַמִּתְפַּלְלִים, הָרַב מָרְדְּכַאי עַטִיָּה לָקַח שְׁנֵי פְּרָקִים מִסֵּפֶר יְשַׁעְיָהוּ, פֶּרֶק לַ"ד וּפֶרֶק לַ"ה. הַפְּרָקִים הָאֵלֶּה מְדַבְּרִים מַמָּשׁ עַל הַתְּקוּפָה שֶׁלָּנוּ, עַל כָּךְ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲנִישׁ אֶת הַגּוֹיִים וְעַל כָּךְ שֶׁעַם יִשְׂרָאֵל יִהְיֶה בְּשִׂמְחָה גְּדוֹלָה בַּגְּאֻלָּה הָאַחֲרוֹנָה. הָרַב מָרְדְּכַאי הָיָה מַדְפִּיס אֶת הַפְּרָקִים הָאֵלֶּה וּמְחַלֵּק אוֹתָם לְכָל הָאֲנָשִׁים שֶׁבָּאוּ לַתְּפִילּוֹת, וְכֻלָּם הָיוּ קוֹרְאִים אֶת הַפְּרָקִים יַחַד.

הַתְּפִילּוֹת הָאֵלֶּה הִשְׁפִיעוּ עַל הָאֲנָשִׁים בְּצוּרָה נִפְלָאָה. אֲנָשִׁים שֶׁהָיוּ שָׁם סִפְּרוּ שֶׁבִּזְמַן הַתְּפִילּוֹת הֵם לֹא פַּחֲדוּ כְּלָל. בְּאוֹתוֹ רֶגַע הֵם הִרְגִישׁוּ אֵיךְ הַקָּדוֹשׁ בָּרוּךְ הוּא מַחְזִיר אוֹתָנוּ לִירוּשָׁלַיִם, מַחְזִיר אוֹתָנוּ לַכֹּתֶל, וְאֵיךְ הַכֹּל טוֹב וְאֵין שׁוּם פַּחַד בַּלֵּב.

הָרַב מָרְדְּכַאי הָיָה מְחַלֵּק אֶת הַפְּרָקִים הָאֵלֶּה גַּם בְּהִזְדַּמְנוּיוֹת נוֹסָפוֹת, כְּמוֹ בִּזְמַן שֶׁפָּתְחוּ אֶת אֲרוֹן הַקֹּדֶשׁ בְּשַׁבָּת וּבְחַגִּים וְגַם בִּבְרִית מִילָה. בְּנוֹסָף, כְּשֶׁהָרַב מָרְדְּכַאי הָיָה פּוֹגֵשׁ יֶלֶד צָעִיר, הוּא הָיָה נוֹתֵן לוֹ אֶת הַפְּרָקִים הָאֵלֶּה וּמְבַקֵּשׁ מִמֶּנּוּ לִלְמֹד אוֹתָם בְּעַל-פֶּה וְלָדַעַת לִקְרֹא אוֹתָם בְּצוּרָה מְדֻיֶּקֶת עִם טַעֲמֵי הַמִּקְרָא, וְהָיָה מְעוֹדֵד אֶת הַיְּלָדִים עַל יְדֵי פְּרָסִים. כָּךְ הִצְלִיחַ הָרַב מָרְדְּכַאי עַטִיָּה לְחַזֵּק אֶת תּוֹשְׁבֵי יְרוּשָׁלַיִם בֶּאֱמוּנָה, בְּשִׂמְחָה וּבְתִקְוָה.
