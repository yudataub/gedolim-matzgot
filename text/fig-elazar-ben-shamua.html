הַבְּכִי בְּצִידוֹן וְהַחֲזָרָה לְאֶרֶץ יִשְׂרָאֵל

הַגְּמָרָא מְסַפֶּרֶת עַל רַבִּי יְהוּדָה בֶּן בְּתֵירָא, שֶׁהָיָה חָכָם גָּדוֹל וְתַלְמִיד חָכָם עָצוּם מְאֹד. הוּא גָּר בְּעִיר שֶׁנִּקְרְאָה נְצִיבִין, שֶׁהָיְתָה מְמוֹקָמָה בְּחוּץ לָאָרֶץ, מִצָּפוֹן לְאֶרֶץ יִשְׂרָאֵל, בַּמָּקוֹם שֶׁנִּקְרָא כַּיּוֹם מְדִינַת לְבָנוֹן. בְּאוֹתָהּ תְּקוּפָה חָיוּ בְּאֶרֶץ יִשְׂרָאֵל שְׁנֵי תַּלְמִידֵי חֲכָמִים חֲשׁוּבִים וִיקָרִים מְאֹד, שֶׁנִּקְרְאוּ רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ וְרַבִּי יוֹחָנָן הַסַּנְדְּלָר. שְׁנֵי הֶחָכָמִים הָאֵלֶּה רָצוּ מְאֹד לִלְמֹד תּוֹרָה מִפִּיו שֶׁל רַבִּי יְהוּדָה בֶּן בְּתֵירָא. מִתּוֹךְ רְצוֹנָם הָעַז וְהַגָּדוֹל לִלְמֹד מִמֶּנּוּ תּוֹרָה, הֶחְלִיטוּ שְׁנֵיהֶם לַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל וְלָלֶכֶת לְחוּץ לָאָרֶץ, כְּדֵי לְהַגִּיעַ עַד לָעִיר נְצִיבִין וְלִלְמֹד שָׁם תּוֹרָה מֵהָרַב הַגָּדוֹל.

קָמוּ שְׁנֵי הֶחָכָמִים, רַבִּי יוֹחָנָן הַסַּנְדְּלָר וְרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ, וְהִתְחִילוּ לָלֶכֶת בַּדֶּרֶךְ כְּלַפֵּי צְפוֹנָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל. יָמִים רַבִּים הָלְכוּ שְׁנֵי הֶחָכָמִים בַּדֶּרֶךְ הָאֲרֻכָּה, עַד שֶׁהִגִּיעוּ לְאֵזוֹר הַגָּלִיל. מִשָּׁם הִמְשִׁיכוּ בְּדַרְכָּם, יָצְאוּ מֵאֶרֶץ יִשְׂרָאֵל וְהָלְכוּ בַּדֶּרֶךְ הַמּוֹלִיכָה לִנְצִיבִין שֶׁבְּחוּץ לָאָרֶץ.

כְּשֶׁהִגִּיעוּ שְׁנֵי הֶחָכָמִים לְמָקוֹם שֶׁנִּקְרָא צִידוֹן, שֶׁמִּמֶּנּוּ עֲדַיִן אֶפְשָׁר הָיָה לִרְאוֹת מֵרָחוֹק אֶת אֶרֶץ יִשְׂרָאֵל, הֵם הָפְכוּ אֶת פְּנֵיהֶם אָחוֹרָה כְּלַפֵּי אֶרֶץ יִשְׂרָאֵל. הֵם הִבִּיטוּ עַל הָאָרֶץ וְאָמְרוּ בְּצַעַר רַב: זֹאת הִיא הַפַּעַם הָאַחֲרוֹנָה שֶׁאָנַחְנוּ רוֹאִים אֶת אֶרֶץ יִשְׂרָאֵל, וּמִי יוֹדֵעַ מָתַי נִזְכֶּה לִרְאוֹת אוֹתָהּ עוֹד פַּעַם. כְּשֶׁהִסְתַּכְּלוּ עַל אֶרֶץ יִשְׂרָאֵל מֵרָחוֹק, הִתְחִילוּ לִזְלֹג מֵעֵינֵיהֶם דְּמָעוֹת רַבּוֹת, וְהֶחָכָמִים הַצַּדִּיקִים הִתְחִילוּ לִבְכּוֹת מֵעֹמֶק הַלֵּב.

הֵם עָמְדוּ שָׁם וְאָמְרוּ בִּדְמָעוֹת: אֵיךְ נַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל, הָאָרֶץ הַקְּדוֹשָׁה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא כָּל כָּךְ אוֹהֵב אוֹתָהּ? אֵיךְ נַעֲזֹב אֶת הָאָרֶץ שֶׁמֹּשֶׁה רַבֵּנוּ כָּל כָּךְ רָצָה לְהִכָּנֵס אֵלֶיהָ וְלֹא זָכָה? אֵיךְ נַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל? אָמְנָם אָנַחְנוּ רוֹצִים מְאֹד לִלְמֹד תּוֹרָה מֵרַבִּי יְהוּדָה בֶּן בְּתֵירָא, הַתַּלְמִיד חָכָם הָעֲצוּם, אֲבָל אֵיךְ נַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל, הָאָרֶץ הַקְּדוֹשָׁה וְהַנִּפְלָאָה, אֶרֶץ זָבַת חָלָב וּדְבָשׁ?

הֵם בָּכוּ וּבָכוּ מִתּוֹךְ אַהֲבָה גְּדוֹלָה לָאָרֶץ. בַּסּוֹף הֶחְלִיטוּ שְׁנֵי הֶחָכָמִים לַחְזֹר לְאֶרֶץ יִשְׂרָאֵל וְלֹא לַעֲזֹב אוֹתָהּ. הֵם חָזְרוּ לְאֶרֶץ יִשְׂרָאֵל וְלָמְדוּ תּוֹרָה מִפִּי חַכְמֵי אֶרֶץ יִשְׂרָאֵל, כְּפִי שֶׁנֶּאֱמַר: כִּי מִצִיּוֹן תֵּצֵא תוֹרָה וּדְבַר ה' מִירוּשָׁלָיִם.

💡 אֶרֶץ יִשְׂרָאֵל הִיא אֶרֶץ קְדוֹשָׁה וַאֲהוּבָה מְאֹד, וּמִצְוָה גְּדוֹלָה לֶאֱהֹב אוֹתָהּ וְלִחְיוֹת בָּהּ.
