הַבְּרָכָה הַמְּיֻחֶדֶת בַּכֹּתֵל

לְאַחַר שֶׁזָּכִינוּ לְשַׁחְרֵר אֶת הַכֹּתֵל הַמַּעֲרָבִי בְּמִלְחֶמֶת שֵׁשֶׁת הַיָּמִים, הָיָה הָרַב אַבְרָהָם אֶלְקָנָה שַׁפִירָא זֵכֶר צַדִּיק וְקָדוֹשׁ לִבְרָכָה מַקְפִּיד בְּכָל שָׁנָה לָלֶכֶת אֶל הַכֹּתֵל הַמַּעֲרָבִי בְּחַג הַשָּׁבוּעוֹת. הָרַב שַׁפִירָא יָצָא בְּרֶגֶל מִישִׁיבַת 'מֶרְכַּז הָרַב' שֶׁנִּמְצֵאת בִּשְׁכוּנַת קִרְיַת מֹשֶׁה בִּירוּשָׁלַיִם, וְצָעַד כָּל הַדֶּרֶךְ עַד לָעִיר הָעַתִּיקָה וְאֶל הַכֹּתֵל הַמַּעֲרָבִי. הָרַב הָיָה צוֹעֵד אֶת הַדֶּרֶךְ הָאֲרֻכָּה הַזֹּאת יַחַד עִם תַּלְמִידֵי הַיְּשִׁיבָה, לַמְרוֹת שֶׁהוּא כְּבָר הָיָה אָדָם מְבוֹגָר מְאֹד. הָרַב שַׁפִירָא הִקְפִּיד מְאֹד לָלֶכֶת בְּכָל שָׁנָה וְשָׁנָה עַד לַכֹּתֵל הַמַּעֲרָבִי, כְּדֵי לְהִתְפַּלֵּל שָׁם יַחַד עִם כָּל הַתַּלְמִידִים אֶת תְּפִלַּת שַׁחֲרִית שֶׁל חַג הַשָּׁבוּעוֹת.

כַּאֲשֶׁר הִגִּיעוּ אֶל הַכֹּתֵל הַמַּעֲרָבִי וְהִתְחִילוּ לְהִתְפַּלֵּל, הָיָה הָרַב מַקְפִּיד מְאֹד שֶׁתְּפִלַּת שַׁחֲרִית תִּהְיֶה מְהִירָה מַמָּשׁ מַמָּשׁ. הָרַב לֹא רָצָה לְעַכֵּב אֶת תַּלְמִידֵי הַיְּשִׁיבָה שֶׁהָיוּ עֵרִים בְּכָל הַלַּיְלָה שֶׁעָבַר וְלָמְדוּ תּוֹרָה בִּשְׁקִידָה. הָרַב חָשַׁב עַל הַתַּלְמִידִים הָעֲיֵפִים, וְלָכֵן רָצָה שֶׁהַתְּפִלָה תִּהְיֶה זְרִיזָה וּמְהִירָה כְּדֵי שֶׁלֹּא לְהַכְבִּיד עֲלֵיהֶם וְלֹא לְעַכֵּב אוֹתָם.

אַךְ כָּל הַתַּלְמִידִים שָׂמוּ לֵב שֶׁהָיָה קֶטַע אֶחָד מְיֻחָד בְּתוֹךְ הַתְּפִלָה, שֶׁבּוֹ הָרַב לֹא מִהֵר כְּלָל. בַּקֶּטַע הַזֶּה הָיָה הָרַב שַׁפִירָא מִתְפַּלֵּל לַאַט לַאַט, בְּכַוָּנָה גְּדוֹלָה וַעֲמֻקָּה מְאֹד. הַקֶּטַע הַזֶּה הָיָה בִּרְכַּת 'אַהֲבָה רַבָּה', אוֹתָהּ אוֹמְרִים בַּתְּפִלָה מַמָּשׁ לִפְנֵי קְרִיאַת שְׁמַע. בְּבִרְכַּת 'אַהֲבָה רַבָּה' הָיָה הָרַב מְכַוֵּן אֶת לִבּוֹ וְאוֹמֵר כָּל מִלָּה וּמִלָּה בְּכַוָּנָה אַדִּירָה וּבְהִתְרַגְּשׁוּת רַבָּה.

פַּעַם אַחַת, נִגַּשׁ אֶל הָרַב אֶחָד מִתַּלְמִידֵי הַיְּשִׁיבָה וְשָׁאַל אוֹתוֹ בְּפֶלֶא: "רַבֵּנוּ, מַדּוּעַ אַתָּה מְכַוֵּן כָּל כָּךְ הַרְבֵּה וּמִתְפַּלֵּל לַאַט בַּתְּפִלָה הַזֹּאת שֶׁל 'אַהֲבָה רַבָּה'?". עָנָה לוֹ הָרַב שַׁפִירָא וְהִסְבִּיר לוֹ אֶת הַסּוֹד הַגָּדוֹל: "בַּתְּפִלָה הַזֹּאת שֶׁל 'אַהֲבָה רַבָּה', אָנוּ מְבַקְשִׁים מֵהַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיִּתֵּן לָנוּ שֵׂכֶל וַהֲבָנָה כְּדֵי לְהָבִין וְלִלְמֹד אֶת הַתּוֹרָה. כָּךְ בְּדִיּוּק אָנוּ מִתְפַּלְלִים וְאוֹמְרִים בַּתְּפִלָה: 'לִשְׁמֹעַ לִלְמֹד וּלְלַמֵּד... וּלְקַיֵּם אֶת כָּל דִּבְרֵי תַלְמוּד תּוֹרָתֶךָ בְּאַהֲבָה'. אָנוּ מְבַקְשִׁים מֵהַשֵּׁם יִתְבָּרַךְ שֶׁיַּעֲנִיק לָנוּ סִיַּעְתָּא דִּשְׁמַיָּא – עֶזְרָה מֵהַשָּׁמַיִם – כְּדֵי שֶׁנַּצְלִיחַ לְהָבִין אֶת הַתּוֹרָה הַקְּדוֹשָׁה, לִלְמֹד אוֹתָהּ בְּאַהֲבָה וְגַם לְלַמֵּד אוֹתָהּ לַאֲחֵרִים".

מֵאָז הֵבִינוּ הַתַּלְמִידִים עַד כַּמָּה חֲשׁוּבָה תְּפִלָה זוֹ עַל לִמּוּד הַתּוֹרָה, וְכַמָּה רַבָּה הָיְתָה אַהֲבָתוֹ שֶׁל הָרַב שַׁפִירָא לַתּוֹרָה וְלַתַּלְמִידִים.

💡 כְּשֶׁאָנוּ מִתְפַּלְלִים בְּכַוָּנָה עַל לִמּוּד הַתּוֹרָה וּמִתַּחֲשְׁבִים בַּזּוּלָת, אָנוּ זוֹכִים לְסִיַּעְתָּא דִּשְׁמַיָּא וּלְאַהֲבַת הַתּוֹרָה.
