שַׁמְגַּר בֶּן עֲנָת וּמַלְמַד הַבָּקָר

לִפְנֵי שָׁנִים רַבּוֹת, בִּתְקוּפַת הַשּׁוֹפְטִים, הָיָה לְעַם יִשְׂרָאֵל מַנְהִיג דָּגוּל וְגִבּוֹר שֶׁנִּקְרָא אֵהוּד בֶּן גֵּרָא. אֵהוּד הוֹשִׁיעַ אֶת יִשְׂרָאֵל וְהֵבִיא שֶׁקֶט לָאָרֶץ. אַךְ כַּאֲשֶׁר נִפְטַר אֵהוּד בֶּן גֵּרָא, עַם יִשְׂרָאֵל נִשְׁאַר לְלֹא מַנְהִיג, וּבְאוֹתָהּ שָׁנָה קָם שׁוֹפֵט חָדָשׁ בְּיִשְׂרָאֵל שֶׁשְּׁמוֹ שַׁמְגַּר בֶּן עֲנָת.

הַיָּמִים הָהֵם הָיוּ יָמִים קָשִׁים מְאֹד עֲבוּר בְּנֵי יִשְׂרָאֵל. הָאוֹיְבִים הַפְּלִשְׁתִּים הֵצֵירוּ לָהֶם וְגָרְמוּ לְפַחַד רַב בְּכָל הָאָרֶץ. בִּגְלַל הַפַּחַד הַגָּדוֹל מֵהַפְּלִשְׁתִּים, בְּנֵי יִשְׂרָאֵל לֹא הֵעֵזוּ לָלֶכֶת בַּדְּרָכִים הָרָאשִׁיּוֹת. כְּפִי שֶׁמְּתֹאָר בַּתַּנָּךְ וּבְשִׁירַת דְּבוֹרָה, "חָדְלוּ אֳרָחוֹת וְהֹלְכֵי נְתִיבוֹת יֵלְכוּ אֳרָחוֹת עֲקַלְקַלּוֹת". כָּל אוֹרֵחַ וְכָל הֵלֶךְ שֶׁרָצָה לַעֲבֹר מִמָּקוֹם לְמָקוֹם נֶאֱלַץ לַעֲקֹף אֶת הַדְּרָכִים הַמְּרֻוָּחוֹת וְהַיְשָׁרוֹת, וְלָלֶכֶת בִּשְׁבִילִים צְדָדִיִּים, עֲקֻמִּים וּמְפֻתָּלִים, כְּדֵי שֶׁהָאוֹיְבִים לֹא יִפְגְּעוּ בּוֹ.

שַׁמְגַּר בֶּן עֲנָת רָאָה אֶת הַצַּעַר הַגָּדוֹל שֶׁל עַם יִשְׂרָאֵל וְאֶת הַפַּחַד שֶׁהִשְׁתַּלֵּט עַל הַדְּרָכִים. הוּא הֶחְלִיט לָקוּם וּלְהוֹשִׁיעַ אֶת בְּנֵי יִשְׂרָאֵל מִיַּד הַפְּלִשְׁתִּים. אַךְ לְשַׁמְגַּר לֹא הָיוּ כְּלֵי נֶשֶׁק רְגִילִים מְפֹאָרִים אוֹ חֲרָבוֹת מְלֻטָּשׁוֹת. בִּמְקוֹם חֶרֶב אוֹ חֲנִית, לָקַח שַׁמְגַּר כְּלִי עֲבוֹדָה חַקְלָאִי פָּשׁוּט – מַלְמַד הַבָּקָר.

מָה הוּא מַלְמַד הַבָּקָר? חֲכָמֵינוּ, וּבָהֶם רַשִׁ"י, מְצֻדַּת דָּוִד וְרַלְבַּ"ג, מִסְבִּירִים כִּי מַלְמַד הַבָּקָר (הַנִּקְרָא גַּם מַרְדֵּעַ, דָּרְבָן אוֹ אֲגוֹלִיָּא) הוּא מַקֵּל עֵץ אָרֹךְ וְחַד שֶׁבּוֹ מִשְׁתַּמְשִׁים הַחַקְלָאִים כְּדֵי לְכַוֵּן וּלְהַנְהִיג אֶת הַשּׁוֹרִים בִּזְמַן חֲרִישַׁת הַשָּׂדֶה.

בְּאֶמְצָעוּת מַקֵּל חַד זֶה בִּלְבַד, יָצָא שַׁמְגַּר בֶּן עֲנָת לַקְּרָב נֶגֶד הָאוֹיְבִים הַפְּלִשְׁתִּים. בִּגְבוּרָה גְּדוֹלָה וּבְעֶזְרַת מַלְמַד הַבָּקָר, הִכָּה שַׁמְגַּר שֵׁשׁ מֵאוֹת אִישׁ מִן הַפְּלִשְׁתִּים. כְּפִי שֶׁנֶּאֱמַר בְּסֵפֶר שׁוֹפְטִים: "וַיַּךְ אֶת פְּלִשְׁתִּים שֵׁשׁ־מֵאוֹת אִישׁ בְּמַלְמַד הַבָּקָר וַיּוֹשַׁע גַּם־הוּא אֶת־יִשְׂרָאֵל".

לְפִי מְפָרְשֵׁי הַתַּנָּךְ, שַׁמְגַּר בֶּן עֲנָת שָׁפַט אֶת יִשְׂרָאֵל בְּמֶשֶׁךְ שָׁנָה אַחַת בִּלְבַד, בְּאוֹתָהּ שָׁנָה שֶׁבָּהּ נִפְטַר אֵהוּד. אַף עַל פִּי שֶׁתְּשׁוּעָתוֹ שֶׁל שַׁמְגַּר לֹא הָיְתָה תְּשׁוּעָה שְׁלֵמָה לִזְמַן רַב כְּמוֹ תְּשׁוּעַת אֵהוּד, וְאַחֲרֵי מוֹת אֵהוּד חָזְרוּ בְּנֵי יִשְׂרָאֵל לַעֲשׂוֹת אֶת הָרַע בְּעֵינֵי ה', מִכָּל מָקוֹם עָשָׂה שַׁמְגַּר תְּשׁוּעָה וְהוֹשִׁיעַ אֶת עַם יִשְׂרָאֵל מֵהַפְּלִשְׁתִּים בְּעֵת צָרָתָם.

💡 גַּם בְּאֶמְצָעִי פָּשׁוּט וּבֶאֱמוּנָה רַבָּה אֶפְשָׁר לַעֲשׂוֹת מַעֲשִׂים גְּדוֹלִים וּלְהוֹשִׁיעַ אֶת זוּלָתֵנוּ.
