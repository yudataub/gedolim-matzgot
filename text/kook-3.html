הָרַב קוּק וְהַסְּלִיחָה הַגְּדוֹלָה

יוֹם אֶחָד חָזַר הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק מִטֶּקֶס בְּרִית מִילָה חֲגִיגִי שֶׁנֶּעֶרְכָה בָּעִיר הָעַתִּיקָה. יַחַד עִם הָרַב צָעֲדוּ עַשְׂרוֹת תַּלְמִידִים וַאֲנָשִׁים רַבִּים שֶׁלִּוּוּ אוֹתוֹ בְּכָבוֹד רַב בַּדֶּרֶךְ. וְהִנֵּה, בְּאֶמְצַע הַדֶּרֶךְ, הִתְנַפֵּל עַל הָרַב קוּמֶץ שֶׁל קַנָּאִים, אֲנָשִׁים שֶׁהִתְנַגְּדוּ לַדֵּעוֹת שֶׁלּוֹ. הַקַּנָּאִים הָאֵלֶּה עָשׂוּ מַעֲשֶׂה רַע וּמַעֲלִיב מְאֹד: הֵם שָׁפְכוּ עַל הָרַב מֵי בִיּוּב מְטוֹנָפִים. הָרַב הַצַּדִּיק הִתְלַכְלֵךְ כֻּלּוֹ מֵהַמַּיִם הַמְּזֹהָמִים שֶׁנִּשְׁפְּכוּ עָלָיו.

הָאֲוִירָה סְבִיב הָרַב הָיְתָה סוֹעֶרֶת מְאֹד וּמְלֵאָה כַּעַס. הָאֲנָשִׁים וְהַתַּלְמִידִים שֶׁלִּוּוּ אֶת הָרַב הָיוּ הֲמוּמִים וְכֹעֲסִים מְאֹד, וְהֵם רָצוּ מִיָּד לִתְפֹּס אֶת הַפּוֹשְׁעִים שֶׁעָשׂוּ אֶת הַמַּעֲשֶׂה הַמְּבַזֶּה הַזֶּה. לְאַחַר מִכֵּן הִגִּיעַ הָרַב קוּק לְבֵיתוֹ. אֲנָשִׁים רַבִּים וַחֲשׁוּבִים מְאֹד הִגִּיעוּ אֶל בֵּית הָרַב כְּדֵי לְהַגִּיד לוֹ שֶׁהֵם מִתְנַגְּדִים בְּתֹקֶף לַמַּעֲשֶׂה הַנּוֹרָא שֶׁנַּעֲשָׂה לוֹ, וְשֶׁהֵם מִצְטַעֲרִים מְאֹד עַל הָאֵרוּעַ הַמַּבְהִיל, הַמְּבִיךְ וְהַמְּבַזֶּה שֶׁקָּרָה.

בֵּין כָּל הַמְּבַקְרִים הַחֲשׁוּבִים שֶׁבָּאוּ לְבֵית הָרַב, הִגִּיעַ גַּם הַיּוֹעֵץ הַמִּשְׁפָּטִי שֶׁל מֶמְשֶׁלֶת הַמַּנְדָּט הַבְּרִיטִי בְּאֶרֶץ יִשְׂרָאֵל בְּאוֹתָם יָמִים, שֶׁשְּׁמוֹ נֹורְמַן בֶּנְטְוִיץ'. הַיּוֹעֵץ הַמִּשְׁפָּטִי בֶּנְטְוִיץ' הִצִּיעַ לָרַב קוּק לְהַגִּישׁ תְּלוֹנָה רִשְׁמִית נֶגֶד אוֹתָם מַבְצֵעֵי הַמַּעֲשֶׂה. הוּא אָמַר לָרַב כִּי הוּא יִדְאַג אִישִׁית שֶׁעוֹנְשָׁם שֶׁל הַפּוֹשְׁעִים יִהְיֶה קָשֶׁה וְשֶׁהֵם יְגֹרְשׁוּ לַלְּחָלוּטִין מִגְּבוּלוֹת אֶרֶץ יִשְׂרָאֵל.

אַךְ הָרַב קוּק הֵשִׁיב לוֹ בְּאֲצִילוּת וּבְאַהֲבָה גְּדוֹלָה. שָׁעָה אַחַת בִּלְבַד לְאַחַר שֶׁאֵרַע הַמַּעֲשֶׂה הַמַּעֲלִיב, אָמַר הָרַב לַיּוֹעֵץ הַמִּשְׁפָּטִי: "אֵין לִי עֵסֶק עִם תְּלוֹנוֹת, מִשְׁפָּטִים אוֹ עֳנָשִׁים. עַל אַף מַה שֶּׁהֵם עָשׂוּ לִי, אֲנִי מָלֵא אַהֲבָה אֲלֵיהֶם. אֲנִי אֲפִילוּ מוּכָן לְחַבְּקָם מֵרֹב הָאַהֲבָה שֶׁיֵּשׁ בְּלִבִּי כְּלַפֵּיהֶם. וְלָכֵן אֲנִי מְבַקֵּשׁ מִמְּךָ שֶׁלֹּא תַּעֲנִישֵׁם".

הַדְּבָרִים הַמְּרַגְּשִׁים הָאֵלֶּה נֶאֶמְרוּ עַל יְדֵי הָרַב קוּק שָׁעָה אַחַת בִּלְבַד לְאַחַר שֶׁהַמַּעֲשֶׂה הִתְרַחֵשׁ. הָרַב הֶרְאָה לְכָל הַמְּלַוִּים וְהַמְּבַקְרִים עַד כַּמָּה הַלֵּב שֶׁלּוֹ מָלֵא בְּאַהֲבָה עֲמֻקָּה לְכָל אָדָם, וְכַמָּה הוּא מִתְרַחֵק מִשִּׂנְאָה, מִמִּשְׁפָּטִים וּמֵעֳנָשִׁים. הַהַנְהָגָה הַמֻּפְלָאָה הַזֹּאת לִמְדָה אֶת כָּל סְבִיבָתוֹ אֶת כֹּחָהּ שֶׁל הָאַהֲבָה וְהַסְּלִיחָה.
