הַגָּאוֹן הַצָּעִיר שֶׁלֹּא הִפְסִיק לִלְמֹד

רַבִּי שַׁבְּתַאי הַכֹּהֵן נוֹלַד בָּעֲיָרָה אָמְסְטִיבּוֹבָה שֶׁבְּפוֹלִין. כְּבָר מֵאָז שֶׁהָיָה יֶלֶד צָעִיר מְאֹד, כֻּלָּם רָאוּ וְיָדְעוּ שֶׁהוּא עִלּוֹי מְיֻחָד בְּמִינוֹ. אָבִיו, רַבִּי מֵאִיר הַכֹּהֵן, לִמֵּד אוֹתוֹ תּוֹרָה בְּאַהֲבָה רַבָּה, וְרַבִּי שַׁבְּתַאי הַקָּטָן שָׁקַד עַל לִמּוּדוֹ בְּשִׂמְחָה. כְּשֶׁהִגִּיעַ לְגִיל שְׁתֵּים עֶשְׂרֵה בִּלְבַד, נִשְׁלַח רַבִּי שַׁבְּתַאי מִבֵּיתוֹ כְּדֵי לִלְמֹד בִּישִׁיבָה רְחוֹקָה. הוּא הִמְשִׁיךְ אֶת לִמּוּדוֹ, וּמִשָּׁם נָסַע לָעִיר לוּבְלִין וְלָמַד תּוֹרָה מִפִּיהֶם שֶׁל גְּדוֹלֵי הָרַבָּנִים. לְאַחַר מִכֵּן עָבַר לִחְיוֹת בָּעִיר וִילְנָה, וְשָׁם יָשַׁב בְּבֵית הַמִּדְרָשׁ וְלָמַד תּוֹרָה יוֹמָם וָלַיְלָה לְלֹא הַפְסָקָה.

כְּשֶׁהָיָה רַבִּי שַׁבְּתַאי בֶּן עֶשְׂרִים וְאַרְבַּע שָׁנִים בִּלְבַד, בִּשְׁנַת הַ'תַּ"ו, הוּא כְּבָר הִדְפִּיס אֶת סִפְרוֹ הַגָּדוֹל "שִׂפְתֵי כֹהֵן" עַל חֵלֶק "יוֹרֶה דֵּעָה" שֶׁבַּשֻּׁלְחָן עָרוּךְ. כָּל גְּדוֹלֵי הַדּוֹר הִתְפַּלְּאוּ מְאֹד מֵחָכְמָתוֹ הָעֲמֻקָּה וְאָמְרוּ: אֵיךְ אֶפְשָׁר שֶׁחָכָם צָעִיר כָּל כָּךְ יִהְיֶה גָּאוֹן כָּל כָּךְ גָּדוֹל! אָבִיו הָאַהוּב, רַבִּי מֵאִיר, זָכָה לִרְאוֹת אֶת הַסֵּפֶר הַנִּפְלָא שֶׁל בְּנוֹ וְאַף כָּתַב לִכְבוֹדוֹ שִׁיר מְיֻחָד. רַבִּי שַׁבְּתַאי זָכָה לְכִנּוּי הַמְפֻרְסָם "הַשָּׁ"ךְ", עַל שֵׁם סִפְרוֹ "שִׂפְתֵי כֹהֵן" – פֵּרוּשׁ חָשׁוּב עַל הַשֻּׁלְחָן עָרוּךְ, בְּחֶלְקֵי "יוֹרֶה דֵּעָה" וְ"חוֹשֶׁן מִשְׁפָּט". סִפְרוֹ הַקָּדוֹשׁ שֶׁל הַשָּׁ"ךְ נִדְפַּס מֵאָז בְּכָל שֻׁלְחָן עָרוּךְ, וְכָל רַב וּפוֹסֵק הֲלָכָה לוֹמֵד אוֹתוֹ עַד הַיּוֹם הַזֶּה. בִּשְׁנַת הַ'תִּי"א, כְּשֶׁהָיָה עֲדַיִן צָעִיר לְיָמִים, מִנּוּ אוֹתוֹ חַכְמֵי הָעִיר לִהְיוֹת דַּיָּן בְּבֵית הַדִּין שֶׁל וִילְנָה.

אֲבָל לְאַחַר מִכֵּן הִגִּיעוּ יָמִים קָשִׁים עַל עַם יִשְׂרָאֵל. בַּשָּׁנִים הַ'תַּ"ח וְהַ'תַּ"ט פָּרְעוּ הַקּוֹזָקִים בִּקְהִילּוֹת יִשְׂרָאֵל, וּבִשְׁנַת הַ'תַּטִּ"ו כָּבְשׁוּ הָרוּסִים אֶת הָעִיר וִילְנָה. רַבִּי שַׁבְּתַאי הַכֹּהֵן נֶאֱלַץ לִבְרֹחַ מֵעִירוֹ, וְהוּא נָדַד מִמָּקוֹם לְמָקוֹם בְּצַעַר וּבְסֵבֶל רָב. אַךְ גַּם בְּתוֹךְ הַנְּדוֹדִים הַקָּשִׁים וְהַצָּרוֹת הָרַבּוֹת, הוּא לֹא הִפְסִיק לִלְמֹד תּוֹרָה וּלְחַבֵּר סְפָרִים. עַל הַצָּרוֹת הַנּוֹרָאוֹת שֶׁעָבְרוּ עַל עַם יִשְׂרָאֵל בְּאוֹתָן שָׁנִים הוּא כָּתַב אֶת "מְגִילַּת עֵיפָה", וְגַם חִבֵּר תְּפִלּוֹת סְלִיחוֹת מְיֻחָדוֹת שֶׁנֶּאֶמְרוּ בְּכָף בְּסִיוָן לְזֵכֶר הַקְּהִילּוֹת שֶׁנֶּחֶרְבוּ.

לְבַסּוֹף הִגִּיעַ רַבִּי שַׁבְּתַאי לָעִיר הֲלִישׁוֹי שֶׁבְּמוֹרַבְיָה. תּוֹשָׁבֵי הָעִיר קִבְּלוּ אוֹתוֹ בְּכָבוֹד וּמִנּוּ אוֹתוֹ לִהְיוֹת רַב הַקְּהִילָה. שָׁם נִפְטַר רַבִּי שַׁבְּתַאי וְהוּא אָדָם צָעִיר, בֶּן אַרְבָּעִים וְאַחַת שָׁנִים בִּלְבַד. אֲבָל בַּשָּׁנִים הַקְּצָרוֹת הָאֵלֶּה הוּא הִסְפִּיק לְחַבֵּר סְפָרִים רַבִּים וּחֲשׁוּבִים, שֶׁכָּל עַם יִשְׂרָאֵל מִמְשִׁיךְ לִלְמֹד מֵהֶם עַד הַיּוֹם.

💡 גַּם בִּזְמַנִּים קָשִׁים וּבְתוֹךְ נְדוֹדִים אֶפְשָׁר לְהַמְשִׁיךְ לִלְמֹד תּוֹרָה, וְגַם צָעִיר יָכוֹל לַעֲשׂוֹת דְּבָרִים גְּדוֹלִים כְּשֶׁהוּא מְנַצֵּל כָּל רֶגַע.
