הַכִּפָּה מִבַּד הָאַדְמוֹ"ר הַצֶּמַח צֶדֶק

רַב רְפָאֵל הָיָה חָסִיד נֶאֱמָן וּמָסוּר מְאֹד לְרַבּוֹ הָאַדְמוֹ"ר הַצֶּמַח צֶדֶק, הָרַבִּי הַקָּדוֹשׁ מֵחַבָּ"ד. בְּכָל עֵת הָיָה רַב רְפָאֵל מַקְשִׁיב בְּשִׁקָּקָה רַבָּה לְדִבְרֵי הַתּוֹרָה הָעֲמֻקִּים שֶׁל הָרַבִּי. הוּא הָיָה חוֹזֵר עֲלֵיהֶם שׁוּב וָשׁוּב, וּמַעֲמִיק בָּהֶם אֶת מַחְשְׁבוֹתָיו בְּאַהֲבָה גְּדוֹלָה. גַּם תְּפִלָּתוֹ שֶׁל רַב רְפָאֵל הָיְתָה בְּכַוָּנָה עֲמֻקָּה וּבְהִתְלַהֲבוּת רַבָּה, כְּדַרְכָּם שֶׁל הַחֲסִידִים הָרִאשׁוֹנִים.

יוֹם אֶחָד הִגִּיעָה אֶל רַב רְפָאֵל וְאֶל שְׁאָר הַחֲסִידִים הַשְּׁמוּעָה הַכּוֹאֶבֶת עַל פְּטִירָתוֹ שֶׁל הָרַבִּי הָאָהוּב. כְּשֶׁשָּׁמַע זֹאת רַב רְפָאֵל, פָּרַץ בִּבְכִי וּבְצַעַר גָּדוֹל מְאֹד. הַחֲסִידִים הָאֲבֵלִים הִתְאַסְפוּ יַחְדָּו, וְנִסּוּ לַחֲקֹק בְּלִבָּם אֶת כָּל הַזִּכְרוֹנוֹת הַיְקָרִים מֵהָרַבִּי. בְּמַהֲלַךְ הַמִּפְגָּשׁ, הוֹצִיא אֶחָד מֵהַחֲסִידִים אֶת מְעִילוֹ שֶׁל הָרַבִּי הַצֶּמַח צֶדֶק, וְהִצִּיעַ לְכָל הַחֲסִידִים לַחֲלֵק אֶת הַמְּעִיל בֵּינֵיהֶם לְמַזְכֶּרֶת. גַּם רַב רְפָאֵל זָכָה לְקַבֵּל חֲתִיכַת בַּד קְטַנָּה מִן הַמְּעִיל הַמְּיֻחָד, וְיַחַד עִמָּהּ גַּם כַּפְתּוֹר קָטָן.

בַּתְּחִלָּה לֹא יָדַע רַב רְפָאֵל מָה יַעֲשֶׂה בַּבַּד הַקָּטָן וּבַכַּפְתּוֹר שֶׁקִּבֵּל. הוּא מִשֵּׁשׁ אוֹתָם בְּיָדָיו, הִבִּיט בָּהֶם בְּהַעֲרָצָה, וְנִזְכַּר בְּכָל אוֹתָן הַשָּׁנִים הַיְפֹת שֶׁבָּהֵן לָמַד אֶת דִּבְרֵי הַתּוֹרָה שֶׁל הָרַבִּי. לְפֶתַע נִצְנַץ בְּלִבּוֹ רַעְיוֹן נִפְלָא. הוּא מִהֵר, יָצָא מִבֵּיתוֹ וּפָסַע בִּזְרִיזוּת רַבָּה אֶל בֵּיתָהּ שֶׁל פֶּרְל זְלָאטָה, שֶׁבֵּן קָטָן נוֹלַד לָהּ חֳדָשִׁים סְפוּרִים קוֹדֶם לָכֵן, וּשְׁמוֹ אַבְרָהָם יִצְחָק.

כְּשֶׁנִּכְנַס רַב רְפָאֵל אֶל הַבַּיִת, אָמַר לָהּ: "פֶּרְל זְלָאטָה, בִּתִּי הַיְקָרָה, רְאִי בַּמֶּה זָכִיתִי. קְחִי נָא אֶת הַבַּד וְאֶת הַכַּפְתּוֹר וְתִפְרִי אוֹתָם בִּבְגָדָיו שֶׁל אַבְרָהָם יִצְחָק הַקָּטָן. וַדַּאי סְגֻלָּה גְּדוֹלָה יֵשׁ בְּכָךְ לְהוֹסִיף קְדֻשָּׁה וְיִרְאַת שָׁמַיִם בַּתִּינוֹק." פֶּרְל זְלָאטָה שָׂמְחָה מְאֹד, וּבְהִתְרַגְּשׁוּת רַבָּה קִבְּלָה אֶת הַמַּתָּנָה הַיְקָרָה מִיָּדָיו שֶׁל רַב רְפָאֵל.

אַחֲרֵי כַּמָּה יָמִים חָזַר רַב רְפָאֵל לְבַקֵּר בְּבֵיתָהּ שֶׁל פֶּרְל זְלָאטָה. הוּא נִכְנַס לַבַּיִת וְהֵרִים בְּאַהֲבָה אֶת אַבְרָהָם יִצְחָק הַקָּטָן. הוּא הִבִּיט בּוֹ וְאָמַר: "אוֹוּ, אַבְרָהָם יִצְחָק, מָה אִמָּא שָׂמָה לְךָ!" עַל רֹאשׁוֹ שֶׁל אַבְרָהָם יִצְחָק הַתִּינוֹק הָיְתָה מוּנַחַת כִּפָּה יָפָה, שֶׁהֵכִינָה הָאֵם מֵהַבַּד הַמְּיֻחָד שֶׁל הָאַדְמוֹ"ר הַצֶּמַח צֶדֶק.

בְּכָל שְׁנוֹת יַלְדוּתוֹ הָיָה אַבְרָהָם יִצְחָק הַקָּטָן לוֹבֵשׁ אֶת הַכִּפָּה הַזּוֹ עַל רֹאשׁוֹ. גַּם בַּלֵּילוֹת, כְּשֶׁהָיָה יָשֵׁן עַל מִטָּתוֹ, אִם הָיְתָה הַכִּפָּה נוֹפֶלֶת מֵרֹאשׁוֹ, הוּא הָיָה מִתְעוֹרֵר מִיָּד כְּדֵי לְהַחֲזִירָהּ לִמְקוֹמָהּ.
