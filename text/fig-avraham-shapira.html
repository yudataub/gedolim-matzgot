הַמְּעִיל וּבִרְכַּת הַהַצְלָחָה

הָרַב אַבְרָהָם שָׁפִירָא, שֶׁרַבִּים כִּנּוּ אוֹתוֹ בְּאַהֲבָה בְּשֵׁם הָרַב אַבְרֹום, הָיָה הָרַב הָרָאשִׁי לְיִשְׂרָאֵל וְגַם רֹאשׁ יְשִׁיבַת "מֶרְכַּז הָרַב" בִּירוּשָׁלַיִם. הוּא הָיָה תַּלְמִיד חָכָם גָּדוֹל מְאֹד וּמַנְהִיג חָשׁוּב, שֶׁכָּל מִי שֶׁפָּגַשׁ אוֹתוֹ הִרְגִּישׁ אֶת הַחֹם, אֶת הַחִיּוּךְ וְאֶת אַהֲבַת הַתּוֹרָה שֶׁלּוֹ.

כַּאֲשֶׁר הִגִּיעַ הָרַב אַבְרֹום לְגִיל זִקְנָה, גּוּפוֹ נִהְיָה חַלָּשׁ וְקָשֶׁה הָיָה לוֹ לַעֲשׂוֹת פְּעֻלּוֹת יוֹמְיוֹמִיּוֹת פְּשׁוּטוֹת בְּכֹחוֹת עַצְמוֹ. בִּפְרָט, הָיָה לוֹ קוֹשִׁי גָּדוֹל לְהַכְנִיס אֶת יָדָיו לְתוֹךְ שַׁרְווּלֵי הַמְּעִיל שֶׁלּוֹ, מִפְּנֵי שֶׁתְּנוּעַת הַזְּרוֹעוֹת הָיְתָה קָשָׁה עָלָיו מְאֹד.

בִּישִׁיבַת "מֶרְכַּז הָרַב", כַּאֲשֶׁר הָרַב אַבְרֹום הָיָה מַגִּיעַ לְבֵית הַמִּדְרָשׁ לִפְנֵי הַתְּפִלָּה וְגַם כַּאֲשֶׁר סִיֵּם אֶת הַתְּפִלָּה, הָיוּ תַּלְמִידִים צְעִירִים וּמְסוּרִים נִגָּשִׁים אֵלָיו בְּשִׂמְחָה כְּדֵי לַעֲזֹר לוֹ לִלְבֹּשׁ אֶת הַמְּעִיל. הַתַּלְמִידִים רָאוּ בְּכָךְ זְכוּת גְּדוֹלָה מְאֹד, כִּי הֵם רָצוּ לְקַיֵּם אֶת מִצְוַת הַתּוֹרָה הַחֲשׁוּבָה שֶׁל "וְהָדַרְתָּ פְּנֵי זָקֵן" וּלְכַבֵּד אֶת רַבָּם הַיָּקָר.

יוֹם אֶחָד, לִפְנֵי שֶׁהִתְחִילָה הַתְּפִלָּה בַּיְשִׁיבָה, נִגַּשׁ תַּלְמִיד אֶחָד אֶל הָרַב אַבְרֹום כְּדֵי לַעֲזֹר לוֹ לִלְבֹּשׁ אֶת הַמְּעִיל שֶׁלּוֹ. הַתַּלְמִיד הָרִאשׁוֹן הִתְחִיל בַּמְּלָאכָה, וְכֵיוָן שֶׁעָזַר בִּזְרִיזוּת, הוּא כְּבָר הִכְנִיס אֶת יָדוֹ הַיְמָנִית שֶׁל הָרַב לְתוֹךְ שַׁרְווּל יָמִין שֶׁל הַמְּעִיל. כָּעֵת עָמַד הַתַּלְמִיד לַעֲזֹר לָרַב לְהַכְנִיס גַּם אֶת יָדוֹ הַשְּׂמָאלִית לְתוֹךְ הַשַּׁרְווּל הַשֵּׁנִי.

בְּדִיּוּק בְּאוֹתוֹ רֶגַע מַמָּשׁ, נִגַּשׁ אֶל הָרַב תַּלְמִיד אַחֵר. הַתַּלְמִיד הַשֵּׁנִי הָיָה נִרְגָּשׁ וּמָתוּחַ מְאֹד, מִפְּנֵי שֶׁבְּאוֹתוֹ יוֹם עָמַד לַעֲשׂוֹת מִבְחָן נְהִיגָה, מָה שֶׁנִּקְרָא בְּשָׂפָה פְּשׁוּטָה מִבְחָן טֶסְט. הַתַּלְמִיד הַשֵּׁנִי רָצָה מְאֹד לְהַצְלִיחַ בַּמִּבְחָן הַחָשׁוּב, וְהוּא בָּא לְבַקֵּשׁ מֵהָרַב אַבְרֹום שֶׁיִּתְפַּלֵּל עָלָיו וִיבָרֵךְ אוֹתוֹ שֶׁיַּעֲבֹר אֶת הַמִּבְחָן בְּהַצְלָחָה רַבָּה.

הָרַב אַבְרֹום הִבִּיט בִּשְׁנֵי הַתַּלְמִידִים בְּחִיּוּךְ חָם, מָלֵא אֲהָבָה וְחָכְמָה. הוּא פָּנָה בִּנְעִימוּת אֶל הַתַּלְמִיד הָרִאשׁוֹן, שֶׁכְּבָר עָזַר לוֹ לָשִׂים אֶת שַׁרְווּל יָמִין, וְאָמַר לוֹ: "רֶגַע, אוּלַי אַתָּה מוֹכָן לְוַתֵּר לַתַּלְמִיד הַשֵּׁנִי שֶׁעוֹשֶׂה הַיּוֹם אֶת הַטֶּסְט, אֶת הַמִּבְחָן? הֲרֵי הוּא רוֹצֶה מְאֹד לְהַצְלִיחַ בַּמִּבְחָן שֶׁלּוֹ. אִם הוּא רוֹצֶה לְהַצְלִיחַ בַּמִּבְחָן, אָז תַּעֲזֹר לוֹ שֶׁהוּא יְקַיֵּם אֶת מִצְוַת 'וְהָדַרְתָּ פְּנֵי זָקֵן', וְהוּא יַעֲזֹר לִי לָשִׂים אֶת הַשַּׁרְווּל הַשֵּׁנִי. בִּזְכוּת זֶה שֶׁהוּא יַעֲשֶׂה מַעֲשֶׂה טוֹב, הוּא יִהְיֶה קָדוֹשׁ יוֹתֵר, וּבִזְכוּת זֶה הוּא יִזְכֶּה גַּם לְהַצְלִיחַ בַּטֶּסְט".

הָרַב אַבְרֹום לִמֵּד אֶת הַתַּלְמִידִים מָה הִיא הַדֶּרֶךְ הַנְּכוֹנָה לִזְכּוֹת בִּבְרָכָה וּבְהַצְלָחָה. הוּא הֶרְאָה לָהֶם כִּי עֶזְרָה לַזּוּלָת וְקִיּוּם מִצְווֹת מְבִיאִים מַעֲשִׂים טוֹבִים שֶׁמְּבָרְכִים אֶת הָאָדָם בְּכָל אֲשֶׁר יַפְנֶה.
