שַׂק הַחִטִּים שֶׁל רַבִּי פִּנְחָס בֶּן יָאִיר

לִפְנֵי שָׁנִים רַבּוֹת, הִגִּיעוּ שְׁנֵי עֲנִיִּים אֶל הַכְּפָר שֶׁבּוֹ גָּר רַבִּי פִּנְחָס בֶּן יָאִיר. הָעֲנִיִּים עָבְרוּ מִבַּיִת לְבַיִת וּבִקְשׁוּ צְדָקָה מֵאַנְשֵׁי הַכְּפָר. אַנְשֵׁי הַכְּפָר הַטּוֹבִים נָתְנוּ לָהֶם מַתָּנוֹת שׁוֹנוֹת, וּבֵין הַדְּבָרִים שֶׁאָסְפוּ הָיָה גַּם שַׂק גָּדוֹל וְכָבֵד מָלֵא בְּחִטִּים. הָעֲנִיִּים הִמְשִׁיכוּ לָלֶכֶת מִבַּיִת לְבַיִת, אַךְ מֵרֹב שֶׁהַשַּׂק הָיָה כָּבֵד, הָיָה לָהֶם קָשֶׁה מְאֹד לְהַמְשִׁיךְ וְלָשֵׂאת אוֹתוֹ עַל גַּבָּם. כַּאֲשֶׁר הִגִּיעוּ אֶל בֵּיתוֹ שֶׁל רַבִּי פִּנְחָס בֶּן יָאִיר, הֵם נָקְשׁוּ בַּדֶּלֶת, וּבִקְשׁוּ מִמֶּנּוּ רְשׁוּת לְהַנִּיחַ אֶצְלוֹ אֶת שַׂק הַחִטִּים הַכָּבֵד. הֵם אָמְרוּ לוֹ שֶׁיָּבוֹאוּ לָקַחַת אֶת הַשַּׂק כְּבָר לְמַחָרָת. רַבִּי פִּנְחָס בֶּן יָאִיר הִסְכִּים בְּשִׂמְחָה וְשָׁמַר עַל הַשַּׂק בְּבֵיתוֹ.

אַךְ הַיָּמִים עָבְרוּ, וְהָעֲנִיִּים לֹא חָזְרוּ. הֵם לֹא בָּאוּ לְמַחָרָת, גַּם לֹא לְמַחֳרָתַיִם, וְגַם לֹא כַּעֲבֹר שָׁבוּעַ אוֹ שְׁבוּעַיִם. רַבִּי פִּנְחָס בֶּן יָאִיר הִבִּיט בְּשַׂק הַחִטִּים וְרָאָה שֶׁהַחִטִּים הָיוּ עֲלוּלוֹת לְהִתְיַבֵּשׁ וּלְהֵהָרֵס. הוּא לֹא רָצָה שֶׁהָעֲנִיִּים יַפְסִידוּ אֶת מַה שֶּׁקִּבְלוּ. מָה עָשָׂה רַבִּי פִּנְחָס בֶּן יָאִיר? הוּא יָצָא אֶל הַשָּׂדֶה, חָפַר בָּאֲדָמָה, וְשָׁתַל בָּהּ אֶת הַחִטִּים מִתּוֹךְ הַשַּׂק. לְאַחַר מִכֵּן הִתְפַּלֵּל אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיֵּרֵד גֶּשֶׁם. הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַע אֶת תְּפִלָּתוֹ, הַגֶּשֶׁם יָרַד, וּמִן הָאֲדָמָה צָמְחוּ חִטִּים רַבּוֹת וִיפֵיוֹת.

כַּאֲשֶׁר הִבְשִׁילוּ הַחִטִּים בַּשָּׂדֶה, יָצָא רַבִּי פִּנְחָס בֶּן יָאִיר וְקָצַר אוֹתָן. סוּסִיקוֹ, עַבְדּוֹ הַנֶּאֱמָן שֶׁל רַבִּי פִּנְחָס בֶּן יָאִיר, סָחַב אֶת כָּל עֲרֵמוֹת הַחִטִּים הַחֲדָשׁוֹת וְהִכְנִיס אוֹתָן אֶל הָאָסָם. וְכָךְ עָשׂוּ רַבִּי פִּנְחָס בֶּן יָאִיר וְסוּסִיקוֹ שָׁנָה אַחַר שָׁנָה. הֵם לֹא עָשׂוּ זֹאת רַק שָׁנָה אַחַת, גַּם לֹא שְׁנָתַיִם, וְגַם לֹא שָׁלֹשׁ שָׁנִים – אֶלָּא שֶׁבַע שָׁנִים תְּמִימוֹת! בְּכָל שָׁנָה זָרְעוּ אֶת הַחִטִּים, קָצְרוּ אוֹתָן, וְאָסְפוּ אֶת כָּל אֲגֻדּוֹת הַחִטִּים הָרַבּוֹת אֶל הָאָסָם.

כַּעֲבֹר שֶׁבַע שָׁנִים, הִגִּיעוּ פִּתְאֹם אוֹתָם שְׁנֵי עֲנִיִּים חֲזָרָה אֶל הַכְּפָר. הֵם נָקְשׁוּ שׁוּב בַּדֶּלֶת שֶׁל רַבִּי פִּנְחָס בֶּן יָאִיר, וְשָׁאֲלוּ אוֹתוֹ: "הֵיכָן שַׂק הַחִטִּים שֶׁהִשְׁאַרְנוּ אֶצְלְךָ?" רַבִּי פִּנְחָס בֶּן יָאִיר חִיֵּךְ אֲלֵיהֶם בְּאַהֲבָה וּבִקֵּשׁ מֵהֶם לָלֶכֶת וּלְהָבִיא חֲמוֹרִים, גְּמַלִּים וְסוּסִים. הָעֲנִיִּים תָּמְהוּ מְאֹד וְלֹא הֵבִינוּ מַדּוּעַ יֵשׁ צֹרֶךְ בִּבְהֵמוֹת רַבּוֹת כָּל כָּךְ בִּשְׁבִיל שַׂק חִטִּים אֶחָד קָטָן.

אַךְ כַּאֲשֶׁר הוֹבִיל אוֹתָם רַבִּי פִּנְחָס בֶּן יָאִיר אֶל הָאָסָם, וְהֶרְאָה לָהֶם אֶת הָאָסָם הַמָּלֵא לְגַמְרֵי בַּאֲגֻדּוֹת חִטִּים רַבּוֹת, הָעֲנִיִּים הֻכּוּ בְּתַדְהֵמָה. הֵם לֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם! הָעֲנִיִּים הַנִּרְגָּשִׁים בִּקְשׁוּ לָתֵת לְרַבִּי פִּנְחָס בֶּן יָאִיר חֲצִי מֵאֲגֻדּוֹת הַחִטִּים בְּתוֹדָה עַל כָּל הָעֲבֹדָה הַקָּשָׁה שֶׁעָשָׂה. אַךְ רַבִּי פִּנְחָס בֶּן יָאִיר סֵרֵב לְהִתְחַלֵּק בַּחִטִּים. הוּא נָתַן לָהֶם בְּשִׂמְחָה אֶת כָּל עֲרֵמוֹת הַחִטִּים לְלֹא כָּל תְּמוּרָה וְשֶׁלֹּא עַל מְנָת לְקַבֵּל פְּרָס, מִתּוֹךְ חֶסֶד טָהוֹר וַאֲמִתִּי.

💡 כְּשֶׁאָנוּ עוֹשִׂים חֶסֶד מִכָּל הַלֵּב וְשֶׁלֹּא עַל מְנָת לְקַבֵּל פְּרָס, אָנוּ מְבִיאִים בְּרָכָה וְשִׂמְחָה גְּדוֹלָה לָעוֹלָם.
