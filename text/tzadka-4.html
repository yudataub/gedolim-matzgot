הַהֲלִיכָה בַּשֶּׁלֶג הָעָמֹק לַיְשִׁיבָה

בִּימֵי הַחֹרֶף הַקָּרִים וְהַסּוֹעֲרִים בִּירוּשָׁלַיִם, כְּשֶׁהָרַב יְהוּדָה צְדָקָה זֵכֶר צַדִּיק לִבְרָכָה הָיָה נַעַר צָעִיר וְתַלְמִיד בַּיְשִׁיבָה, הוּא לָמַד תּוֹרָה בְּשִׁקְדָנוּת גְּדוֹלָה וּבְאַהֲבָה רַבָּה. אֲפִילוּ בִּימוֹת הַחֹרֶף הַקָּרִים וְהַמּוּשְׁלָגִים בְּיוֹתֵר, הוּא לֹא הֶחְסִיר מֵעוֹלָם וְלֹא אֵחַר, אֶלָּא הִקְפִּיד תָּמִיד לָבוֹא אֶל בֵּית הַיְשִׁיבָה בַּשָּׁעָה הַקְּבוּעָה.

יוֹם אֶחָד שָׁרַר בַּחוּץ קוֹר עַז מְאֹד. פְּתִיתֵי שֶׁלֶג רַבִּים יָרְדוּ מֵהַשָּׁמַיִם בְּלִי הַפְסָקָה וְנֶעֶרְמוּ לְמַכְבִּיר בְּכַמּוּיוֹת גְּדוֹלוֹת לְאֹרֶךְ כָּל הַדֶּרֶךְ הָאֲרֻכָּה שֶׁהוֹלִיכָה מִדִּירַת הוֹרָיו וְעַד אֶל בֵּית הַיְשִׁיבָה. מֶזֶג הָאֲוָויר בַּחוּץ הָיָה סַגְרִירִי, קָפוּא וְקָשֶׁה מְאֹד. בְּתוֹךְ דִּירַת הוֹרָיו הָיָה נָעִים, חַם וּמֻסָּק, וְלֹא הָיָה קַל בִּכְלָל לָצֵאת מֵהַבַּיִת הַחַם וְהַמָּגֵן אֶל הַקּוֹר הַגָּדוֹל וְאֶל הַסּוּפָה שֶׁהָיוּ בַּחוּץ.

אַךְ הַנַּעַר יְהוּדָה לֹא הָיָה מִתְפַּנֵּק בְּעִנְיָנִים מִסּוּג זֶה. נַפְשׁוֹ חָשְׁקָה בַּתּוֹרָה הַקְּדוֹשָׁה בְּכָל לִבּוֹ, וְהוּא הָלַךְ אֶל בֵּית הַשֵּׁם גַּם בְּמֶזֶג אֲוָויר שֶׁל רֶגֶ"שׁ – שֶׁהוּא נוֹטְרִיקוֹן שֶׁל בָּרָד, רוּחַ, גֶּשֶׁם וְשֶׁלֶג. הוּא צָעַד בַּשֶּׁלֶג הָעָמֹק וּבַקּוֹר הָעַז, מִבְּלִי לְהִתְחַשֵּׁב בַּסֶּבֶל אוֹ בַּמִּכְשׁוֹלִים שֶׁהָיוּ כְּרוּכִים בַּדֶּרֶךְ, כִּי רְצוֹנוֹ הַיְחִידִי הָיָה לְהַגִּיעַ אֶל הַיְשִׁיבָה וְלִלְמֹד תּוֹרָה.

כַּאֲשֶׁר הִגִּיעַ אֶל בֵּית הַיְשִׁיבָה, הָיָה הַנַּעַר יְהוּדָה הָרִאשׁוֹן מִכָּל הַתַּלְמִידִים שֶׁנִּכְנַס אֶל הֵיכַל הַקֹּדֶשׁ. הוּא הִתְיַשֵּׁב בִּמְקוֹמוֹ בְּשִׂמְחָה וְהִתְכּוֹנֵן לַלִּמּוּד. וְהִנֵּה, כַּעֲבֹר זְמַן קָצָר הִגִּיעַ אֶל הַיְשִׁיבָה גַּם רֹאשׁ הַיְשִׁיבָה, הַגָּאוֹן הָרַב עֶזְרָא עַטִיָּה זֵכֶר צַדִּיק לִבְרָכָה. כְּשֶׁרָאָה רֹאשׁ הַיְשִׁיבָה אֶת הַנַּעַר יְהוּדָה כְּבָר יוֹשֵׁב בַּיְשִׁיבָה, הוּא פָּנָה אֵלָיו בְּהֲנָאָה מְרֻבָּה וּבְשִׂמְחָה גְּדוֹלָה, וְאָמַר לוֹ מִלִּים מְרוֹמְמוֹת וּמְחַזְּקוֹת.

רֹאשׁ הַיְשִׁיבָה הָרַב עֶזְרָא עַטִיָּה אָמַר לוֹ בְּחִיּוּךְ וּבְאַהֲבָה: "רְאֵה בְּנִי, מָה טוֹב חֶלְקֵנוּ, שֶׁזָּכִינוּ לָבֹא אֶל הַיְשִׁיבָה אַף בְּיוֹם חָרְפִּי קָשֶׁה כָּזֶה, וּמָה נָעִים גּוֹרָלֵנוּ, שֶׁנִּלְמַד תּוֹרָה עַל אַף הַמִּכְשׁוֹלִים וְהַסֵּבֶל שֶׁהָיוּ כְּרוּכִים בְּבוֹאֵנוּ הֵנָּה. וְעַכְשָׁו אוֹר הַתּוֹרָה יָגִיהַּ חֶשְׁכֵּנוּ, וְאֵשׁ הַתּוֹרָה תְּחַמֵּם אֶת אֵיבָרֵינוּ הַקְּפוּאִים!"

הַנַּעַר יְהוּדָה שָׁמַע אֶת הַדְּבָרִים הַמְּרַגְּשִׁים שֶׁל רֹאשׁ הַיְשִׁיבָה וְהִרְגִּישׁ כֵּיצַד חֹם הַתּוֹרָה הַקְּדוֹשָׁה מְחַמֵּם אֶת אֵיבָרָיו הַקְּפוּאִים וְאֶת לִבּוֹ. מִתּוֹךְ כָּךְ הוּא הֵבִין שִׁעוּר חָשׁוּב וְנִפְלָא לְכָל יְמֵי חַיָּיו: כַּאֲשֶׁר אָדָם מִתְמַאֲמֵץ וּמִתְגַּבֵּר עַל כָּל הַקְּשָׁיִים כְּדֵי לִלְמֹד תּוֹרָה וְלַעֲשׂוֹת אֶת רְצוֹן הַשֵּׁם, הַתּוֹרָה הַקְּדוֹשָׁה מְחַמֶּמֶת אֶת הַנְּשָׁמָה, מְאִירָה אֶת הַחֹשֶׁךְ וּמְבִיאָה שִׂמְחָה גְּדוֹלָה וְאוֹר לַלֵּב.

💡 כְּשֶׁמִּתְגַּבְּרִים עַל קְשָׁיִים כְּדֵי לִלְמֹד תּוֹרָה, אוֹר הַתּוֹרָה מֵאִיר אֶת הַדֶּרֶךְ וּמְחַמֵּם אֶת הַלֵּב.
