הַשִּׁיר שֶׁנּוֹלַד מִתּוֹךְ הַתִּקְוָה

לִפְנֵי אֶלֶף וּמָאתַיִם שָׁנִים חַי בִּסְפָרַד רַבִּי דּוּנַשׁ בֶּן לַבְרָט, שֶׁהָיָה רֹאשׁ יְשִׁיבָה חָכָם וַחֲשׁוּב. רַבִּי דּוּנַשׁ הָיָה מְדַבֵּר תָּמִיד נֶגֶד דָּת הָאִסְלָאם. וְהִנֵּה, יוֹם אֶחָד קָרָה דָּבָר קָשֶׁה מְאֹד: הַבֵּן שֶׁל רַבִּי דּוּנַשׁ הֶחְלִיט לַעֲזֹב וְהִתְאַסְלֵם בַּעֲצְמוֹ. כְּשֶׁרַבִּי דּוּנַשׁ שָׁמַע עַל כָּךְ, הוּא נִמְלָא עֶצֶב גָּדוֹל מְאֹד. הוּא נִסְגַּר בְּתוֹךְ הַחֶדֶר שֶׁלּוֹ, הָיָה עָצוּב מְאֹד וְלֹא יָכֹל לָצֵאת מֵהַחֶדֶר בְּמַהֲלַךְ יָמִים רַבִּים. הָאֲנָשִׁים סְבִיבוֹ דָּאֲגוּ לִשְׁלוֹמוֹ מְאֹד וּפָחֲדוּ שֶׁהוּא מְסַכֵּן אֶת נַפְשׁוֹ מֵרֹב צַעַר.

הָאֲנָשִׁים בַּקְּהִלָּה לֹא יָדְעוּ מָה לַעֲשׂוֹת. לְבַסּוֹף הֵם אָמְרוּ: נִשְׁלַח מִכְתָּב וְנִשְׁלַח מִשְׁלַחַת אֶל רַבֵּנוּ סְעַדְיָה גָּאוֹן. רַבֵּנוּ סְעַדְיָה גָּאוֹן חַי אָז בְּבָבֶל, שֶׁהִיא בְּעִירָאק. זֶה הָיָה מָקוֹם רָחוֹק מְאֹד מִסְפָרַד, בְּעֶרֶךְ חֲמֵשֶׁת אֲלָפִים קִילֹמֶטְרִים. הָאֲנָשִׁים שָׁלְחוּ אִגֶּרֶת עִם שָׁלִיחַ, וְהִסְבִּירוּ לָרַב שֶׁיֵּשׁ בְּעָיָה חֲמוּרָה וְשֶׁרַבִּי דּוּנַשׁ בֶּן לַבְרָט כַּנִּרְאֶה נִמְצָא בְּסַכָּנָה. כְּשֶׁרַבֵּנוּ סְעַדְיָה גָּאוֹן שָׁמַע אֶת הַדְּבָרִים, הוּא אָמַר מִיָּד: מִיָּד תֶּאֶרְזוּ לִי אֶת הַחֲפָצִים, אֲנִי עַכְשָׁו יוֹצֵא! הוּא יָצָא עִם סוּס, יָצָא עִם הַדְּבָרִים שֶׁלּוֹ בִּמְהִירוּת רַבָּה, וְהָלַךְ בָּרֶגֶל עַד סְפָרַד.

כְּשֶׁרַבֵּנוּ סְעַדְיָה גָּאוֹן הִגִּיעַ לִסְפָרַד, הוּא דָּפַק בַּדֶּלֶת. הָאֲנָשִׁים בַּבַּיִת אָמְרוּ לְרַבִּי דּוּנַשׁ: בָּא לְבַקֵּר אוֹתְךָ רַבֵּנוּ סְעַדְיָה גָּאוֹן! מִיָּד רַבִּי דּוּנַשׁ בֶּן לַבְרָט יָצָא מֵהַחֶדֶר, הִסְתַּדֵּר קְצָת, פָּתַח אֶת הַדֶּלֶת וְקָרָא בְּהִתְרַגְּשׁוּת: רַבֵּנוּ סְעַדְיָה! מָה אַתָּה עוֹשֶׂה כָּאן? הוּא נִשֵּׁק אֶת הַיָּדַיִם שֶׁל רַבֵּנוּ סְעַדְיָה גָּאוֹן. אָמַר לוֹ רַבֵּנוּ סְעַדְיָה גָּאוֹן: תִּשְׁמַע, שָׁמַעְתִּי שֶׁיֵּשׁ לְךָ בַּת כְּלִילַת הַמַּעֲלוֹת, וְלִי יֵשׁ בֵּן תַּלְמִיד חָכָם, אוּלַי נְשַׁדֵּךְ בֵּינֵיהֶם? רַבִּי דּוּנַשׁ שָׂמַח וְעָנָה: בֶּטַח בְּוַדַּאי! נַכִּיר בֵּינֵיהֶם. הַבַּת וְהַבֵּן פָּגְשׁוּ זֶה אֶת זֶה, מָצְאוּ חֵן אֶחָד בְּעֵינֵי הַשֵּׁנִי וְהֶחְלִיטוּ לְהִתְחַתֵּן.

לִכְבוֹד שַׁבַּת הֶחָתָן, כָּתַב רַבִּי דּוּנַשׁ בֶּן לַבְרָט שִׁיר מְיֻחָד, הוּא הַפִּיּוּט הַמְּפֻרְסָם דְּרוֹר יִקְרָא. בַּשִּׁיר הוּא כָּתַב אֶת הַמִּלִּים: דְּרוֹר יִקְרָא לְבֵן עִם בַּת. כְּמוֹ שֶׁהַבַּת זָכְתָה לַגְּדֻלָּה הַזֹּאת, לְהִתְחַתֵּן עִם תַּלְמִיד חָכָם, הַבֵּן שֶׁל רַבֵּנוּ סְעַדְיָה גָּאוֹן, כָּךְ הוּא בִּקֵּשׁ מֵהַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיַּעֲזֹר גַּם לַבֵּן שֶׁלּוֹ, שֶׁלְּצַעֲרוֹ הִתְאַסְלֵם, שֶׁגַּם הוּא יִתְקָרֵב אֲלֵיהֶם קְצָת. בַּשִּׁיר הוּא כָּתַב גַּם: וְגַם בָּבֶל אֲשֶׁר גָּבְרָה, לְזֵכֶר בָּבֶל שֶׁמִּמֶּנָּה הִגִּיעַ רַבֵּנוּ סְעַדְיָה גָּאוֹן. כָּךְ נוֹלַד הַשִּׁיר הַמְּפֻרְסָם לִכְבוֹד שַׁבַּת הַחָתָן.

💡 גַּם כְּשֶׁעוֹבְרִים זְמַנִּים קָשִׁים וְעֲצוּבִים, חָבֵר טוֹב וֶאֱמוּנָה יְכוֹלִים לְהָבִיא תִּקְוָה וְשִׂמְחָה חֲדָשָׁה.
