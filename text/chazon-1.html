הַחֲזוֹן אִישׁ וּמִדַּת הָאֱמֶת

הַחֲזוֹן אִישׁ הָיָה מַנְהִיג רוּחָנִי גָּדוֹל, חָכָם וְצַדִּיק מְאֹד. הוּא הָיָה נוֹהֵג לְהִתְפַּלֵּל תְּפִלַּת מִנְחָה גְּדוֹלָה בְּכָל יוֹם בְּשָׁעָה שְׁתֵים עֶשְׂרֵה וָחֵצִי בַּצָּהֳרַיִם. הַתְּפִלָּה הָיְתָה מִתְקַיֶּמֶת בְּבֵית הַכְּנֶסֶת שֶׁהָיָה מְמֻקָּם בְּתוֹךְ בֵּיתוֹ. הַחֲזוֹן אִישׁ אָהַב מְאֹד אֶת הַתְּפִלָּה וְאֶת עֲבוֹדַת הַשֵּׁם, וְהִקְפִּיד תָּמִיד לְהִתְפַּלֵּל בְּצִבּוּר יַחַד עִם מִנְיָן שֶׁל עֲשָׂרָה אֲנָשִׁים בַּשָּׁעָה הַקְּבוּעָה.

פַּעַם אַחַת אֵרַע מַעֲשֶׂה מְעַנְיֵן בְּבֵיתוֹ שֶׁל הַחֲזוֹן אִישׁ. בְּאוֹתוֹ יוֹם הָיָה קָשֶׁה מְאֹד לֶאֱסֹף עֲשָׂרָה אֲנָשִׁים כְּדֵי לְהַשְׁלִים מִנְיָן לִתְפִלַּת הַמִּנְחָה. הַזְּמַן חָלַף, הַשָּׁעָה שְׁתֵים עֶשְׂרֵה וָחֵצִי עָבְרָה, וַעֲדַיִן לֹא הָיוּ מַסְפִּיק אֲנָשִׁים בְּבֵית הַכְּנֶסֶת. רַק בַּשָּׁעָה שְׁתֵים עֶשְׂרֵה וְאַרְבָּעִים וַחֲמִשָּׁה רְגָעִים, כְּלוֹמַר בְּרֶבַע לְאַחַת בַּצָּהֳרַיִם, נִמְצָא לְבַסּוֹף הָאִישׁ הָעֲשִׂירִי שֶׁהִשְׁלִים אֶת הַמִּנְיָן הַמְּקֻוֶּה.

בְּאוֹתָם רְגָעִים, מַמָּשׁ לִפְנֵי שֶׁהִתְחִילוּ לְהִתְפַּלֵּל, פָּנָה אֶל הַחֲזוֹן אִישׁ גִּיסוֹ, הָרַב רַבִּי שְׁמוּאֵל גְרֵינְמָן. רַבִּי שְׁמוּאֵל גְרֵינְמָן שָׁאַל אֶת הַחֲזוֹן אִישׁ שְׁאֵלָה חֲשׁוּבָה מְאֹד. הוּא סִפֵּר לוֹ: הִזְמַנְתִּי לְבֵיתִי לְשָׁעָה אַחַת בַּצָּהֳרַיִם אִישׁ אֶחָד, פְּלוֹנִי, כְּדֵי לְקַיֵּם עִמּוֹ פְּגִישָׁה. וְהִנֵּה, בִּגְלַל שֶׁתְּפִלַּת הַמִּנְחָה הִתְאַחֲרָה כָּל כָּךְ וְתַתְחִיל רַק עַכְשָׁיו, בְּרֶבַע לְאַחַת, אֲנִי בְּוַדַּאי אֶתְאַחֵר לְהוֹפִיעַ בַּפְּגִישָׁה בְּבֵיתִי. כְּתוֹצָאָה מִכָּךְ, יִהְיֶה מַצָּב שֶׁבּוֹ הָאִישׁ הַהוּא יַמְתִּין לִי בְּבֵיתִי לֹא כְּפִי שֶׁקָּבַעְנוּ וְלֹא כְּפִי שֶׁהִבְטַחְתִּי לוֹ מֵרֹאשׁ.

כְּשֶׁשָּׁמַע הַחֲזוֹן אִישׁ אֶת דִּבְרֵי גִּיסוֹ רַבִּי שְׁמוּאֵל גְרֵינְמָן, הוּא הֵשִׁיב לוֹ מִיָּד בְּבֵרוּר וּבְתַקִּיפוֹת: מִי שֶׁדָּבוּק בְּמִדַּת הָאֱמֶת, אֵין אֶצְלוֹ כָּאן שׁוּם צֵל שֶׁל שְׁאֵלָה! הַחֲזוֹן אִישׁ אָמַר לוֹ שֶׁעָדִיף שֶׁיְּבֻטַּל הַמִּנְיָן לַתְּפִלָּה, אֲבָל מִדִּבּוּר הָאֱמֶת לֹא תָזוּז! הַחֲזוֹן אִישׁ לִמֵּד אוֹתָנוּ יְסוֹד עָמוֹק: מִדַּת הָאֱמֶת, הַשְּׁמִירָה עַל הַדִּבּוּר וְהָעֲמִידָה בַּזְּמַן שֶׁנִּקְבַּע הֵם דְּבָרִים עֶלְיוֹנִים וַחֲשׁוּבִים מְאֹד. אָסוּר לִגְרֹם לְאָדָם אַחֵר לְחַכּוֹת לֹא כְּפִי שֶׁהֻבְטַח לוֹ.

וְאָכֵן, מִיָּד לְאַחַר שֶׁאָמַר הַחֲזוֹן אִישׁ אֶת דְּבָרָיו הַחֲשׁוּבִים, הַקָּהָל שֶׁנֶּאֱסַף בְּבֵית הַכְּנֶסֶת הִתְפַּזֵּר לְבָתֵּיהֶם, וְהַמִּנְיָן לִתְפִלַּת הַמִּנְחָה הִתְבַּטֵּל. הַחֲזוֹן אִישׁ וְגִיסוֹ רַבִּי שְׁמוּאֵל גְרֵינְמָן הֶרְאוּ לְכָל הָאֲנָשִׁים עַד כַּמָּה חָשׁוּב לִהְיוֹת אֲנָשִׁים שֶׁדְּבוּקִים בֶּאֱמֶת, שׁוֹמְרִים עַל מִלָּתָם וְלֹא זָזִים מֵהָאֱמֶת לְעוֹלָם.

💡 עָלֵינוּ לִהְיוֹת תָּמִיד נֶאֱמָנִים לַמִּלָּה שֶׁלָּנוּ, לַעֲמֹד בַּהַבְטָחוֹת שֶׁלָּנוּ וְלֹא לָזוּז מֵהָאֱמֶת.
