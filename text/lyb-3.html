מִשְׁלוֹחַ הַמָּנוֹת שֶׁל רַבִּי לֵוִי יִצְחָק

בָּעֲיָרָה בֶּרְדִיטְשׁוֹב הִתְכַּנְסוּ תּוֹשָׁבֵי הָעִיר – גְּבָרִים, נָשִׁים וִילָדוֹת – בְּבֵית הַכְּנֶסֶת וּבְעֶזְרַת הַנָּשִׁים בְּיוֹם תַּעֲנִית אֶסְתֵּר, כְּדֵי לִשְׁמֹעַ אֶת קְרִיאַת הַמְּגִילָה. רַבִּי לֵוִי יִצְחָק מִבֶּרְדִיטְשׁוֹב, הַנִּקְרָא גַּם "סַנְגוֹרָם שֶׁל יִשְׂרָאֵל", עָמַד לְהַתְחִיל בִּתְפִילַּת עַרְבִית. מַמָּשׁ לִפְנֵי הַתְּפִילָּה, נִגַּשׁ שַׁמַּשׁ הָרַב וְלָחַשׁ מַשֶּׁהוּ בְּאָזְנוֹ. רַבִּי לֵוִי יִצְחָק קָם מִיָּד וְיָצָא אֶל חֶדֶר סָמוּךְ הַנִּמְצָא לְיַד בֵּית הַכְּנֶסֶת.

בַּחֶדֶר הַזֶּה חִכְּתָה אִשָּׁה עֲנִיָּה, שֶׁשְּׁמָהּ בֶּלָה. הִיא הֶחְזִיקָה בְּיָדֶיהָ תַּרְנְגוֹל שֶׁקָּנְתָה בְּכַסְפָּהּ הָאַחֲרוֹן, כְּדֵי לְהָכִין מִמֶּנּוּ אֲרוּחָה לִשְׁבִירַת הַצּוֹם עֲבוּר יְלָדֶיהָ וַעֲבוּר בַּעְלָהּ הַחוֹלֶה. הָאִשָּׁה שָׁאֲלָה אֶת הָרַב בִּדְאָגָה הַאִם הַתַּרְנְגוֹל כָּשֵׁר לַאֲכִילָה. הָרַב הִבִּיט בַּתַּרְנְגוֹל, עִיֵּן בּוֹ, וְאָמַר לָהּ בְּצַעַר רַב: "אֲנִי מִצְטַעֵר, אַךְ הַתַּרְנְגוֹל הַזֶּה אֵינוֹ כָּשֵׁר, וְאִי אֶפְשָׁר לֶאֱכֹל אוֹתוֹ".

כְּשֶׁשָּׁמְעָה זֹאת הָאִשָּׁה, הִיא פָּרְצָה בְּבֶכִי מַר, כִּי לֹא הָיָה לָהּ מָה לָתֵת לִילָדֶיהָ וּלְבַעְלָהּ הַחוֹלֶה. הָרַב נִהֵל אִתָּהּ שִׂיחָה קְצָרָה וְאָמַר לָהּ בְּרַכּוּת וּבְחֹם: "אַל תִּדְאֲגִי, הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹזֵר. תִּכְנְסִי כָּעֵת אֶל עֶזְרַת הַנָּשִׁים, תִּתְפַּלְלִי, תִּשְׁמְעִי אֶת קְרִיאַת הַמְּגִילָה, וַה' יַעֲזֹר".

מִיָּד לַאֲחַר מִכֵּן, יָצָא הָרַב מִבֵּית הַכְּנֶסֶת וְהָלַךְ בִּמְהִירוּת לְבֵיתוֹ. כְּשֶׁהִגִּיעַ לַבַּיִת, לָקַח הָרַב אֶת הַמַּפָּה הַלְּבָנָה שֶׁהָיְתָה עַל הַשֻּׁלְחָן, וְצָרַר בְּתוֹכָהּ אֶת כָּל הַמַּאֲכָלִים הַטְּעִימִים שֶׁהוּכְנוּ לַחַג: חַלּוֹת טְרִיּוֹת, לֶחֶם, יַיִן, פֵּרוֹת וְגֶפִילְטֶה פִישׁ (דָּג מְמֻלָּא). הוּא נָשָׂא אֶת הַחֲבִילָה הַכְּבֵדָה וְהָלַךְ אֶל בֵּיתָהּ שֶׁל הָאִשָּׁה הָעֲנִיָּה.

כְּשֶׁנִּכְנַס הָרַב בַּסֵּתֶר אֶל בֵּיתָהּ שֶׁל הָאִשָּׁה, הוּא שָׁמַע קוֹל חָלוּשׁ מֵהַחֶדֶר הַפְּנִימִי. הָיָה זֶה בַּעְלָהּ הַחוֹלֶה שֶׁחָשַׁב שֶׁאִשְׁתּוֹ חָזְרָה, וְשָׁאַל: "בֶּלָה, נוּ, שָׁאַלְתְּ אֶת הָרַב? מָה הָרַב אָמַר, אֶפְשָׁר לֶאֱכֹל אֶת הַתַּרְנְגוֹל?". רַבִּי לֵוִי יִצְחָק הֵשִׁיב לוֹ בְּשֶׁקֶט וּבְאַהֲבָה: "שָׁלוֹם, ה' אוֹהֵב אֶתְכֶם, ה' דָּאַג לָכֶם, בְּבַקָּשָׁה, ה' דָּאַג לָכֶם לְמִשְׁלוֹחַ מָנוֹת". הָרַב הִנִּיחַ אֶת כָּל הָאוֹכֶל עַל הַשֻּׁלְחָן וְיָצָא מִיָּד בַּסֵּתֶר.

רַבִּי לֵוִי יִצְחָק חָזַר בִּמְהִירוּת לְבֵית הַכְּנֶסֶת, הִתְפַּלֵּל תְּפִילַּת עַרְבִית עִם הַצִּבּוּר וְקָרָא אֶת הַמְּגִילָה בִּקְרִיאָה מְיֻחֶדֶת בְּמִינָהּ שֶׁרִגְּשָׁה מְאֹד אֶת כָּל הַמִּתְפַּלְלִים. כְּשֶׁחָזַר הָרַב לְבֵיתוֹ בְּתוֹם הַתְּפִילָּה, רָאֲתָה אִשְׁתּוֹ הָרַבָּנִית שֶׁכָּל הַמַּאֲכָלִים שֶׁהֵכִינָה נֶעֶלְמוּ מֵהַשֻּׁלְחָן. הִיא הִבִּיטָה בְּפָנָיו הַקּוֹרְנוֹת מֵאֹשֶׁר וּמִשִּׂמְחָה, וְהֵבִינָה מִיָּד אֶת גֹּדֶל הַחֶסֶד שֶׁעָשָׂה. בְּשִׂמְחָה רַבָּה הִיא נִגְּשָׁה לַמִּטְבָּח וְהֵכִינָה אֲרוּחָה חֲדָשָׁה עֲבוּרָם.

בַּמַּקְבִּיל, הָאִשָּׁה הִגִּיעָה לְבֵיתָהּ מִבֵּית הַכְּנֶסֶת וְרָאֲתָה אֶת הַשֻּׁלְחָן הָעָמוֹס בְּכָל טוּב. הִיא הָיְתָה נִרְגֶּשֶׁת וּמֻפְתָּעָה מְאֹד, וְסִפְּרָה לַחֲבֵרוֹתֶיהָ שֶׁאֵלִיָּהוּ הַנָּבִיא הִגִּיעַ לְבֵיתָהּ וְהֵבִיא לָהּ מִשְׁלוֹחַ מָנוֹת. אַךְ אַנְשֵׁי בֶּרְדִיטְשׁוֹב הֵבִינוּ מִיָּד שֶׁזֶּה הָיָה רַבִּי לֵוִי יִצְחָק, שֶׁנִּכְנַס בַּסֵּתֶר וְהֵבִיא אֶת הָאוֹכֶל לַמִּשְׁפָּחָה הָעֲנִיָּה.

💡 כְּשֶׁאָנוּ עוֹזְרִים לַאֲחֵרִים בַּסֵּתֶר וּבְאַהֲבָה, אָנוּ מְבִיאִים אֶת הָאוֹר, הַחֶסֶד וְהַשִּׂמְחָה שֶׁל ה' לָעוֹלָם.
