הַצִּרְעָה בַּמֵּצַח וְכוֹחַ הָאִפּוּק שֶׁל מַר בּוּסְתְּנַאי

בִּתְקוּפַת הַסְּבוֹרָאִים חַי בְּבָבֶל מַנְהִיג מְיֻחָד מִזֶּרַע דָּוִד הַמֶּלֶךְ, וּשְׁמוֹ מַר בּוּסְתְּנַאי. מְסֻפָּר שֶׁבְּאוֹתָם יָמִים גָּזַר מֶלֶךְ פָּרַס לְהַשְׁמִיד אֶת כָּל בֵּית דָּוִד, וְלֹא נִשְׁאַר מֵהֶם אֶלָּא עֻבָּר אֶחָד בְּבֶטֶן אִמּוֹ שֶׁהִסְתַּתְּרָה. לַיְלָה אֶחָד חָלַם הַמֶּלֶךְ חֲלוֹם מַבְהִיל: הוּא רָאָה אֶת עַצְמוֹ נִכְנָס לְגַן עֵצִים מְפֹאָר – שֶׁנִּקְרָא בִּלְשׁוֹן פָּרַס "בּוּסְתָּאן" – וְהוּא עוֹקֵר וּמַשְׁחִית בְּגַרְזֶן אֶת כָּל עֲצֵי הַגַּן. כְּשֶׁבִּקֵּשׁ הַמֶּלֶךְ לִכְרֹת גַּם שֹׁרֶשׁ קָטָן אֶחָד שֶׁנִּשְׁאַר פּוֹרֵחַ, הוֹפִיעַ לְפָנָיו אִישׁ זָקֵן וְהִכָּה אוֹתוֹ בְּמִצְחוֹ עַד שֶׁזָּב דָּמוֹ, וְהִזְהִיר אוֹתוֹ שֶׁלֹּא יִגַּע בַּשֹּׁרֶשׁ הָאַחֲרוֹן. הַמֶּלֶךְ הִתְחַנֵּן עַל נַפְשׁוֹ וְהִבְטִיחַ לִנְצֹר אֶת הַשֹּׁרֶשׁ וּלְטַפְּחוֹ. כְּשֶׁהִתְעוֹרֵר הַמֶּלֶךְ בְּבֶהָלָה, פָּתְרוּ לוֹ חַכְמֵי יִשְׂרָאֵל כִּי הַזָּקֵן הָיָה דָּוִד הַמֶּלֶךְ, וְהַשֹּׁרֶשׁ שֶׁנִּשְׁאַר הוּא הַתִּינוֹק שֶׁעָתִיד לְהִוָּלֵד. הַמֶּלֶךְ הֶחֱזִיק בְּהַבְטָחָתוֹ וְשָׁמַר עַל הָאֵם בְּאַרְמוֹנוֹ, וּכְשֶׁנּוֹלַד הַבֵּן קָרְאוּ לוֹ בּוּסְתְּנַאי, עַל שֵׁם אוֹתוֹ הַבּוּסְתָּן שֶׁרָאָה הַמֶּלֶךְ בַּחֲלוֹמוֹ.

מְסֻפָּר שֶׁכַּאֲשֶׁר גָּדַל בּוּסְתְּנַאי, לָמַד בִּשְׁקִידָה תּוֹרָה וּמִשְׁנָה, תַּלְמוּד, הֲלָכָה וּשְׁאָר חָכְמוֹת, וְהָיָה מָלֵא יִרְאַת שָׁמַיִם וּמִדּוֹת טוֹבוֹת, עַד שֶׁשִּׁבְחוֹ נוֹדַע בְּכָל הָאָרֶץ. הַמֶּלֶךְ הִשְׁתּוֹקֵק מְאֹד לִפְגֹּשׁ אוֹתוֹ, וּכְשֶׁבָּא בּוּסְתְּנַאי לָאַרְמוֹן, מָצָא חֵן וָחֶסֶד רַב בְּעֵינָיו. בּוּסְתְּנַאי נִכְנַס אֶל הֵיכַל הַמַּלְכוּת, וְעָמַד שָׁם בְּיִרְאַת כָּבוֹד עֲמֻקָּה מֵהַבֹּקֶר וְעַד הָעֶרֶב, בְּלִי לְהַזִּיז אֶת רַגְלָיו וּבְלִי לְנַעְנֵעַ אֶת רֹאשׁוֹ כְּלָל.

בְּעוֹד בּוּסְתְּנַאי עוֹמֵד דּוּמָם לִפְנֵי הַמֶּלֶךְ, עָפָה לְעֶבְרוֹ צִרְעָה – שֶׁיֵּשׁ מְקוֹרוֹת הַמְּכַנִּים אוֹתָהּ זְבוּב – וְנָחֲתָה יָשָׁר עַל מִצְחוֹ. הִיא נָשְׁכָה וְעָקְצָה אוֹתוֹ שׁוּב וָשׁוּב בְּעָצְמָה רַבָּה, עַד שֶׁהַדָּם הֵחֵל לִנְטֹף מִמִּצְחוֹ עַל פָּנָיו. הַכְּאֵב הָיָה חַד וְצוֹרֵב מְאֹד, אַךְ בּוּסְתְּנַאי הֶפְגִּין כּוֹחַ אִפּוּק מֻפְלָא: הוּא לֹא זָז מִמְּקוֹמוֹ, לֹא צָעַק, וַאֲפִילוּ לֹא הֵרִים אֶת יָדוֹ כְּדֵי לְגָרֵשׁ אוֹתָהּ. הַמֶּלֶךְ, שֶׁרָאָה אֶת הַדָּם הַיּוֹרֵד עַל פָּנָיו וְאֶת הַשֶּׁקֶט הַמֻּחְלָט שֶׁלּוֹ, הִתְפַּלֵּא עַד מְאֹד וְשָׁאַל: "לָמָּה אֵינְךָ מַבְרִיחַ אֶת הַצִּרְעָה מֵעַל מִצְחֲךָ?".

הֵשִׁיב לוֹ בּוּסְתְּנַאי בְּעִנְוְתָנוּת וּבְנִימוּס: מִנְהָג וּמוּסָר זֶה יָרַשְׁנוּ מֵאֲבוֹתֵינוּ מִבֵּית דָּוִד הַמֶּלֶךְ; כַּאֲשֶׁר אָנוּ עוֹמְדִים לִפְנֵי הַמֶּלֶךְ בְּהֵיכָלוֹ, אֵין אָנוּ מְדַבְּרִים, אֵין אָנוּ צוֹחֲקִים, וְאֵין אָנוּ מְרִימִים יָד שֶׁלֹּא לְצֹרֶךְ. כְּשֶׁשָּׁמַע הַמֶּלֶךְ אֶת הַדְּבָרִים, נִפְעַם מְאֹד מִגְּבוּרַת רוּחוֹ, מִכּוֹחַ הָאִפּוּק שֶׁלּוֹ וּמִיִּרְאַת הַכָּבוֹד שֶׁהֶרְאָה. הַדָּבָר מָצָא חֵן עָצוּם בְּעֵינָיו, וְהוּא צִוָּה מִיָּד לָתֵת לוֹ מַתָּנוֹת יְקָרוֹת, לְהַרְכִּיב אוֹתוֹ בְּמֶרְכֶּבֶת הַמִּשְׁנֶה שֶׁל הַמַּלְכוּת, וּמִנָּה אוֹתוֹ לִהְיוֹת רֹאשׁ גָּלֻיּוֹת יִשְׂרָאֵל. מְסֻפָּר כִּי לְזֵכֶר אוֹתוֹ מַעֲשֶׂה מֻפְלָא שֶׁל שְׁלִיטָה עַצְמִית וְכָבוֹד, נִקְבַּע צִיּוּר שֶׁל צִרְעָה אוֹ זְבוּב בַּחוֹתָם הָרִשְׁמִי שֶׁל בֵּית רֹאשׁ הַגּוֹלָה.

💡 כּוֹחַ הָאִפּוּק וְהַשְּׁלִיטָה הָעַצְמִית מְעִידִים עַל גְּבוּרָה אֲמִתִּית וּמְבִיאִים כָּבוֹד וּבְרָכָה.
