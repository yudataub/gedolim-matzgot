הַמִּכְתָּב הַגָּדוֹל שֶׁל רַב שְׁרִירָא גָּאוֹן

בְּאֶרֶץ בָּבֶל הָרְחוֹקָה חַי מַנְהִיג רוּחָנִי דָּגוּל וְחָכָם מְפֻרְסָם שֶׁנִּקְרָא רַב שְׁרִירָא גָּאוֹן, בְּנוֹ שֶׁל רַב חֲנַנְיָה וְנֶכְדוֹ שֶׁל רַב יְהוּדָה. רַב שְׁרִירָא גָּאוֹן הָיָה רֹאשׁ יְשִׁיבַת פּוּמְבְּדִיתָא הַדְּגוּלָה, וְיַחַד עִמּוֹ הִנְהִיג אֶת בֵּית הַדִּין בְּנוֹ, רַב הַאי הַדַּיָּן. בְּאוֹתוֹ זְמַן, בִּצְפוֹן אַפְרִיקָה הָרְחוֹקָה, בָּעִיר קֵירוּאָן, יָשְׁבוּ חַכְמֵי יִשְׂרָאֵל שֶׁלָּמְדוּ תּוֹרָה בִּשְׁקִידָה וּבְאַהֲבָה. חַכְמֵי קֵירוּאָן רָצוּ מְאֹד לְהָבִין לְעֹמֶק אֶת תּוֹלְדוֹתֶיהָ שֶׁל הַתּוֹרָה שֶׁבְּעַל פֶּה, כֵּיצַד הִגִּיעוּ אֲלֵיהֶם סִפְרֵי הַקֹּדֶשׁ, וְאֵיךְ נִכְתְּבוּ הַמִּשְׁנָה וְהַתַּלְמוּד מִדּוֹר לְדוֹר. לָכֵן הֵם הֶחְלִיטוּ לִשְׁלֹחַ מִכְתָּב עִם שְׁאֵלוֹת רַבּוֹת וַחֲשׁוּבוֹת אֶל בֵּית הַדִּין שֶׁל רַב שְׁרִירָא גָּאוֹן בְּבָבֶל.

בְּמִכְתָּבָם שָׁאֲלוּ חַכְמֵי קֵירוּאָן כֵּיצַד נִכְתְּבָה הַמִּשְׁנָה, וְהַאִם אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה הֵם שֶׁהִתְחִילוּ לִכְתֹּב אוֹתָהּ, וְחַכְמֵי כָּל דּוֹר וָדוֹר הוֹסִיפוּ מְעַט עַד שֶׁבָּא רַבִּי יְהוּדָה הַנָּשִׂיא וְחָתַם אוֹתָהּ. הֵם שָׁאֲלוּ לָמָה רוֹב הַמִּשְׁנָיוֹת הַסְּתָמִיּוֹת הֵן כְּדַעַת רַבִּי מֵאִיר, וּמַדּוּעַ רוֹב הַחֲכָמִים הַמְּפֹרָשִׁים בָּהּ, כְּגוֹן רַבִּי מֵאִיר, רַבִּי יְהוּדָה, רַבִּי יוֹסֵי וְרַבִּי שִׁמְעוֹן, הָיוּ כֻּלָּם תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא בְּסוֹף יְמֵי בַּיִת שֵׁנִי. הֵם תָּמְהוּ מַדּוּעַ הַחֲכָמִים הָרִאשׁוֹנִים הִשְׁאִירוּ דְּבָרִים רַבִּים כָּל כָּךְ לַחֲכָמִים הָאַחֲרוֹנִים, וּבִפְרָט אִם לֹא נִכְתַּב דָּבָר עַד יְמֵי רַבִּי יְהוּדָה הַנָּשִׂיא. עוֹד שָׁאֲלוּ עַל סֵדֶר הַמַּסַּכְתוֹת, לָמָה הִקְדִּימוּ אֶת מַסֶּכֶת יוֹמָא לְמַסֶּכֶת שְׁקָלִים, וְאֶת מַסֶּכֶת סֻכָּה לְמַסֶּכֶת יוֹם טוֹב, וְלָמָה הִקְדִּימוּ אֶת שְׁתֵּיהֶן לְמַסֶּכֶת רֹאשׁ הַשָּׁנָה. הֵם בִּקְשׁוּ לְהָבִין גַּם אֵיךְ נִכְתְּבָה הַתּוֹסֶפְתָּא עַל יְדֵי רַבִּי חִיָּא, וְלָמָה רַבִּי יְהוּדָה הַנָּשִׂיא לֹא כָּתַב אֶת הַדְּבָרִים הָאֵלֶּה בַּמִּשְׁנָה עַצְמָהּ. לְבַסּוֹף שָׁאֲלוּ אֵיךְ נִכְתְּבוּ הַבָּרַיְתּוֹת וְהַתַּלְמוּד, כֵּיצַד נִסְדְּרוּ הַחֲכָמִים הַסַּבּוֹרָאִים אַחֲרֵי רָבִינָא, מִי מָלַךְ אַחֲרֵיהֶם וְכַמָּה שָׁנִים עָבְרוּ מֵאָז וְעַד הַיּוֹם.

כַּאֲשֶׁר הִגִּיעַ הַמִּכְתָּב הֶחָשׁוּב לְבָבֶל, יָשַׁב רַב שְׁרִירָא גָּאוֹן וְכָתַב עֲבוֹרָם תְּשׁוּבָה מְפֹרָטֶת וּמֻפְלָאָה שֶׁנִּקְרֵאת "אִגֶּרֶת רַב שְׁרִירָא גָּאוֹן". בְּאִגֶּרֶת זוֹ סִפֵּר לָהֶם אֶת כָּל סֵדֶר הַדּוֹרוֹת הָאֵיתָן שֶׁל מְסִירַת הַתּוֹרָה מֵאַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה, דֶּרֶךְ הַתַּנָּאִים וְהָאָמוֹרָאִים, וְעַד לַסַּבּוֹרָאִים וְלַגְּאוֹנִים שֶׁבְּדוֹרוֹ. הוּא הֶרְאָה לָהֶם אֵיךְ הַתּוֹרָה נִשְׁמְרָה בִּדְיוּק רַב מִדּוֹר לְדוֹר. בְּסוֹף הַמִּכְתָּב סִפֵּר רַב שְׁרִירָא גָּאוֹן כִּי מִנָּה אֶת בְּנוֹ, רַב הַאי, לִהְיוֹת אָב בֵּית דִּין, וּבֵרֵךְ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִכְתֹּב אֶת כֻּלָּם לְחַיִּים, יִזְכֶּה אוֹתָם לְהַנְהִיג אֶת יִשְׂרָאֵל בָּאֱמֶת, וְיָבִיא אֶת מְשִׁיחַ בֶּן דָּוִד בִּמְהֵרָה בְּיָמֵינוּ.
