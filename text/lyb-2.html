הַמַּמְתָּק שֶׁל לֵוִי יִצְחָק וּתְפִלַּת הַנְּעִילָה

דָּבָר מַדְהִים קָרָה בְּעִיר בֶּרְדִיצֶ'ב, בְּבֵית הַמִּדְרָשׁ שֶׁל הַצַּדִּיק רַבִּי לֵוִי יִצְחָק. בְּכָל שָׁנָה, בְּסוֹף יוֹם הַכִּפּוּרִים, עָבַר רַבִּי לֵוִי יִצְחָק לִפְנֵי הַתֵּבָה כְּחַזָּן בִּתְפִלַּת הַנְּעִילָה הַקְּדוֹשָׁה, וְכָל הַקָּהָל סִיֵּם אִתּוֹ אֶת הַיּוֹם בְּהִתְרַגְּשׁוּת גְּדוֹלָה. גַּם בְּאוֹתָהּ שָׁנָה עָמַד רַבִּי לֵוִי יִצְחָק וְהִתְפַּלֵּל בְּכַוָּנָה רַבָּה, וְהַתְּפִלָּה כְּבָר עָמְדָה לְהִסְתַּיֵּם. הוּא הִגִּיעַ עַד הַבְּרָכָה הַכִּמְעַט אַחֲרוֹנָה, וּפִתְאוֹם נֶעֱצַר. רַבִּי לֵוִי יִצְחָק עָמַד בְּשֶׁקֶט מֻחְלָט, הִשְׁתַּתֵּק וְלֹא הוֹצִיא מִלָּה מִפִּיו. הַקָּהָל כֻּלּוֹ חִכָּה בְּמֶתַח. אַף אֶחָד לֹא הֵבִין מָה קָרָה, אַךְ כֻּלָּם יָדְעוּ שֶׁאִם הָרַב עוֹמֵד וְשׁוֹתֵק, כַּנִּרְאֶה שֶׁמַּשֶׁהוּ רְצִינִי מְאֹד מִתְרַחֵשׁ.

אַחֲרֵי דַּקּוֹת אֲרֻכּוֹת שֶׁל שְׁתִיקָה, פָּתַח רַבִּי לֵוִי יִצְחָק אֶת פִּיו בְּהִתְלַהֲבוּת גְּדוֹלָה וּבֵרֵךְ אֶת הַבְּרָכָה: "בָּרוּךְ אַתָּה ה' מֶלֶךְ מוֹחֵל וְסוֹלֵחַ לַעֲוֹנוֹתֵינוּ וְלַעֲווֹנוֹת עַמּוֹ בֵּית יִשְׂרָאֵל, וּמַעֲבִיר אַשְׁמוֹתֵינוּ בְּכָל שָׁנָה וְשָׁנָה, מֶלֶךְ עַל כָּל הָאָרֶץ, מְקַדֵּשׁ יִשְׂרָאֵל וְיוֹם הַכִּפּוּרִים." הוּא סִיֵּם אֶת הַתְּפִלָּה בְּהִתְרַגְּשׁוּת, הִמְשִׁיךְ לִתְפִלַּת עַרְבִית, הַבְּדָלָה וּתְקִיעַת שׁוֹפָר גְּדוֹלָה בְּשִׂמְחָה רַבָּה.

בְּסוֹף יוֹם הַכִּפּוּרִים נִגְּשׁוּ אֵלָיו תַּלְמִידָיו וְשָׁאֲלוּ: "רַבֵּנוּ, מָה הָיְתָה הָעֲצִירָה הַזֹּאת?" הִסְבִּיר לָהֶם רַבִּי לֵוִי יִצְחָק: "הִרְגַּשְׁתִּי שֶׁגְּזֵרָה קָשָׁה נִגְזְרָה עַל עַם יִשְׂרָאֵל, וְהַתְּפִלּוֹת שֶׁלָּנוּ לֹא מַצְלִיחוֹת לְבַטֵּל אוֹתָהּ. זָעַקְתִּי בְּתוֹךְ לִבִּי וְהִתְחַנַּנְתִּי לַקָּדוֹשׁ בָּרוּךְ הוּא, אַךְ שוּם דָּבָר לֹא עָזַר. וְאָז, פִּתְאוֹם נִזְכַּרְתִּי בְּמַעֲשֶׂה מִיְּמֵי יַלְדוּתִי. כְּשֶׁהָיִיתִי יֶלֶד קָטָן וְשׁוֹבָב, אָהַבְתִּי מְאֹד מַמְתָּקִים. בְּבֵית הוֹרַי הָיְתָה מְגֵרַת מַמְתָּקִים מְיֻחֶדֶת, וְקִבַּלְנוּ מַמְתָּק רַק פַּעַם אַחַת בַּשָּׁבוּעַ, לִכְבוֹד שַׁבַּת קֹדֶשׁ. בְּאֶחָד מִימֵי הַשָּׁבוּעַ הִשְׁתَوْקַקְתִּי מְאֹד לְמַמְתָּק וְהִתְחַנַּנְתִּי לְאִמָּא, אַךְ הִיא לֹא הִסְכִּימָה."

"מֵרֹב רָצוֹן וְשׁוֹבָבוּת, עָמַדְתִּי לְיַד הַמְּגֵרָה וּבֵרַכְתִּי בְּקוֹל: 'בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ'. אִמִּי הָאוֹהֶבֶת שָׁמְעָה אוֹתִי מְבָרֵךְ אֶת שֵׁם ה', וּמִיָּד קָפְצָה, פָּתְחָה אֶת הַמְּגֵרָה וְנָתְנָה לִי מַמְתָּק. הִיא אָמְרָה לִי: 'לֵוִי יִצְחָק שֶׁלִּי, קַח מַמְתָּק, הָעִקָּר שֶׁלֹּא תֹּאמַר בְּרָכָה לְבַטָּלָה'. הִיא כָּל כָּךְ אָהֲבָה אוֹתִי וְדָאֲגָה שֶׁלֹּא אֶעֱשֶׂה עֲבֵרָה."

"כְּשֶׁנִּזְכַּרְתִּי בָּזֶה בִּתְפִלַּת הַנְּעִילָה, אָמַרְתִּי לְעַצְמִי: אָבִינוּ שֶׁבַּשָּׁמַיִם אוֹהֵב אוֹתָנוּ אֲפִלּוּ יוֹתֵר! אִם אֲנִי אַתְחִיל לְבָרֵךְ בְּשֵׁם וּמַלְכוּת 'מֶלֶךְ מוֹחֵל וְסוֹלֵחַ לַעֲוֹנוֹתֵינוּ', הַקָּדוֹשׁ בָּרוּךְ הוּא בְּוַדַּאי לֹא יִרְצֶה שֶׁנְּבָרֵךְ בְּרָכָה לְבַטָּלָה, וְהוּא יִסְלַח לָנוּ וִיבַטֵּל אֶת הַגְּזֵרָה! וְכָךְ בְּדִיּוּק קָרָה – בֵּרַכְתִּי, וְהִרְגַּשְׁתִּי שֶׁהַגְּזֵרָה הַקָּשָׁה נִקְרְעָה. כָּכָה אָבִינוּ שֶׁבַּשָּׁמַיִם אוֹהֵב אוֹתָנוּ."

💡 הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹהֵב אוֹתָנוּ מְאֹד כְּמוֹ אָב אָהוּב, וְתָמִיד רוֹצֶה לִסְלֹחַ לָנוּ וּלְקַבֵּל אֶת תְּפִלּוֹתֵינוּ.
