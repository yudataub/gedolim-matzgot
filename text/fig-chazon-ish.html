שַׁבָּת שָׁלוֹם לָעֵץ וְלָאֲדָמָה

זֶהוּ סִפּוּר מְרַגֵּשׁ וּמְאַלֵּף מְאֹד עַל שְׁנֵי רַבָּנִים גְּדוֹלִים וַחֲשׁוּבִים מִגְּדוֹלֵי הַדּוֹר שֶׁלָּנוּ. הָרַב הָרִאשׁוֹן הַמּוֹפִיעַ בַּסִּפּוּר הוּא הַחֲזוֹן אִי"שׁ, רַבִּי אַבְרָהָם יְשַׁעְיָהוּ קַרְלִיץ זֵכֶר צַדִּיק לִבְרָכָה. הָרַב הַשֵּׁנִי הוּא הָרַב מִפּוֹנוֹבִיץ', הָרַב יוֹסֵף כַּהֲנְמַן זֵכֶר צַדִּיק לִבְרָכָה, שֶׁנִּקְרָא גַּם בְּפִי כֹּל הָרַב כַּהֲנְמַן. הַסִּפּוּר הַמְּיֻחָד וְהַמְּרוֹמֵם הַזֶּה הִתְרַחֵשׁ בְּאֶרֶץ יִשְׂרָאֵל עוֹד לִפְנֵי קוּם הַמְּדִינָה, בִּשְׁנַת תָּשָׁ"ה. הַמָּקוֹם שֶׁבּוֹ הִתְרַחֵשׁ הַסִּפּוּר הַמּוֹפְתִי הוּא קִבּוּץ 'חֵפֶץ חַיִּים'. בַּקִּבּוּץ הַזֶּה הִתְגּוֹרְרוּ חַקְלָאִים יְהוּדִים יְקָרִים שֶׁהִקְפִּידוּ מְאֹד לִשְׁמֹר עַל מִצְוַת הַשְּׁמִיטָה כְּהִלְכָתָהּ.

כֻּלָּנוּ יוֹדְעִים וּמְבִינִים שֶׁזֶּה בִּכְלָל לֹא פָּשׁוּט לִשְׁמֹר עַל מִצְוַת הַשְּׁמִיטָה וְלֹא לַעֲבֹד אֶת הָאֲדָמָה בְּמֶשֶׁךְ שָׁנָה תְּמִימָה. הַחַקְלָאִים שֶׁהִתְגּוֹרְרוּ בַּקִּבּוּץ בֶּאֱמֶת חָשְׁשׁוּ וְדָאֲגוּ מְאֹד מֵהַמַּצָּב הַזֶּה. הֵם שָׁאֲלוּ אֶת עַצְמָם בִּדְאָגָה עֲמֻקָּה: מֵאֵיפֹה הֵם יִתְפַּרְנְסוּ וְכֵיצַד יַכְנִיסוּ אוֹכֶל לְבָתֵיהֶם כַּאֲשֶׁר הֵם לֹא יַעַבְדוּ בַּשָּׂדֶה בְּמֶשֶׁךְ כָּל שְׁנַת הַשְּׁמִיטָה?. כַּאֲשֶׁר הַחֲזוֹן אִי"שׁ וְהָרַב מִפּוֹנוֹבִיץ', הָרַב כַּהֲנְמַן, שָׁמְעוּ עַל הַחֲשָׁשׁוֹת וְעַל הַדְּאָגוֹת שֶׁל הַחַקְלָאִים הַמַּאֲמִינִים, הֵם הֶחְלִיטוּ לִנְסֹעַ בַּמְּיֻחָד אֶל קִבּוּץ 'חֵפֶץ חַיִּים'. הַמַּטָּרָה שֶׁל שְׁנֵי הָרַבָּנִים הַגְּדוֹלִים הָיְתָה לְחַזֵּק אֶת רוּחָם שֶׁל הַחַקְלָאִים הָאֵלֶּה וּלְעוֹדֵד אֶת לִבָּם בִּשְׁמִירַת הַמִּצְוָה הַקְּדוֹשָׁה.

כַּאֲשֶׁר שְׁנֵי גְּדוֹלֵי הַדּוֹר הָאֵלֶּה הִגִּיעוּ לַקִּבּוּץ וְעָמְדוּ מוּל הַשָּׂדוֹת שֶׁלֹּא עוֹבְדִים בִּשְׁנַת הַשְּׁמִיטָה, הֵם מְאֹד הִתְרַגְּשׁוּ מֵהַמַּרְאֶה הַקָּדוֹשׁ וְהַמְּרוֹמֵם. הֵם שָׁמְחוּ מְאֹד לִרְאוֹת אֶת הַשָּׂדוֹת הַנָּחִים מֵעֲבוֹדָה, וְחִיּוּךְ רָחָב וּמָלֵא אוֹר עָלָה עַל פְּנֵיהֶם שֶׁל שְׁנֵי הָרַבָּנִים הַצַּדִּיקִים. הֵם רָאוּ אֵיךְ הָאֲדָמָה וְהָעֵצִים שׁוֹבְתִים מִכָּל עֲבוֹדָה. אָז הַחֲזוֹן אִי"שׁ הִבִּיט עַל הָעֵץ הָעוֹמֵד בַּשָּׂדֶה, וְאָמַר לוֹ בְּהִתְרַגְּשׁוּת רַבָּה: "שַׁבָּת שָׁלוֹם לְךָ עֵץ!".

וְהָרַב מִפּוֹנוֹבִיץ', הָרַב כַּהֲנְמַן, עָשָׂה מַעֲשֶׂה מְרַגֵּשׁ וּמְיֻחָד עוֹד יוֹתֵר. הוּא הִתְכּוֹפֵף מַטָּה, גָּחַן עַל הַקַּרְקַע שֶׁל הַשָּׂדֶה, וְנִשַּׁק אֶת הָאֲדָמָה בְּחֹם וּבְאַהֲבָה גְּדוֹלָה. הוּא אָמַר לָאֲדָמָה בְּרֹךְ וּבְהִתְרַגְּשׁוּת: "שַׁבָּת שָׁלוֹם לָךְ אֲדָמָה". עֶצֶם הַמַּחְשָׁבָה עַל כָּךְ שֶׁהָרַב הַגָּדוֹל מִתְכּוֹפֵף, גּוֹחֵן, מְנַשֵּׁק אֶת הָאֲדָמָה וְאוֹמֵר לָהּ "שַׁבָּת שָׁלוֹם" מְעוֹרֶרֶת בָּנוּ צַמְרֹמֶרֶת גְּדוֹלָה שֶׁל הִתְרַגְּשׁוּת וּקְדֻשָּׁה.

💡 כַּאֲשֶׁר אָנוּ שוֹמְרִים עַל מִצְוֹת הַשֵּׁם בְּשִׂמְחָה וּבְאַהֲבָה, אָנוּ מַרְגִּישִׁים קְדֻשָּׁה וְשַׁבָּת שָׁלוֹם בְּכָל הָעוֹלָם.
