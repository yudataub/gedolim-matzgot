הַנֵּס שֶׁל רַבִּי אֶלְעָזָר

רַבִּי אֶלְעָזָר אִישׁ בַּרְתּוֹתָא הָיָה חָכָם וְצַדִּיק גָּדוֹל שֶׁאָהַב מְאֹד לָתֵת צְדָקָה לַנִּזְקָקִים. בְּכָל פַּעַם שֶׁגַּבָּאֵי הַצְּדָקָה הָיוּ רוֹאִים אוֹתוֹ בָּרְחוֹב, הֵם הָיוּ בּוֹרְחִים מִמֶּנּוּ וּמִתְחַבְּאִים. הַגַּבָּאִים יָדְעוּ שֶׁרַבִּי אֶלְעָזָר נוֹתֵן לָהֶם אֶת כָּל הַכֶּסֶף שֶׁיֵּשׁ לוֹ בַּכִּיס, וְהֵם הִרְגִּישׁוּ רַע מְאֹד עַל כָּךְ שֶׁלְּאִשְׁתּוֹ וּלְבִתּוֹ לֹא נִשְׁאַר מַה לֶאֱכֹל בַּבַּיִת.

יוֹם אֶחָד יָצָא רַבִּי אֶלְעָזָר אֶל הַשּׁוּק. הָיָה אֶצְלוֹ סְכוּם כֶּסֶף שֶׁמָּצָא בִּשְׁבִיל לִקְנוֹת נְדוּנְיָה וְצָרְכֵי חֲתֻנָּה עֲבוּר בִּתּוֹ. כְּשֶׁרָאוּ אוֹתוֹ גַּבָּאֵי הַצְּדָקָה בַּשּׁוּק, הֵם שׁוּב בָּרְחוּ מִמֶּנּוּ כְּדֵי שֶׁלֹּא יִקַּח מִמֶּנּוּ אֶת הַכֶּסֶף. אַךְ רַבִּי אֶלְעָזָר לֹא וִיתֵר. הוּא רָדַף אַחֲרֵיהֶם בִּמְהִירוּת, תָּפַס אוֹתָם וְהִשְׁבִּיעַ אוֹתָם לְגַלּוֹת לוֹ: "תַּגִּידוּ לִי, בִּשְׁבִיל מָה אַתֶּם אוֹסְפִים עַכְשָׁיו צְדָקָה?"

הַגַּבָּאִים עָנוּ לוֹ: "אָנוּ אוֹסְפִים כֶּסֶף עֲבוּר יָתוֹם וִיתוֹמָה שֶׁעוֹמְדִים לְהִתְחַתֵּן." כְּשֶׁשָּׁמַע זֹאת רַבִּי אֶלְעָזָר, אָמַר לָהֶם מִיָּד: "הַבַּת שֶׁלִּי אֵינָהּ יְתוֹמָה, יֵשׁ לָהּ אַבָּא שֶׁיִּדְאַג לָהּ, אֲבָל הַיָּתוֹם וְהַיְּתוֹמָה הָאֵלֶּה – אֵין מִי שֶׁיִּדְאַג לָהֶם!" בְּלִי לְהַסֵּס, נָתַן לָהֶם אֶת כָּל הַכֶּסֶף שֶׁהָיָה אֶצְלוֹ, וְנִשְׁאַר בְּיָדוֹ רַק זוּז אֶחָד בִּלְבַד.

אֶת הַזּוּז הַיָּחִיד הַזֶּה לָקַח רַבִּי אֶלְעָזָר וְקָנָה בּוֹ מְעַט חִטָּה. הוּא הֵבִיא אֶת הַחִטָּה לְבֵיתוֹ, שָׂם אוֹתָהּ בַּמַּחְסָן, וְהָלַךְ לִלְמֹד תּוֹרָה בְּבֵית הַמִּדְרָשׁ.

כַּעֲבֹר זְמַן מַה, בָּאָה אִשְׁתּוֹ שֶׁל רַבִּי אֶלְעָזָר וְשָׁאֲלָה אֶת בִּתָּהּ: "מָה קָנָה אַבָּא בַּשּׁוּק לַנְּדוּנְיָה שֶׁלָּךְ?" עָנְתָה הַבַּת: "אֲנִי לֹא יוֹדַעַת מָה הוּא קָנָה, אֲבָל כָּל מָה שֶׁהוּא קָנָה הוּא שָׂם בַּמַּחְסָן." הָלְכוּ הָאֵם וְהַבַּת לִרְאוֹת מָה יֵשׁ בַּמַּחְסָן. כְּשֶׁהִגִּיעוּ לְשָׁם, גִּילוּ דָּבָר מַפְלִיא וּמַרְהִיב! הַמַּחְסָן, שֶׁהָיָה רֵיק קוֹדֵם לָכֵן, הָיָה כָּעֵת מָלֵא וּמֻצָּף בְּחִטִּים רַבּוֹת עַד שֶׁלֹּא יָכְלוּ אֲפִילוּ לִפְתֹּחַ אֶת דֶּלֶת הַמַּחְסָן.

הַבַּת הֵבִינָה מִיָּד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה לָהֶם נֵס גָּדוֹל. הִיא רָצָה מְהֵרָה אֶל בֵּית הַמִּדְרָשׁ, קָרְאָה לְאָבִיהָ וְאָמְרָה לוֹ: "אַבָּא, בֹּא תִּרְאֶה מָה עָשָׂה לְךָ אוֹהַבְךָ!" כְּשֶׁבָּא רַבִּי אֶלְעָזָר לַבַּיִת וְרָאָה אֶת הַנֵּס וְאֶת הַחִטִּים הָרַבּוֹת, הוּא הִשְׁבִּיעַ אֶת אִשְׁתּוֹ וְאֶת בִּתּוֹ שֶׁלֹּא יִקְחוּ מֵהֶן שׁוּם דָּבָר מֵעֵבֶר לְמָה שֶׁנּוֹתְנִים לַאֲנָשִׁים אֲחֵרִים, אֶלָּא יִתְחַלְקוּ בַּחִטִּים וּבַכֶּסֶף יַחַד עִם כָּל הָעֲנִיִּים.
