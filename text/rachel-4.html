הַדְּמָעוֹת שֶׁהִצִּילוּ אֶת קֶבֶר רָחֵל

בְּקַיִץ שְׁנַת תַּשְׁנָ"ה (1995), בְּמַהֲלַךְ הַדִּיּוּנִים הַמְּדִינִיִּים עַל הֶסְכְּמֵי אוֹסְלוֹ בּ', עָמְדָה מֶמְשֶׁלֶת יִשְׂרָאֵל לְהַעֲבִיר אֶת הַשְּׁלִיטָה בְּאֵזוֹר קֶבֶר רָחֵל אִמֵּנוּ לִידֵי הָרָשׁוּת הַפַּלַסְטִינִית. כַּאֲשֶׁר שָׁמְעוּ עַל הַכַּוָּנָה הַזֹּאת חַבֵּר הַכְּנֶסֶת חָנָן פּוֹרָת וְהָרַב מְנַחֵם פְּרוּשׁ, הֵם הֶחְלִיטוּ לִפְעֹל יַחַד בְּכָל כֹּחָם כְּדֵי לִמְנֹעַ אֶת הַעֲבָרַת הַמָּקוֹם הַקָּדוֹשׁ וּלְהַשְׁאִיר אֶת הַשְּׁלִיטָה בְּקֶבֶר רָחֵל בִּידֵי מְדִינַת יִשְׂרָאֵל.

לְשֵׁם כָּךְ, חָנָן פּוֹרָת יָזַם פְּגִישָׁה מְיֻחֶדֶת וּדְחוּפָה עִם רֹאשׁ הַמֶּמְשָׁלָה יִצְחָק רָבִין. אֶל הַפְּגִישָׁה הֵבִיא עִמּוֹ חָנָן תַּצְלוֹם אֲוִיר גָּדוֹל וּמְפֹרָט שֶׁל מִתְחַם קֶבֶר רָחֵל וְהָאֵזוֹר שֶׁסְּבִיבוֹ. הוּא סִמֵּן עַל גַּבֵּי הַתַּצְלוֹם קַוִּים בְּרוּרִים, כְּדֵי לְהַמְחִישׁ לְרָבִין אֶת הַמֶּרְחָקִים הַקְּצָרִים וְאֶת טְוָחֵי הַיְרִי הַמְּסֻכָּנִים שֶׁבֵּין שְׁכוּנַת גִּילֹה הַיְרוּשַׁלְמִית לְבֵין הָעִיר בֵּית לֶחֶם. בְּנוֹסָף, שָׁאַל חָנָן פּוֹרָת אֶת רָבִין כְּאַתָּגָר, הַאִם הָיָה מַסְכִּים לְהַעֲבִיר גַּם אֶת קֶבֶר דָּוִד בֶּן-גּוּרִיּוֹן לְשְׁלִיטַת הָרָשׁוּת הַפַּלַסְטִינִית. בְּאֶמְצָעוּת שְׁאָלָה זוֹ הוּא הִמְחִישׁ לְרֹאשׁ הַמֶּמְשָׁלָה אֶת הַמַּשְׁמָעוּת הַעֲמֻקָּה וְהַחֲשׁוּבָה שֶׁל הַמָּקוֹם עֲבוּר הָעָם הַיְהוּדִי כֻּלּוֹ, וְלֹא רַק עֲבוּר הַצִּבּוּר הַדָּתִי.

יַחַד עִם חָנָן פּוֹרָת נִכְנַס לְלִשְׁכַּת רֹאשׁ הַמֶּמְשָׁלָה גַּם הָרַב מְנַחֵם פְּרוּשׁ. בַּמַּהֲלָךְ הַפְּגִישָׁה, מִתּוֹךְ כְּאֵב עָמֹק וְאַהֲבָה גְּדוֹלָה לְאִמָּא רָחֵל, פָּרַץ הָרַב פְּרוּשׁ בִּבְכִי רַעַד נִרְגָּשׁ, אָחַז בְּבִגְדוֹ שֶׁל רָבִין וְזָעַק בְּאָזְנוֹ מִכָּל הַלֵּב: "זוֹ אִמָּא רָחֵל! אֵיךְ אַתָּה נֹתֵן אֶת הַקֶּבֶר שֶׁל אִמָּא?!" רֹאשׁ הַמֶּמְשָׁלָה יִצְחָק רָבִין נִרְגַּשׁ מְאֹד מֵהַבְּכִי הַעַז וּמֵהַשְּׁאֵלָה הַנּוֹגַעַת לַלֵּב. לְפִי הָעֵדוּיוֹת מֵהַפְּגִישָׁה, בְּאוֹתוֹ מַעֲמָד מַמָּשׁ, הוֹדִיעַ רֹאשׁ לִשְׁכָּתוֹ שֶׁל רָבִין לְחָנָן פּוֹרָת וְלָרַב פְּרוּשׁ כִּי הַהַחְלָטָה לְהַעֲבִיר אֶת הַמִּתְחָם לְשְׁלִיטָה פַּלַסְטִינִית בֻּטְלָה. בִּזְכוּת פְּעֻלָּתָם, קֶבֶר רָחֵל נִשְׁאַר בְּשְׁלִיטָה יִשְׂרְאֵלִית.

לְאַחַר רֶצַח יִצְחָק רָבִין בְּחֹדֶשׁ חֶשְׁוָן תַּשְׁנָ"ו (נוֹבֶמְבֶּר 1995) בִּידֵי יְהוּדִי, סִפֵּר חָנָן פּוֹרָת שֶׁהוּא חָשׁ שֶׁרָחֵל אִמֵּנוּ בּוֹכָה בְּאוֹתָם יָמִים בְּכִי מְיֻחָד, בְּכִי תַּמְרוּרִים, עַל בְּנָהּ יִצְחָק רָבִין שֶׁלֹּא נֶהֱרַג בִּידֵי אוֹיֵב אֶלָּא בִּידֵי יְהוּדִי. חָנָן פּוֹרָת הִסְבִּיר כִּי גַּם רָבִין, עַל אַף חִילוֹנִיּוֹתוֹ וְעַל אַף הַמַּחֲלֹקֶת הַגְּדוֹלָה שֶׁהָיְתָה סְבִיבוֹ, נֶחְשָׁב בְּעֵינָיו לְאֶחָד מִבָּנֶיהָ שֶׁל רָחֵל אִמֵּנוּ, שֶׁהִיא בּוֹכָה עָלָיו וּמִתְאַבֶּלֶת עַל לְכָתוֹ.

💡 כַּאֲשֶׁר מְדַבְּרִים וּפֹעֲלִים מִתּוֹךְ אַהֲבָה אֲמִתִּית וּמְסִירוּת לְעַם יִשְׂרָאֵל וְלַמְּקוֹמוֹת הַקְּדוֹשִׁים, יְכוֹלִים לִנְגֹּעַ בַּלֵּב וּלְשַׁנּוֹת גְּזֵרוֹת קָשׁוֹת.
