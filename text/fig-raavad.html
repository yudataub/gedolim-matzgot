רוּחַ הַקֹּדֶשׁ בְּבֵית הַמִּדְרָשׁ שֶׁל הָרָאָבַ"ד

לִפְנֵי כִּשְׁמֹנֶה מֵאוֹת וַחֲמִשִּׁים שָׁנָה, בָּעִיר פּוֹשְׁקִירָא שֶׁבִּדְרוֹם צָרְפַת, חַי חָכָם גָּדוֹל וְקָדוֹשׁ שֶׁנִּקְרָא רַבִּי אַבְרָהָם בֵּן דָּוִד, וּבְקִצּוּר – הָרָאָבַ"ד. הָרָאָבַ"ד עָמַד בְּרֹאשׁ יְשִׁיבָה גְּדוֹלָה מְאֹד, לִמֵּד תּוֹרָה בְּאַהֲבָה, וְגַם דָּאַג לְכָל הַצְּרָכִים שֶׁל תַּלְמִידָיו. בְּבֵית הַמִּדְרָשׁ שֶׁלּוֹ לָמְדוּ הַחֲכָמִים וְהַתַּלְמִידִים בִּשְׁקִידָה נִפְלָאָה, וְחִפְּשׂוּ תָּמִיד לְהָבִין אֶת הַהֲלָכָה הַנְּכוֹנָה עַד עֻמְקָהּ.

כַּמָּה שָׁנִים לִפְנֵי שֶׁהִגִּיעַ אֲלֵיהֶם סֵפֶר הֲלָכָה חָדָשׁ, עָסְקוּ הָרָאָבַ"ד וְתַלְמִידָיו בְּהִלְכוֹת אַרְבַּעַת הַמִּינִים שֶׁל חַג הַסֻּכּוֹת. הֵם הִתְעַמְּקוּ מְאֹד בִּשְׁאֵלָה חֲשׁוּבָה עַל עֲנַף הַהֲדַס: מָה הַדִּין אִם רֹאשׁוֹ שֶׁל הַהֲדַס נִקְטַם, כְּלוֹמַר נַחְתַּךְ? הַאִם מֻתָּר לְבָרֵךְ עָלָיו בַּחַג וְהוּא כָּשֵׁר לַמִּצְוָה, אוֹ שֶׁהוּא פָּסוּל? הָרָאָבַ"ד וְחַכְמֵי יְשִׁיבָתוֹ עִיְּנוּ בַּמִּשְׁנָה וּבַמְּקוֹרוֹת הַקְּדוֹשִׁים, וְלָמְדוּ בְּעֹמֶק רַב מִתּוֹךְ רָצוֹן לְהַגִּיעַ לָאֱמֶת. וְאָז קָרָה דָּבָר מֻפְלָא וּמִתְרַגֵּשׁ: רוּחַ הַקֹּדֶשׁ הוֹפִיעָה בְּבֵית הַמִּדְרָשׁ שֶׁלָּהֶם! מִן הַשָּׁמַיִם סִיְּעוּ לָהֶם לְהָבִין אֶת הַתּוֹרָה, וְהֵם הִגִּיעוּ לַמַּסְקָנָה הַבְּרוּרָה שֶׁהֲדַס שֶׁנִּקְטַם רֹאשׁוֹ הוּא פָּסוּל לְמִצְוָה, כְּמוֹ שֶׁמִּתְבָּרֵר מִסְּתָם הַמִּשְׁנָה. הֵם גַּם הִסְבִּירוּ שֶׁדִּבְרֵי רַבִּי טַרְפוֹן בַּמִּשְׁנָה מְדַבְּרִים עַל עִנְיָן אַחֵר לַחֲלוּטִין וְלֹא עַל הֲדַס שֶׁנִּקְטַם רֹאשׁוֹ, וְאֶת כָּל הַבֵּרוּר הַזֶּה כָּתַב הָרָאָבַ"ד בַּחִבּוּר שֶׁלּוֹ.

עָבְרוּ שָׁנִים, וְאֶל הָרָאָבַ"ד הִגִּיעַ הַסֵּפֶר הַמְּפֻרְסָם "מִשְׁנֵה תּוֹרָה" שֶׁכָּתַב הָרַמְבַ"ם, רַבִּי מֹשֶׁה בֵּן מַיְמוֹן. הָרָאָבַ"ד לָמַד אֶת הַסֵּפֶר בְּעִיּוּן רַב. הוּא הֶעֱרִיךְ מְאֹד אֶת הַסֵּפֶר, וּבְכָל מָקוֹם שֶׁחָשַׁב אַחֶרֶת אוֹ רָצָה לְהוֹסִיף בֵּרוּר, כָּתַב הַעֲרָה מְיֻחֶדֶת. כְּשֶׁהִגִּיעַ לְהִלְכוֹת לוּלָב, רָאָה שֶׁהָרַמְבַ"ם כָּתַב הֲלָכָה כָּזֹאת: "הֲדַס שֶׁנִּקְטַם רֹאשׁוֹ – כָּשֵׁר".

כְּשֶׁקָּרָא הָרָאָבַ"ד אֶת הַמִּלִּים הָאֵלֶּה, נִזְכַּר מִיָּד בַּלִּמּוּד הֶעָמֹק שֶׁלָּמַד עִם תַּלְמִידָיו בְּבֵית הַמִּדְרָשׁ כַּמָּה שָׁנִים קֹדֶם לָכֵן, וּבָרוּחַ הַקֹּדֶשׁ שֶׁהוֹפִיעָה אֶצְלָם. הוּא כָּתַב מִיָּד הַעֲרָה חֲשׁוּבָה עַל דִּבְרֵי הָרַמְבַ"ם, שֶׁנִּקְרֵאת "הַשָּׂגָה". בְּהַשָּׂגָתוֹ הִסְבִּיר הָרָאָבַ"ד שֶׁכְּבָר לִפְנֵי כַּמָּה שָׁנִים הוֹפִיעָה רוּחַ הַקֹּדֶשׁ בְּבֵית מִדְרָשָׁם וְהֵם הֶעֱלוּ שֶׁהַהֲדַס פָּסוּל, וְכִי מִן הַשָּׁמַיִם הִשְׁאִירוּ לוֹ מָקוֹם לְבָרֵר וּלְגַלּוֹת אֶת הָאֱמֶת הַזֹּאת.

הַהַעֲרוֹת הָאֵלֶּה שֶׁל הָרָאָבַ"ד נִקְרָאוֹת "הַשָּׂגוֹת הָרָאָבַ"ד", וְהֵן מֻדְפָּסוֹת עַד הַיּוֹם הַזֶּה בְּכָל סִפְרֵי "מִשְׁנֵה תּוֹרָה" שֶׁל הָרַמְבַ"ם. חַכְמֵי יִשְׂרָאֵל וְתַלְמִידֵי חֲכָמִים בְּכָל הַדּוֹרוֹת לוֹמְדִים אֶת דִּבְרֵי הָרַמְבַ"ם וְאֶת הַשָּׂגוֹת הָרָאָבַ"ד יַחַד בְּשִׂמְחָה, וּמִתְרַגְּשִׁים לִרְאוֹת אֵיךְ הַשֵּׁם יִתְבָּרַךְ עוֹזֵר לְלוֹמְדֵי הַתּוֹרָה וּמֵאִיר אֶת עֵינֵיהֶם בִּשְׁקִידָתָם.

💡 כְּשֶׁלּוֹמְדִים תּוֹרָה בִּשְׁקִידָה וּבֶאֱמֶת, הַשֵּׁם יִתְבָּרַךְ עוֹזֵר וּמֵאִיר אֶת הָעֵינַיִם לְהָבִין אֶת הַתּוֹרָה.
