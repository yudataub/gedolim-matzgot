רְפוּאָה בִּזְכוּת הַתּוֹרָה שֶׁל הַטַּז

מָעָשֶׂה בְּאִשָּׁה אַחַת שֶׁהָיְתָה לָהּ יַלְדָּה חֲמוּדָה. הַיַּלְדָּה תָּמִיד הִתְנַהֲגָה יָפֶה מְאֹד וְשִׂמְּחָה אֶת אִמָּהּ. אֲבָל פִּתְאֹם, כַּאֲשֶׁר הַיַּלְדָּה הִגִּיעָה לְגִיל שְׁתֵּים עֶשְׂרֵה וָחֵצִי, מַשּׁَهُו מְוּזָר קָרָה לָהּ. הִיא הִתְחִילָה לִצְעֹק תָּמִיד בַּבַּיִת, לְהִשְׁתַּולֵל לְלֹא הַפְסָקָה, וְאֲפִילוּ זָרְקָה חֲפָצִים וּדְבָרִים שׁוֹנִים לְכָל עֵבֶר. הִיא הִתְנַהֲגָה מַמָּשׁ מַמָּשׁ לֹא יָפֶה. הָאֵם הַדּוֹאַגְתְּ נִסְתָה מְאֹד לְדַבֵּר עִמָּהּ בְּנַחַת וּלְבַקֵּשׁ מִמֶּנָּה לְהִתְנַהֵג יָפֶה, אֲבָל הַיַּלְדָּה פַּשׁוּט לֹא הִקְשִׁיבָה לָאִמָּא כְּלָל.

הָאֵם וּבְנֵי מִשְׁפַּחְתָּהּ נִסּוּ לִמְצֹא פַּתְרוֹן לַבְּעָיָה הַקָּשָׁה. בְּאֵיזֶהוּ שָׁלָב, הֵם הָלְכוּ לְרוֹפְאִים שׁוֹנִים, וְגַם הָלְכוּ לְכָל מִינֵי אֲנָשִׁים אֲחֵרִים כְּדֵי לְבַקֵּשׁ עֶזְרָה, אֲבָל אַף אֶחָד לֹא עָזַר לָהֶם וְלֹא מָצָא רְפוּאָה לַיַּלְדָּה. הָאֵם הָיְתָה עֲצוּבָה מְאֹד. וְהִנֵּה, בָּאָה אֶל הָאֵם אַחַת מֵהַחֲבָרוֹת שֶׁלָּהּ, רָאֲתָה אֶת הַצַּעַר הַגָּדוֹל, וְאָמְרָה לָהּ: "אוּלַי תְּנַסִּי לָלֶכֶת אֶל הַטַּז? אוּלַי הַטַּז יוּכַל לַעֲזֹר לָךְ?". הָאֵם, אַחֲרֵי שֶׁנִּסְתָה כְּבָר אֶת כָּל הָאֶפְשָׁרוֹיוֹת וְלֹא יָדְעָה מָה עוֹד לַעֲשׂוֹת, הֶחְלִיטָה לָלֶכֶת אֶל הַטַּז, הָרַב הַגָּדוֹל.

הָאֵם הִגִּיעָה אֶל הַטַּז, הִסְתַּכְּלָה עָלָיו וְאָמְרָה לוֹ בְּכֶאֶב: "כְּבוֹד הָרַב, הַבַּת שֶׁלִּי הִיא כָּךְ וְכָךְ, כָּךְ וְכָךְ קָרָה אִתָּהּ בַּזְּמַן הָאַחֲרוֹן. אֲנִי לֹא יוֹדַעַת מַה קָּרָה לָהּ. נִסֵּינוּ לָלֶכֶת לְרוֹפְאִים, נִסֵּינוּ לָלֶכֶת לְכָל מִינֵי אֲנָשִׁים, אֲבָל אַף אֶחָד לֹא עָזַר לָנוּ. אוּלַי תּוּכַל אַתָּה לְרַפֵּא אֶת הַבַּת שֶׁלִּי?".

הַטַּז הִסְתַּכֵּל עָלֶיהָ וְאָמַר לָהּ בִּפְלִיאָה: "אֲבָל אֲנִי לֹא רוֹפֵא, אֲנִי רַב!". אָמְרָה לוֹ הָאִשָּׁה בֶּאֱמוּנָה גְּדוֹלָה: "כֵּן, אֲנִי יוֹדַעַת, אֲבָל בִּזְכוּת הַתּוֹרָה שֶׁלְּךָ אוּלַי תַּצְלִיחַ לְרַפֵּא אֶת הַבַּת שֶׁלִּי". הַטַּז יָשַׁב מְעַט, חָשַׁב עֲמֻקּוֹת, וְאָז אָמַר לָהּ: "תִּרְאִי, הַיּוֹם בַּבֹּקֶר אֲנִי לָמַדְתִּי גְּמָרָא, וְהָיְתָה לִי קֻשְׁיָא חֲזָקָה וְקָשָׁה עַל דִּבְרֵי רַשִּׁ"י. לֹא הֵבַנְתִּי אֶת הָרַשִּׁ"י, וְהִתְפַּלַּלְתִּי מִכָּל הַלֵּב לַהַ' שֶׁיִּשְׁלַח לִי תְּשׁוּבָה וַהֲבָנָה לַדָּבָר שֶׁאֲנִי לֹא מֵבִין בְּרַשִּׁ"י. וְהַהֵ' בֶּאֱמֶת שָׁלַח לִי אֶת הַתְּשׁוּבָה הַנְּכוֹנָה. יְהִי רָצוֹן שֶׁבִּזְכוּת הַתְּשׁוּבָה הַזֹּאת, בִּזְכוּת הַתֵּרוּץ הַזֶּה שֶׁמָּצָאתִי בַּתּוֹרָה, יִהְיֶה לַבַּת שֶׁלָּךְ רְפוּאָה שְׁלֵמָה!".

הָאִשָּׁה, שֶׁרַק שָׁמְעָה אֶת הַמִּלִּים "רְפוּאָה שְׁלֵמָה" יוֹצְאוֹת מִפִּיו שֶׁל הָרַב, שָׂמְחָה שִׂמְחָה גְּדוֹלָה מְאֹד וְהָלְכָה מִיָּד חֲזָרָה לַבַּיִת שֶׁלָּהּ. כְּשֶׁהִיא מַגִּיעָה לַבַּיִת שֶׁלָּהּ, הִיא רוֹאָה אֶת הַיַּלְדָּה יְשֵׁנָה. הַיַּלְדָּה יָשְׁנָה שֵׁנָה חֲזָקָה מְאֹד. וּבְאֵיזֶהוּ שָׁלָב הַיַּלְדָּה קָמָה, וְאֵיךְ שֶׁהִיא קָמָה מִשְּׁנָתָהּ, הִיא מִסְתַּכֶּלֶת עַל סְבִיבָתָהּ וְאוֹמֶרֶת: "אִמָּא, לָמָּה הַבַּיִת פֹּה כָּל כָּךְ מְבֻלָּגָן? בּוֹאִי נַתְחִיל לְסַדֵּר אוֹתוֹ!". הִיא מִתְחִילָה מִיָּד לְסַדֵּר אֶת הַחֲפָצִים וּמִתְחִילָה לְהַחְזִיר אֶת כָּל הַדְּבָרִים לַמָּקוֹם, וְהִיא כְּבָר מְדַבֶּרֶת בְּנַחַת וּבְרֹגַע. פִּתְאֹם, הָאִשָּׁה הֵבִינָה בְּשִׂמְחָה שֶׁבִּזְכוּת הַתּוֹרָה שֶׁהַטַּז לָמַד, הַיַּלְדָּה שֶׁלָּהּ הִתְרַפְּאָה.
