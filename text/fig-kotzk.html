אֵיפֹה נִמְצָא הַקָּדוֹשׁ בָּרוּךְ הוּא?

לִפְנֵי שָׁנִים רַבּוֹת, בְּאֶרֶץ פּוֹלִין, חַי רַב מְפֻרְסָם וְקָדוֹשׁ שֶׁנִּקְרָא רַבִּי מְנַחֵם מֶנְדֶל מִקּוֹצְק. עוֹד כְּשֶׁהָיָה יֶלֶד וְנַעַר צָעִיר, כֻּלָּם קָרְאוּ לוֹ "הָעִלּוּי מִגּוֹרַיי", כִּי הוּא יָדַע לִלְמֹד תּוֹרָה בְּצוֹרָה נִפְלָאָה וַחֲכָמָה מְאֹד. כְּשֶׁגָּדַל, הָפַךְ לִהְיוֹת תַּלְמִיד שֶׁל גְּדוֹלֵי הַדּוֹר, כְּמוֹ "הַחוֹזֶה מִלּוּבְּלִין", "הַיְּהוּדִי הַקָּדוֹשׁ" מִפְּשִׁיסְחָא, וְרַבִּי שִׁמְחָה בּוֹנֶם מִפְּשִׁיסְחָא. לְאַחַר מִכֵּן הָפַךְ רַבִּי מְנַחֵם מֶנְדֶל לְמַנְהִיג חֲסִידִים גָּדוֹל בָּעִיר קוֹצְק. הָרַבִּי מִקּוֹצְק דָּרַשׁ מֵהַתַּלְמִידִים וּמֵהַחֲסִידִים שֶׁלּוֹ דָּבָר אֶחָד חָשׁוּב מֵעַל הַכֹּל: לִהְיוֹת אֲנָשִׁים שֶׁל אֱמֶת. הוּא לֹא הִסְכִּים שֶׁאֲנָשִׁים יַעֲשׂוּ דְּבָרִים רַק כְּדֵי לַעֲשׂוֹת רֹשֶׁם עַל אֲחֵרִים אוֹ לְהַעֲמִיד פָּנִים, אֶלָּא רָצָה שֶׁכָּל דָּבָר יִהְיֶה אֲמִתִּי מִתּוֹךְ הַלֵּב. בֵּין תַּלְמִידָיו הָרַבִּים הָיָה גַּם רַבִּי יִצְחָק מֵאִיר אַלְתֶּר, שֶׁנִּקְרָא לְאַחַר מִכֵּן "הַחִדּוּשֵׁי הָרִי"ם" מִגּוֹר.

יוֹם אֶחָד, יָשְׁבוּ אֵצֶל הָרַבִּי מִקּוֹצְק תַּלְמִידֵי חֲכָמִים רַבִּים וְלָמְדוּ תּוֹרָה. פִּתְאֹם פָּנָה אֲלֵיהֶם הָרַבִּי וְשָׁאַל אוֹתָם שְׁאֵלָה מַפְתִּיעָה: "אֵיפֹה נִמְצָא הַקָּדוֹשׁ בָּרוּךְ הוּא?". הַתַּלְמִידִים הִתְפַּלְאוּ מְאֹד מֵהַשְּׁאֵלָה הַזֹּאת. הֵם חָשְׁבוּ לְעַצְמָם: מָה הַשְּׁאֵלָה הַזֹּאת? הֲרֵי כָּל יֶלֶד וְכָל תַּלְמִיד יוֹדֵעַ שֶׁכָּתוּב בַּפָּסוּק "מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ", וְהַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא בְּכָל מָקוֹם בָּעוֹלָם! הֵם עָנוּ לָרַבִּי מִיָּד: "הַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא בְּכָל מָקוֹם!".

הִבִּיט בָּהֶם הָרַבִּי מִקּוֹצְק וְאָמַר לָהֶם אֶת תְּשׁוּבָתוֹ הָעֲמֻקָּה וְהַנִּפְלָאָה: "הַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא בַּמָּקוֹם שֶׁנּוֹתְנִים לוֹ לִכְנֹס!". הָרַבִּי הִסְבִּיר לָהֶם אֶת הַכַּוָּנָה שֶׁל דְּבָרָיו: נָכוֹן שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא בְּכָל הָעוֹלָם כֻּלּוֹ, אֲבָל בְּתוֹךְ הַלֵּב שֶׁל הָאָדָם, הַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא רַק כְּשֶׁהָאָדָם פּוֹתֵחַ לוֹ אֶת הַלֵּב. כְּשֶׁהָאָדָם מִתְפַּלֵּל בֶּאֱמֶת, כְּשֶׁהוּא לוֹמֵד תּוֹרָה בֶּאֱמֶת, וּכְשֶׁהוּא עוֹשֶׂה מִצְוֹת בְּכָל הַלֵּב – הוּא מַכְנִיס אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא לְתוֹךְ חַיָּיו.

עוֹד סֻפַּר עַל הָרַבִּי מִקּוֹצְק, שֶׁלִּפְנֵי פְּטִירָתוֹ אָמַר מִשְׁפָּט מְעוֹרֵר הַשְׁרָאָה: "כָּל מִי שֶׁיַּנִּיחַ אֶת רֹאשׁוֹ בַּתּוֹרָה – אֲנִי עָרֵב בַּעֲדוֹ". הָרַבִּי מִקּוֹצְק לֹא הִשְׁאִיר אַחֲרָיו סְפָרִים שֶׁכָּתַב בְּעַצְמוֹ, אֲבָל הָאֲמָרוֹת הַקְּצָרוֹת, הַחַדּוֹת וְהַנִּפְלָאוֹת שֶׁלּוֹ עָבְרוּ מִפֶּה לְאֹזֶן, מֵחָסִיד לְחָסִיד, וְנִכְתְבוּ בִּסְפָרִים שׁוֹנִים. עַד הַיּוֹם, כָּל מִי שֶׁשּׁוֹמֵעַ אֶת אֲמָרוֹתָיו הַקְּדוֹשׁוֹת שֶׁל הָרַבִּי מִקּוֹצְק, מַרְגִּישׁ כֵּיצַד הֵן נוֹגְעוֹת יָשָׁר בְּתוֹךְ לִבּוֹ וּמְעוֹרְרוֹת אוֹתוֹ לְאַהֲבַת הַשֵּׁם וּלְעַשִׂיַּת טוֹב.

💡 אָנוּ יְכוֹלִים לִפְתֹחַ אֶת הַלֵּב לַקָּדוֹשׁ בָּרוּךְ הוּא בְּכָל רֶגַע עַל יְדֵי תְּפִלָּה אֲמִתִּית, לִמּוּד תּוֹרָה וַעֲשִׂיַּת מִצְוֹת בְּשִׂמְחָה.
