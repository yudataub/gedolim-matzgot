רַבִּינָא וַחֲתָמַת הַתַּלְמוּד

לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, בְּאֶרֶץ בָּבֶל הָרְחוֹקָה, חַי חָכָם גָּדוֹל וְקָדוֹשׁ שֶׁשְּׁמוֹ הָיָה רַבִּינָא. שְׁמוֹ הַמָּלֵא וְהָרִשְׁמִי שֶׁל רַבִּינָא הָיָה רַב אֲבִינָא בְּרֵיהּ דְּרַב הוּנָא, כְּפִי שֶׁמְּסַפֶּרֶת לָנוּ אִגֶּרֶת רַב שְׁרִירָא גָּאוֹן. רַבִּינָא הָיָה הֶחָכָם הָאַחֲרוֹן מִבֵּין כָּל חַכְמֵי הַתַּלְמוּד הַבַּבְלִי הַמְּפֹאָרִים, אֵלּוּ הַחֲכָמִים הַקְּדוֹשִׁים שֶׁלָּמְדוּ וְלִמְדוּ אֶת הַתּוֹרָה בִּשְׁקִידָה עֲצוּמָה.

בְּמֶשֶׁךְ דּוֹרוֹת רַבִּים מְאֹד, דּוֹר אַחַר דּוֹר, לָמְדוּ חַכְמֵי יִשְׂרָאֵל אֶת הַתּוֹרָה, הוֹסִיפוּ בֵּרוּרִים וְהוֹרוּ אֶת הַהֲלָכוֹת הַקְּדוֹשׁוֹת לְעַם יִשְׂרָאֵל. כָּל דּוֹר שֶׁעָבַר הוֹסִיף עוֹד חָכְמָה וְעוֹד הַסְבָּרוֹת יְקָרוֹת לַהוֹרָאָה שֶׁל הַתּוֹרָה. אַךְ הַחָכְמָה הָיְתָה רַבָּה וּרְחָבָה מְאֹד, וְהָיָה צֹרֶךְ גָּדוֹל לֶאֱסֹף אֶת כָּל הַדִּבְרִים הַקְּדוֹשִׁים הָאֵלֶּה וּלְסַדֵּר אוֹתָם בְּצוּרָה מְסֻדֶּרֶת וּבְרוּרָה לְכָל עַם יִשְׂרָאֵל.

הֶחָכָם הַגָּדוֹל, רַב אָשִׁי, הָיָה זֶה שֶׁהִתְחִיל אֶת הַמְּלָאכָה הָעֲצוּמָה וְהַקְּדוֹשָׁה הַזֹּאת. הוּא הִתְחִיל לֶאֱסֹף וּלְסַדֵּר אֶת כָּל הַדִּיּוּנִים, הַמַּאֲמָרִים וְהַהֶסְבֵּרִים שֶׁל חַכְמֵי הַגְּמָרָא אֶל תּוֹךְ מְפַעַל עֲנָקִי שֶׁנִּקְרָא הַתַּלְמוּד הַבַּבְלִי. אַךְ הַמְּלָאכָה הָיְתָה עֲנָקִית וּמוּרְכָבָה מְאֹד, וְלֹא הִסְתַּיְמָה בִּימֵי רַב אָשִׁי.

אָז הִגִּיעַ רַבִּינָא. בְּצִדְקוּתוֹ, בִּשְׁקִידָתוֹ הַנִּפְלָאָה וּבְאַהֲבַת הַתּוֹרָה הָעֲמוּקָה שֶׁלּוֹ, הִמְשִׁיךְ רַבִּינָא אֶת עֲבוֹדָתוֹ שֶׁל רַב אָשִׁי, סִדֵּר אֶת כָּל הַדְּבָרִים וְהִשְׁלִים אֶת חֲתָמַת הַתַּלְמוּד הַבַּבְלִי כֻּלּוֹ.

בְּמַסֶּכֶת בָּבָא מְצִיעָא בַּתַּלְמוּד הַבַּבְלִי, מְבִיאִים חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה מַאֲמָר מְפֻרְסָם וְחָשׁוּב מְאֹד: "רַבִּי וְרַבִּי נָתָן – סוֹף מִשְׁנָה. רַב אָשִׁי וְרַבִּינָא – סוֹף הוֹרָאָה". חֲכָמֵינוּ מַסְבִּירִים כִּי כְּפִי שֶׁרַבִּי יְהוּדָה הַנָּשִׂיא (שֶׁנִּקְרָא 'רַבִּי') וְרַבִּי נָתָן הָיוּ אֵלּוּ שֶׁסִּיְּמוּ וְחָתְמוּ אֶת תְּקוּפַת הַמִּשְׁנָה, כָּךְ רַב אָשִׁי וְרַבִּינָא הָיוּ אֵלּוּ שֶׁסִּיְּמוּ וְחָתְמוּ אֶת תְּקוּפַת הַהוֹרָאָה וְהַתַּלְמוּד. דּוֹר אַחַר דּוֹר הִתְרַחֲבָה הַהוֹרָאָה, עַד יְמֵי רַבִּינָא, וּלְאַחַר רַבִּינָא נֶפְסְקָה תְּקוּפָה זוֹ.

תַּפְקִידָם הַקָּדוֹשׁ שֶׁל רַב אָשִׁי וְרַבִּינָא הָיָה רָשׁוּם וּמְכֻוָּן מִשָּׁמַיִם עוֹד מִבְּרִיאַת הָעוֹלָם! אִגֶּרֶת רַב שְׁרִירָא גָּאוֹן מְסַפֶּרֶת לָנוּ שֶׁהֶחָכָם שְׁמוּאֵל יַרְחִינָאָה רָאָה כָּתוּב בְּ"סִפְרָא דְּאָדָם הָרִאשׁוֹן", מִלִּים קְדוֹשׁוֹת שֶׁנִּכְתְּבוּ עוֹד בִּתְחִלַּת יְמֵי הָעוֹלָם: "רַב אָשִׁי וְרַבִּינָא סוֹף הוֹרָאָה". הַקָּדוֹשׁ בָּרוּךְ הוּא יָדַע מֵרֹאשׁ שֶׁרַב אָשִׁי וְרַבִּינָא יִהְיוּ אֵלּוּ שֶׁיַּחְתְּמוּ אֶת הַתַּלְמוּד.

בְּיוֹם רְבִיעִי, יִג-בִּכִסְלֵו שְׁנַת ד'ר"ס (שֶׁהִיא שְׁנַת חֲמֵשׁ מֵאוֹת לַסְּפִירָה, וּשְׁנַת תַּתְיָ"א לְמִנְיַן הַשְּׁטָרוֹת), נִפְטַר רַבִּינָא הַצַּדִּיק. בְּאוֹתוֹ יוֹם הוּבָאָה לְסִיּוּם תְּקוּפַת הַהוֹרָאָה שֶׁל חַכְמֵי הַתַּלְמוּד.

מֵאָז וְעַד הַיּוֹם הַזֶּה, בְּכָל מָקוֹם וּמָקוֹם בָּעוֹלָם – בְּבָתֵּי כְּנֵסִיּוֹת, בְּבָתֵּי מִדְרָשׁ, בִּישִׁיבוֹת וּבְבָתֵּי סֵפֶר – לוֹמְדִים יְלָדִים, נְעָרִים וַחֲכָמִים אֶת הַתַּלְמוּד הַבַּבְלִי שֶׁנֶּחְתַּם בִּימֵי רַבִּינָא, וּמַמְשִׁיכִים אֶת שַׁרְשֶׁרֶת הַתּוֹרָה הַקְּדוֹשָׁה לָנֶצַח.
