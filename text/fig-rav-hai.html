הַבֵּן שֶׁעָמַד לְצַד אָבִיו

בִּזְמַן הַגְּאוֹנִים הָיְתָה יְשִׁיבַת פּוּמְבְּדִיתָא אַחַת מִן הַיְשִׁיבוֹת הַגְּדוֹלוֹת וְהַחֲשׁוּבוֹת בְּיוֹתֵר שֶׁל עַם יִשְׂרָאֵל. בְּרֹאשׁ הַיְשִׁיבָה עָמַד חָכָם גָּדוֹל וּמְפֻרְסָם שֶׁשְּׁמוֹ רַב שְׁרִירָא גָּאוֹן. לְרַב שְׁרִירָא גָּאוֹן הָיָה בֵּן חָכָם וְצַדִּיק שֶׁשְּׁמוֹ רַב הַאי גָּאוֹן. רַב הַאי אָהַב אֶת הַתּוֹרָה מְאֹד, לָמַד בִּשְׁקִידָה, וְכִבֵּד אֶת אָבִיו בְּכָל לִבּוֹ. כַּאֲשֶׁר רַב שְׁרִירָא הָיָה לְגָאוֹן וּלְרֹאשׁ הַיְשִׁיבָה, הוּא הֶחְלִיט לְמַנּוֹת אֶת בְּנוֹ רַב הַאי לְתַפְקִיד חָשׁוּב מְאֹד – אַב בֵּית דִּין בַּיְשִׁיבָה.

מֵאָז שֶׁרַב הַאי מוּנָה לְאַב בֵּית דִּין, הוּא עָמַד לְצַד אָבִיו בְּמֶשֶׁךְ שָׁנִים רַבּוֹת. הָאָב וְהַבֵּן עָבְדוּ יַחְדָּו בְּאַהֲבָה, בְּשָׁלוֹם וּבְשִׁתּוּף פְּעֻלָּה מָלֵא. שְׁנֵיהֶם יַחְדָּו הִנְהִיגוּ אֶת הַיְשִׁיבָה, לִמְדוּ תּוֹרָה אֶת הַתַּלְמִידִים, וְעָנוּ עַל שְׁאֵלוֹת רַבּוֹת שֶׁנִּשְׁלְחוּ אֲלֵיהֶם מִקְּהִלּוֹת יְהוּדִיּוֹת רְחוֹקוֹת. הַיְהוּדִים בְּכָל מְקוֹמוֹת מוֹשְׁבוֹתֵיהֶם יָדְעוּ כִּי בִּיְשִׁיבַת פּוּמְבְּדִיתָא יוֹשְׁבִים שְׁנֵי חֲכָמִים גְּדוֹלִים, רַב שְׁרִירָא גָּאוֹן וּבְנוֹ רַב הַאי, וְכִי אֶפְשָׁר לִשְׁלֹחַ אֲלֵיהֶם כָּל שְׁאֵלָה בַּתּוֹרָה וּבַהֲלָכָה.

אַחַת הַקְּהִלּוֹת הָרְחוֹקוֹת בְּיוֹתֵר שֶׁפָּנוּ אֲלֵיהֶם הָיְתָה הַקְּהִלָּה בָּעִיר קֵירוּאָן שֶׁבִּצְפוֹן אַפְרִיקָה. חַכְמֵי קֵירוּאָן רָצוּ לָדַעַת כֵּיצַד לִנְהֹג בְּעִנְיָנִים שׁוֹנִים בַּהֲלָכָה, וְלָכֵן שָׁלְחוּ מִכְתָּבִים וּשְׁאֵלוֹת רַבּוֹת אֶל יְשִׁיבַת פּוּמְבְּדִיתָא. הַשְּׁאֵלוֹת הִגִּיעוּ אֶל בֵּית הַדִּין שֶׁל רַב שְׁרִירָא גָּאוֹן וְשֶׁל בְּנוֹ רַב הַאי, הַדַּיָּן שֶׁל הַיְשִׁיבָה. הָאָב וְהַבֵּן יָשְׁבוּ יַחְדָּו, לָמְדוּ אֶת הַשְּׁאֵלוֹת הַקָּשׁוֹת בְּכֹבֶד רֹאשׁ, וְכָתְבוּ עֲלֵיהֶן תְּשׁוּבוֹת בְּרוּרוֹת וּמְפֹרָטוֹת.

בָּאִגֶּרֶת הַמְפֻרְסֶמֶת שֶׁכָּתַב רַב שְׁרִירָא גָּאוֹן לְחַכְמֵי קֵירוּאָן, נִכְתְּבָה כּוֹתֶרֶת מְיֻחֶדֶת שֶׁמְּסַפֶּרֶת כִּי הַשְּׁאֵלוֹת הִגִּיעוּ אֶל בֵּית הַדִּין שֶׁל מָרָנָא רַבָּנָא שְׁרִירָא וְאֶל בְּנוֹ רַב הַאי. בְּסוֹף הָאִגֶּרֶת סִפֵּר רַב שְׁרִירָא גָּאוֹן כִּי שְׁנָתַיִם לְאַחַר שֶׁהוּא עַצְמוֹ מוּנָה לְגָאוֹן, הוּא סָמַךְ וּמִנָּה אֶת בְּנוֹ רַב הַאי לִהְיוֹת אַב בֵּית דִּין. רַב שְׁרִירָא גָּאוֹן הוֹסִיף וְהִתְפַּלֵּל בְּכָל לִבּוֹ אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיַּנְהִיגוּ אֶת עַם יִשְׂרָאֵל בֶּאֱמֶת, כָּרָאוּי וְכַהֹגֶן. כְּמוֹ כֵן הִתְפַּלֵּל שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יָבִיא אֶת מְשִׁיחַ בֶּן דָּוִד בִּמְהֵרָה בְּיָמֵינוּ וּבְחַיֵּי כָּל בֵּית יִשְׂרָאֵל.

לְאַחַר פְּטִירָתוֹ שֶׁל אָבִיו רַב שְׁרִירָא גָּאוֹן, עָמַד רַב הַאי גָּאוֹן בְּרֹאשׁ יְשִׁיבַת פּוּמְבְּדִיתָא וְהָיָה לַגָּאוֹן הֶחָדָשׁ. רַב הַאי גָּאוֹן הִמְשִׁיךְ אֶת הַדֶּרֶךְ הַגְּדוֹלָה שֶׁל אָבִיו, לִמֵּד תּוֹרָה, וְשָׁלַח תְּשׁוּבוֹת הֲלָכָתִיּוֹת לְכָל קְהִלּוֹת יִשְׂרָאֵל בְּכָל רַחֲבֵי הָעוֹלָם. רַב הַאי גָּאוֹן נֶחְשָׁב עַד הַיּוֹם לָאַחֲרוֹן מִבֵּין הַגְּאוֹנִים הַגְּדוֹלִים שֶׁהִנְהִיגוּ אֶת עַם יִשְׂרָאֵל.

💡 כַּאֲשֶׁר יְלָדִים מְכַבְּדִים אֶת הוֹרֵיהֶם וְעוֹזְרִים לָהֶם, הֵם זוֹכִים לִגְדֹּל בַּתּוֹרָה וּלְהָאִיר אֶת הָעוֹלָם.
