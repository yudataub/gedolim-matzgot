עַבְדּוֹן בֶּן הִלֵּל וְשִׁבְעִים הָעֲיָרִים

בִּימֵי קֶדֶם, בִּתְקוּפַת הַשּׁוֹפְטִים הַקְּדוּמָה שֶׁל עַם יִשְׂרָאֵל, עָמַד מַנְהִיג מְכֻבָּד וְשׁוֹפֵט חָשׁוּב שֶׁשְּׁמוֹ עַבְדּוֹן בֶּן הִלֵּל הַפִּרְעָתוֹנִי. עַבְדּוֹן הָיָה אִישׁ מִשֵּׁבֶט אֶפְרַיִם. הוּא נִבְחַר לְכַהֵן כְּשׁוֹפֵט וּמַנְהִיג שֶׁל כָּל עַם יִשְׂרָאֵל אַחֲרֵי הַשּׁוֹפֵט אֵילוֹן הַזְּבוּלֹנִי, שֶׁהִנְהִיג אֶת הָעָם לְפָנָיו. עַבְדּוֹן בֶּן הִלֵּל הִנְהִיג אֶת עַם יִשְׂרָאֵל, דָּן אוֹתָם בְּמִשְׁפָּט וְשָׁפַט אֶת הָעָם בְּמֶשֶׁךְ שְׁמֹנֶה שָׁנִים תְּמִימוֹת.

לַשּׁוֹפֵט עַבְדּוֹן בֶּן הִלֵּל הָיְתָה מִשְׁפָּחָה גְּדוֹלָה, מְפֹאָרָה וּמְבֹרֶכֶת מְאֹד. הָיוּ לוֹ אַרְבָּעִים בָּנִים וְגַם שְׁלֹשִׁים נְכָדִים, שֶׁהֵם בְּנֵי בָנָיו. כַּאֲשֶׁר מְחַשְּׁבִים אֶת כָּל אַרְבָּעִים הַבָּנִים יַחַד עִם שְׁלֹשִׁים הַנְּכָדִים, מַגִּיעִים לְמִסְפָּר שֶׁל שִׁבְעִים צֶאֱצָאִים מְכֻבָּדִים בְּמִשְׁפַּחְתּוֹ שֶׁל הַשּׁוֹפֵט. כָּל שִׁבְעִים הַבָּנִים וְהַנְּכָדִים שֶׁל עַבְדּוֹן בֶּן הִלֵּל הָיוּ רוֹכְבִים יַחַד עַל שִׁבְעִים עֲיָרִים.

מַה הֵם עֲיָרִים? הַמִּלָּה עֲיָרִים מְפֹרֶשֶׁת כַּחֲמוֹרִים צְעִירִים. בָּאֹתָם יָמִים קְדוּמִים, הָרְכִיבָה עַל שִׁבְעִים חֲמוֹרִים צְעִירִים וַחֲשׁוּבִים אֵלֶּה הָיְתָה סִימָן מֻבְהָק שֶׁל כָּבוֹד, מַעֲלָה וַחֲשִׁיבוּת רַבָּה מְאֹד. כְּפִי שֶׁמַּסְבִּיר הַמְּפָרֵשׁ מְצוּדַת דָּוִד עַל מִקְרָא זֶה, הָרְכִיבָה עַל הָעֲיָרִים הָיְתָה דָּבָר שֶׁל חֲשִׁיבוּת גְּדוֹלָה מְאֹד בָּעָם. הָרְכִיבָה הַמְּשֻׁתֶּפֶת שֶׁל שִׁבְעִים הַבָּנִים וְהַנְּכָדִים עַל שִׁבְעִים הָעֲיָרִים הֶרְאֲתָה לְכָל עַם יִשְׂרָאֵל אֶת הַכָּבוֹד הָרַב, אֶת הַחֲשִׁיבוּת וְאֶת הַמַּעֲמָד הַמְּכֻבָּד שֶׁל מִשְׁפַּחַת הַשּׁוֹפֵט עַבְדּוֹן.

כָּל עַם יִשְׂרָאֵל שֶׁרָאָה אֶת שִׁבְעִים הַבָּנִים וְהַנְּכָדִים רוֹכְבִים עַל שִׁבְעִים הָעֲיָרִים, הִכִּיר בַּחֲשִׁיבוּתָם הַגְּדוֹלָה וּבְמַנְהִיגוּתוֹ שֶׁל עַבְדּוֹן. עַבְדּוֹן בֶּן הִלֵּל הַפִּרְעָתוֹנִי הִמְשִׁיךְ לְהַנְהִיג אֶת יִשְׂרָאֵל, לִשְׁפֹּט אֶת הָעָם בְּצֶדֶק וּלְכַהֵן כְּשׁוֹפֵט מְכֻבָּד עַל כָּל יִשְׂרָאֵל בְּמֶשֶׁךְ כָּל שְׁמֹנֵה הַשָּׁנִים.

לַאֲחַר שֶׁסִּיֵּם שְׁמֹנֵה שָׁנִים שֶׁל הַנְהָגָה וּשְׁפִיטָה עַל עַם יִשְׂרָאֵל, עַבְדּוֹן בֶּן הִלֵּל הַפִּרְעָתוֹנִי מֵת וְנִפְטַר מִן הָעוֹלָם. אַנְשֵׁי עַם יִשְׂרָאֵל הֵבִיאוּ אֶת הַשּׁוֹפֵט עַבְדּוֹן בֶּן הִלֵּל לִקְבוּרָה בָּעִיר פִּרְעָתוֹן שֶׁבְּאֶרֶץ אֶפְרַיִם, שֶׁמִּמֶּנָּה הִגִּיעַ. הוּא נִקְבַּר בְּמָקוֹם מְיֻחָד בְּאֶרֶץ אֶפְרַיִם, שֶׁנִּקְרָא בַּסֵּפֶר הַר הָעֲמָלֵקִי. כָּךְ נִזְכַּר עַבְדּוֹן בֶּן הִלֵּל הַפִּרְעָתוֹנִי מִשֵּׁבֶט אֶפְרַיִם כְּשׁוֹפֵט חָשׁוּב, מְכֻבָּד וּמְפֹאָר בְּתּוֹלְדוֹת עַם יִשְׂרָאֵל.
