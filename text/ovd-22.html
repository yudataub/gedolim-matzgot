הַהַצָּלָה הַנִּסִּית בַּעֲלִיַּת הַגַּג

בִּשְׁכוּנַת הַבּוּכָרִים שֶׁבִּירוּשָׁלַיִם עָמַד בֵּית כְּנֶסֶת מְפֹאָר וּשְׁמוֹ "שׁוֹשַׁנִּים לְדָוִד". בַּעֲלִיַּת הַגַּג הַקְּטַנָּה שֶׁל בֵּית הַכְּנֶסֶת קָבַע אֶת מוֹשָׁבוֹ הַחָכָם הַגָּדוֹל, רַבִּי יַעֲקֹב חַיִּים סוֹפֵר זֵכֶר צַדִּיק לִבְרָכָה, שֶׁהָיָה מִתַּלְמִידָיו שֶׁל מָרָן הַ"בֵּן אִישׁ חַי". רַבִּי יַעֲקֹב חַיִּים הָיָה יוֹשֵׁב בַּעֲלִיַּת הַגַּג שָׁעוֹת אֲרֻכּוֹת, יָמִים וְלֵילוֹת, וְהוֹגֶה בִּסְפָרִים קְדוֹשִׁים. שָׁם גַּם חִבֵּר אֶת סִפְרוֹ הַמְּפֻרְסָם וְהַמַּקִּיף "כַּף הַחַיִּים" עַל הַשֻּׁלְחָן עָרוּךְ.

לְצִדּוֹ שֶׁל הַגָּאוֹן הַגָּדוֹל יָשַׁב נַעַר צָעִיר וְשַׁקְדָן שֶׁשְּׁמוֹ עוֹבַדְיָה. עוֹבַדְיָה הַקָּטָן הִשְׁתּוֹקֵק עוֹד מִיְּמֵי יַלְדוּתוֹ לִהְיוֹת קָרוֹב אֶל גְּדוֹלֵי יִשְׂרָאֵל וְלִלְמֹד מִתּוֹרָתָם הָעֲמֻקָּה. בְּכָל יוֹם הָיָה עוֹבַדְיָה יוֹשֵׁב בַּעֲלִיַּת הַגַּג לְצַד רַבִּי יַעֲקֹב חַיִּים סוֹפֵר, מַקְשִׁיב בְּקֶשֶׁב רַב לְקוֹל הַלִּמּוּד הַמָּתוֹק, וְלוֹמֵד גַּם הוּא מִתּוֹךְ סֵפֶר שֶׁהָיָה פָּתוּחַ לְפָנָיו בְּאַהֲבָה רַבָּה.

וְהִנֵּה, בְּאַחַד הַיָּמִים, הִתְרַחֵשׁ מְאֹרָע מַפְחִיד וְנוֹרָא. לְפֶתַע נִשְׁמַע קוֹל רַעַשׁ גָּדוֹל וְהַרְעָדָה חֲזָקָה בְּכָל שְׁכוּנַת הַבּוּכָרִים. הָיְתָה זוֹ רְעִידַת אֲדָמָה קָשָׁה! הַדְּרָכִים כֻּלָּן הִזְדַּעְזְעוּ, הַבָּתִּים הָרַבִּים נָעוּ עַל מְקוֹמָם, וַעֲלִיּוֹת גַּג רַבּוֹת קָרְסוּ בְּמַהֲרוּת כְּמוֹ מִגְדְּלֵי קְלָפִים. גַּם עֲלִיַּת הַגַּג שֶׁל בֵּית הַכְּנֶסֶת "שׁוֹשַׁנִּים לְדָוִד" לֹא עָמְדָה בָּרַעַשׁ וְקָרְסָה תַּחְתֶּיהָ. אֲבָנִים גְּדוֹלוֹת וְכְבֵדוֹת נָשְׁרוּ מֵהַתִּקְרָה וְהִתְגַּלְגְּלוּ לְכָל עֵבֶר, וְעַנְנֵי אָבָק סְמִיכִים הִתְרוֹמְמוּ מִן הַקַּרְקַע.

מִן הָרְחוֹב נִשְׁמְעוּ צְוָחוֹת עַזּוֹת וְחֲרָדָה גְּדוֹלָה. אֲנָשִׁים מְבֹהָלִים מְאֹד הִתְרוֹצְצוּ בַּדְּרָכִים מִכָּל עֵבֶר, כְּשֶׁהֵם מְחַפְּשִׂים אֶת יַקִּירֵיהֶם וְאֶת בְּנֵי מִשְׁפַּחְתָּם בֵּין הַהֲרִיסוֹת, קוֹרְאִים בְּקוֹל לְעֶזְרָה וּמְשַׁוְּעִים לְהַצָּלָה. יוֹשְׁבֵי בֵּית הַמִּדְרָשׁ שֶׁל "שׁוֹשַׁנִּים לְדָוִד" נִזְעֲקוּ בְּבֶהָלָה רַבָּה, וְקָרְאוּ: "רַבֵּנוּ נִמְצָא בַּעֲלִיַּת הַגַּג! רַבֵּנוּ נִמְצָא בַּעֲלִיַּת הַגַּג!". הֵם מִהֲרוּ בַּעֲבוֹדָה מְהִירָה כְּדֵי לְחַלֵּץ אֶת הַגָּאוֹן הַגָּדוֹל רַבִּי יַעֲקֹב חַיִּים סוֹפֵר מִבֵּין הָאֲבָנִים וְהַהֲרִיסוֹת.

אַךְ כַּאֲשֶׁר הִגִּיעוּ הַמְּחַלְּצִים אֶל רַבִּי יַעֲקֹב חַיִּים, הוּא לֹא מִהֵר לָצֵאת הַחוּצָה לְמָקוֹם בָּטוּחַ. הוּא פָּנָה אֲלֵיהֶם וְאָמַר: "רַק רֶגַע! אֲנִי שׁוֹמֵעַ כָּאן אֶת קוֹל בְּכִיָּתוֹ שֶׁל עוֹבַדְיָה הַקָּטָן!". עוֹבַדְיָה הַנַּעַר הָיָה מְבֹהָל עַד עָמְקֵי נִשְׁמָתוֹ מֵהָרַעַשׁ וּמֵהָאֲבָנִים הַנּוֹפְלוֹת, וְהוּא פָּרַץ בִּבְכִי נוֹרָא וּמַר. רַבִּי יַעֲקֹב חַיִּים לֹא חָשַׁב עַל עַצְמוֹ, אֶלָּא שָׁלַח אֶת יָדוֹ מִיָּד, אָחַז בִּזְרוֹעוֹ שֶׁל עוֹבַדְיָה הַקָּטָן, וּמָשַׁךְ אוֹתוֹ הַחוּצָה בִּזְהִירוּת רַבָּה מִבֵּין הָאֲבָנִים. בְּחַסְדֵי הַשֵּׁם, חֻלְּצוּ שְׁנֵיהֶם בְּשָׁלוֹם מִמְּקוֹם הַסַּכָּנָה.

💡 אָהֲבַת הַתּוֹרָה וְהַדְּאָגָה לַזּוּלַת מְגִנּוֹת עַל הָאָדָם וּמַצִּילוֹת אוֹתוֹ בְּעֵת צָרָה.
