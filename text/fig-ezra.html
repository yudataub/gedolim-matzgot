עֶזְרָא הַסּוֹפֵר וְהַדֶּרֶךְ לִירוּשָׁלַיִם

עֶזְרָא הַסּוֹפֵר, בֶּן שְׂרָיָה בֶּן עֲזַרְיָה בֶּן חִלְקִיָּה הַכֹּהֵן, הָיָה סוֹפֵר מָהִיר וְחָכָם בְּתוֹרַת מֹשֶׁה אֲשֶׁר נָתַן הַשֵּׁם אֱלֹהֵי יִשְׂרָאֵל. עֶזְרָא הֵכִין אֶת לְבָבוֹ בְּאַהֲבָה רַבָּה לִדְרֹשׁ אֶת תּוֹרַת הַשֵּׁם, לַעֲשׂוֹת אוֹתָהּ בְּעַצְמוֹ וּלְלַמֵּד בְּיִשְׂרָאֵל חֹק וּמִשְׁפָּט. הוּא גָּר בְּבָבֶל, וּבִשְׁנַת שֶׁבַע לְאַרְתַּחְשַׁסְתְּא מֶלֶךְ פָּרָס, הֶחְלִיט לַעֲלוֹת מִבָּבֶל אֶל עִיר הַקֹּדֶשׁ יְרוּשָׁלַיִם.

יַחַד עִם עֶזְרָא הִתְאַרְגְּנָה קְבוּצָה גְּדוֹלָה שֶׁל עֹלִים מִבְּנֵי יִשְׂרָאֵל, וּבָהֶם כֹּהֲנִים, לְוִיִּם, מְשֹׁרְרִים, שֹׁעֲרִים וּנְתִינִים. הַמֶּלֶךְ אַרְתַּחְשַׁסְתְּא נָתַן לְעֶזְרָא אֶת כָּל בַּקָּשָׁתוֹ, כִּי יַד הַשֵּׁם אֱלֹהָיו הָיְתָה עָלָיו לְטוֹבָה. הֵם הֶחְלִיטוּ לָצֵאת לַדֶּרֶךְ מִבָּבֶל בָּאֶחָד לַחֹדֶשׁ הָרִאשׁוֹן, כְּדֵי לָלֶכֶת אֶל יְרוּשָׁלַיִם.

הַמַּסָּע מִבָּבֶל לִירוּשָׁלַיִם הָיָה אָרֹךְ וְקָשֶׁה מְאֹד, וְהַדְּרָכִים הָיוּ מְלֵאוֹת בְּסַכָּנוֹת, כִּי בַּדֶּרֶךְ הָיוּ אוֹיְבִים וְאוֹרְבִים שֶׁהָיוּ יְכוֹלִים לִפְגֹּעַ בָּהֶם וְלִגְנֹב אֶת רְכוּשָׁם. עֶזְרָא הָיָה יָכוֹל לִבְקֹשׁ מֵהַמֶּלֶךְ אַרְתַּחְשַׁסְתְּא שֶׁיִּשְׁלַח עִמָּם חַיִל וּפָרָשִׁים, כָּלוֹמַר חַיָּלִים וְרוֹכְבֵי סוּסִים, כְּדֵי שֶׁיַּעַזְרוּ לָהֶם וְיָגֵנוּ עֲלֵיהֶם מִפְּנֵי הָאוֹיֵב בַּדֶּרֶךְ. אַךְ עֶזְרָא הִתְבַּיֵּשׁ מְאֹד לִשְׁאֹל מִן הַמֶּלֶךְ חַיָּלִים וּפָרָשִׁים.

הַסִּבָּה שֶׁעֶזְרָא הִתְבַּיֵּשׁ הָיְתָה שֶׁקּוֹדֵם לָכֵן הוּא אָמַר לַמֶּלֶךְ אֶת הַדְּבָרִים הָאֵלֶּה: "יַד אֱלֹהֵינוּ עַל כָּל מְבַקְשָׁיו לְטוֹבָה, וְעֻזּוֹ וְאַפּוֹ עַל כָּל עֹזְבָיו". עֶזְרָא סִפֵּר לַמֶּלֶךְ שֶׁהַשֵּׁם אֱלֹהֵי יִשְׂרָאֵל שׁוֹמֵר בְּאַהֲבָה עַל כָּל מִי שֶׁמְּבַקֵּשׁ אוֹתוֹ. לָכֵן הִרְגִּישׁ עֶזְרָא שֶׁאִם הוּא יְבַקֵּשׁ כָּעֵת חַיָּלִים בְּשָׂר וָדָם מֵהַמֶּלֶךְ, יֵרָאֶה כְּאִלּוּ הוּא אֵינוֹ סוֹמֵךְ עַל עֶזְרַת הַשֵּׁם.

בִּמְקוֹם לִבְקֹשׁ צָבָא מֵהַמֶּלֶךְ, קָרָא עֶזְרָא צוֹם לְכָל הָעָם עַל נְהַר אַהֲוָא. הֵם הִתְעַנּוּ לִפְנֵי אֱלֹהֵינוּ כְּדֵי לְבַקֵּשׁ מִמֶּנּוּ דֶּרֶךְ יְשָׁרָה וּבְטוּחָה עֲבוּרָם, עֲבוּר טַפָּם (הַיְלָדִים הַקְּטַנִּים) וַעֲבוּר כָּל רְכוּשָׁם. הֵם צָמוּ וּבִקְשׁוּ מֵאֱלֹהֵינוּ עַל זֹאת בֶּאֱמוּנָה שְׁלֵמָה, וְהַשֵּׁם שָׁמַע אֶת תְּפִלָּתָם וְנַעֲתַר לָהֶם.

בִּשְׁנֵים עָשָׂר לַחֹדֶשׁ הָרִאשׁוֹן נָסְעוּ מִנְּהַר אַהֲוָא לָלֶכֶת אֶל יְרוּשָׁלַיִם. יַד אֱלֹהֵינוּ הָיְתָה עֲלֵיהֶם לְטוֹבָה כָּל הַדֶּרֶךְ, וְהַשֵּׁם הִצִּיל אוֹתָם מִכַּף כָּל אוֹיֵב וְאוֹרֵב עַל הַדֶּרֶךְ. הֵם צָעֲדוּ בְּבִטָּחוֹן, וּבָאוּ אֶל יְרוּשָׁלַיִם בָּאֶחָד לַחֹדֶשׁ הַחֲמִישִׁי, בַּשָּׁנָה הַשְּׁבִיעִית לַמֶּלֶךְ, כְּיַד אֱלֹהָיו הַטּוֹבָה עָלָיו. הֵם הִגִּיעוּ בְּשָׁלוֹם וְיָשְׁבוּ שָׁם שְׁלֹשָׁה יָמִים.

💡 כְּשֶׁאָנוּ סוֹמְכִים עַל הַשֵּׁם וּמִתְפַּלְּלִים אֵלָיו בֶּאֱמֶת, הַשֵּׁם שׁוֹמֵר עָלֵינוּ וּמַצִּיל אוֹתָנוּ מִכָּל סַכָּנָה.
