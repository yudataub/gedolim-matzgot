רַבָּה בַּר נַחְמָנִי וְהַתְּשׁוּבָה שֶׁבִּקְּשׁוּ בַּשָּׁמַיִם

רַבָּה בַּר נַחְמָנִי הָיָה חָכָם גָּדוֹל מְאֹד וְרֹאשׁ יְשִׁיבַת פּוּמְבְּדִיתָא בְּבָבֶל בְּמֶשֶׁךְ עֶשְׂרִים וּשְׁתַּיִם שָׁנִים. הוּא נִקְרָא "עוֹקֵר הָרִים" בִּגְלָל חֲרִיפוּת חָכְמָתוֹ. בְּכָל שָׁנָה, בְּחָדְשֵׁי הַכַּלָּה שֶׁבַּקַּיִץ וּבַחֹרֶף, בָּאוּ אֵלָיו שְׁנֵים עָשָׂר אֶלֶף תַּלְמִידִים כְּדֵי לִלְמֹד תּוֹרָה. אֲנָשִׁים רָעִים הָלְכוּ לַמֶּלֶךְ וְהִלְשִׁינוּ עַל רַבָּה. הֵם אָמְרוּ לַמֶּלֶךְ שֶׁיֵּשׁ אָדָם אֶחָד בִּיהוּדָה שֶׁגּוֹרֵם לִשְׁנֵים עָשָׂר אֶלֶף יְהוּדִים לֹא לַעֲבֹד וְלֹא לְשַׁלֵּם אֶת מַס הַמֶּלֶךְ בִּזְמַן הַלִּמּוּד. הַמֶּלֶךְ כָּעַס וְשָׁלַח אֶת הַשָּׁלִיחַ שֶׁלּוֹ כְּדֵי לִתְפֹּס אֶת רַבָּה.

רַבָּה בָּרַח מִמָּקוֹם לְמָקוֹם כְּדֵי שֶׁלֹּא יִתְפְּסוּ אוֹתוֹ: מִפּוּמְבְּדִיתָא לְאַקְרָא, מֵאַקְרָא לַאֲגַם הַמַּיִם, וּמִשָּׁם לִשְׁחִין, לִצְרִיפָא, לְעֵין הַמַּיִם וּבַחֲזָרָה לְפוּמְבְּדִיתָא. הַשָּׁלִיחַ שֶׁל הַמֶּלֶךְ הִגִּיעַ לַפֻּנְדָּק שֶׁבּוֹ הִתְאַכְסֵן רַבָּה. הַמְּאָרְחִים הֵגִישׁוּ לַשָּׁלִיחַ אֹכֶל וּשְׁתֵּי כּוֹסוֹת יַיִן, וּכְשֶׁהֵרִימוּ אֶת הַשֻּׁלְחָן, הִסְתּוֹבְבוּ פָּנָיו שֶׁל הַשָּׁלִיחַ לַאֲחוֹרַיִם. הָאֲנָשִׁים נִבְהֲלוּ וְשָׁאֲלוּ אֶת רַבָּה מָה לַעֲשׂוֹת. רַבָּה אָמַר לָהֶם: הַגִּישׁוּ לוֹ שׁוּב אֹכֶל וְכוֹס אַחַת שֶׁל יַיִן, וְאָז הֲרִימוּ אֶת הַשֻּׁלְחָן. הֵם עָשׂוּ כָּךְ, וְהַשָּׁלִיחַ הִתְרַפֵּא. הַשָּׁלִיחַ הֵבִין שֶׁהָאָדָם שֶׁהוּא מְחַפֵּשׂ נִמְצָא שָׁם. הוּא חִפֵּשׂ וּמָצָא אֶת רַבָּה, הִכְנִיס אוֹתוֹ לַחֶדֶר וְנָעַל אֶת הַדֶּלֶת. רַבָּה בִּקֵּשׁ רַחֲמִים מֵהַקָּדוֹשׁ בָּרוּךְ הוּא, הַקִּיר נִפְרַץ, וְרַבָּה בָּרַח אֶל הָאֲגַם. הוּא יָשַׁב עַל גֶּזַע דֶּקֶל וְלָמַד תּוֹרָה בְּלִי לְהַפְסִיק.

בְּאוֹתוֹ זְמַן הָיְתָה מַחֲלֹקֶת בַּיְּשִׁיבָה שֶׁל מַעְלָה בַּשָּׁמַיִם בְּהִלְכוֹת נְגָעִים. הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר שֶׁהַדָּבָר טָהוֹר, וְכָל חַכְמֵי הַיְּשִׁיבָה בַּשָּׁמַיִם אָמְרוּ שֶׁהוּא טָמֵא. אָמְרוּ בַּשָּׁמַיִם: מִי יַכְרִיעַ בַּמַּחֲלֹקֶת? רַבָּה בַּר נַחְמָנִי!. כִּי רַבָּה הָיָה מֻמְחֶה גָּדוֹל וְאָמַר עַל עַצְמוֹ: "אֲנִי יָחִיד בִּנְגָעִים". שָׁלְחוּ מַלְאָךְ כְּדֵי לְהָבִיא אֶת נִשְׁמָתוֹ לַשָּׁמַיִם, אַךְ מַלְאַךְ הַמָּוֶת לֹא יָכֹל לְהִתְקָרֵב אֵלָיו כִּי פִּיו לֹא פָסַק מִלִּמּוּד הַתּוֹרָה.

לְפֶתַע נָשְׁבָה רוּחַ וְהִרְעִישָׁה אֶת הַקָּנִים בָּאֲגַם. רַבָּה חָשַׁב שֶׁהַפָּרָשִׁים שֶׁל הַמֶּלֶךְ בָּאוּ לִתְפֹּס אוֹתוֹ. הוּא אָמַר: יָמוּת הָאִישׁ הַהוּא וְאַל יִמָּסֵר בִּידֵי הַמַּלְכוּת. כְּשֶׁיָּצְאָה נִשְׁמָתוֹ אָמַר: "טָהוֹר, טָהוֹר!". יָצְאָה בַּת קוֹל מֵהַשָּׁמַיִם וְאָמְרָה: אַשְׁרֶיךָ רַבָּה בַּר נַחְמָנִי שֶׁגּוּפְךָ טָהוֹר וְיָצְאָה נִשְׁמָתְךָ בְּטָהוֹר. מֵהַשָּׁמַיִם נָפַל מִכְתָּב בְּפוּמְבְּדִיתָא: רַבָּה בַּר נַחְמָנִי נִתְבַּקֵּשׁ בִּיְשִׁיבָה שֶׁל מַעְלָה.

אַבָּיֵי, רָבָא וְכָל הַחֲכָמִים יָצְאוּ לְחַפֵּשׂ אוֹתוֹ. הֵם הִגִּיעוּ לָאֲגַם וְרָאוּ צִפֳּרִים עוֹמְדוֹת וּמְצִלּוֹת בְּכַנְפֵיהֶן עַל מָקוֹם אֶחָד, וְכָךְ יָדְעוּ שֶׁשָּׁם הוּא נִמְצָא. הֵם סָפְדוּ לוֹ שִׁבְעָה יָמִים. בְּיוֹם פְּטִירָתוֹ עָלְתָה סְעָרָה גְּדוֹלָה מְאֹד. אָדָם אֶחָד רָכוּב עַל גָּמָל הֻפְתַּע כְּשֶׁהָרוּחַ הֶעֱבִירָה אוֹתוֹ מִצַּד אֶחָד שֶׁל הַנָּהָר לַצַּד הַשֵּׁנִי. כְּשֶׁשָּׁמַע שֶׁרַבָּה נִפְטַר, הִתְפַּלֵּל לַה' וְאָמַר: רִבּוֹנוֹ שֶׁל עוֹלָם, כָּל הָעוֹלָם שֶׁלְּךָ וְרַבָּה שֶׁלְּךָ, לָמָּה לְהַחֲרִיב אֶת הָעוֹלָם? וּמִיָּד נָחָה הַסְּעָרָה.

💡 לִמּוּד הַתּוֹרָה וְהַטָּהֳרָה הֵם הַדְּבָרִים הַחֲשׁוּבִים בְּיוֹתֵר, וְהֵם מְגִנִּים עַל הָאָדָם וּמְבִיאִים אוֹר לָעוֹלָם.
