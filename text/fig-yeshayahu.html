מַרְאֵה הַנְּבוּאָה שֶׁל יְשַׁעְיָהוּ הַנָּבִיא

יְשַׁעְיָהוּ בֶּן אָמוֹץ הָיָה נָבִיא קָדוֹשׁ וְצַדִּיק אֲשֶׁר חָזָה אֶת חֶזְיוֹנוֹ וְנִבָּא בָּעִיר יְרוּשָׁלַיִם עַל אֶרֶץ יְהוּדָה וְעַל יוֹשְׁבֵי יְרוּשָׁלַיִם. יְשַׁעְיָהוּ הַנָּבִיא נִבָּא וּפָעַל בִּימֵיהֶם שֶׁל אַרְבָּעָה מַלְכֵי יְהוּדָה: בִּימֵי עֻזִּיָּהוּ הַמֶּלֶךְ, בִּימֵי יוֹתָם הַמֶּלֶךְ, בִּימֵי אָחָז הַמֶּלֶךְ, וּבִימֵי יְחִזְקִיָּהוּ הַמֶּלֶךְ.

בִּשְׁנַת מוֹתוֹ שֶׁל הַמֶּלֶךְ עֻזִּיָּהוּ, זָכָה יְשַׁעְיָהוּ הַנָּבִיא לִרְאוֹת חָזוֹן וּמַרְאֶה נְבוּאִי קָדוֹשׁ וְנִפְלָא מְאֹד. בַּחָזוֹן הַזֶּה הוּא רָאָה אֶת ה' יוֹשֵׁב עַל כִּסֵּא רָם וְנִשָּׂא, וְשׁוּלָיו שֶׁל הַכִּסֵּא הַקָּדוֹשׁ מִלְאוּ אֶת כָּל הַהֵיכָל. מֵעַל הַכִּסֵּא עָמְדוּ מַלְאָכִים קְדוֹשִׁים הַנִּקְרָאִים שְׂרָפִים. לְכָל מַלְאָךְ שָׂרָף הָיוּ שֵׁשׁ כְּנָפַיִם: בִּשְׁתֵּי כְּנָפַיִם הוּא הָיָה מְכַסֶּה אֶת פָּנָיו, בִּשְׁתֵּי כְּנָפַיִם הוּא הָיָה מְכַסֶּה אֶת רַגְלָיו, וּבִשְׁתֵּי כְּנָפַיִם הוּא הָיָה עָף וּמְעוֹפֵף.

הַמַּלְאָכִים הַשְּׂרָפִים הַקְּדוֹשִׁים קָרְאוּ זֶה אֶל זֶה בְּקוֹל גָּדוֹל וְאָמְרוּ: "קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ ה' צְבָאוֹת מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ". מֵאָז הַמַּרְאֶה הַקָּדוֹשׁ הַזֶּה וְעַד עַצְמֶם שֶׁל יָמִים אֵלֶּה, גַּם אֲנַחְנוּ אוֹמְרִים אֶת הַמִּלִּים הַקְּדוֹשׁוֹת הָאֵלֶּה בְּכָל יוֹם בִּתְפִלַּת הַקְּדֻשָּׁה.

כַּאֲשֶׁר הַמַּלְאָכִים הַשְּׂרָפִים קָרְאוּ אֶת הַקְּרִיאָה הַקְּדוֹשָׁה הַזֹּאת, הָיָה הַקּוֹל חָזָק כָּל כָּךְ עַד שֶׁזָּזוּ וְזָעוּ אֲמּוֹת הַסִּפִּים שֶׁל הַפְּתָחִים, וְכָל הַבַּיִת נִמְלָא עָשָׁן. כְּשֶׁרָאָה יְשַׁעְיָהוּ הַנָּבִיא אֶת הַמַּרְאֶה הַגָּדוֹל הַזֶּה, הוּא נִבְהַל מְאֹד וְאָמַר: אוֹי לִי כִּי נִדְמֵיתִי וְאָבַדְתִּי, כִּי אֲנִי אִישׁ אֲשֶׁר שְׂפָתָיו אֵינָן טְהוֹרוֹת, וַאֲנִי יוֹשֵׁב בְּתוֹךְ עַם שֶׁשְּׂפָתָיו אֵינָן טְהוֹרוֹת, וְאֶת הַמֶּלֶךְ ה' צְבָאוֹת רָאוּ עֵינַי.

אָז עָף אֶל יְשַׁעְיָהוּ הַנָּבִיא אֶחָד מִן הַשְּׂרָפִים, וּבְיָדוֹ הָיְתָה גַּחַלֶת אֵשׁ הַנִּקְרֵאת רִצְפָּה, שֶׁאוֹתָהּ הוּא לָקַח בְּמֶלְקַחַיִם מֵעַל הַמִּזְבֵּחַ. הַמַּלְאָךְ הַשָּׂרָף נָגַע בַּגַּחַלֶת עַל פִּיו וְעַל שְׂפָתָיו שֶׁל יְשַׁעְיָהוּ וְאָמַר לוֹ: הִנֵּה נָגַע זֶה עַל שְׂפָתֶיךָ, וְסָר עֲוֹנְךָ וְחַטָּאתְךָ תְּכֻפָּר.

מִיָּד אַחַר כָּךְ שָׁמַע יְשַׁעְיָהוּ אֶת קוֹלוֹ שֶׁל ה' שׁוֹאֵל: אֶת מִי אֶשְׁלַח וּמִי יֵלֶךְ לָנוּ? כַּאֲשֶׁר שָׁמַע יְשַׁעְיָהוּ הַנָּבִיא אֶת שְׁאֵלָתוֹ שֶׁל ה', הוּא לֹא הִסֵּס וְלֹא חָשַׁב פַּעֲמַיִם, אֶלָּא עָנָה מִיָּד בְּשִׂמְחָה וּבִמְסִירוּת רַבָּה: הִנְנִי, שְׁלָחֵנִי! הוּא הָיָה מוּכָן וּמְזֻמָּן לָלֶכֶת בְּאַהֲבָה וְלִהְיוֹת שָׁלִיחַ נֶאֱמָן לִמְסֹר אֶת דִּבְרֵי ה'.

💡 כַּאֲשֶׁר קוֹרְאִים לָנוּ לַעֲשׂוֹת מִצְוָה אוֹ מַעֲשֶׂה טוֹב, אָנוּ צְרִיכִים לַעֲנוֹת מִיָּד בְּשִׂמְחָה: "הִנְנִי!", וְלָלֶכֶת בְּאַהֲבָה לַעֲזֹר.
