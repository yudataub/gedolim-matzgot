הַשֻּׁלְחָן עָרוּךְ הַקָּטָן בַּחֲצַר מַצּוֹת הַמִּצְוָה

בְּעֶרֶב חַג הַפֶּסַח, לְאַחַר חֲצוֹת הַיּוֹם, הָיְתָה תְּכוּנָה רַבָּה וְהִתְרַגְּשׁוּת גְּדוֹלָה בַּחֲצַר מִשְׁפַּחַת לֵוִי שֶׁבִּשְׁכוּנַת "בֵּית יִשְׂרָאֵל" בִּירוּשָׁלַיִם. חַכְמֵי בֵּית הַכְּנֶסֶת "שֶׁמֶשׁ צְדָקָה", וּבְרֹאשָׁם הָרַב צְדָקָה חוּצִ'ין, הִתְאַסְּפוּ כְּדֵי לֶאֱפוֹת מַצּוֹת מְהֻדָּרוֹת שֶׁל מִצְוָה לִכְבוֹד הַחַג. עַל כָּל מְלֶאכֶת הָאֲפִיָּה הִשְׁגִּיחַ בִּקְפֵדָה רַבָּה הַגָּאוֹן הַמְּקֻבָּל חָכָם סַלְמָאן מוּצָפִי. הוּא דָּאַג שֶׁלֹּא יִמָּצֵא אֲפִלּוּ פֵּרוּר קָטָן שֶׁל חָמֵץ בַּמַּצּוֹת, וְזֵרֵז אֶת הָאֲנָשִׁים לַעֲשׂוֹת אֶת הָעֲבוֹדָה בִּזְרִיזוּת רַבָּה. הָאֲנָשִׁים בַּחָצֵר קָרְאוּ זֶה לָזֶה: "הָבִיאוּ קֶמַח!", "הָבִיאוּ מַיִם שֶׁלָּנוּ!", "לָלוּשׁ מַהֵר!", וְהַמַּצּוֹת הֻכְנְסוּ אַחַת אַחַר הַשְּׁנִיָּה אֶל הַתַּנּוּר הַלּוֹהֵט.

וְהִנֵּה, בְּאֶמְצַע הָעֲבוֹדָה הַנִּמְרֶצֶת, הוֹפִיעַ בְּפֶתַח הַחָצֵר הָרַב עֹבַדְיָה יוֹסֵף. הוּא הָיָה אָז אַבְרֵךְ צָעִיר כְּבֶן עֶשְׂרִים וָחָמֵשׁ שָׁנִים, אַךְ כְּבָר נִכַּר כְּמִי שֶׁמַּקְפִּיד וּמְהַדֵּר מְאֹד בְּמִצְווֹת הַתּוֹרָה. הָרַב עֹבַדְיָה נִגַּשׁ בַּעֲדִינוּת אֶל הָאוֹפֶה, הוֹצִיא מִחֵיקוֹ שַׂקִּיק קָטָן וּבוֹ חֲצִי קִילוֹגְרָם קֶמַח, וּבִקֵּשׁ מִמֶּנּוּ בְּנַחַת: "הַאִם תּוּכַל לֶאֱפוֹת עֲבוּרִי שָׁלוֹשׁ מַצּוֹת מִצְוָה מֵהַקֶּמַח שֶׁהֵבֵאתִי?" אֲבָל הָאוֹפֶה, שֶׁהָיָה לָחוּץ מֵרֹב הָעֲבוֹדָה, כָּעַס וְלֹא הִסְכִּים לַעֲזֹר לוֹ. הוּא צָעַק עַל הָרַב עֹבַדְיָה הַצָּעִיר וְאָמַר: "הַאִם אֵינְךָ רוֹאֶה שֶׁיֵּשׁ כָּאן חֲכָמִים שֶׁאֹופִים? בְּבַקָּשָׁה, אַל תַּפְרִיעַ!"

הָרַב עֹבַדְיָה לֹא נֶעֱלַב, לֹא כָּעַס וְלֹא הִתְוַכֵּחַ כְּלָל. הוּא עָנָה בְּשַׁלְוָה וּבְנַחַת: "אַמְתִּין בְּסַבְלָנוּת". הוּא צָעַד לְאִטּוֹ אֶל פִּנַּת הַחָצֵר וְהִתְיַשֵּׁב עַל כִּסֵּא. אָז הוֹצִיא מִכִּיסוֹ סֵפֶר "שֻׁלְחָן עָרוּךְ" קָטָן, בְּגֹדֶל שֶׁל סִדּוּר תְּפִלָּה, עִם פֵּרוּשׁ "בָּאֵר הֵיטֵב". הָרַב עֹבַדְיָה הִנִּיחַ אֶת הַסֵּפֶר לְפָנָיו וְהֵחֵל לִלְמֹד וּלְשַׁנֵּן בִּשְׁקִידָה עֲמֻקָּה וּבְרִכּוּז רַב אֶת הִלְכוֹת הַפֶּסַח, הֲלָכָה אַחַר הֲלָכָה וְסִימָן אַחַר סִימָן. הוּא הָיָה שָׁקוּעַ כֻּלּוֹ בַּלִּמּוּד וְלֹא שָׂם לֵב לַנַּעֲשֶׂה סְבִיבוֹ.

חָכָם סַלְמָאן מוּצָפִי, שֶׁעָמַד בַּחָצֵר וְהִשְׁגִּיחַ עַל אֲפִיַּת הַמַּצּוֹת, הִבִּיט מִן הַצַּד בָּאַבְרֵךְ הַצָּעִיר שֶׁיָּשַׁב וְלָמַד בִּשְׁקִידָה נִפְלָאָה כָּזוֹ בְּאֶמְצַע הַהֲמֻלָּה. חָכָם סַלְמָאן הִתְפַּעֵל מְאֹד מֵאַהֲבַת הַתּוֹרָה וּמֵהַרִכּוּז הָעָמֹק שֶׁל הָרַב עֹבַדְיָה. הוּא הִתְבּוֹנֵן בּוֹ בְּהַעֲרָצָה גְּדוֹלָה, וּמִלְמֵל לְעַצְמוֹ בְּהִתְרַגְּשׁוּת: "הוּא יִהְיֶה גְּדוֹל הַדּוֹר! הוּא יִהְיֶה גְּדוֹל הַדּוֹר, אֵין סָפֵק!" לְאַחַר שָׁנִים רַבּוֹת, כַּאֲשֶׁר שְׁמוֹ שֶׁל הָרַב עֹבַדְיָה יָצָא לַמֶּרְחַקִּים וְהוּא הָיָה לְאַחַד מִפּוֹסְקֵי הַהֲלָכָה הַגְּדוֹלִים בְּיוֹתֵר, סִפֵּר חָכָם סַלְמָאן מוּצָפִי לִבְנוֹ, הָרַב בֶּן צִיּוֹן מוּצָפִי, כֵּיצַד יָדַע כְּבָר בְּאוֹתוֹ עֶרֶב פֶּסַח בַּחָצֵר כִּי הָאַבְרֵךְ הַצָּעִיר יִגְדַּל לִהְיוֹת מַנְהִיגָם וּרְבָּם שֶׁל כָּל יִשְׂרָאֵל.

💡 כַּאֲשֶׁר אָנוּ מִתְנַהֲגִים בְּסַבְלָנוּת, בְּעַנְוָה וּבְאַהֲבָה גְּדוֹלָה לַתּוֹרָה, אָנוּ יְכוֹלִים לִלְמֹד בְּכָל מָקוֹם וּלְהַגִּיעַ לִגְדֻלָּה שֶׁאֵין כְּמוֹתָהּ. 📖 **רוֹצֶה שֶׁאֶעֱזֹר לְךָ לְעַבֵּד סִפּוּר נוסָף מֵהַמְּקוֹרוֹת לְטֶקְסְט מְנֻקָּד עֲבוּר הַתַּלְמִידִים?**
