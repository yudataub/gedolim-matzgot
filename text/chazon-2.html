הַחֲזוֹן אִישׁ וְהַיְהוּדִי שְׁבוּר הַלֵּב

בְּאֶחָד מֵהָעֲרָבִים, עָבַד וְלָמַד הַצַּדִּיק הַגָּדוֹל, הַחֲזוֹן אִישׁ, בְּחַדְרוֹ. מִחַדְרוֹ שֶׁל הַחֲזוֹן אִישׁ יָצָא תַּלְמִידוֹ, רַבִּי שְׁרָגָא פַיְבֶל שְׁטַיְנְבֶּרְג, לְאַחַר שֶׁשּׂוֹחַח עִמּוֹ. כַּאֲשֶׁר יָצָא רַבִּי שְׁרָגָא אֶל הַפְּרוֹזְדוֹר הַמּוֹלִיךְ אֶל הַחֶדֶר, הוּא רָאָה שָׁם אָדָם אֶחָד שֶׁעָמַד וְחִכָּה בְּסַבְלָנוּת. הָאִישׁ הַזֶּה הִגִּיעַ בַּמְּיֻחָד כְּדֵי לְהִתְיָעֵץ עִם הַחֲזוֹן אִישׁ בְּעִנְיָן חָשׁוּב וּמַדְאִיג עַל אִישׁ חוֹלֶה שֶׁנִּזְקַק לְרַחֲמֵי שָׁמַיִם וּלְעֵצָה טוֹבָה.

רַבִּי שְׁרָגָא פַיְבֶל רָאָה אֶת הָאִישׁ הַמַּמְתִּין בַּפְּרוֹזְדוֹר, וְחָזַר מִיָּד פְּנִימָה אֶל חַדְרוֹ שֶׁל הַחֲזוֹן אִישׁ כְּדֵי לְהוֹדִיעַ לוֹ שֶׁיֵּשׁ יְהוּדִי שֶׁמְּבַקֵּשׁ לְהִכָּנֵס וּלְדַבֵּר עִמּוֹ. אַךְ כַּאֲשֶׁר נִכְנַס רַבִּי שְׁרָגָא אֶל הַחֶדֶר, הוּא גִּלָּה שֶׁהַחֲזוֹן אִישׁ מִתְכּוֹנֵן בְּאוֹתָן רְגָעִים מַמָּשׁ לֶאֱכוֹל אֶת סְעֻדַּת פַּת הָעֶרֶב שֶׁלּוֹ. הַחֲזוֹן אִישׁ כְּבָר הֶחְזִיק בְּיָדָיו אֶת הַנְּטִלָה וְאֶת הַמַּגֶּבֶת כְּדֵי לִטּוֹל יָדַיִם לַסְּעֻדָּה לְאַחַר יוֹם לִמּוּד אָרֹךְ.

כַּאֲשֶׁר רָאָה רַבִּי שְׁרָגָא שֶׁהָרַב עוֹמֵד לֶאֱכוֹל, הוּא נִרְתַּע מִיָּד לַאֲחוֹרָיו. הוּא חָשַׁב שֶׁהָרַב עָיֵף וְזָקוּק לָאֹכֶל, וְעַל כֵּן אָמַר וְהִטְעִים לַחֲזוֹן אִישׁ שֶׁהוּא יֵצֵא אֶל הַפְּרוֹזְדוֹר וְיֹאמַר לְאוֹתוֹ יְהוּדִי לְהַמְתִּין מִחוּץ לַחֶדֶר עַד שֶׁהָרַב יְסַיֵּם לֶאֱכוֹל אֶת סְעֻדָתוֹ. רַבִּי שְׁרָגָא רָצָה לִשְׁמֹר עַל מְנוּחָתוֹ שֶׁל הָרַב כְּדֵי שֶׁיּוּכַל לֶאֱכוֹל בְּשֶׁקֶט וּבְנַחַת.

אוּלָם הַחֲזוֹן אִישׁ לֹא הִסְכִּים לְכָךְ, וַחֲזַרְתּוֹ שֶׁל רַבִּי שְׁרָגָא אָחוֹרָה כְּבָר הָיְתָה מְאֻחֶרֶת מִדַּי. בָּרֶגַע שֶׁהַחֲזוֹן אִישׁ שָׁמַע שֶׁיֵּשׁ יְהוּדִי שֶׁמְּבַקֵּשׁ עֵצָה עַל אִישׁ חוֹלֶה, הוּא הֶחְזִיר מִיָּד אֶת הַנְּטִלָה לִמְקוֹמָהּ וְלֹא נָטַל אֶת יָדָיו. הָרַב צִוָּה עַל רַבִּי שְׁרָגָא לְהַכְנִיס אֶת הָאִישׁ אֶל הַחֶדֶר מִיָּד וּלְלֹא כָּל שִׁהוּי.

כַּאֲשֶׁר נִכְנַס הָאִישׁ אֶל הַחֶדֶר, הַחֲזוֹן אִישׁ הִקְשִׁיב לִדְבָרָיו בְּהִתְעַנְיְנוּת מְרֻבָּה מְאֹד. הוּא מָצָא אֶת הַזְּמַן וְחָקַר בְּסַבְלָנוּת גְּדוֹלָה לְכָל הַפְּרָטִים שֶׁל הַמַּחֲלָה. הָרַב יָעַץ לוֹ עֵצוֹת טוֹבוֹת, דִּבֵּר עִמּוֹ בְּרַכּוּת וְהִשְׁתַּדֵּל בְּכָל כֹּחוֹ לְהַרְגִּיעַ אֶת רוּחוֹ הַסּוֹעֶרֶת וְאֶת לִבּוֹ הַדּוֹאֵג, עַד שֶׁבַּסּוֹף פָּטַר אוֹתוֹ לְשָׁלוֹם כְּשֶׁהוּא מְנֻחָם וְרָגוּעַ.

רַק לְאַחַר שֶׁהָאִישׁ יָצָא מֵהַחֶדֶר, חָזַר הַחֲזוֹן אִישׁ לִטּוֹל אֶת יָדָיו וְלֶאֱכוֹל אֶת סְעֻדַת הָעֶרֶב שֶׁלּוֹ. אוּלָם אֲכִילָתוֹ שֶׁל הָרַב לְאַחַר מִכֵּן לֹא הָיְתָה כְּבְכָל יוֹם, אֶלָּא הָיְתָה מְהוּלָה וּמְעֹרֶבֶת בְּצַעֲרוֹ הֶעָמֹק שֶׁל אוֹתוֹ חוֹלֶה. הַחֲזוֹן אִישׁ הִרְגִּישׁ אֶת הַכְּאֵב שֶׁל הַזּוּלָת בְּתוֹךְ לִבּוֹ, וְלָכֵן מַאֲכָלוֹ הָיָה מְלֻוֶּה בִּתְפִלָּה וּבִדְאָגָה לִרְפוּאַת הַחוֹלֶה.

💡 לָמַדְנוּ מֵהַחֲזוֹן אִישׁ לָשִׂים אֶת הַצַּעַר וְהַדְּאָגָה שֶׁל הַזּוּלָת לִפְנֵי הַנּוֹחוּת וְהַסְּעֻדָּה שֶׁלָּנוּ.
