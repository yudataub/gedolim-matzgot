רַבִּי עֲקִיבָא וְהַתַּלְמִיד הַחוֹלֶה

לְרַבִּי עֲקִיבָא הָיוּ הָמוֹן הָמוֹן הָמוֹן תַּלְמִידִים שֶׁבָּאוּ לִלְמֹד אֶצְלוֹ תּוֹרָה. רַבִּי עֲקִיבָא הָיָה רַב גָּדוֹל גָּדוֹל וַחֲכָם מְאֹד, וְכָל הַתַּלְמִידִים אָהֲבוּ לִלְמֹד מִמֶּנּוּ. יוֹם אֶחָד, כְּשֶׁרַבִּי עֲקִיבָא הִבִּיט סְבִיבוֹ, הוּא שָׂם לֵב שֶׁאֶחָד מֵהַתַּלְמִידִים הַקְּבוּעִים שֶׁלּוֹ חָסֵר וְאֵינוֹ יוֹשֵׁב בִּמְקוֹמוֹ. רַבִּי עֲקִיבָא הִתְעַנְיֵן מִיָּד וְשָׁאַל אֶת הַתַּלְמִידִים הָאֲחֵרִים: מָה קוֹרֶה עִם הַתַּלְמִיד הַזֶּה? לָמָה הוּא לֹא הִגִּיעַ לַלִּמּוּד? עָנוּ לוֹ הַתַּלְמִידִים הָאֲחֵרִים: אָה, הוּא חֹלֶה וְשָׁכוּב בַּחֶדֶר שֶׁלּוֹ. כְּשֶׁשָּׁמַע זֹאת רַבִּי עֲקִיבָא, אָמַר לָהֶם: אוֹיְשׁ, טוֹב, תְּאַחֲלוּ לוֹ רְפוּאָה שְׁלֵמָה.

לְמָחֳרָת, כְּשֶׁהִתְחִיל הַלִּמּוּד, רָאָה רַבִּי עֲקִיבָא שֶׁעֲדַיִן אוֹתוֹ תַּלְמִיד לֹא הִגִּיעַ. וְכָךְ עָבַר עוֹד יוֹם, וְעוֹד יוֹם עָבַר, וְהַתַּלְמִיד עֲדַיִן הָיָה חָסֵר וְלֹא יָכֹל לָבוֹא. בְּשָׁלָב מְסֻיָּם הֵבִין רַבִּי עֲקִיבָא שֶׁהַתַּלְמִיד מַמְשִׁיךְ לִהְיוֹת חֹלֶה בְּחַדְרוֹ, וְשֶׁלֹּא דַּי בְּכָךְ שֶׁרַק יְאַחֲלוּ לוֹ רְפוּאָה שְׁלֵמָה מֵרָחוֹק. אָמַר רַבִּי עֲקִיבָא: מָה קוֹרֶה עִם הַתַּלְמִיד הַזֶּה? צָרִיךְ לִדְאֹג לוֹ!

רַבִּי עֲקִיבָא לֹא חִכָּה שֶׁאֲנָשִׁים אֲחֵרִים יֵלְכוּ אֶל הַתַּלְמִיד הַחוֹלֶה. לַמְרוֹת שֶׁהוּא הָיָה רַב גָּדוֹל גָּדוֹל וּמְפֻרְסָם, קָם רַבִּי עֲקִיבָא בִּכְבוֹדוֹ וּבְעַצְמוֹ וְהָלַךְ אֶל הַחֶדֶר שֶׁבּוֹ שָׁכַב הַתַּלְמִיד. רַבִּי עֲקִיבָא נִכְנַס לַחֶדֶר, וְאֵיךְ שֶׁהַתַּלְמִיד רָאָה אֶת רַבִּי עֲקִיבָא נִכְנַס אֵלָיו לַחֶדֶר, בַּהַתְחָלָה אוּלַי הוּא קְצָת נִבְהַל וְהֻפְתַּע. אַךְ מִיָּד לְאַחַר מִכֵּן הֵבִין הַתַּלְמִיד שֶׁהָרַב הַגָּדוֹל הִגִּיעַ אֵלָיו בִּכְבוֹדוֹ וּבְעַצְמוֹ כְּדֵי לְבַקֵּר אוֹתוֹ, לִבְדֹּק מָה שְׁלוֹמוֹ וְלִרְאוֹת שֶׁהוּא בְּסֵדֶר.

כְּשֶׁרַבִּי עֲקִיבָא נִכְנַס, הוּא רָאָה שֶׁהַחֶדֶר שֶׁל הַתַּלְמִיד הָיָה זָקוּק לְטִפּוּל וּלְסֵדֶר. רַבִּי עֲקִיבָא הֶחְלִיט לִדְאֹג לוֹ מִיָּד. הוּא נִקָּה אֶת הַחֶדֶר שֶׁל הַתַּלְמִיד, סִדֵּר אֶת הַדְּבָרִים בַּחֶדֶר, טִאטֵא אֶת הָרִצְפָּה, פָּתַח אֶת הַחַלּוֹן וְאִוֵּר אֶת הַחֶדֶר בַּאֲוִיר נָקִי. יֵשׁ שֶׁאוֹמְרִים שֶׁכְּשֶׁהַתַּלְמִידִים הָאֲחֵרִים רָאוּ שֶׁרַבִּי עֲקִיבָא הוֹלֵךְ בִּכְבוֹדוֹ וּבְעַצְמוֹ לְבַקֵּר אֶת הַתַּלְמִיד הַחוֹלֶה, הֵם מִיָּד הִצְטָרְפוּ אֵלָיו, בָּאוּ גַּם הֵם לַחֶדֶר וְסִיְּעוּ לְסַדֵּר וּלְנַקּוֹת.

הַגְּמָרָא בְּמַסֶּכֶת נְדָרִים מְסַפֶּרֶת עַל הַמַּעֲשֶׂה הַזֶּה וְאוֹמֶרֶת שֶׁבִּזְכוֹת זֶה שֶׁרַבִּי עֲקִיבָא נִכְנַס לְבַקְרוֹ, כִּבְדוּ וְרִבְּצוּ לְפָנָיו, הַתַּלְמִיד חָיָה וְהִבְרִיא. בִּזְכוֹת זֶה שֶׁסִּדְרוּ לוֹ אֶת הַחֶדֶר, אִוֵּרוּ אֶת הָאֲוִיר, וּבִכְלָל הִתְעַנְיְנוּ בִּשְׁלוֹמוֹ וְדָאֲגוּ לוֹ מִכָּל הַלֵּב, הַתַּלְמִיד הַחוֹלֶה הִרְגִּישׁ הַרְבֵּה יוֹתֵר טוֹב וְחָזַר לְאֵיתָנוֹ.

💡 כַּאֲשֶׁר חָבֵר שֶׁלָּנוּ חֹלֶה, עָלֵינוּ לְבַקֵּר אוֹתוֹ, לִדְאֹג לוֹ וּלְסַיֵּעַ לוֹ, כִּי אִכְפַּתִיּוּת וְעֶזְרָה מֵהַלֵּב יְכוֹלוֹת לַעֲזֹר לוֹ לְהַבְרִיא.
