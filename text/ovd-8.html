הַכִּסֵּא הַמִּתְדַּבֵּק וְהַכָּבוֹד לַתּוֹרָה

הָרַב עוֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק לִבְרָכָה אָהַב אֶת הַתּוֹרָה הַקְּדוֹשָׁה בְּכָל נַפְשׁוֹ וּבְכָל מְאֹדוֹ, וְשָׁקַד עַל לִמּוּדָהּ יוֹמָם וָלַיְלָה בְּאַהֲבָה רַבָּה. כָּל רֶגַע שֶׁל לִמּוּד הָיָה יָקָר עֲבוּרוֹ כְּמוֹ יַהֲלֹם מַבְרִיק, וְהוּא הִשְׁתַּדֵּל מְאֹד שֶׁלֹּא לְבַטֵּל תּוֹרָה אֲפִלּוּ לִזְמַן קָצָר בְּיוֹתֵר. כַּאֲשֶׁר עָבַר הָרַב עוֹבַדְיָה לְהִתְגּוֹרֵר בְּבִנְיָן מְגוּרִים חָדָשׁ, הֶחְלִיטוּ כָּל שְׁכֵנֵי הַבִּנְיָן לַעֲרֹךְ אֲסֵפַת דַּיָּרִים מְשֻׁתֶּפֶת כְּדֵי לְשַׂוחֵחַ יַחְדָּו עַל עִנְיְנֵי הַבַּיִת. הָרַב רָצָה מְאֹד לְהַמְשִׁיךְ וְלִלְמֹד תּוֹרָה בִּשְׁקִידָה עֲמֻקָּה בְּחַדְרוֹ, וְלָכֵן חָשַׁב בְּטוּב חָכְמָתוֹ כֵּיצַד יוּכַל גַּם לְכַבֵּד אֶת כָּל הַשְּׁכֵנִים וְגַם לֹא לְאַבֵּד זְמַן יָקָר שֶׁל לִמּוּד תּוֹרָה. הוּא הֶחְלִיט לִשְׁלֹחַ אֶת הַנַּהָג הַמָּסוּר שֶׁלּוֹ שֶׁיֵּרֵד אֶל אֲסֵפַת הַדַּיָּרִים בִּמְקוֹמוֹ, כְּדֵי שֶׁהַנַּהָג יְיַצֵּג אוֹתוֹ בְּכָל מָה שֶׁיִּצְטָרְכוּ הַשְּׁכֵנִים.

הַנַּהָג יָרַד אֶל מְקוֹם הַמִּפְגָּשׁ שֶׁל דַּיָּרֵי הַבִּנְיָן, אַךְ כַּאֲשֶׁר הִגִּיעַ אֲלֵיהֶם, אַחַת הַשְּׁכֵנוֹת לֹא הָיְתָה מְרוּצָה מִכָּךְ כְּלָל. הִיא מָחֲתָה וְדָרְשָׁה בְּתֹקֶף שֶׁהָרַב עוֹבַדְיָה עַצְמוֹ יֵרֵד אֶל אֲסֵפַת הַדַּיָּרִים, יֵשֵׁב עִמָּהֶם וְיִהְיֶה נָכַח בַּמִּפְגָּשׁ בְּאֹפֶן אִישִׁי, וְאֵינוֹ יָכוֹל לְהִסְתַּפֵּק בִּשְׁלִיחַת הַנַּהָג בִּמְקוֹמוֹ. הַנַּהָג עָלָה חֲזָרָה אֶל בֵּית הָרַב וְסִפֵּר לוֹ בְּשֶׁקֶט עַל דְּרִישָׁתָהּ הַתַּקִיפָה שֶׁל הַשְּׁכֵנָה. הָרַב עוֹבַדְיָה, שֶׁהָיָה עָנָו גָּדוֹל וְאוֹהֵב שָׁלוֹם, לֹא רָצָה לִפְגֹּעַ בִּשְׁכֶנְתּוֹ וְלֹא רָצָה לִגְרֹם לְמַחֲלֹקֶת בַּבִּנְיָן. הוּא סָגַר אֶת הַסֵּפֶר הַקָּדוֹשׁ בְּצַעַר עַל בִּטּוּל הַתּוֹרָה, יָרַד בְּעַצְמוֹ אֶל אֲסֵפַת הַדַּיָּרִים וְיָשַׁב עִם כָּל הַשְּׁכֵנִים כְּפִי שֶׁדָּרְשָׁה הַשְּׁכֵנָה.

כַּאֲשֶׁר הִסְתַּיְמָה אֲסֵפַת הַדַּיָּרִים, קָמוּ כָּל הַשְּׁכֵנִים מִמְּקוֹמוֹתֵיהֶם כְּדֵי לַחֲזֹר לְבָתֵיהֶם. גַּם הַשְּׁכֵנָה שֶׁדָּרְשָׁה מֵהָרַב לָרֶדֶת בִּקְשָׁה לָקוּם מִכִּסֵּאָהּ, אַךְ כַּאֲשֶׁר נִכְנַס הָרַב חֲזָרָה אֶל בֵּיתוֹ, גִּלְתָה הַשְּׁכֵנָה לְתִמְהוֹנָהּ הָרַב כִּי אֵינָהּ מְסֻגֶּלֶת לָקוּם. בְּאֹפֶן מֻפְלָא וּמַפְתִּיעַ בְּיוֹתֵר, הִיא נִדְבְּקָה לַכִּסֵּא וְלֹא הִצְלִיחָה לְהִתְנַתֵּק מִמֶּנּוּ כְּלָל וְכָלַל. כָּל הַשְּׁכֵנִים שֶׁעָמְדוּ מִסָּבִיב נִדְהֲמוּ מֵהַמַּחֲזֶה הַמּוּזָר וְהֵבִינוּ מִיָּד כִּי דָּבָר עַל-טִבְעִי אֵרַע כָּאן. הַשְּׁכֵנָה הֵבִינָה מִיָּד בְּלִבָּהּ כִּי פָּגְעָה בִּכְבוֹדוֹ שֶׁל תַּלְמִיד חָכָם עָצוּם וְגָרְמָה לוֹ לְבִטּוּל תּוֹרָה יָקָר. הִיא הִרְגִּישָׁה חֲרָטָה עֲמֻקָּה מְאֹד וּבִקְשָׁה מְחִילָה מֵהָרַב עוֹבַדְיָה עַל הַפְּגִיעָה בִּכְבוֹדוֹ. הָרַב עוֹבַדְיָה, שֶׁהָיָה מָלֵא אַהֲבָה וְרַחֲמִים לְכָל אָדָם, מָחַל לָהּ מִכָּל הַלֵּב. רַק לְאַחַר שֶׁהָרַב מָחַל לָהּ, הִצְלִיחָה הַשְּׁכֵנָה לָקוּם מֵהַכִּסֵּא בְּקַלּוּת וְלַחֲזֹר לְבֵיתָהּ בְּשָׁלוֹם.

💡 עָלֵינוּ לְכַבֵּד אֶת חַכְמֵי הַתּוֹרָה וּלְהִזָּהֵר מִפְּגִיעָה בִּכְבוֹדָם.
