הַשְּׁתִיל בְּגִינַת הַמֶּלֶךְ

בִּתְקוּפַת הַסְּבוֹרָאִים, לִפְנֵי שָׁנִים רַבּוֹת, חָיוּ יְהוּדִים רַבִּים בְּאֶרֶץ בָּבֶל תַּחַת שִׁלְטוֹן מַלְכֵי פָּרַס. בְּאוֹתָם יָמִים הִנְהִיגוּ אֶת הָעָם רָאשֵׁי הַגּוֹלָה, שֶׁהָיוּ צֶאֱצָאִים מְיֻחָסִים מִזֶּרַע דָּוִד הַמֶּלֶךְ. וְהִנֵּה קָם מֶלֶךְ פָּרַס וְגָזַר גְּזֵרָה קָשָׁה לְהַשְׁמִיד אֶת כָּל בֵּית דָּוִד בְּמַמְלַכְתּוֹ. מְסֻפָּר בַּמְּקוֹרוֹת שֶׁרַק כַּלָּה צְעִירָה אַחַת מִבֵּית דָּוִד נִשְׁאֲרָה בַּחַיִּים, וְהִיא הָיְתָה בְּהֵרָיוֹן.

עוֹד מְסֻפָּר שֶׁבְּאוֹתוֹ הַזְּמַן חָלַם הַמֶּלֶךְ חֲלוֹם שֶׁהִבְהִיל אוֹתוֹ מְאֹד. בַּחֲלוֹמוֹ הוּא עָמַד בְּתוֹךְ גִּנַּת בֵּיתוֹ הַמְּפֹאֶרֶת, וּבְיָדוֹ גַּרְזֶן גָּדוֹל. הַמֶּלֶךְ הֵחֵל לַהֲרֹס אֶת עֲצֵי הַפְּרִי וְעָקַר אֶת כָּל אִילָנוֹת הַגַּן, עַד שֶׁנִּשְׁאַר בָּאֲדָמָה רַק שֹׁרֶשׁ קָטָן אֶחָד שֶׁפָּרַח. כְּשֶׁהֵרִים הַמֶּלֶךְ אֶת הַגַּרְזֶן כְּדֵי לְהַשְׁחִית גַּם אֶת הַשֹּׁרֶשׁ הָאַחֲרוֹן, נִגְלָה לְפָנָיו פִּתְאוֹם אִישׁ זָקֵן וְהָדָר בְּפָנָיו. הַזָּקֵן חָטַף מִיָּדוֹ אֶת הַגַּרְזֶן, הִכָּה בְּמִצְחוֹ שֶׁל הַמֶּלֶךְ וְגָעַר בּוֹ עַל שֶׁבָּא לַעֲקֹר גַּם אֶת הַשֹּׁרֶשׁ הַזֶּה שֶׁנִּשְׁאַר לִפְלֵיטָה. דָּם רַב נִשְׁפַּךְ עַל פְּנֵי הַמֶּלֶךְ. הַמֶּלֶךְ הַנִּבְהָל הִתְחַנֵּן עַל חַיָּיו וְאָמַר כִּי לֹא יָדַע אֶת הַדָּבָר, וְהִבְטִיחַ לַזָּקֵן שֶׁמֵּעַתָּה יַשְׁקֶה אֶת הַשֹּׁרֶשׁ, יִנְצֹר אוֹתוֹ וִיגַדֵּל מִמֶּנּוּ עֲנָפִים חֲדָשִׁים עַד שֶׁיָּשׁוּב הַגַּן לִפְרֹחַ.

כְּשֶׁהֵקִיץ הַמֶּלֶךְ מִשְּׁנָתוֹ תָּמַהּ וְנִבְהַל, כִּי רָאָה שֶׁדָּם אֲמִתִּי שׁוֹתֵת עַל מִצְחוֹ. הוּא אָסַף אֶת כָּל חַכְמֵי פָּרַס, אַךְ אִישׁ מֵהֶם לֹא יָדַע לִפְתֹּר אֶת פֵּשֶׁר הַחֲלוֹם. אָז אָמַר לוֹ אֶחָד מִמְּשָׁרְתָיו כִּי חַכְמֵי הַיְּהוּדִים מְפֻרְסָמִים בְּחָכְמָתָם וּבִפְתִירוֹת חֲלוֹמוֹת, וּמִקְצָתָם נִמְצָאִים בְּבֵית הָאֲסוּרִים. צִוָּה הַמֶּלֶךְ לְהוֹצִיא חָכָם יָשִׁישׁ מִן הַכֶּלֶא וּלְהַבִיאוֹ לָאַרְמוֹן.

הֶחָכָם הַיְּהוּדִי שָׁמַע אֶת הַחֲלוֹם וּפָתַר אוֹתוֹ בִּבְהִירוּת: הַגַּן הוּא מִשְׁפַּחַת בֵּית דָּוִד, וְהָעֵצִים הֵם בְּנֵי הַמְּלוּכָה שֶׁהֻכְרְתוּ. הַשֹּׁרֶשׁ שֶׁנִּשְׁאַר בַּקַּרְקַע הוּא כַּלָּה צְעִירָה מִבֵּית דָּוִד שֶׁעוֹמֶדֶת לָלֶדֶת תִּינוֹק. הַזָּקֵן שֶׁהִכָּה אוֹתְךָ הוּא דָּוִד הַמֶּלֶךְ עַצְמוֹ, וְהַבִּטָּחוֹן שֶׁנָּתַתָּ מְחַיֵּב אוֹתְךָ לִשְׁמֹר מִכָּל מִשְׁמָר עַל הָאִשָּׁה וְעַל הַיֶּלֶד שֶׁיִּוָּלֵד לָהּ. הַמֶּלֶךְ רָאָה כִּי אֱמֶת הַדָּבָר, וּבִקֵּשׁ לָדַעַת הֵיכָן הָאִשָּׁה. אָז פָּרַץ הֶחָכָם בִּבְכִי וְאָמַר שֶׁהִיא בִּתּוֹ הָאֲהוּבָה.

מְסֻפָּר שֶׁהַמֶּלֶךְ קִיֵּם אֶת הַבְטָחָתוֹ, שִׁחְרֵר אֶת הָאֲסִירִים וְשָׁמַר עַל הָאִשָּׁה בְּאַרְמוֹנוֹ בְּכָבוֹד רַב. שָׁם נוֹלַד הַתִּינוֹק, וְהַמֶּלֶךְ קָרָא אֶת שְׁמוֹ בּוּסְתְּנַאי, עַל שֵׁם הַגַּן שֶׁנִּקְרָא בְּלָשׁוֹן פָּרַס "בּוּסְתָּאן". בּוּסְתְּנַאי גָּדַל, לָמַד תּוֹרָה, מִשְׁנָה וְתַלְמוּד, וְנַעֲשָׂה תַּלְמִיד חָכָם מֻפְלָג. בְּבוֹא הַיָּמִים הוּא נִתְמַנָּה לְרֹאשׁ גָּלוּיוֹת יִשְׂרָאֵל, שָׁמַר עַל הַגַּחֶלֶת שֶׁל בֵּית דָּוִד וְהִנְהִיג אֶת עַמּוֹ בְּחָכְמָה וּבִקְדֻשָּׁה.

💡 הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר תָּמִיד עַל עַם יִשְׂרָאֵל וְעַל מַנְהִיגָיו מִבֵּית דָּוִד, וּמַצְמִיחַ יְשׁוּעָה גַּם מִתּוֹךְ מַצָּב קָשֶׁה.
