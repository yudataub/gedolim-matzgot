הַחַיָּל שֶׁשָּׁב לִרְאוֹת

בִּמְלֶאכֶת הַקֹּדֶשׁ שֶׁל הַנְהָגַת עַם יִשְׂרָאֵל, כִּהֵן מָרָן הָרַב עוֹבַדְיָה יוֹסֵף זַצַ"ל כַּ"רִאשׁוֹן לְצִיּוֹן" וְכָרַב הָרָאשִׁי לְיִשְׂרָאֵל. בְּאוֹתָם יָמִים פָּרְצָה מִלְחֶמֶת יוֹם כִּפּוּר הַקָּשָׁה, שֶׁהֵבִיאָה עִמָּהּ צַעַר גָּדוֹל וּדְאָגָה רַבָּה לְכָל בָּתֵּי יִשְׂרָאֵל. חַיָּלִים רַבִּים נָפְלוּ בַּקְּרָבוֹת, וַאֲלָפִים נוֹסָפִים נִפְצְעוּ. לִבּוֹ הָרַחוּם שֶׁל הָרַב עוֹבַדְיָה נִקְרַע מִכְּאֵב עַל סֵבֶל הַחַיָּלִים וּמִשְׁפְּחוֹתֵיהֶם. בְּמַקְבִּיל לְבִקּוּרָיו הַמְּסֻכָּנִים בַּחֲזִית הַמִּלְחָמָה, נָסַע הָרַב לְבָתֵּי הַחוֹלִים כְּדֵי לְבַקֵּר אֶת הַחַיָּלִים הַפְּצוּעִים. הוּא עָבַר מִמִּטָּה לְמִטָּה, חִבֵּק, חִזֵּק, בֵּרַךְ וְהֶעְתִּיר דִּבְרֵי נֶחָמָה חַמִּים עַל כָּל חַיָּל וְחַיָּל.

בְּאֶחָד מִבִּקּוּרָיו הִגִּיעַ הָרַב עוֹבַדְיָה אֶל מִטָּתוֹ שֶׁל חַיָּל צָעִיר שֶׁנִּפְצַע קָשֶׁה בַּקְּרָב. הָרוֹפְאִים שֶׁטִּפְּלוּ בּוֹ קָבְעוּ בַּעֲצָבוֹת כִּי הַחַיָּל אִבֵּד אֶת מְאוֹר עֵינָיו וְלֹא יוּכַל עוֹד לִרְאוֹת דָּבָר. הַחַיָּל הַפָּצוּעַ שָׁכַב בְּמִטָּתוֹ מְדֻכָּא וּמְיֻסָּר מְאֹד. דִּמְעוֹת כְּאֵב זָלְגוּ מֵעֵינָיו הַחֲבוּשׁוֹת, מִתּוֹךְ פַּחַד וְצַעַר עַל כָּךְ שֶׁלֹּא יִרְאֶה עוֹד אֶת אוֹר הַשֶּׁמֶשׁ וְאֶת פְּנֵיהֶם הַצּוֹחֲקוֹת שֶׁל יְלָדָיו הַחֲמוּדִים.

כַּאֲשֶׁר נִכְנַס הָרַב עוֹבַדְיָה לַחֶדֶר וְרָאָה אֶת הַחַיָּל בְּסִבְלוֹ, נִסָּה לְדַבֵּר עַל לִבּוֹ וּלְנַחֲמוֹ. אַךְ לְפֶתַע, מֵרֹב צַעַר וְרַחֲמִים, פָּרַץ הָרַב עַצְמוֹ בִּבְכִי מַר וְנִרְגָּשׁ. לִבּוֹ הַטָּהוֹר לֹא יָכוֹל הָיָה לָשֵׂאת אֶת הַכְּאֵב הַנּוֹרָא שֶׁל הַחַיָּל. הָרַב בִּקֵּשׁ כִּסֵּא, יָשַׁב לְצַד הַמִּטָּה, הֵלִיט אֶת פָּנָיו בְּיָדָיו וּבָכָה יַחַד עִם הַחַיָּל שָׁעָה אֲרֻכָּה.

כַּאֲשֶׁר נִרְגַּע הָרַב מְעַט, חִזֵּק אֶת עַצְמוֹ, עָמַד לְצַד הַמִּטָּה וְהִנִּיחַ אֶת יָדָיו עַל רֹאשׁוֹ שֶׁל הַחַיָּל. בְּעוֹד דִּמְעוֹתָיו שֶׁל הָרַב מִתְעָרְבוֹת בְּדִמְעוֹתָיו שֶׁל הַחַיָּל הַפָּצוּעַ, בֵּרַךְ אוֹתוֹ הָרַב בְּכַוָּנָה עֲמֻקָּה וּבַאֲרִיכוּת שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁלַח לוֹ רְפוּאָה שְׁלֵמָה וְיָשִׁיב לוֹ אֶת מְאוֹר עֵינָיו. הַמִּפְגָּשׁ הַמְּרַגֵּשׁ הַזֶּה נָגַע בְּנַפְשׁוֹ שֶׁל הָרַב עַד כְּדֵי כָּךְ שֶׁהוּא לֹא הָיָה מְסֻגָּל לְהַמְשִׁיךְ בְּבִקּוּרֵי פְּצוּעִים אֲחֵרִים בְּאוֹתוֹ יוֹם, וְרַק לַמָּחֳרָת שָׁב לְבֵית הַחוֹלִים.

כַּעֲבֹר כַּמָּה שָׁבוּעוֹת, חָזַר הָרַב עוֹבַדְיָה שׁוּב לְבֵית הַחוֹלִים כְּדֵי לְחַזֵּק אֶת רוּחָם שֶׁל פְּצוּעִים נוֹסָפִים. בְּעוֹדוֹ עוֹבֵר בְּמִסְדְּרוֹן בֵּית הַחוֹלִים, פָּגַשׁ אֶת הָרוֹפֵא הַבָּכִיר שֶׁהִתְלַוָּה אֵלָיו בַּבִּקּוּר הַקּוֹדֵם. הָרוֹפֵא פָּנָה אֶל הָרַב וְאָמַר לוֹ: "זָכוּר לִי כִּי בַּפַּעַם הַקּוֹדֶמֶת, כַּאֲשֶׁר הָרַב בִּקֵּר אֶת הַחַיָּל שֶׁאִבֵּד אֶת מְאוֹר עֵינָיו, הַדָּבָר הִכְאִיב מְאֹד לִכְבוֹדוֹ". הָרַב הִנְהֵן בְּרֹאשׁוֹ, וְסֵבֶל שׁוּב עָלָה עַל פָּנָיו.

אָז הִכְרִיז הָרוֹפֵא בְּשִׂמְחָה רַבָּה וּבְקוֹל חֲגִיגִי: "אֲנִי שָׂמֵחַ מְאֹד לְבַשֵּׂר לָרַב כִּי בִּרְכַתְכֶם הִתְקַיְּמָה! הִצְלַחְנוּ לְהַצִּיל אֶת עֵינָיו שֶׁל הַחַיָּל, וְהוּא שָׁב לִרְאוֹת בְּבֵרוּר!". הִתְרַגְּשׁוּתוֹ שֶׁל הָרַב עוֹבַדְיָה הָיְתָה עֲצוּמָה וּבִלְתִּי נִתֶּנֶת לְתֵאוּר. בְּקוֹל רוֹעֵד מִשִּׂמְחָה וּמֵהוֹדָיָה לַה', קָרָא הָרַב: "הַמַּלְאָךְ רְפָאֵל עוֹמֵד לְצִדְּכֶם!", וּבְמֶשֶׁךְ דַּקּוֹת אֲרֻכּוֹת הֶעְתִּיר בְּרָכוֹת חַמּוֹת וּמְפֹרָשׁוֹת עַל הָרוֹפֵא וְעַל כָּל הַצֶּוֶת הָרְפוּאִי.
