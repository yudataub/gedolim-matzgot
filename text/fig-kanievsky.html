בִּזְכוּת הַשְּׁתִיקָה וְהַבְּרָכָה

מְסֻפָּר עַל הָרַב הַגָּאוֹן הָרַב חַיִּים קַנְיֶבְסְקִי זֵכֶר צַדִּיק וְקָדוֹשׁ לִבְרָכָה, שֶׁהֲמוֹן אֲנָשִׁים הָיוּ בָּאִים אֵלָיו כְּדֵי לְקַבֵּל מִמֶּנּוּ בְּרָכוֹת. הָאֲנָשִׁים הָיוּ מַגִּיעִים מִכָּל הַמְּקוֹמוֹת בִּזְכוּת הַקְּדֻשָּׁה הַגְּדוֹלָה שֶׁלּוֹ, וּבִזְכוּת הַתּוֹרָה הַרַבָּה שֶׁלָּמַד יוֹם וָלַיְלָה. הָרַב חַיִּים קַנְיֶבְסְקִי הָיָה מְבָרֵךְ אֶת כָּל אֶחָד מֵהֶם בַּמָּה שֶׁהָיָה צָרִיךְ, וְהַבְּרָכוֹת שֶׁלּוֹ הָיוּ מִתְקַיְּמוֹת.

פַּעַם אַחַת הִגִּיעַ אֶל הָרַב אִישׁ אֶחָד. הָאִישׁ הַזֶּה וְאִשְׁתּוֹ הָיוּ נְשׂוּאִים כְּבָר שָׁנִים רַבּוֹת, וְלֹא הָיוּ לָהֶם יְלָדִים. הוּא פָּנָה אֶל הָרַב וּבִקֵּשׁ מִמֶּנּוֹ: "הָרַב, בְּבַקָּשָׁה תְּבָרֵךְ אוֹתִי וְאֶת אִשְׁתִּי שֶׁיִּהְיוּ לָנוּ יְלָדִים." הִסְתַּכֵּל עָלָיו הָרַב חַיִּים קַנְיֶבְסְקִי וְאָמַר לוֹ: "אֲנִי לֹא אֲבָרֵךְ אוֹתְךָ, כִּי צָרִיךְ בִּשְׁבִילְכֶם זְכוּת גְּדוֹלָה. יְבָרֵךְ אוֹתְךָ אָדָם שֶׁפָּגְעוּ בּוֹ – וְהוּא לֹא עָנָה! לֵךְ תְּחַפֵּשׁ אָדָם כָּזֶה..."

הָאִישׁ יָצָא לַדֶּרֶךְ וְתָמַהּ בְּלִבּוֹ: אֵיךְ אֶפְשָׁר לִמְצֹא אָדָם כָּזֶה?! הֲרֵי בְּדֶרֶךְ כְּלָל אֲנָשִׁים לֹא מַעֲלִיבִים, וְגַם מִי שֶׁמַּעֲלִיבִים אוֹתוֹ – בְּדֶרֶךְ כְּלָל עֹנֶה וּמַחֲזִיר. טוֹב, הָאִישׁ הָלַךְ לְדַרְכּוֹ וְחִפֵּשׁ. הוּא מְחַפֵּשׁ וּמְחַפֵּשׁ בְּכָל מָקוֹם, וְהִנֵּה יוֹם אֶחָד הוּא רוֹאֶה בְּאֵיזֶה מָקוֹם מִישֶׁהוּ שֶׁמַּעֲלִיבִים אוֹתוֹ וְהוּא שׁוֹתֵק. אָז הָאִישׁ אָמַר בְּלִבּוֹ: מַהֵר מַהֵר, אֲנִי אֶבְקַשׁ מִמֶּנּוּ שֶׁלֹּא יַעֲנֶה כְּדֵי שֶׁתִּהְיֶה לִי אֶת הַזְּכוּת הַזֹּאת!

הוּא רָץ אֵלָיו מַהֵר, וּבְשֶׁקֶט הִתְחַנֵּן אֵלָיו וְאָמַר לוֹ: "אֲנִי מְבַקֵּשׁ מִמְּךָ, אַל תַּעֲנֶה לוֹ, אַל תַּעֲנֶה לוֹ. בְּבַקָּשָׁה, אַל תַּעֲנֶה לוֹ, אֲנִי מִתְחַנֵּן, אֲנִי אַגִּיד לְךָ עוֹד מְעַט לָמָּה..." הִסְתַּכֵּל הָאִישׁ שֶׁפָּגְעוּ בּוֹ וְאָמַר: "טוֹב, אֲנִי לֹא אֶעֱנֶה." וְאָכֵן, הָאִישׁ הַפּוֹגֵעַ אָמַר אֶת הַמִּלִּים הַקָּשׁוֹת שֶׁלּוֹ, וְאַחַר כָּךְ הוּא הָלַךְ לְדַרְכּוֹ.

וְאָז אָמַר לוֹ הָאִישׁ בְּהִתְרַגְּשׁוּת: "הַגָּאוֹן הָרַב חַיִּים קַנְיֶבְסְקִי אָמַר לִי שֶׁרַק אַתָּה יָכוֹל לְבָרֵךְ אוֹתִי. לִי וּלְאִשְׁתִּי כְּבָר כַּמָּה שָׁנִים אֵין יְלָדִים, וּבִקַּשְׁתִּי מֵהָרַב שֶׁיְּבָרֵךְ אוֹתִי, אֲבָל הוּא אָמַר לִי שֶׁרַק אַתָּה – אָדָם שֶׁפָּגְעוּ בּוֹ וְלֹא עָנָה, הַזְּכוּת שֶׁלְּךָ לְבָרֵךְ אוֹתִי הִיא הַזְּכוּת שֶׁתַּעֲזֹר לָנוּ." הָאִישׁ שֶׁשָּׁתַק אָמַר: "טוֹב", וּבֵרַךְ אוֹתוֹ. וּבֶאֱמַת, כַּעֲבֹר שָׁנָה נוֹלַד לָהֶם תִּינוֹק.

💡 כְּשֶׁאָדָם שֶׁפָּגְעוּ בּוֹ שׁוֹתֵק וְאֵינוֹ עֹנֶה, הַזְּכוּת הַגְּדוֹלָה שֶׁלּוֹ יְכוֹלָה לְהָבִיא בְּרָכָה וִישׁוּעָה.
