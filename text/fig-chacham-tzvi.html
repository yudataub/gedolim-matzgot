הַחָכָם צְבִי וְהַתַּרְנְגוֹלֶת שֶׁבְּלִי לֵב

רַבִּי צְבִי הִירְשׁ אַשְׁכְּנַזִּי, שֶׁנִּקְרָא "הַחָכָם צְבִי", הָיָה חָכָם גָּדוֹל וְרַב חָשׁוּב וּמְפֻרְסָם מְאֹד. הוּא נוֹלַד בָּעִיר טְרֶבִּיטְשׁ שֶׁבְּמוֹרַבְיָה. כְּבָר כְּשֶׁהָיָה יֶלֶד צָעִיר בֶּן שְׁמֹנֶה שָׁנִים בִּלְבַד, נָסַע לִלְמֹד תּוֹרָה אֵצֶל סַבּוֹ, רַבִּי אֶפְרַיִם הַכֹּהֵן, שֶׁהָיָה רַבָּהּ שֶׁל הָעִיר אוֹבֶן שֶׁבְּהוּנְגַרְיָה, וּמְאוּחָר יוֹתֵר נָסַע לִלְמֹד גַּם בָּעִיר סָלוֹנִיקִי אֵצֶל חַכְמֵי הַסְּפָרַדִּים. בְּמַהֲלַךְ חַיָּיו כִּהֵן כְּרַב בֶּעָרִים הַחֲשׁוּבוֹת אַלְטוֹנָה, אַמְסְטֶרְדָּם וְלָבּוֹב, וּבְנוֹ הַדָּגוּל הָיָה רַבִּי יַעֲקֹב עֶמְדִּין, הַמּُכָּר בְּשֵׁם הַיַּעֲבֶ"ץ. הוּא נִקְרָא "הַחָכָם צְבִי" עַל שֵׁם סֵפֶר הַשְּׁאֵלוֹת וְהַתְּשׁוּבוֹת הַמְּפֻרְסָם שֶׁחִבֵּר.

יוֹם אֶחָד הֵבִיאוּ לִפְנֵי הָרַב שְׁאֵלָה מוּזָרָה וּמַפְתִּיעָה מְאֹד: אֲנָשִׁים פָּתְחוּ תַּרְנְגוֹלֶת שֶׁנִּשְׁחֲטָה כַּדִּין – וּלְתִמְהוֹנָם הָרַב לֹא מָצְאוּ בְּתוֹכָהּ לֵב! הָיוּ אֲנָשִׁים וְרַבָּנִים שֶׁרָאוּ זֹאת וְאָמְרוּ מִיָּד בְּבִטָּחוֹן: "תַּרְנְגוֹלֶת בְּלִי לֵב הִיא תַּרְנְגוֹלֶת טְרֵפָה! מָה פִּתְאֹם לֶאֱכֹל אוֹתָהּ? אָסוּר לֶאֱכֹל אוֹתָהּ בְּשׁוּם אֹפֶן!"

אֲבָל הַחָכָם צְבִי לֹא מִהֵר לְהִבָּהֵל מִמַּה שֶּׁנִּרְאָה מוּזָר. הוּא חָשַׁב בְּעֹמֶק, לָמַד אֶת הַסּוּגְיָה הֵיטֵב, וְהֵשִׁיב בְּחָכְמָה עֲמֻקָּה: אִי אֶפְשָׁר שֶׁבַּעַל חַיִּים כָּלְשֶׁהוּ יִחְיֶה אֲפִילוּ רֶגַע אֶחָד בְּלִי לֵב! אִם הַתַּרְנְגוֹלֶת הַזֹּאת הָלְכָה, אָכְלָה וְחָיְתָה כְּרָגִיל עַד שֶׁנִּשְׁחֲטָה – בְּוַדַּאי וּבְבִטָּחוֹן הָיָה לָהּ לֵב. הוּא הִסְבִּיר כִּי כַּנִּרְאֶה הַלֵּב נָפַל אוֹ נִלְקַח בִּזְמַן שֶׁהֵכִינוּ אֶת הַתַּרְנְגוֹלֶת לַבִּשּׁוּל – לְמָשָׁל, אוּלַי חָתוּל חָטַף אוֹתוֹ פִּתְאֹם בִּמְהִירוּת – וְהָאֲנָשִׁים פְּשׁוּט לֹא שָׂמוּ לֵב לְכָךְ. לָכֵן פָּסַק הַחָכָם צְבִי בְּבִטָּחוֹן מָלֵא: הַתַּרְנְגוֹלֶת כְּשֵׁרָה לַאֲכִילָה!

הַחָכָם צְבִי כָּתַב אֶת פְּסַק הַהֲלָכָה הַזֶּה בְּסִפְרוֹ, וְהִסְבִּיר אֶת דְּבָרָיו הֵיטֵב מִתּוֹךְ דִּבְרֵי הַגְּמָרָא וּמִתּוֹךְ הַכָּרַת הַטֶּבַע הַנִּפְלָא שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא. הָיוּ רַבָּנִים שֶׁחָלְקוּ עָלָיו בְּחָרִיפוּת רַבָּה, וְהַוִּכּוּחַ הַזֶּה גָּרַם לוֹ צַעַר רָב מְאֹד. אֲבָל הוּא לֹא נִכְנַע, אֶלָּא עָמַד בִּגְבוּרָה עַל הָאֱמֶת כְּפִי שֶׁהֵבִין אוֹתָהּ מִתּוֹךְ הַתּוֹרָה הַקְּדוֹשָׁה.

סִפְרוֹ הַחָשׁוּב, "שׁוּת חָכָם צְבִי", נִדְפַּס בָּעִיר אַמְסְטֶרְדָּם, וּגְדוֹלֵי הַדּוֹרוֹת הַבָּאִים לָמְדוּ מִמֶּנּוּ וְהִתְפַּעֲלוּ מֵחָכְמָתוֹ. בְּנֵי הַקְּהִילּוֹת בְּאַמְסְטֶרְדָּם – גַּם הָאַשְׁכְּנַזִּים וְגַם הַסְּפָרַדִּים – כִּבְדוּ וְהֶעֱרִיכוּ אוֹתוֹ מְאֹד.

💡 כְּדֵי לִפְסֹק הֲלָכָה וּלְהָבִין אֶת הָאֱמֶת צָרִיךְ לַחְשֹׁב לְעֹמֶק, לְהַכִּיר אֶת הַבְּרִיאָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא, וְלַעֲמֹד עַל הָאֱמֶת בִּגְבוּרָה.
