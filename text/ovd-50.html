הַלִּמּוּד לְאוֹר פָּנַס הָרְחוֹב

בְּאֶחָד מֵהָעֲרָבִים הָיְתָה הַפְסָקַת חַשְׁמַל גְּדוֹלָה וּפִתְאוֹמִית בְּכָל בָּתֵּי הָעִיר יְרוּשָׁלַיִם. בְּכָל הַבָּתִּים בָּעִיר שָׁרַר חֹשֶׁךְ רַב, וְלֹא הָיָה אֶפְשָׁר לְהַדְלִיק בָּהֶם אֶת הָאוֹר כְּלָל. אַךְ מִחוּץ לַבָּתִּים, בִּרְחוֹבֵי הָעִיר יְרוּשָׁלַיִם, הָיוּ פַּנְּסֵי הָרְחוֹב דּוֹלְקִים וּמְאִירִים אֶת הַדֶּרֶךְ בָּאוֹר הַנָּעִים שֶׁלָּהֶם.

הָרַב עוֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק לִבְרָכָה אָהַב מְאֹד מְאֹד אֶת לִמּוֹד הַתּוֹרָה הַקְּדוֹשָׁה, וּבִקֵּשׁ לְהַמְשִׁיךְ וְלִלְמֹד תּוֹרָה גַּם כַּאֲשֶׁר הָיְתָה הַפְסָקַת חַשְׁמַל בַּבָּתִּים. הָרַב עוֹבַדְיָה לֹא רָצָה לְהַפְסִיק אֶת הַלִּמּוּד שֶׁלּוֹ, וְלֹא רָצָה לְבַטֵּל מִלִּמּוּד הַתּוֹרָה בִּגְלַל שֶׁהָיָה חֹשֶׁךְ בַּבַּיִת. מָה עָשָׂה הָרַב עוֹבַדְיָה? הוּא יָצָא מִבֵּיתוֹ אֶל הָרְחוֹב, הָלַךְ וְנֶעֱמַד לְיַד אֶחָד מִפַּנְּסֵי הָרְחוֹב שֶׁהָיוּ דּוֹלְקִים בָּרְחוֹב, וְהֵחֵל לִלְמֹד תּוֹרָה מִתּוֹךְ הַסֵּפֶר שֶׁלּוֹ לְאוֹרוֹ שֶׁל אוֹתוֹ פָּנַס רְחוֹב.

בְּאוֹתוֹ הָעֶרֶב עָבַר שָׁם אִישׁ אֶחָד שֶׁהָיָה בְּדַרְכּוֹ חֲזָרָה אֶל בֵּיתוֹ. כַּאֲשֶׁר עָבַר הָאִישׁ בָּרְחוֹב, הוּא רָאָה אֶת הָרַב עוֹבַדְיָה עוֹמֵד לְיַד פָּנַס הָרְחוֹב הַדּוֹלֵק, אוֹחֵז בַּסֵּפֶר וְלוֹמֵד תּוֹרָה בְּחֶשְׁקָה וּבִשְׁקִידָה רַבָּה. הָאִישׁ הִמְשִׁיךְ בַּדַּרְכּוֹ אֶל בֵּיתוֹ, אָכַל אֶת אֲרוּחַת הָעֶרֶב שֶׁלּוֹ, הָלַךְ לִישֹׁן בַּלַּיְלָה בְּמִטָּתוֹ, וְכַאֲשֶׁר הִגִּיעַ הַבֹּקֶר הַמֻּקְדָּם, קָם הָאִישׁ מִמִּטָּתוֹ כְּדֵי לָלֶכֶת אֶל עֲבוֹדָתוֹ.

כַּאֲשֶׁר יָצָא הָאִישׁ מִבֵּיתוֹ בַּבֹּקֶר הַמֻּקְדָּם וְהָלַךְ בָּרְחוֹב בַּדֶּרֶךְ לַעֲבוֹדָתוֹ, עָבַר שׁוּב לְיַד אוֹתוֹ פָּנַס רְחוֹב. הָאִישׁ הִבִּיט לְעֵבֶר הַפָּנָס וְהִתְפַּלֵּא מְאוֹד! הוּא הָיָה מוּפְתָע בְּיוֹתֵר לִרְאוֹת שֶׁהָרַב עוֹבַדְיָה עֲדַיִן עוֹמֵד שָׁם לְיַד אוֹתוֹ פָּנַס רְחוֹב, וְעוֹדֶנּוּ שָׁקוּעַ כֻּלּוֹ בְּלִמּוֹד הַתּוֹרָה מִתּוֹךְ הַסֵּפֶר, בַּדִּיּוּק כְּפִי שֶׁעָמַד שָׁם וְלָמַד בָּעֶרֶב שֶׁלִּפְנֵי כֵן.

הָאִישׁ הֵבִין כִּי הָרַב עוֹבַדְיָה עָמַד שָׁם בָּרְחוֹב לְיַד הַפָּנָס וְלָמַד תּוֹרָה בְּמֶשֶׁךְ כָּל הַלַּיְלָה כֻּלּוֹ, מִבְּלִי לְהַרְגִּישׁ בִּכְלָל שֶׁהַלַּיְלָה עוֹבֵר וְחָלַף. מֵרֹב שֶׁהָרַב עוֹבַדְיָה הָיָה שָׁקוּעַ כָּל כָּךְ בְּלִמּוֹד הַתּוֹרָה הַקְּדוֹשָׁה שֶׁכָּל כָּךְ אָהַב, הוּא לֹא שָׂם לֵב לַזְּמַן שֶׁעָבַר, וְלֹא הִרְגִּישׁ כֵּיצַד חָלַף לוֹ כָּל הַלַּיְלָה בִּלְמִידַת הַתּוֹרָה. אֲהָבָתוֹ הָעֲצֻמָה שֶׁל הָרַב לַתּוֹרָה הַקְּדוֹשָׁה מִלְּאָה אֶת לִבּוֹ בְּשִׂמְחָה רַבָּה, וְהֵאִירָה אֶת כָּל הַלַּיְלָה בְּאוֹר גָּדוֹל שֶׁל תּוֹרָה.

💡 כַּאֲשֶׁר אוֹהֲבִים אֶת הַתּוֹרָה מִכָּל הַלֵּב, לִמּוּד הַתּוֹרָה מֵבִיא שִׂמְחָה גְּדוֹלָה וּמִתְגַּבֵּר עַל כָּל קוֹשִׁי.
