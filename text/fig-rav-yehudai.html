הַחָכָם הַמְּאִיר אֶת הָעֵינַיִם

לִפְנֵי שָׁנִים רַבּוֹת, בִּתְקוּפָתָם הַמְּפֹאָרָה שֶׁל הַגְּאוֹנִים בְּבָבֶל, חַי וּפָעַל חָכָם גָּדוֹל וְדָגוּל מְאֹד שֶׁשְּׁמוֹ רַב יְהוּדַאי בַּר רַב נַחְמָן. רַב יְהוּדַאי הָיָה מֵחַכְמֵי יְשִׁיבַת פּוּמְבְּדִיתָא הַדְּגוּלָה, שֶׁהָיְתָה אֶחָד מִמֶּרְכְּזֵי הַתּוֹרָה הַחֲשׁוּבִים וְהַמְּפֻרְסָמִים בְּיוֹתֵר בְּכָל הָעוֹלָם הַיְּהוּדִי. בִּיְשִׁיבָה זוֹ לָמְדוּ תַּלְמִידֵי חֲכָמִים רַבִּים תּוֹרָה בִּשְׁקִידָה גְּדוֹלָה, וְרַב יְהוּדַאי בָּלַט בֵּינֵיהֶם בְּחָכְמָתוֹ הַעֲמֻקָּה, בִּבְקִיאוּתוֹ הָרַבָּה וּבִתְבוּנָתוֹ הַגְּדוֹלָה.

דָּבָר מְיֻחָד בְּמִינוֹ הָיָה בְּרַב יְהוּדַאי: הוּא הָיָה עִוֵּר וְלֹא יָכֹל לִרְאוֹת בְּעֵינָיו הַגַּשְׁמִיּוֹת. בִּלְשׁוֹן חֲכָמִים יְקָרָה וּנְקִיָּה כִּנּוּ אוֹתוֹ בַּתֹּאַר "מְאוֹר עֵינַיִם". אַף עַל פִּי שֶׁלֹּא רָאָה בְּעֵינָיו, עֵינֵי רוּחוֹ, נִשְׁמָתוֹ וּשְׂכָלוֹ הָיוּ פְּתוּחוֹת וּמְאִירוֹת בְּאוֹר גָּדוֹל. הוּא זָכַר אֶת כָּל הַתּוֹרָה בְּבֵרוּר וּבְדִיּוּק רַב, וְהֵאִיר אֶת דַּרְכָּם שֶׁל כָּל לוֹמְדֵי הַתּוֹרָה שֶׁבָּאוּ לִשְׁמֹעַ אֶת דְּבָרָיו הַחֲכָמִים.

בְּאוֹתָם יָמִים הָיְתָה בְּבָבֶל יְשִׁיבָה מְפֹאָרָה נוֹסֶפֶת שֶׁנִּקְרְאָה יְשִׁיבַת סוּרָא. מַנְהִיגֵי יְשִׁיבַת סוּרָא חִפְּשׂוּ חָכָם גָּדוֹל שֶׁיַּנְהִיג אֶת הַיְּשִׁיבָה וְיִהְיֶה לָהֶם לְרֹאשׁ וּלְמוֹרֶה, אַךְ בִּיְשִׁיבַת סוּרָא לֹא נִמְצָא אָז שׁוּם חָכָם שֶׁהִגִּיעַ לְרָמַת חָכְמָתוֹ הַמֻּפְלָגָה וְהָעֲמֻקָּה שֶׁל רַב יְהוּדַאי מִיְּשִׁיבַת פּוּמְבְּדִיתָא.

כְּשֶׁרָאָה זֹאת רֹאשׁ הַגּוֹלָה, שְׁלֹמֹה הַנָּשִׂיא, שֶׁהָיָה הַמַּנְהִיג הַכְּלָלִי שֶׁל יְהוּדֵי בָּבֶל, הוּא הֵבִין אֶת הַחֲשִׁיבוֹת הַגְּדוֹלָה שֶׁל הַעֲמָדַת מַנְהִיג רוּחָנִי דָּגוּל בְּרֹאשׁ יְשִׁיבַת סוּרָא. לָכֵן, שְׁלֹמֹה הַנָּשִׂיא לָקַח אֶת רַב יְהוּדַאי מִיְּשִׁיבַת פּוּמְבְּדִיתָא וּמִנָּה אוֹתוֹ לִהְיוֹת הַגָּאוֹן וְרֹאשׁ הַיְּשִׁיבָה בְּסוּרָא. רַב יְהוּדַאי גָּאוֹן הִנְהִיג אֶת יְשִׁיבַת סוּרָא בְּחָכְמָה, בְּרוֹמְמָה וּבִמְסִירוּת רַבָּה בְּמֶשֶׁךְ שָׁלֹשׁ שָׁנִים וָחֵצִי.

בְּאוֹתוֹ הַפֶּרֶק מַמָּשׁ, הָיָה גַּם אָחִיו שֶׁל רַב יְהוּדַאי, שֶׁשְּׁמוֹ מַר רַב דּוּדַאי בַּר מַר רַב נַחְמָן, חָכָם גָּדוֹל מְאֹד. מַר רַב דּוּדַאי כִּהֵן כִּגְאוֹן יְשִׁיבַת פּוּמְבְּדִיתָא. כָּךְ אֵרַע דָּבָר מֻפְלָא וְרַב חֲשִׁיבוֹת: שְׁנֵי אַחִים, בְּנֵי מַר רַב נַחְמָן, כִּהֲנוּ בְּאוֹתוֹ הַזְּמָן מַמָּשׁ כִּגְאוֹנִים וּכְרָאשֵׁי שְׁתֵּי הַיְּשִׁיבוֹת הַגְּדוֹלוֹת, רַב יְהוּדַאי בְּסוּרָא וְרַב דּוּדַאי בְּפּוּמְבְּדִיתָא, וְיַחְדָּו הֵאִירוּ אֶת עַם יִשְׂרָאֵל בְּתוֹרָתָם.

בִּמְהַלַּךְ שְׁנוֹת הַנְהָגָתוֹ וְלִמּוּדוֹ, חִבֵּר רַב יְהוּדַאי גָּאוֹן אֶת הַסֵּפֶר הַחָשׁוּב וְהַמְּפֻרְסָם "הִלְכוֹת פְּסוּקוֹת". סֵפֶר זֶה הָיָה אֶחָד מִסִּפְרֵי הַהֲלָכָה הָרִאשׁוֹנִים מַמָּשׁ שֶׁנִּכְתְּבוּ לַאֲחַר חֲתִימַת הַתַּלְמוּד, וְהוּא סִיַּע לִיהוּדִים רַבִּים לָדַעַת וּלְהָבִין אֶת הַהֲלָכָה הַפְּסוּקָה וְהַבְּרוּרָה לְמַעֲשֶׂה. אֶת כָּל הַפְּרָטִים הַחֲשׁוּבִים הָאֵלֶּה עַל חַיָּיו, חָכְמָתוֹ, מִנּוּיוֹ וְהַנְהָגָתוֹ שֶׁל רַב יְהוּדַאי גָּאוֹן, אָנוּ לוֹמְדִים מִתּוֹךְ "אִגֶּרֶת רַב שְׁרִירָא גָּאוֹן", שֶׁשָּׁמְרָה עַל הַזִּכָּרוֹן הַהִיסְטוֹרִי הַמְּפֹאָר זֶה לְדוֹרוֹת.
