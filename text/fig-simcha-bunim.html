הַכַּשְׁרוּת שֶׁל הַפֶּה

הַיְהוּדִי הַקָּדוֹשׁ מִפְּשִׁיסְחָא, רַבִּי יַעֲקֹב יִצְחָק מִפְּשִׁיסְחָא, פָּנָה אֶל תַּלְמִידוֹ, רַבִּי שִׁמְחָה בּוֹנִים מִפְּשִׁיסְחָא, וְאָמַר לוֹ: "קָדִימָה, צֵא לַדֶּרֶךְ!". רַבִּי שִׁמְחָה בּוֹנִים לֹא שָׁאַל שׁוּם שְׁאֵלוֹת. הוּא אָסַף כַּמָּה חֲסִידִים וְאָמַר לָהֶם: "קָדִימָה! הָרַב אָמַר יוֹצְאִים לַדֶּרֶךְ, בּוֹאוּ נֵצֵא לַדֶּרֶךְ!". הַחֲסִידִים וְרַבִּי שִׁמְחָה בּוֹנִים עָלוּ עַל עֲגָלָה וְיָצְאוּ לַדֶּרֶךְ, מִבְּלִי לָדַעַת כְּלָל מָה הוּא יַעֲדָם וּלְאָן הֵם נוֹסְעִים.

הֵם נָסְעוּ וְנָסְעוּ בַּדֶּרֶךְ הָאֲרֻכָּה, עַד שֶׁהִגִּיעוּ לִכְפָר אֶחָד. בַּכְּפָר הֵם עָצְרוּ לְיַד בַּיִת כְּדֵי לְהִכָּנֵס וְלֶאֱכֹל מַשֶּׁהוּ. הֵם דָּפְקוּ עַל הַדֶּלֶת וְשָׁאֲלוּ אֶת בַּעַל הַבַּיִת אִם אֶפְשָׁר לֶאֱכֹל אֶצְלוֹ. בַּעַל הַבַּיִת קִבֵּל אוֹתָם בְּשִׂמְחָה רַבָּה וְאָמַר לָהֶם: "כֵּן, בּוֹאוּ, בּוֹאוּ תִּכָּנְסוּ, בְּרוּכִים הַבָּאִים!".

הַחֲסִידִים הָיוּ אֲנָשִׁים שֶׁנִּזְהֲרוּ מְאֹד בִּכְשְׁרוּת הַמַּאֲכָלִים. מִכֵּיוָן שֶׁלֹּא יָדְעוּ מָה הִיא כַּשְׁרוּת הַבָּשָׂר וְהַשְּׁחִיטָה בַּמָּקוֹם הַזֶּה, הֵם בִּקְשׁוּ מִבַּעַל הַבַּיִת מַאֲכָלֵי חָלָב. בַּעַל הַבַּיִת הֵשִׁיב לָהֶם כִּי אֵין לוֹ מַאֲכָלֵי חָלָב הַמַּסְפִּיקִים לִסְעוּדָה גְּדוֹלָה כָּזוֹ, אֶלָּא רַק מַאֲכָלֵי בָּשָׂר.

כַּאֲשֶׁר שָׁמְעוּ הַחֲסִידִים שֶׁיֵּשׁ רַק מַאֲכָלֵי בָּשָׂר, הֵם הִתְחִילוּ לַחְקֹר אֶת בַּעַל הַבַּיִת בִּשְׁאֵלוֹת הֲלָכָתִיּוֹת מְפֹרָטוֹת מְאֹד. הֵם שָׁאֲלוּ אוֹתוֹ: "אֵיךְ? מִי שָׁחַט? מִי עָשָׂה אֶת הַשְּׁחִיטָה? וְהַבְּהֵמָה הָיְתָה כְּשֵׁרָה? בָּדַקְתָּ טוֹב לִמְהַדְרִין? וּמָה עִם הַהַמְלָחָה? מָה עִם הַהַדָּחָה?". הֵם הִמְשִׁיכוּ וְשָׁאֲלוּ שְׁאֵלוֹת רַבּוֹת עַל הַכַּשְׁרוּת שֶׁל הַבָּשָׂר כְּדֵי לִהְיוֹת בְּטוּחִים שֶׁהַמַּאֲכָל כָּשֵׁר לַמְהַדְרִין.

תּוֹךְ כְּדֵי שֶׁהַחֲסִידִים הִמְשִׁיכוּ לִשְׁאֹל אֶת כָּל הַשְּׁאֵלוֹת הַמְפֹרָטוֹת, פִּתְאֹם הֵם שָׁמְעוּ קוֹל מִפִּנַּת הַבַּיִת. הֵם פָּנוּ לְשָׁם וְרָאוּ אִישׁ עָנִי לָבוּשׁ בְּגָדִים בְּלוֹיִים שֶׁיָּשַׁב לְיַד הַתַּנּוּר. הָאִישׁ הָעָנִי פָּנָה אֲלֵיהֶם וְאָמַר: "חֲסִידִים יְקָרִים! אַשְׁרֵיכֶם! עַל כָּל דָּבָר שֶׁאַתֶּם מַכְנִיסִים אֶל הַפֶּה אַתֶּם שׁוֹאֲלִים שְׁאֵלוֹת רַבּוֹת, כָּל הַכָּבוֹד! אֲבָל תַּגִּידוּ: הַאִם כָּךְ אַתֶּם נוֹהֲגִים גַּם בִּדְבָרִים שֶׁאַתֶּם מוֹצִיאִים מִן הַפֶּה? הַאִם אַתֶּם שׁוֹמְרִים גַּם עַל הַכַּשְׁרוּת שֶׁל הַדְּבָרִים שֶׁאַתֶּם מוֹצִיאִים מֵהַפֶּה וּמְדַבְּרִים?".

כְּשֶׁשָּׁמַע זֹאת רַבִּי שִׁמְחָה בּוֹנִים מִפְּשִׁיסְחָא, הוּא הֵבִין מִיָּד אֶת כָּל הָעִנְיָן. הוּא הֵבִין שֶׁבִּשְׁבִיל לִשְׁמֹעַ אֶת הַמִּשְׁפָּט הַזֶּה וְאֶת הַשְּׁאֵלָה הַזֹּאת מִפִּי הָעָנִי, שָׁלַח אוֹתוֹ רַבּוֹ, הַיְהוּדִי הַקָּדוֹשׁ מִפְּשִׁיסְחָא, וּבִשְׁבִיל הַמַּסָּר הַחָשׁוּב הַזֶּה נַעֲשָׂה כָּל הַמַּסָּע הָאָרֹךְ שֶׁלָּהֶם.

💡 כְּפִי שֶׁאָנוּ נִזְהָרִים מְאֹד בְּמָה שֶׁאָנוּ מַכְנִיסִים לַפֶּה, כָּךְ עָלֵינוּ לְהִזָּהֵר מְאֹד גַּם בַּמִּלִּים שֶׁאָנוּ מוֹצִיאִים מֵהַפֶּה.
