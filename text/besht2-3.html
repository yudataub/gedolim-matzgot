הָרִקּוּד שֶׁבִּטֵּל אֶת הַגְּזֵרָה

בְּעֶרֶב יוֹם הַכִּפּוּרִים, לְאַחַר סִיּוּם תְּפִלַּת הַשַּׁחֲרִית, פָּנָה הַבַּעַל שֵׁם טוֹב אֶל כָּל אַנְשֵׁי בֵּית הַכְּנֶסֶת וּבִקֵּשׁ מֵהֶם לְהִשָּׁאֵר בִּמְקוֹמָם. הוּא גִּלָּה לָהֶם בְּדַאֲגָה כִּי גְּזֵרָה קָשָׁה מְרַחֶפֶת עַל כָּל הַיְּהוּדִים בָּעַיָּרָה. הַבַּעַל שֵׁם טוֹב בִּקֵּשׁ מִכָּל הָאֲנָשִׁים וְהַתַּלְמִידִים לַעֲמֹד בִּתְפִלָּה נִרְגֶּשֶׁת עוֹד לִפְנֵי יוֹם כִּפּוּר, וּלְכַוֵּן מִכָּל הַלֵּב שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יְבַטֵּל אֶת הַגְּזֵרָה. אַנְשֵׁי בֵּית הַכְּנֶסֶת הִתְכַּנְסוּ וְהִתְפַּלְּלוּ בִּתְפִלַּת מִנְחָה שֶׁל עֶרֶב יוֹם כִּפּוּר בְּכַוָּנָה גְּדוֹלָה וּבִדְמָעוֹת מְרֻבּוֹת, אַךְ לְצַעֲרָם הַרַב הַגְּזֵרָה לֹא בֻּטְּלָה. הַבַּעַל שֵׁם טוֹב בִּקֵּשׁ מֵהֶם לְהַמְשִׁיךְ וּלְהִתְפַּלֵּל גַּם בְּלֵיל יוֹם כִּפּוּר, וְעַדַיִן הַגְּזֵרָה לֹא בֻּטְּלָה. כָּל הַלַּיְלָה יָשְׁבוּ כָּל אַנְשֵׁי בֵּית הַכְּנֶסֶת וְלָמְדוּ תּוֹרָה בְּעֶרֶב יוֹם כִּפּוּר, וְעַדַיִן הַגְּזֵרָה לֹא בֻּטְּלָה.

בַּבֹּקֶר שֶׁל יוֹם הַכִּפּוּרִים הֵם הִתְפַּלְּלוּ תְּפִלַּת שַׁחֲרִית בְּכַוָּנָה גְּדוֹלָה וּבִדְמָעוֹת, וְהַגְּזֵרָה עֲדַיִן לֹא בֻּטְּלָה. עָמַד הַבַּעַל שֵׁם טוֹב וְאָמַר לָהֶם כִּי עוֹד מְעַט מַגִּיעָה תְּפִלַּת מִנְחָה וְאַחַר כָּךְ תְּפִלַּת נְעִילָה שֶׁל יוֹם כִּפּוּר, וּבִקֵּשׁ מֵהֶם מְאֹד לְהִתְפַּלֵּל בְּכַוָּנָה רַבָּה. הַתַּלְמִידִים וְאַנְשֵׁי בֵּית הַכְּנֶסֶת הִתְפַּלְּלוּ בְּכָל כֹּחָם, אַךְ גַּם לְאַחַר תְּפִלַּת עַרְבִית וְגַם לְאַחַר צֵאת הַצּוֹם, הַגְּזֵרָה לֹא בֻּטְּלָה.

לְאַחַר צֵאת הַצּוֹם נִכְנַס הַבַּעַל שֵׁם טוֹב לַחֲדָרוֹ שֶׁלּוֹ. הוּא עָמַד שָׁם לְבַדּוֹ בִּתְפִלָּה וּבִדְמָעוֹת וְאָמַר: ה', תְּבַטֵּל אֶת הַגְּזֵרָה! הַתַּלְמִידִים עָמְדוּ בַּחוּץ וְהִתְפַּלְּלוּ גַּם הֵם בְּכָל לִבָּם, וּבְכָל זֹאת הַגְּזֵרָה לֹא בֻּטְּלָה.

לְאַחַר מִכֵּן יָצָא הַבַּעַל שֵׁם טוֹב מֵהַחֶדֶר שֶׁלּוֹ וְאָמַר לַתַּלְמִידִים שֶׁעַכְשָׁיו זֶה זְמַן בִּרְכַּת הַלְּבָנָה, אַף עַל פִּי שֶׁכְּלָל לֹא רָאוּ אֶת הַלְּבָנָה בַּשָּׁמַיִם. הַבַּעַל שֵׁם טוֹב אָמַר לַתַּלְמִידִים: כֻּלְּכֶם עַכְשָׁיו עֹמְדִים אֶחָד לַשֵּׁנִי קָרוֹב קָרוֹב מְחֻבָּקִים, וּמְקַבְּלִים עֲלֵיכֶם מִצְוָה, מִצְוַת עֲשֵׂה שֶׁל "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ", שֶׁכָּל אֶחָד אוֹהֵב אֶת הַשֵּׁנִי בְּכָל נַפְשׁוֹ וּבְכָל מְאֹדוֹ. הַתַּלְמִידִים עָמְדוּ צְפוּפִים וּמְחֻבָּקִים בְּאַהֲבָה רַבָּה, וְהִתְחִילוּ לָשִׁיר וְלִרְקֹד יַחַד.

וְאָז אֵרַע דָּבָר מֻפְלָא! הָעֲנָנִים שֶׁהִסְתִּירוּ עַד כֹּה אֶת הַלְּבָנָה הִתְחִילוּ לְהִתְפַּזֵּר. הַלְּבָנָה הִתְגַּלְתָה בַּשָּׁמַיִם, וְהַבַּעַל שֵׁם טוֹב הִתְחִיל לִרְקֹד וְלִשְׂמֹחַ בְּשִׂמְחָה גְּדוֹלָה, וְאָמַר לְכֻלָּם שֶׁהַגְּזֵרָה בֻּטְּלָה!

💡 כַּאֲשֶׁר יְהוּדִים מִתְאַחֲדִים בְּאַהֲבַת יִשְׂרָאֵל, מְחַבְּקִים זֶה אֶת זֶה וְשָׂמְחִים יַחַד, שַׁעֲרֵי שָׁמַיִם נִפְתָּחִים וְכָל הַגְּזֵרוֹת הָרָעוֹת מִתְבַּטְּלוֹת.
