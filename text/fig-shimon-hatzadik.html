שִׁמְעוֹן הַצַּדִּיק וּשְׁלֹשֶׁת עַמּוּדֵי הָעוֹלָם

בִּימֵי בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי חַי בִּירוּשָׁלַיִם חָכָם גָּדוֹל וְקָדוֹשׁ שֶׁשְּׁמוֹ שִׁמְעוֹן הַצַּדִּיק. הוּא הָיָה מֵאַחֲרוֹנֵי הַחֲכָמִים שֶׁל אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה, וְשִׁמֵּשׁ בְּתַפְקִיד כֹּהֵן גָּדוֹל בְּמֶשֶׁךְ אַרְבָּעִים שָׁנָה. כֻּלָּם קָרְאוּ לוֹ "הַצַּדִּיק", מִפְּנֵי שֶׁהָיָה מָלֵא בְּאַהֲבַת הַתּוֹרָה, בְּמַעֲשֵׂי צְדָקָה וּבַחֲסָדִים גְּדוֹלִים לְכָל אָדָם.

שִׁמְעוֹן הַצַּדִּיק לִמֵּד אֶת כָּל עַם יִשְׂרָאֵל יְסוֹד חָשׁוּב מְאוֹד: הָעוֹלָם כֻּלּוֹ עוֹמֵד עַל שְׁלֹשָׁה עַמּוּדִים חֲזָקִים, וּבִלְעֲדֵיהֶם הוּא אֵינוֹ יָכוֹל לְהִתְקַיֵּם. הָעַמּוּד הָרִאשׁוֹן הוּא הַתּוֹרָה הַקְּדוֹשָׁה, שֶׁהִיא הַבְּרִית שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא עִם הָעוֹלָם, וְכַאֲשֶׁר יְלָדִים לוֹמְדִים תּוֹרָה, הֵם מְאִירִים אֶת הַבְּרִיאָה כֻּלָּהּ. הָעַמּוּד הַשֵּׁנִי הוּא הָעֲבוֹדָה, שֶׁהָיְתָה עֲבוֹדַת הַקָּרְבָּנוֹת בְּבֵית הַמִּקְדָּשׁ, וְהַיּוֹם הִיא הַתְּפִלָּה הַיָּפָה שֶׁאָנוּ מִתְפַּלְּלִים בְּכַוָּנָה מִתּוֹךְ הַלֵּב. הָעַמּוּד הַשְּׁלִישִׁי הוּא גְּמִילוּת חֲסָדִים – לַעֲזֹר לַחֲבֵרִים, לְבַקֵּר חוֹלִים, לְשַׂמֵּחַ אֲנָשִׁים וְלַעֲשׂוֹת מַעֲשִׂים טוֹבִים בְּכָל יוֹם.

בִּזְכוּת צִדְקוּתוֹ שֶׁל שִׁמְעוֹן הַצַּדִּיק וְאַהֲבַת ה' הַטְּהוֹרָה שֶׁלּוֹ, שָׁרְתָה בְּרָכָה נִסִּית בְּבֵית הַמִּקְדָּשׁ בְּמֶשֶׁךְ כָּל שְׁנוֹת כְּהֻנָּתוֹ. הַנֵּר הַמַּעֲרָבִי שֶׁבִּמְנוֹרַת הַזָּהָב דָּלַק תָּמִיד וְלֹא כָּבָה מֵעַצְמוֹ. בְּיוֹם הַכִּפּוּרִים, הַגּוֹרָל לַה' עָלָה תָּמִיד בְּיַד יָמִין, שֶׁהִיא סִימָן לְחֶסֶד וּלְרָצוֹן טוֹב מִשָּׁמַיִם. הַחוּט הָאָדֹם הָיָה הוֹפֵךְ בְּאֹפֶן נִסִּי לְלָבָן כְּדֵי לְהַרְאוֹת שֶׁה' סָלַח לְעַם יִשְׂרָאֵל, וַאֲפִלּוּ מְעַט מִלֶּחֶם הַפָּנִים הִסְפִּיק כְּדֵי לְהַשְׂבִּיעַ אֶת הַכֹּהֲנִים בִּבְרָכָה רַבָּה.

לְפִי הַמָּסֹרֶת, מְסֻפָּר שֶׁבִּימָיו הִגִּיעַ הַמֶּלֶךְ הַיְּוָנִי אַלֶכְּסַנְדֶּר מוֹקְדוֹן עִם צְבָאוֹ, וְרָצָה לַהֲרֹס אֶת בֵּית הַמִּקְדָּשׁ בִּגְלַל דִּבְרֵי מִרְמָה שֶׁסִּפְּרוּ לוֹ שׂוֹנְאֵי יִשְׂרָאֵל. כַּאֲשֶׁר שָׁמַע עַל כָּךְ שִׁמְעוֹן הַצַּדִּיק, לָבַשׁ אֶת בִּגְדֵי הַכְּהֻנָּה הַקְּדוֹשִׁים וְיָצָא לִקְרָאתוֹ בַּלַּיְלָה יַחַד עִם נִכְבְּדֵי יְרוּשָׁלַיִם, כְּשֶׁבִּידֵיהֶם לַפִּידֵי אֵשׁ. מְסֻפָּר שֶׁכַּאֲשֶׁר רָאָה הַמֶּלֶךְ אֶת שִׁמְעוֹן הַצַּדִּיק, הוּא יָרַד מִמֶּרְכַּבְתּוֹ וְהִשְׁתַּחֲוָה לְפָנָיו בְּכָבוֹד רַב. הַמֶּלֶךְ הִסְבִּיר לְשָׂרָיו הַתְּמֵהִים שֶׁדְּמוּתוֹ הַקְּדוֹשָׁה שֶׁל הַכֹּהֵן מַגִּיעָה אֵלָיו בַּחֲלוֹמוֹת וּבַקְּרָבוֹת, וְהִיא זוֹ שֶׁעוֹזֶרֶת לוֹ לְנַצֵּחַ. כָּךְ בִּטֵּל הַמֶּלֶךְ אֶת הַגְּזֵרָה וְהֵגֵן עַל הַמִּקְדָּשׁ.

עוֹד מְסֻפָּר בַּמְּקוֹרוֹת, שֶׁשִּׁמְעוֹן הַצַּדִּיק הִקְפִּיד מְאוֹד עַל צְנִיעוּת וְעַל כַּוָּנַת הַלֵּב הַטְּהוֹרָה. פַּעַם אַחַת הִגִּיעַ אֵלָיו רוֹעֶה צָעִיר וִיפֵה תֹּאַר מֵהַדָּרוֹם, שֶׁהֶחְלִיט לִהְיוֹת נָזִיר וּלְגַלֵּחַ אֶת שְׂעָרוֹ כְּדֵי לְהִתְגַּבֵּר עַל הַגַּאֲוָה וְלַעֲשׂוֹת נַחַת רוּחַ לַבּוֹרֵא. כְּשֶׁשָּׁמַע שִׁמְעוֹן הַצַּדִּיק אֶת כַּוָּנָתוֹ הַנְּקִיָּה לְשֵׁם שָׁמַיִם, נִשֵּׁק אוֹתוֹ עַל רֹאשׁוֹ וְשִׁבֵּחַ אוֹתוֹ עַל נִצְחוֹנוֹ עַל הַיֵּצֶר. כָּךְ הֵאִיר שִׁמְעוֹן אֶת הָעוֹלָם כֻּלּוֹ בְּתוֹרָה, בִּתְפִלָּה וּבְאַהֲבַת חֲבֵרִים.

💡 כְּשֶׁאָנוּ לוֹמְדִים בְּשִׂמְחָה, מִתְפַּלְּלִים מֵהַלֵּב וְעוֹשִׂים חֶסֶד עִם חֲבֵרִים, אָנוּ מַחֲזִיקִים אֶת הָעוֹלָם כֻּלּוֹ.
