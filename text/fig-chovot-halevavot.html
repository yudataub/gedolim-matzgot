הַחַכָּם שֶׁכִּמְעַט וִתֵּר עַל הַסֵּפֶר

לִפְנֵי כִּתְשַׁע מֵאוֹת שָׁנִים, בְּאֶרֶץ סְפָרַד, חַי חָכָם גָּדוֹל וְצַדִּיק שֶׁנִּקְרָא רַבֵּנוּ בַּחְיֵי אִבְּן פַּקוּדָה. רַבֵּנוּ בַּחְיֵי אָהַב מְאֹד אֶת הַשֵּׁם וְאֶת הַתּוֹרָה, וְהוּא הִרְהֵר הַרְבֵּה בַּמִּצְוֹת שֶׁקְּשׁוּרוֹת לַלֵּב שֶׁל הָאָדָם, כְּמוֹ אֱמוּנָה פְּשׁוּטָה, בִּטָּחוֹן בַּהַשֵּׁם, עֲנָוָה, תְּשׁוּבָה וְאַהֲבַת הַשֵּׁם. הוּא לָמַד עַל הַמִּצְוֹת הָאֵלֶּה וְנִסָּה לְקַיֵּם אוֹתָן בְּכָל לִבּוֹ. כְּכָל שֶׁלָּמַד יוֹתֵר, כָּךְ הֵבִין עַד כַּמָּה הַנּוֹשָׂא הַזֶּה רָחָב וְעָמֹק, וְהוּא הֵחֵל לַחְשׁוֹשׁ שֶׁאִם לֹא יִכְתֹּב אֶת הַדְּבָרִים, הוּא עָלוּל לִשְׁכֹּחַ אֶת כָּל הַמַּחְשָׁבוֹת הַטּוֹבוֹת שֶׁעָלוּ בְּרַעְיוֹנוֹ.

בַּדּוֹר שֶׁלּוֹ לֹא הָיוּ הַרְבֵּה אֲנָשִׁים שֶׁעָסְקוּ בְּמִצְוֹת הַלֵּב, וְרַבֵּנוּ בַּחְיֵי רָצָה מְאֹד לַעֲזֹר גַּם לַאֲחֵרִים לָדַעַת אֵיךְ לַעֲבֹד אֶת הַשֵּׁם בֶּאֱמֶת. לָכֵן הוּא הֶחְלִיט לִכְתֹּב סֵפֶר מְיֻחָד שֶׁיֶּאֱסֹף אֶת כָּל הַכְּלָלִים הַחֲשׁוּבִים הָאֵלֶּה, כְּדֵי שֶׁיִּהְיֶה כְּמוֹ נֵר שֶׁיָּאִיר אֶת הַדֶּרֶךְ לִבְנֵי הָאָדָם. אֲבָל כַּאֲשֶׁר הוּא יָשַׁב לִכְתֹּב אֶת הַסֵּפֶר, הֵחֵלָה לַעֲלוֹת בְּלִבּוֹ דְּאָגָה גְּדוֹלָה. הוּא אָמַר לְעַצְמוֹ שֶׁהוּא אֵינוֹ אִישׁ מַסְפִּיק חָכָם וְרָאוּי לְחַבֵּר סֵפֶר כָּזֶה חָשׁוּב. הוּא חָשַׁב שֶׁהַכֹּחַ שֶׁלּוֹ מְעַט, שֶׁהוּא אֵינוֹ מֵבִין הַכֹּל לְגַמְרֵי, וְשֶׁהוּא עָלוּל לִטְעוֹת בַּכְּתִיבָה. הוּא גַּם דָּאַג שֶׁהַשָּׂפָה הָעֲרָבִית שֶׁבָּהּ רָצָה לִכְתֹּב, כְּדֵי שֶׁאַנְשֵׁי דּוֹרוֹ יְבִינוּ, אֵינָהּ מַסְפִּיק צְחוֹרָה וִיפֵהפִיָּה בְּפִיו. מֵרֹב פַּחַד וַחֲשָׁשׁ שֶׁלֹּא יַצְלִיחַ, הֶחְלִיט רַבֵּנוּ בַּחְיֵי לְוַתֵּר עַל הַכְּתִיבָה וְלַחֲזֹר בּוֹ מֵהַתָּכְנִית שֶׁלּוֹ.

אֲבָל אָז הוּא הִתְחִיל לִבְדֹּק אֶת עַצְמוֹ וְלִשְׁאֹל: לָמָה אֲנִי בּוֹרֵחַ? אוּלַי אֲנִי פַּשְׁטוּת מַחְפֵּשׂ אֶת הַמְּנוּחָה וְהָעַצְלוּת? הוּא הֵבִין שֶׁהַפַּחַד עָלוּל לִגְרֹם לָאָדָם לְהַפְסִיד דְּבָרִים טוֹבִים רַבִּים. הוּא נִזְכַּר בַּמַּאֲמָר שֶׁאוֹמֵר שֶׁמֵּרֹב זְהִירוּת שֶׁלֹּא לִטְעוֹת, אָדָם עָלוּל לֹא לַעֲשׂוֹת כְּלָל. הוּא אָמַר לְעַצְמוֹ: אִם כָּל אָדָם שֶׁרוֹצֶה לַעֲשׂוֹת דָּבָר טוֹב יַחֲשֹׁב שֶׁהוּא חַיָּב לִהְיוֹת מֻשְׁלָם וְיִשְׁתֹּק, אַף אֶחָד לֹא יְדַבֵּר דִּבְרֵי תּוֹרָה וְלֹא יַעֲשֶׂה חֶסֶד בָּעוֹלָם! הוּא הֵבִין שֶׁהַיֵּצֶר הָרַע מְנַסֶּה לְפַתּוֹת אֶת הָאָדָם לָשֶׁבֶת בְּעַצְלוּת וְלִמְצֹא תֵּרוּצִים לָמָה לֹא לִפְעֹל.

לָכֵן הֶחְלִיט רַבֵּנוּ בַּחְיֵי לְהִתְגַּבֵּר עַל הַפַּחַד וְעַל הַחֲשָׁשׁ. הוּא הֶחְלִיט לִסְמֹךְ עַל הַשֵּׁם שֶׁיַּעֲזֹר לוֹ, וְלִכְתֹּב אֶת הַסֵּפֶר בְּכָל כֹּחוֹ בַּשָּׂפָה הֲכִי בְּרוּרָה שֶׁהוּא יוֹדֵעַ. הוּא חִלֵּק אֶת הַסֵּפֶר לַעֲשָׂרָה שְׁעָרִים חֲשׁוּבִים שֶׁמְּלַמְּדִים אֶת הָאָדָם אֵיךְ לְתַקֵּן אֶת לִבּוֹ, וְקָרָא לוֹ "חוֹבוֹת הַלְּבָבוֹת". בְּעֶזְרַת הַשֵּׁם, הַסֵּפֶר שֶׁכִּמְעַט לֹא נִכְתַּב הָפַךְ לְסֵפֶר קָדוֹשׁ וְחָשׁוּב שֶׁיְּהוּדִים לוֹמְדִים בּוֹ עַד הַיּוֹם הַזֶּה.

💡 כְּשֶׁאֲנַחְנוּ רוֹצִים לַעֲשׂוֹת דָּבָר טוֹב, אָסוּר לָנוּ לְפַחֵד שֶׁלֹּא נִהְיֶה מֻשְׁלָמִים, אֶלָּא לְהִתְגַּבֵּר עַל הַחֲשָׁשׁ וְלַעֲשׂוֹת בְּכָל הַכֹּחַ.
