יֹאשִׁיָּהוּ הַמֶּלֶךְ וְסֵפֶר הַתּוֹרָה הַנִּפְלָא

יֹאשִׁיָּהוּ הָיָה יֶלֶד צָעִיר בֶּן שְׁמֹנֶה שָׁנִים בִּלְבַד כַּאֲשֶׁר עָלָה לִמְלֹךְ עַל מַמְלֶכֶת יְהוּדָה בִּירוּשָׁלַיִם, וְהוּא מָלַךְ שְׁלֹשִׁים וְאַחַת שָׁנָה. שֵׁם אִמּוֹ הָיָה יְדִידָה בַּת עֲדָיָה מִבָּצְקַת. יֹאשִׁיָּהוּ עָשָׂה אֶת הַיָּשָׁר בְּעֵינֵי הַשֵּׁם, הָלַךְ בְּכָל דֶּרֶךְ דָּוִד אָבִיו וְלֹא סָר מִמֶּנָּה יָמִין אוֹ שְׂמֹאל.

כַּאֲשֶׁר הִגִּיעַ יֹאשִׁיָּהוּ לִשְׁנַת הַמִּמְשָׁל הַשְּׁמִינִית עֶשְׂרֵה שֶׁלּוֹ, שָׁלַח הַמֶּלֶךְ אֶת שָׁפָן בֶּן אֲצַלְיָהוּ בֶן מְשֻׁלָּם הַסּוֹפֵר אֶל בֵּית הַשֵּׁם. הַמֶּלֶךְ צִוָּה עָלָיו לַעֲלוֹת אֶל חִלְקִיָּהוּ הַכֹּהֵן הַגָּדוֹל, כְּדֵי לֶאֱסֹף אֶת כָּל הַכֶּסֶף שֶׁאָסְפוּ שׁוֹמְרֵי הַסַּף מֵאֵת הָעָם, וְלִתֵּן אוֹתוֹ לְעוֹשֵׂי הַמְּלָאכָה הַמֻּפְקָדִים בְּבֵית הַשֵּׁם. כֶּסֶף זֶה יֹעַד לַחָרָשִׁים, לַבּוֹנִים וְלַגּוֹדְרִים, וְגַם לִקְנִיַּת עֵצִים וְאַבְנֵי מַחְצֵב כְּדֵי לְחַזֵּק אֶת בֶּדֶק הַבַּיִת. עֹשֵׂי הַמְּלָאכָה עָשׂוּ אֶת עֲבוֹדָתָם בֶּאֱמוּנָה רַבָּה, וְלָכֵן לֹא הָיָה צֹרֶךְ לַעֲשׂוֹת עִמָּהֶם חֶשְׁבּוֹן עַל הַכֶּסֶף שֶׁנִּתַּן לָהֶם.

בְּמַהֲלָךְ הָעֲבוֹדָה בְּבֵית הַשֵּׁם, אָמַר חִלְקִיָּהוּ הַכֹּהֵן הַגָּדוֹל אֶל שָׁפָן הַסּוֹפֵר: "סֵפֶר הַתּוֹרָה מָצָאתִי בְּבֵית הַשֵּׁם!". חִלְקִיָּהוּ מָסַר אֶת הַסֵּפֶר לְשָׁפָן, וְשָׁפָן קָרָא בּוֹ. לְאַחַר מִכֵּן בָּא שָׁפָן הַסּוֹפֵר אֶל הַמֶּלֶךְ, דִּוֵּחַ לוֹ שֶׁהַכֶּסֶף הוּתַךְ וְנִמְסַר לְעֹשֵׂי הַמְּלָאכָה, וְאָז סִפֵּר לוֹ שֶׁחִלְקִיָּהוּ הַכֹּהֵן נָתַן לוֹ סֵפֶר. שָׁפָן קָרָא אֶת דִּבְרֵי הַסֵּפֶר לִפְנֵי הַמֶּלֶךְ.

כְּשֶׁשָּׁמַע יֹאשִׁיָּהוּ הַמֶּלֶךְ אֶת דִּבְרֵי סֵפֶר הַתּוֹרָה, הוּא קָרַע אֶת בְּגָדָיו מִצַּעַר רַב. הַמֶּלֶךְ צִוָּה עַל חִלְקִיָּהוּ הַכֹּהֵן, אֲחִיקָם בֶּן שָׁפָן, עַכְבּוֹר בֶּן מִיכָיָה, שָׁפָן הַסּוֹפֵר וַעֲשָׂיָה עֶבֶד הַמֶּלֶךְ לֵאמֹר: "לְכוּ דִרְשׁוּ אֶת הַשֵּׁם בַּעֲדִי, בְּעַד הָעָם וּבְעַד כָּל יְהוּדָה עַל דִּבְרֵי הַסֵּפֶר הַנִּמְצָא הַזֶּה, כִּי גְדוֹלָה חֲמַת הַשֵּׁם אֲשֶׁר נִצְּתָה בָנוּ, עַל אֲשֶׁר לֹא שָׁמְעוּ אֲבוֹתֵינוּ עַל דִּבְרֵי הַסֵּפֶר הַזֶּה לַעֲשׂוֹת כְּכָל הַכָּתוּב עָלֵינוּ".

לְאַחַר מִכֵּן אָסַף הַמֶּלֶךְ אֶת כָּל זִקְנֵי יְהוּדָה וִירוּשָׁלַיִם. יֹאשִׁיָּהוּ עָלָה אֶל בֵּית הַשֵּׁם וְעִמּוֹ כָּל אַנְשֵׁי יְהוּדָה, יוֹשְׁבֵי יְרוּשָׁלַיִם, הַכֹּהֲנִים, הַנְּבִיאִים וְכָל הָעָם מִקָּטֹן וְעַד גָּדוֹל. הַמֶּלֶךְ קָרָא בְאָזְנֵיהֶם אֶת כָּל דִּבְרֵי סֵפֶר הַבְּרִית הַנִּמְצָא בְּבֵית הַשֵּׁם. הוּא עָמַד עַל הָעַמּוּד וְכָרַת אֶת הַבְּרִית לִפְנֵי הַשֵּׁם, לָלֶכֶת אַחַר הַשֵּׁם, לִשְׁמֹר מִצְוֹתָיו, עֵדְוֹתָיו וְחֻקֹּתָיו בְּכָל לֵב וּבְכָל נֶפֶשׁ, וְכָל הָעָם עָמַד בַּבְּרִית.

הַמֶּלֶךְ צִוָּה עַל כָּל הָעָם לַעֲשׂוֹת פֶּסַח לַהַשֵּׁם כַּכָּתוּב עַל סֵפֶר הַבְּרִית. פֶּסַח גָּדוֹל כָּזֶה לֹא נַעֲשָׂה מִימֵי הַשּׁוֹפְטִים וְכֹל יְמֵי מַלְכֵי יִשְׂרָאֵל וּמַלְכֵי יְהוּדָה, אֶלָּא רַק בִּשְׁנַת הַמִּמְשָׁל הַשְּׁמִינִית עֶשְׂרֵה לַמֶּלֶךְ יֹאשִׁיָּהוּ. בְּנוֹסָף, בִּעֵר יֹאשִׁיָּהוּ אֶת הָאֹבוֹת, הַיִּדְּעֹנִים, הַתְּרָפִים, הַגִּלּוּלִים וְכָל הַשִּׁקּוּצִים בְּאֶרֶץ יְהוּדָה וּבִירוּשָׁלַיִם, כְּדֵי לְהָקִים אֶת דִּבְרֵי הַתּוֹרָה הַכְּתוּבִים עַל הַסֵּפֶר. כָּמֹהוּ לֹא הָיָה לְפָנָיו מֶלֶךְ אֲשֶׁר שָׁב אֶל הַשֵּׁם בְּכָל לְבָבוֹ, בְּכָל נַפְשׁוֹ וּבְכָל מְאֹדוֹ כְּכֹל תּוֹרַת מֹשֶׁה, וְאַחֲרָיו לֹא קָם כָּמֹהוּ.

💡 כַּאֲשֶׁר אָנוּ לוֹמְדִים אֶת דִּבְרֵי הַתּוֹרָה וְחוֹזְרִים אֶל הַשֵּׁם בְּכָל לִבֵּנוּ, אָנוּ מְקַיְּמִים אֶת הַמִּצְוֹת בְּשִׂמְחָה וּמְאִירִים אֶת הָעוֹלָם.
