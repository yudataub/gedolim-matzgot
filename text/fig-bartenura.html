הָרַב הַגָּדוֹל וְעֶשֶׂר מַטְבְּעוֹת הַכֶּסֶף

רַבִּי עוֹבַדְיָה מִבַּרְטְנוּרָא נוֹלַד בְּאִיטַלְיָה, כָּתַב אֶת הַפֵּרוּשׁ הַמְּפֻרְסָם עַל הַמִּשְׁנָה, וּבִשְׁנַת רמ"ח עָלָה לְאֶרֶץ יִשְׂרָאֵל וְהִגִּיעַ לִירוּשָׁלַיִם. הֶחָכָם הַגָּדוֹל הַזֶּה עָמַד בְּרֹאשׁ הַקְּהִילָה הַיְּהוּדִית בָּעִיר. בָּאוֹתָם יָמִים, בָּחוּר צָעִיר מֵאִיטַלְיָה עָזַב אֶת מִשְׁפַּחְתּוֹ וְאֶת אֶרֶץ מוֹלַדְתּוֹ וְעָלָה גַּם הוּא לִירוּשָׁלַיִם כְּדֵי לִלְמֹד תּוֹרָה מִפִּיו. כַּאֲשֶׁר הִגִּיעַ הַתַּלְמִיד הַצָּעִיר לָעִיר, הוּא הִרְגִּישׁ מְאֹד לֹא מְלֻמָּד וְרָצָה לַחֲסוֹת תַּחַת צֵל כַּנְפֵי הָרַב. רַבִּי עוֹבַדְיָה קִבֵּל אוֹתוֹ בְּרַחֲמִים רַבִּים וּבְאַהֲבָה גְּדוֹלָה, וְהִבְטִיחַ לִדְאֹג לוֹ כְּמוֹ לְבֵן יָקִיר.

רַבִּי עוֹבַדְיָה הָיָה אָדָם גָּדוֹל מְאֹד, וְכָל אַנְשֵׁי הָאָרֶץ וְגַם אַנְשֵׁי מִצְרַיִם שָׁמְעוּ לִדְבָרָיו. אֲפִילוּ הַיִּשְׁמְעֵאלִים כִּבְּדוּ אוֹתוֹ וְיָרְאוּ מִמֶּנּוּ. יַחַד עִם זֹאת, הָיָה הָרַב עָנָיו מְאֹד, שְׁפַל רוּחַ, וְדִבֵּר בְּנֹעַם עִם כָּל בְּנֵי הָאָדָם. הַתַּלְמִיד הַצָּעִיר מָצָא קְשָׁיִים רַבִּים בִּמְצִיאַת דִּירָה לָגוּר בָּהּ בִּירוּשָׁלַיִם, כִּי אֲנָשִׁים רַבִּים בָּאוּ לָעִיר. הוּא רָצָה מְאֹד לָגוּר בֶּחָצֵר אַחַת עִם תַּלְמִידִים נוֹסָפִים שֶׁלָּמְדוּ תּוֹרָה יוֹמָם וָלַיְלָה. כְּשֶׁשָּׁמַע רַבִּי עוֹבַדְיָה עַל רְצוֹנוֹ שֶׁל הַתַּלְמִיד, הוּא הִשְׁתַּדֵּל בְּמַאֲמָצִים רַבִּים וַעֲזָרוֹ לִמְצֹא מָקוֹם בְּאוֹתָהּ חָצֵר, וְאַף הִבְטִיחַ שֶׁבְּקָרוֹב יִמְצָא לָהֶם מָקוֹם בֶּחָצֵר שֶׁלּוֹ מַמָּשׁ.

מַעֲשֶׂה יָפֶה נוֹסָף הֶרְאָה אֶת חַסְדּוֹ הַגָּדוֹל שֶׁל הָרַב עִם הַתַּלְמִיד הַגֵּר. בְּכָל שָׁנָה הָיוּ תּוֹשְׁבֵי הָעִיר מְשַׁלְמִים מַס כְּדֵי לַעֲשׂוֹת יַיִן. הָעֲשִׁירִים שִׁלְמוּ עֶשְׂרִים מַטְבְּעוֹת כֶּסֶף לְכָל קַנְטַר יַיִן, הָאֲנָשִׁים הַבֵּינוֹנִיִּים שִׁלְמוּ שִׁשָּׁה עָשָׂר, וְהָעֲנִיִּים שִׁלְמוּ שְׁנֵים עָשָׂר מַטְבְּעוֹת כֶּסֶף. אֲבָל הַמְּמוֻנִּים עַל גְּבִיַּת הַמַּס קָבְעוּ עַל הַתַּלְמִיד הַצָּעִיר מַס גָּבוֹהַּ מְאֹד שֶׁל אַרְבָּעִים מַטְבְּעוֹת כֶּסֶף עַל שְׁנֵי קַנְטָרִים וָחֵצִי שֶׁל יַיִן. הַתַּלְמִיד הִרְגִּישׁ שֶׁהֶעֱמִיסוּ עָלָיו יוֹתֵר מִמַּה שֶּׁמַּגִּיעַ לוֹ, וְסִפֵּר עַל כָּךְ לְרַבִּי עוֹבַדְיָה.

כְּשֶׁשָּׁמַע רַבִּי עוֹבַדְיָה אֶת דִּבְרֵי הַתַּלְמִיד, הוּא עָזַב מִיָּד אֶת כָּל עֲסָקָיו הָרַבִּים. הָרַב הַגָּדוֹל יָצָא בִּכְבוֹדוֹ וּבְעַצְמוֹ הַחוּצָה, הָלַךְ אֶל הַגּוֹבֶה, לָקַח מִיָּדוֹ עֶשֶׂר מַטְבְּעוֹת כֶּסֶף וְהֶחֱזִיר אוֹתָן לַתַּלְמִיד. רַבִּי עוֹבַדְיָה כָּעַס עַל הַגּוֹבִים מִפְּנֵי שֶׁלָּקְחוּ מֵהַתַּלְמִיד מָמוֹן שֶׁלֹּא כַּדִּין, רַק מִפְּנֵי שֶׁהָיָה גֵּר, יָחִיד וּבוֹדֵד בָּעִיר. הָרַב אָמַר שֶׁהַתַּלְמִידִים הֵם עַמּוֹ וְהוּא יִהְיֶה לָהֶם לְמוֹשִׁיעַ. הַתַּלְמִידִים הָאֲחֵרִים הִשְׁתּוֹמְמוּ וְהִתְפַּלְאוּ מְאֹד עַל הַחֶסֶד וְהָאַהֲבָה הַגְּדוֹלָה שֶׁהֶרְאָה הָרַב לַתַּלְמִיד.

בְּכָל יוֹם, אַחֲרֵי הַתְּפִלָּה וְהַדְּרָשָׁה, הָיוּ הָאֲנָשִׁים יוֹשְׁבִים בְּבֵית הַמִּדְרָשׁ שֶׁל רַבִּי עוֹבַדְיָה וְלוֹמְדִים מִשְׁנָה וְתַלְמוּד כְּשָׁלֹשׁ שָׁעוֹת. אַחַר כָּךְ הָיוּ הַלּוֹמְדִים הוֹלְכִים לְבַקֵּר חוֹלִים וּלְחַלֵּק מַתָּנוֹת וּצְדָקָה לַעֲנִיִּים, וְכָל אָדָם נָתַן מִכָּל לִבּוֹ. הַתַּלְמִיד הַצָּעִיר הִתְפַּלֵּל אֶל ה' שֶׁיִּשְׁמֹר אֶת רַבִּי עוֹבַדְיָה וִימַלֵּא אֶת יָמָיו בְּטוֹב וּבִנְעִימִים.

💡 תַּלְמִיד חָכָם אֲמִתִּי דּוֹאֵג לְכָל אָדָם בְּאַהֲבָה וּבְחֶסֶד, וּמַגֵּן עַל הַחַלָּשׁ וְהַבּוֹדֵד.
