הַנֵּס בַּבּוֹר וְכֹחָהּ שֶׁל הַתְּפִלָּה

לִפְנֵי אַלְפֵי שָׁנִים בָּעִיר הַקְּדוֹשָׁה יְרוּשָׁלַיִם, חַי אִישׁ צַדִּיק וְחָשׁוּב שֶׁקָּרְאוּ לוֹ נְחוֹנְיָא חוֹפֵר שִׁיחִין. נְחוֹנְיָא עָסַק כָּל יָמָיו בְּמִצְוָה גְּדוֹלָה וּמְפֹאָרָה מְאֹד: הוּא הָיָה חוֹפֵר בּוֹרוֹת מַיִם רַבִּים, כְּדֵי שֶׁבִּימֵי הַחֹרֶף הֵם יִתְמַלְּאוּ בְּמֵי גְּשָׁמִים טְהוֹרִים. הוּא עָשָׂה זֹאת בְּאַהֲבָה רַבָּה, כְּדֵי שֶׁכַּאֲשֶׁר כָּל בְּנֵי יִשְׂרָאֵל יַעֲלוּ לִירוּשָׁלַיִם בָּרְגָלִים בֶּחָגִים, יִהְיוּ לָהֶם מַיִם מְתוּקִים וּרְעַנַנִּים לִשְׁתּוֹת בַּדֶּרֶךְ הָאֲרֻכָּה. נְחוֹנְיָא הָיָה אִישׁ צַדִּיק וְזָהִיר מְאֹד, וְהוּא תָּמִיד דָּאַג לְכַסּוֹת אֶת כָּל הַבּוֹרוֹת שֶׁחָפַר כְּדֵי שֶׁהֵם לֹא יְהַווּ סַכָּנָה לָאֲנָשִׁים וְלָעוֹבְרִים וְשָׁבִים.

אַךְ יוֹם אֶחָד קָרָה דָּבָר מַפְחִיד וְנוֹרָא: בּוֹר אֶחָד הוּא שָׁכַח לְכַסּוֹת, וְהוּא לֹא כִּסָּה אוֹתוֹ. הַבּוֹר הַזֶּה נִשְׁאַר פָּתוּחַ וְגָלוּי. בִּתּוֹ הַקְּטַנָּה שֶׁל רַבִּי נְחוֹנְיָא חוֹפֵר שִׁיחִין הָלְכָה וְעָבְרָה לְתֻמָּהּ בֵּין הַבּוֹרוֹת, וּפִתְאֹם הִיא מָעֲדָה וְנָפְלָה לְתוֹךְ הַבּוֹר הָאֶחָד שֶׁלֹּא הָיָה מְכֻסֶּה.

כָּל הָאֲנָשִׁים שֶׁהָיוּ בַּסְּבִיבָה רָאוּ אֶת הָאָסוֹן וְנִבְהֲלוּ מְאֹד. הֵם מִהֲרוּ וּבָאוּ מִיָּד כְּדֵי לִקְרֹא לְעֶזְרָה. הָאֲנָשִׁים הַדּוֹאַגִים הֶחְלִיטוּ לִקְרֹא לָאִישׁ הַצַּדִּיק, רַבִּי חֲנִינָא בֶּן דּוֹסָא, כְּדֵי שֶׁיַּעֲמֹד בִּתְפִלָּה זַכָּה לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא לְמַעַן בִּתּוֹ שֶׁל רַבִּי נְחוֹנְיָא חוֹפֵר שִׁיחִין.

רַבִּי חֲנִינָא בֶּן דּוֹסָא עָמַד וְהִתְפַּלֵּל וְהִתְפַּלֵּל בְּכַוָּנָה גְּדוֹלָה מִכָּל הַלֵּב. פִּתְאֹם הוּא עָצַר אֶת תְּפִלָּתוֹ וְאָמַר לְכָל הָאֲנָשִׁים הַדּוֹאַגִים שֶׁהַיַּלְדָּה בְּסֵדֶר וְהַכֹּל בְּסֵדֶר. לַאֲחַר מִכֵּן, הוּא עָמַד וְהִתְפַּלֵּל וְהִתְפַּלֵּל וְהִתְפַּלֵּל שׁוּב, וְאָמַר שׁוּב לְכֻלָּם בְּבִטָּחוֹן: 'אַל דְּאָגָה, הַיַּלְדָּה בְּסֵדֶר'. רַבִּי חֲנִינָא בֶּן דּוֹסָא עָמַד וְהִתְפַּלֵּל שָׁלֹשׁ פְּעָמִים שְׁלֵמוֹת עַל הַיַּלְדָּה.

וּבְסוֹפוֹ שֶׁל דָּבָר קָרָה נֵס גָּדוֹל, וְהַיַּלְדָּה חָזְרָה לְהוֹרֶיהָ בְּשָׁלוֹם וּבְשִׂמְחָה גְּדוֹלָה. הִיא יָצְאָה מֵהַבּוֹר הָעָמֹק וְסִפְּרָה לְכָל מִי שֶׁשָּׁאַל אוֹתָהּ שֶׁמִּישֶׁהוּ מְבֻגָּר עִם זָקָן לָבָן עָזַר לָהּ לָצֵאת מֵהַבּוֹר.

הַנֵּס הַמֻּפְלָא הַזֶּה הִתְרַחֵשׁ בִּזְכוּת תְּפִלָּתוֹ הַחֲזָקָה שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא. רַבִּי חֲנִינָא בֶּן דּוֹסָא הִסְבִּיר לְכָל הָאֲנָשִׁים אֶת סוֹד הַתְּפִלָּה וְאָמַר: 'אִם הַתְּפִלָּה שְׁגוּרָה בְּפִי – כְּלוֹמַר אִם אֲנִי מִתְפַּלֵּל בְּרֶצֶף וְלֹא מַפְסִיק, בְּלִי שׁוּם הַפְסָקוֹת בָּאֶמְצַע, זֶהוּ סִימָן מֻבְהָק לְכָךְ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא קִבֵּל אֶת הַתְּפִלָּה שֶׁלִּי'.

💡 כַּאֲשֶׁר אָנוּ מִתְפַּלְלִים לַקָּדוֹשׁ בָּרוּךְ הוּא בְּרֶצֶף וּמִכָּל הַלֵּב, ה' שׁוֹמֵעַ אֶת תְּפִלָּתֵנוּ וְעוֹזֵר לָנוּ.
