הָאַבָּא שֶׁל בְּנֵי הַיְּשִׁיבוֹת בְּוִילְנָה

בָּעֲיָרָה הַקְּטַנָּה אִיוְיָה שֶׁבְּמְדִינַת לִיטָא נוֹלַד יֶלֶד מְיֻחָד וְחָכָם מְאֹד, וּשְׁמוֹ חַיִּים עוֹזֵר גְּרוֹדְזֶנְסְקִי. כְּבָר כְּשֶׁהָיָה יֶלֶד צָעִיר, כֻּלָּם רָאוּ אֶת הַחָכְמָה הַגְּדוֹלָה שֶׁלּוֹ וְאֶת הָאַהֲבָה הָעֲמֻקָּה שֶׁלּוֹ לַתּוֹרָה, וְלָכֵן קָרְאוּ לוֹ בְּחִבָּה "הָעִלּוּי מֵאִיוְיָה". כְּשֶׁגָּדַל מְעַט, הָלַךְ לִלְמֹד בִּישִׁיבַת וָאלוֹזִ'ין הַמְפֻרְסֶמֶת. שָׁם פָּגַשׁ אֶת רַבִּי חַיִּים מִבְּרִיסְק, וְהַשְּׁנַיִם הָיוּ לַחֲבֵרִים קְרוֹבִים מְאֹד לְכָל יְמֵי חַיֵּיהֶם.

אַף עַל פִּי שֶׁהָיָה עֲדַיִן צָעִיר בַּשָּׁנִים, נִבְחַר רַבִּי חַיִּים עוֹזֵר לִהְיוֹת הַמַּנְהִיג הָרוּחָנִי שֶׁל כָּל יַהֲדוּת הָעִיר הַגְּדוֹלָה וִילְנָה. הוּא לָמַד תּוֹרָה בִּשְׁקִידָה וְחִבֵּר אֶת סֵפֶר הַשְּׁאֵלוֹת וְהַתְּשׁוּבוֹת הֶחָשׁוּב "אֲחִיעֶזֶר". בְּבֵית הַמִּדְרָשׁ שֶׁלּוֹ בְּוִילְנָה הוּא לִמֵּד קְבוּצָה מְיֻחֶדֶת שֶׁל תַּלְמִידִים צְעִירִים וּמֻכְשָׁרִים, שֶׁנִּקְרְאָה "הַקִּבּוּץ שֶׁל רַבִּי חַיִּים עוֹזֵר", מִמֶּנָּה יָצְאוּ לְאַחַר מִכֵּן רָאשֵׁי יְשִׁיבוֹת וְרַבָּנִים חֲשׁוּבִים רַבִּים. בְּנוֹסָף לְכָךְ, הָיָה רַבִּי חַיִּים עוֹזֵר מִמְּיַסְּדֵי "וַעַד הַיְּשִׁיבוֹת", אִרְגּוּן שֶׁדָּאַג לְכָל הַיְּשִׁיבוֹת בְּלִיטָא וּבְפּוֹלִין.

רַבִּי חַיִּים עוֹזֵר הָיָה תַּלְמִיד חָכָם עָצוּם, וְעָנָה עַל שְׁאֵלוֹת קָשׁוֹת בַּהֲלָכָה שֶׁנִּשְׁלְחוּ אֵלָיו מִכָּל רַחֲבֵי הָעוֹלָם. כָּל רַב בְּכָל מָקוֹם יָדַע: אִם יֵשׁ שְׁאֵלָה קָשָׁה וּמְסֻבֶּכֶת – שׁוֹאֲלִים אֶת רַבִּי חַיִּים עוֹזֵר בְּוִילְנָה, וְהוּא יָדַע לַעֲנוֹת אֶת הַתְּשׁוּבוֹת הַנְּכוֹנוֹת. אֲבָל רַבִּי חַיִּים עוֹזֵר לֹא דָּאַג רַק לִשְׁאֵלוֹת בַּהֲלָכָה. הוּא דָּאַג לַיְּשִׁיבוֹת וּלְבַחוּרֵי הַיְּשִׁיבוֹת מַמָּשׁ כְּמוֹ אָבָא אוֹהֵב וְרַחוּם.

בִּשְׁנַת הַ'תַּרְצַ"ט פָּרְצָה מִלְחֶמֶת הָעוֹלָם הַשְּׁנִיָּה. יְשִׁיבוֹת שְׁלֵמוֹת מִפּוֹלִין בָּרְחוּ מִפְּנֵי הַמִּלְחָמָה, וְרַבּוֹת מֵהֶן הִגִּיעוּ אֶל הָעִיר וִילְנָה. אַלְפֵי בַּחוּרֵי יְשִׁיבָה הִגִּיעוּ לָעִיר כְּשֶׁהֵם רְחוֹקִים מֵהַבַּיִת וּמֵהַהוֹרִים שֶׁלָּהֶם, בְּלִי כֶּסֶף, בְּלִי אֹכֶל, וּבְלִי מָקוֹם מְסֻדָּר לִישׁוֹן בּוֹ.

בְּאוֹתָהּ תְּקוּפָה, רַבִּי חַיִּים עוֹזֵר כְּבָר הָיָה אָדָם זָקֵן וְחוֹלֶה מְאֹד. אֲבָל רַבִּי חַיִּים עוֹזֵר לֹא נָח וְלֹא נָפַשׁ אֲפִלּוּ לְרֶגַע אֶחָד. הוּא עָבַד יוֹמָם וָלַיְלָה כְּדֵי שֶׁלְּכָל יְשִׁיבָה שֶׁהִגִּיעָה יִהְיֶה מָקוֹם לִלְמֹד בּוֹ, וּכְדֵי שֶׁלְּכָל בָּחוּר וּבָחוּר יִהְיֶה מָקוֹם לִישׁוֹן וְאֹכֶל לֶאֱכֹל. הוּא כָּתַב מִכְתָּבִים רַבִּים, בִּקֵּשׁ עֶזְרָה מִיְּהוּדִים בְּכָל רַחֲבֵי הָעוֹלָם, וְדָאַג בְּכָל כֹּחוֹ שֶׁלִּמּוּד הַתּוֹרָה לֹא יִפָּסֵק גַּם בַּיָּמִים הַקָּשִׁים בְּיוֹתֵר. כָּךְ, בִּזְכוּתוֹ, הִמְשִׁיכוּ אַלְפֵי בַּחוּרִים לִלְמֹד תּוֹרָה בְּשַׁלְוָה וּבִשְׁקִידָה גַּם בִּזְמַן הַמִּלְחָמָה.

רַבִּי חַיִּים עוֹזֵר נִפְטַר בְּוִילְנָה בְּהַ' בְּאָב שְׁנַת הַ'תַּשָ"ם. אֲבָל רַבִּים מִבַּחוּרֵי הַיְּשִׁיבוֹת שֶׁהוּא דָּאַג לָהֶם וְהִצִּיל אוֹתָם, זָכוּ לְהִנָּצֵל מֵהַמִּלְחָמָה, וְהֵם אֵלֶּה שֶׁבָּנוּ מֵחָדָשׁ אֶת עוֹלַם הַתּוֹרָה הַמְּפֹאָר בְּאֶרֶץ יִשְׂרָאֵל וּבְכָל הָעוֹלָם כֻּלּוֹ.

💡 מַנְהִיג אֲמִתִּי דּוֹאֵג לְכָל אֶחָד גַּם כְּשֶׁהוּא חוֹלֶה וְעָיֵף, וּמִי שֶׁדּוֹאֵג לַתּוֹרָה בּוֹנֶה אֶת עֲתִיד עַם יִשְׂרָאֵל.
