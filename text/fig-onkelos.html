אוֹנְקְלוֹס וְחַיָּלֵי אַדְרִיָאנוֹס

אוֹנְקְלוֹס הָיָה פַּעַם גּוֹי. כְּשֶׁהוּא שָׁמַע עַל עַם יִשְׂרָאֵל וְעַל הַתּוֹרָה, הוּא הֶחְלִיט לְהִתְגַּיֵּר וְנִהְיָה יְהוּדִי. הוּא עָבַר לִחְיוֹת בִּירוּשָׁלַיִם. לְאוֹנְקְלוֹס הָיָה דּוֹד גּוֹי רָשָׁע מְאֹד שֶׁקָּרְאוּ לוֹ אַדְרִיָאנוֹס. כְּשֶׁאַדְרִיָאנוֹס שָׁמַע שֶׁאוֹנְקְלוֹס הִתְגַּיֵּר, הוּא כָּעַס מְאֹד. הוּא אָמַר: אֲנִי אֶשְׁלַח אֵלָיו חַיָּלִים לִירוּשָׁלַיִם שֶׁיָּבִיאוּ אוֹתוֹ לְכָאן חֲזָרָה, שֶׁיַּחֲזֹר לִהְיוֹת גּוֹי, וְאִם הוּא לֹא יַסְכִּים – אֲנִי אֶתֵּן לוֹ עֹנֶשׁ!

אַדְרִיָאנוֹס שָׁלַח מִשְׁלַחַת רִאשׁוֹנָה שֶׁל חַיָּלִים. הַחַיָּלִים הִגִּיעוּ לְבֵיתוֹ שֶׁל אוֹנְקְלוֹס בִּירוּשָׁלַיִם. אוֹנְקְלוֹס פָּתַח לָהֶם אֶת הַדֶּלֶת בְּשִׂמְחָה, הוֹשִׁיב אוֹתָם לְיָדוֹ עַל הַשֻּׁלְחָן וְהִתְחִיל לְדַבֵּר אִתָּם בְּדִבְרֵי תוֹרָה. הַחַיָּלִים כָּל כָּךְ הִתְלַהֲבוּ מֵהַדְּבָרִים שֶׁהוּא אָמַר לָהֶם, וּבַסּוֹף הֵם הֶחְלִיטוּ שֶׁגַּם הֵם רוֹצִים לִהְיוֹת יְהוּדִים, וְאוֹנְקְלוֹס גִּיֵּר אוֹתָם.

כְּשֶׁאַדְרִיָאנוֹס רָאָה שֶׁהַחַיָּלִים לֹא חוֹזְרִים, הוּא כָּעַס מְאֹד וְשָׁלַח חַיָּלִים נוֹסָפִים. הוּא הִזְהִיר אוֹתָם: אַתֶּם לֹא מְדַבְּרִים אִתּוֹ, אַתֶּם רַק לוֹקְחִים אוֹתוֹ מֵהַבַּיִת! כְּשֶׁהַחַיָּלִים הִגִּיעוּ, הֵם סֵרְבוּ לִשְׁמֹעַ דִּבְרֵי תוֹרָה וְאָמְרוּ לוֹ: אֲנַחְנוּ רַק בָּאנוּ לָקַחַת אוֹתְךָ, בּוֹא אִתָּנוּ מִיָּד. אוֹנְקְלוֹס יָצָא אִתָּם, וּכְשֶׁהִתְחִילוּ לָלֶכֶת הוּא שָׁאַל אוֹתָם: תַּגִּידוּ, כְּשֶׁהַמֶּלֶךְ שֶׁלָּכֶם הוֹלֵךְ לְטַיֵּל בַּלַּיְלָה, הוּא מַחֲזִיק אִתּוֹ פָּנָס? הַחַיָּלִים הֱשִׁיבוּ: מַה פִּתְאֹם שֶׁהַמֶּלֶךְ יַחֲזִיק אֶת הַפָּנָס?! הַמְּשָׁרְתִים מַחֲזִיקִים אֶת הַפָּנָס שֶׁל הַמֶּלֶךְ! עָנָה לָהֶם אוֹנְקְלוֹס: תֵּדְעוּ לָכֶם שֶׁאֵצֶל עַם יִשְׂרָאֵל זֶה לֹא כָּכָה. כְּשֶׁיָּצָאנוּ מִמִּצְרַיִם, הַקָּדוֹשׁ בָּרוּךְ הוּא הֵאִיר לָנוּ אֶת הַדֶּרֶךְ, הָיָה עַמּוּד אֵשׁ לְפָנֵינוּ כְּדֵי שֶׁאֲנַחְנוּ נִרְאֶה לְאָן צָרִיךְ לָלֶכֶת. הַחַיָּלִים הִתְפַּעֲלוּ מְאֹד, אָמְרוּ: גַּם אֲנַחְנוּ רוֹצִים לִהְיוֹת יְהוּדִים! הֵם לָמְדוּ עַד שֶׁהִתְגַּיְּרוּ.

אַדְרִיָאנוֹס כָּעַס שׁוּב וְשָׁלַח מִשְׁלַחַת שְׁלִישִׁית שֶׁל חַיָּלִים. הוּא הוֹרָה לָהֶם: עַכְשָׁו אַתֶּם הוֹלְכִים, אַתֶּם לֹא מְדַבְּרִים אִתּוֹ עַל שׁוּם דָּבָר! אַתֶּם פְּשׁוּט לוֹקְחִים אוֹתוֹ וּמְבִיאִים אוֹתוֹ לְפֹה. הַחַיָּלִים סָגְרוּ אֶת פִּיהֶם וְהִגִּיעוּ לְבֵית אוֹנְקְלוֹס. כְּשֶׁהֵם לָקְחוּ אוֹתוֹ וְיָצְאוּ, וּכְשֶׁהִגִּיעוּ לְפֶתַח הַדֶּלֶת, אוֹנְקְלוֹס הֵרִים אֶת הַיָּד וְנִשַּׁק אֶת הַמְּזוּזָה. הַחַיָּלִים תָּמְהוּ וְשָׁאֲלוּ אוֹתוֹ מַה הוּא עוֹשֶׂה. הִסְבִּיר לָהֶם אוֹנְקְלוֹס: אֲנִי יוֹדֵעַ שֶׁהַמֶּלֶךְ שֶׁלָּכֶם יוֹשֵׁב בֶּחָדָר שֶׁלּוֹ בִּפְנִים, בְּתוֹךְ הָאַרְמוֹן, וְכָל הַחַיָּלִים שׁוֹמְרִים מִבַּחוּץ. אֲבָל אֵצֶל עַם יִשְׂרָאֵל זֶה לֹא כָּכָה. אֵצֶל עַם יִשְׂרָאֵל אֲנַחְנוּ בִּפְנִים, בְּתוֹךְ הַבַּיִת, וְהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר עָלֵינוּ עַל יְדֵי הַמְּזוּזָה שֶׁבַּפֶּתַח. כְּשֶׁשָּׁמְעוּ זֹאת הַחַיָּלִים, רָצוּ גַּם הֵם לִהְיוֹת יְהוּדִים וְהִתְגַּיְּרוּ.

אַדְרִיָאנוֹס רָאָה שֶׁהוּא לֹא מַצְלִיחַ, הֵבִין שֶׁכְּבָר אֵין לוֹ בְּרֵרָה וְהוּא יוֹתֵר לֹא יִשְׁלַח חַיָּלִים, כִּי הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹהֵב אֶת עַם יִשְׂרָאֵל.
