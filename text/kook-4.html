כֹּתֶרֶת: הָרַב קוּק מַצִּיל אֶת בַּחוּרֵי הַיְשִׁיבָה בְּיוֹם פּוּרִים

בְּבֹקֶר אֶחָד שֶׁל חַג הַפּוּרִים, עָמַד הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק, הָרַאֲיָה קוּק, בְּבֵית הַכְּנֶסֶת. הָרַב הָיָה בְּאֶמְצַע קְרִיאַת מְגִלַּת אֶסְתֵּר לִפְנֵי כָּל הַקָּהָל שֶׁבָּא לִשְׁמֹעַ אֶת הַקְּרִיאָה. הָרַב הָיָה עָטוּף בְּטַלִּית וְעָטוּר בִּתְפִלִּין, כְּפִי שֶׁנּוֹהֲגִים בִּתְפִלַּת הַבֹּקֶר שֶׁל הַחַג. כָּל הָאֲנָשִׁים הִקְשִׁיבוּ בְּשִׂמְחָה וּבְרִכּוּז רַב לְכָל מִלָּה וּמִלָּה שֶׁקָּרָא הָרַב מִתּוֹךְ הַמְּגִלָּה.

וְהִנֵּה, בְּאֶמְצַע קְרִיאַת הַמְּגִלָּה, נִכְנַס אָדָם לְבֵית הַכְּנֶסֶת וּמָסַר לָרַב מִכְתָּב דְּחוּף מְאֹד. הַמִּכְתָּב הַזֶּה הִגִּיעַ מֵהַגָּאוֹן רַבִּי חַיִּים עוֹזֵר, שֶׁהָיָה תַּלְמִיד חָכָם גָּדוֹל מֵהָעִיר קוֹבְנָה שֶׁבְּלִיטָא. הָרַב קוּק פָּתַח אֶת הַמִּכְתָּב וְקָרָא אֶת הַדְּבָרִים הַקָּשִׁים וְהַמַּדְאִיגִים שֶׁכָּתַב לוֹ רַבִּי חַיִּים עוֹזֵר.

בַּמִּכְתָּב סֻפַּר עַל שְׁלֹשָׁה בַּחוּרֵי יְשִׁיבָה צְעִירִים שֶׁבָּרְחוּ לְפּוֹלִין מִשִּׁלְטוֹנוֹת רוּסְיָה הַנּוֹקְשִׁים. הַשִּׁלְטוֹנוֹת הַנּוֹקְשִׁים בְּרוּסְיָה רָצוּ לְגַיֵּס אֶת הַבַּחוּרִים הָאֵלֶּה בְּכֹחַ לַצָּבָא הָרוּסִי, וְלָכֵן בָּרְחוּ הַבַּחוּרִים מֵהֶם כְּדֵי לְהִמָּלֵט מֵהַגְּזֵרָה. אַךְ לְרֹעַ הַמַּזָּל, אוֹתָם שְׁלֹשָׁה בַּחוּרֵי יְשִׁיבָה נִתְפְּסוּ עַל יְדֵי הַחַיָּלִים הָרוּסִיִּים.

רַבִּי חַיִּים עוֹזֵר הִסְבִּיר לָרַב קוּק בַּמִּכְתָּב כִּי אִם לֹא יִשְׁלְחוּ אֲלֵיהֶם מִיָּד "סֶרְטִיפִיקָטִים" – שֶׁהֵם אִישׁוּרֵי עֲלִיָּה רְשְׁמִיִּים לְאֶרֶץ יִשְׂרָאֵל בִּזְמַן הַשִּׁלְטוֹן הַבְּרִיטִי בָּאָרֶץ – יַחֲזִירוּ אוֹתָם הַחַיָּלִים הָרוּסִיִּים אֶל רוּסְיָה. שָׁם, בְּרוּסְיָה, הֵם יִהְיוּ בְּסַכָּנַת חַיִּים מַמָּשִׁית!

כְּשֶׁקָּרָא הָרַב קוּק אֶת הַמִּכְתָּב, הוּא הֵבִין מִיָּד שֶׁמְּדוּבָּר בְּפִקּוּחַ נֶפֶשׁ מַמָּשׁ וְשֶׁהַבַּחוּרִים נִמְצָאִים בְּסַכָּנָה גְּדוֹלָה מְאֹד. הוּא יָדַע שֶׁהַצָּלַת חַיִּים קוֹדֶמֶת לְכָל מִצְוָה אַחֶרֶת, וְשֶׁאָסוּר לְחַכּוֹת אֲפִילוּ רֶגַע אֶחָד.

לָכֵן, מִיָּד הִפְסִיק הָרַב קוּק אֶת קְרִיאַת הַמְּגִלָּה בְּאֶמְצָעָהּ. הוּא לֹא חִכָּה לְסִיּוּם הַתְּפִלָּה, אֶלָּא כְּשֶׁהוּא עֲדַיִן עָטוּף בְּטַלִּית וְעָטוּר בִּתְפִלִּין, הוּא יָצָא וְנָסַע בִּמְהִירוּת אֶל מִשְׂרְדֵי הַמַּנְדָּט הַבְּרִיטִי כְּדֵי לְסַדֵּר אֶת הָעִנְיָן וּלְהַשִּׂיג אֶת אִישׁוּרֵי הָעֲלִיָּה הַמַּצִּילִים.

הַפְּעֻלָּה הַדְּחוּפָה וְהַמְּהִירָה שֶׁל הָרַב קוּק הִצְלִיחָה: עוֹד בְּאוֹתוֹ יוֹם נִשְׁלְחוּ בִּדְחִיפוּת שְׁלֹשָׁה אִישׁוּרֵי כְּנִיסָה לְאֶרֶץ יִשְׂרָאֵל אֶל הַנַּצִּיג הַבְּרִיטִי שֶׁשָּׁהָה בְּפּוֹלִין. בִּזְכוּת זֶה נִצְּלוּ שְׁלֹשֶׁת בַּחוּרֵי הַיְשִׁיבָה מֵהַסַּכָּנָה הַנּוֹרָאָה.

הַבַּחוּרִים הַצְּעִירִים שֶׁנִּצְּלוּ עָלוּ לְאֶרֶץ יִשְׂרָאֵל, וּכְשֶׁגָּדְלוּ הָיוּ לְרַבָּנִים חֲשׁוּבִים וּמְפֻרְסָמִים: הָרַב אַבְרָהָם שַׁדְמִי, הָרַב דָּוִד סָלוֹמוֹן, וְהָרַב שָׁאוּל יִשְׂרָאֵלִי זֵכֶר צַדִּיקִים לִבְרָכָה.

💡 הַצָּלַת חַיִּים שֶׁל אָדָם הִיא הַמִּצְוָה הַחֲשׁוּבָה בְּיוֹתֵר, וְהִיא קוֹדֶמֶת לְכָל דָּבָר.
