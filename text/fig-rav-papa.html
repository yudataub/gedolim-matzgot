רַב פָּפָּא וְהַמַּדְרֵגָה

רַב פָּפָּא הָיָה חָכָם גָּדוֹל, נְשִׂיא יְשִׁיבָה בָּעִיר נַרְשׁ, וְתַלְמִידָם שֶׁל הַחֲכָמִים הַמְּפֻרְסָמִים אָבַיֵי וְרָבָא. רַב פָּפָּא שָׁקַד עַל לִמּוּד הַתּוֹרָה בִּמְסִירוּת רַבָּה וְלִמֵּד תַּלְמִידִים רַבִּים, וְיַחַד עִמּוֹ לָמַד גַּם הֶחָכָם חִיָּיא בַּר רַב מִדִּפְתִּי.

יוֹם אֶחָד, כַּאֲשֶׁר רַב פָּפָּא הָיָה עוֹלֶה בַּמַּדְרֵגוֹת, אֵרַע מַעֲשֶׂה שֶׁהִבְהִיל אוֹתוֹ מְאֹד. פִּתְאֹם רַגְלוֹ הֶחֱלִיקָה עַל אַחַת הַמַּדְרֵגוֹת, וְהוּא כִּמְעַט נָפַל מַטָּה. רַב פָּפָּא נֶאֱחַז בַּמָּקוֹם וְהִצְלִיחַ לִמְנֹעַ נְפִילָה קָשָׁה, אַךְ הוּא עָצַר מִיָּד וְהִתְחִיל לַחְשֹׁב עַל מָה שֶׁקָּרָה לוֹ, כִּי יָדַע שֶׁכָּל דָּבָר שֶׁמִּתְרַחֵשׁ אֵינוֹ קוֹרֶה סְתָם כָּךְ לְלֹא סִבָּה.

רַב פָּפָּא נִבְהַל מְאֹד וְאָמַר מִתּוֹךְ דְּאָגָה עֲמֻקָּה: הַאִים נִתְחַיַּבְנוּ עֹנֶשׁ חָמוּר כְּמוֹ אֲנָשִׁים שֶׁמְּחַלְּלִים אֶת הַשַּׁבָּת אוֹ כְּמוֹ אֲנָשִׁים שֶׁעוֹבְדִים עֲבוֹדָה זָרָה, שֶׁנִּגְזְרָה עֲלֵיהֶם מִיתָה? רַב פָּפָּא דָּאַג מְאֹד וְחָשַׁשׁ שֶׁמָּא נִכְשַׁל בַּעֲבֵרָה חֲמוּרָה מְאֹד שֶׁגָּרְמָה לְכָךְ שֶׁכִּמְעַט נָפַל וְנִפְגַּע.

כְּשֶׁשָּׁמַע זֹאת חִיָּיא בַּר רַב מִדִּפְתִּי, הוּא פָּנָה אֶל רַב פָּפָּא כְּדֵי לְהַרְגִּיעַ אוֹתוֹ וּלְהַסְבִּיר לוֹ אֶת מַשְׁמָעוּת הַדָּבָר. הוּא אָמַר לוֹ: שֶׁמָּא עָנִי בָּא לְיָדְךָ וְלֹא פִּרְנַסְתּוֹ? חִיָּיא בַּר רַב מִדִּפְתִּי הִסְבִּיר לוֹ שֶׁאֵין מְדֻבָּר בַּעֲבֵרָה חֲמוּרָה שֶׁל חִלּוּל שַׁבָּת אוֹ עֲבוֹדָה זָרָה מַמָּשׁ, אֶלָּא יִתָּכֵן שֶׁהִגִּיעַ אֵלָיו אָדָם עָנִי הַזָּקוּק לְעֶזְרָה וּלְמָזוֹן, וְרַב פָּפָּא לֹא נָתַן לוֹ צְדָקָה אוֹ לֹא סִיֵּעַ לוֹ בִּמְזוֹנוֹתָיו כָּרָאוּי.

כְּדֵי לְבָאֵר אֶת דְּבָרָיו, הֵבִיא חִיָּיא בַּר רַב מִדִּפְתִּי אֶת דִּבְרֵי הַחֲכָמִים מִן הַבְּרַיְתָא. הוּא הִזְכִּיר אֶת דִּבְרֵי רַבִּי יְהוֹשֻׁעַ בֶּן קָרְחָה, שֶׁאָמַר שֶׁכָּל הַמַּעֲלִים עֵינָיו מִן הַצְּדָקָה וְאֵינוֹ עוֹזֵר לְעָנִי נֶחְשָׁב כְּאִלּוּ עוֹבֵד עֲבוֹדָה זָרָה, מִפְּנֵי שֶׁבִּשְׁנֵיהֶם כָּתוּב הַלָּשׁוֹן "בְּלִיַּעַל". בַּצְּדָקָה כָּתוּב "הִשָּׁמֶר לְךָ פֶּן יִהְיֶה דָבָר עִם לְבָבְךָ בְּלִיַּעַל", וּבַעֲבוֹדָה זָרָה כָּתוּב "יָצְאוּ אֲנָשִׁים בְּנֵי בְלִיַּעַל".

כְּמוֹ כֵן, הוּא הֵבִיא אֶת דִּבְרֵי רַבִּי אֶלְעָזָר בְּרַבִּי יוֹסֵי, שֶׁאָמַר שֶׁכָּל צְדָקָה וָחֶסֶד שֶׁבְּנֵי יִשְׂרָאֵל עוֹשִׂים בָּעוֹלָם הַזֶּה מְבִיאָה שָׁלוֹם גָּדוֹל וּמְלִיצֵי יֹשֶׁר גְּדוֹלִים, שֶׁהֵם סְנֵגוֹרִים טוֹבִים, בֵּין יִשְׂרָאֵל לְבֵין אֲבִיהֶם שֶׁבַּשָּׁמַיִם. הַחֶסֶד הוּא גְּמִילוּת חֲסָדִים וְהָרַחֲמִים הֵם הַצְּדָקָה. מִסִּפּוּר זֶה לָמְדוּ הַחֲכָמִים עַד כַּמָּה גְדוֹלָה, חֲשׁוּבָה וִיקָרָה מִצְוַת הַצְּדָקָה וְהַחֶסֶד.
