הַתְּפִלָּה הַמְּיֻחֶדֶת שֶׁל מָרָן

מַעֲשֶׂה בְּאִישׁ אֶחָד שֶׁזָּכָה לַחֲזֹר בִּתְשׁוּבָה שְׁלֵמָה וּלְשַׁנּוֹת אֶת חַיָּיו מִקָּצֶה לְקָצֶה. הוּא הֵחֵל לִלְמֹד תּוֹרָה, לִשְׁמֹר מִצְווֹת וְלָלֶכֶת בְּדֶרֶךְ הַיַּהֲדוּת בְּשִׂמְחָה רַבָּה. אַךְ מַצָּבוֹ בְּבֵיתוֹ לֹא הָיָה קַל, כִּי רַעְיָתוֹ נָתְרָה מֵאָחוֹר, לֹא הָיְתָה מוּכָנָה לְהִצְטַרֵף אֵלָיו וְסֵרְבָה לַחֲזֹר בִּתְשׁוּבָה. הָאִישׁ נִסָּה לְדַבֵּר עִם אִשְׁתּוֹ בְּסַבְלָנוּת וּבְנַחַת, כְּדֵי שֶׁתַּכִּיר גַּם הִיא אֶת אוֹר הַתּוֹרָה וְאֶת יָפְיָן שֶׁל הַמִּצְווֹת, אַךְ הִיא הִמְשִׁיכָה בְּדַרְכָּהּ וְלֹא רָצְתָה לַעֲשׂוֹת שׁוּם שִׁנּוּי.

הַדָּבָר הִכְאִיב לָאִישׁ מְאֹד וְגָרַם לוֹ צַעַר גָּדוֹל, כִּי חַיָּיו הָיוּ מְלֵאִים קְשָׁיִים וְהוּא הִרְגִּישׁ שֶׁהוּא אֵינוֹ יָכוֹל לְהַמְשִׁיךְ כָּךְ. בְּצָרָתוֹ הַגְּדוֹלָה, הֶחְלִיט הָאִישׁ לָלֶכֶת אֶל מָרָן הָרַב עוֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק לִבְרָכָה, כְּדֵי לְבַקֵּשׁ מִמֶּנּוּ עֵצָה וּבְרָכָה. הוּא הִגִּיעַ אֶל לִשְׁכָּתוֹ שֶׁל הָרַב, הִמְתִּין בַּתּוֹר בְּסַבְלָנוּת רַבָּה עַד שֶׁהִגִּיעַ תּוֹרוֹ, וּכְשֶׁנִּכְנַס לְחַדְרוֹ שֶׁל הָרַב, סִפֵּר לוֹ אֶת כָּל הַצַּעַר הַכָּבֵד שֶׁסָּבַל בִּגְלַל שֶׁאִשְׁתּוֹ מְסָרֶבֶת לַחֲזֹר בִּתְשׁוּבָה.

לְשֵׁמַע הַדְּבָרִים הַכּוֹאֲבִים, מָרָן הָרַב עוֹבַדְיָה יוֹסֵף הִבִּיט בָּאִישׁ בְּרַחֲמִים רַבִּים וְהֶחְלִיט לְעוֹדֵד אֶת רוּחוֹ. הָרַב אָמַר לוֹ שֶׁהוּא חָשׁוּב וְיָקָר מְאֹד, כְּמוֹ יוֹסֵף הַצַּדִּיק שֶׁעָמַד בְּנִסָּיוֹן גָּדוֹל. הָרַב הִרְגִּיעַ אוֹתוֹ, בִּקֵּשׁ מִמֶּנּוּ שֶׁלֹּא יִבְכֶּה, וְאָמַר לוֹ שֶׁעָלָיו לְהִתְאַזֵּר בְּסַבְלָנוּת וְשֶׁבְּעֶזְרַת הַשֵּׁם אִשְׁתּוֹ תָּשׁוּב בִּתְשׁוּבָה שְׁלֵמָה.

אָז לָקַח מָרָן הָרַב עוֹבַדְיָה יוֹסֵף דַּף נְיָר וְעֵט, וְכָתַב בִּכְתַב יָדוֹ נֻסַּח תְּפִלָּה מְיֻחֶדֶת שֶׁחִבֵּר לַחֲזָרָה בִּתְשׁוּבָה. הָרַב מָסַר אֶת הַנְּיָר לְיָדָיו שֶׁל הָאִישׁ וּבִקֵּשׁ מִמֶּנּוּ שֶׁלֹּא יַרְאֶה אֶת הַתְּפִלָּה לְשׁוּם אָדָם. הָרַב הוֹרָה לוֹ שֶׁבְּכָל יוֹם יִתְפַּלֵּל עַל רַעְיָתוֹ מִתּוֹךְ הַמִּלִּים הַמְּיֻחָדוֹת שֶׁכָּתַב עַל הַדַּף, וְהִבְטִיחַ לוֹ שֶׁבְּעוֹד חֹדֶשׁ אֶחָד אִשְׁתּוֹ תָּשׁוּב בִּתְשׁוּבָה וְהַשֵּׁם יִשְׁלַח לוֹ יְשׁוּעָה.

הָאִישׁ יָצָא מִבֵּיתוֹ שֶׁל הָרַב כְּשֶׁהוּא שָׂמֵחַ, מְנֻחָם וּמְעוֹדָד מְאֹד. הוּא הֶאֱמִין בֶּאֱמוּנָה שְׁלֵמָה בְּבִרְכָתוֹ וּבְדִבְרָיו שֶׁל הַצַּדִּיק. הוּא הִקְפִּיד לַעֲשׂוֹת בְּדִיּוּק כְּפִי שֶׁהוֹרָה לוֹ מָרָן, וּבְכָל יוֹם מִדֵּי יוֹם הִתְפַּלֵּל בְּכַוָּנָה עֲמֻקָּה וּבְרֶגֶשׁ רַב מִתּוֹךְ הַדַּף הַמְּיֻחָד שֶׁקִּבֵּל.

וְהִנֵּה, פֶּלֶא גָּדוֹל אֵרַע! כְּפִי שֶׁהִבְטִיחַ מָרָן הָרַב עוֹבַדְיָה יוֹסֵף, כָּךְ בְּדִיּוּק קָרָה. כַּעֲבֹר חֹדֶשׁ יָמִים אֶחָד, הִתְעוֹרֵר לִבָּהּ שֶׁל אִשְׁתּוֹ, הִיא זָכְתָה לַחֲזֹר בִּתְשׁוּבָה שְׁלֵמָה וְהֵחֵלָה לִשְׁמֹר תּוֹרָה וּמִצְווֹת בְּשִׂמְחָה. הַבַּיִת שֶׁלָּהֶם נִמְלָא אוֹר, שָׁלוֹם וְשִׂמְחָה גְּדוֹלָה, וְכֻלָּם רָאוּ אֶת כֹּחָן הַגָּדוֹל שֶׁל הַתְּפִלָּה וְשֶׁל בִּרְכַת הַצַּדִּיק.

💡 תְּפִלָּה יְשָׁרָה מֵהַלֵּב מִתְקַבֶּלֶת בַּשָּׁמַיִם וִיכוֹלָה לְהָבִיא יְשׁוּעָה וְשָׁלוֹם לְכָל הַמִּשְׁפָּחָה.
