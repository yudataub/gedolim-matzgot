הָרַ"י מִיגָאשׁ – הַחָכָם שֶׁהָרַמְבַּ"ם הִתְפַּעֵל מִמֶּנּוּ

לִפְנֵי כְּתֵשַׁע מֵאוֹת שָׁנִים, בְּאֶרֶץ סְפָרַד, חַי חָכָם גָּדוֹל וּמְפֻרְסָם שֶׁשְּׁמוֹ רַבִּי יוֹסֵף הַלֵּוִי אֶבֶן מִיגָאשׁ, וּבְקִצּוּר קָרְאוּ לוֹ הָרַ"י מִיגָאשׁ. עוֹד מִהְיוֹתוֹ יֶלֶד וְנַעַר צָעִיר, שָׁקַד עַל לִמּוּד הַתּוֹרָה בְּאַהֲבָה רַבָּה וּבְמַאֲמָץ גָּדוֹל. הוּא לָמַד אֵצֶל רַבּוֹ הַמֻּבְהָק, הָרַב הַגָּדוֹל רַבִּי יִצְחָק אַלְפָסִי, הַמֻּכְנָה בְּשֵׁם הָרִי"ף, בָּעִיר אֱלִיסָאנָה, שֶׁנִּקְרְאָה גַּם לוּסֵינָה. הָרַ"י מִיגָאשׁ הִשְׁקִיעַ אֶת כָּל כֹּחוֹ בַּלִּמּוּד, וְהֵבִין אֶת דִּבְרֵי הַתַּלְמוּד בְּעֹמֶק רַב וּבִתְבוּנָה נִפְלָאָה.

הָרִי"ף רָאָה אֶת הַחָכְמָה הָעֲצוּמָה וְאֶת הַשְׁקִידָה הַגְּדוֹלָה שֶׁל תַּלְמִידוֹ הַצָּעִיר. כַּאֲשֶׁר הִגִּיעַ זְמַנּוֹ שֶׁל הָרִי"ף לְהִפָּטֵר מִן הָעוֹלָם, הוּא יָדַע כִּי הָרַ"י מִיגָאשׁ הוּא הָאִישׁ הַרָאוּי בְּיוֹתֵר לְהַנְהִיג אֶת הַדּוֹר. לָכֵן מִנָּה אוֹתוֹ הָרִי"ף לַעֲמֹד אַחֲרָיו בְּרֹאשׁ הַיְּשִׁיבָה הַגְּדוֹלָה. הָרַ"י מִיגָאשׁ קִבֵּל אֶת הַתַּפְקִיד הֶחָשׁוּב, וְעָמַד בְּרֹאשׁ הַיְּשִׁיבָה בְּמֶשֶׁךְ שְׁלֹשִׁים וּשְׁמוֹנֶה שָׁנִים. בְּכָל הַשָּׁנִים הָאֵלֶּה הִרְבָּה לְלַמֵּד תּוֹרָה, לְהָבִין אֶת סוֹדוֹת הַתַּלְמוּד, וּלְגַדֵּל תַּלְמִידֵי חֲכָמִים רַבִּים שֶׁהֵפִיצוּ אֶת אוֹר הַתּוֹרָה.

אֶחָד מִתַּלְמִידָיו הַבּוֹלְטִים וְהַחֲשׁוּבִים בַּיְּשִׁיבָה הָיָה רַבִּי מַיְמוֹן. רַבִּי מַיְמוֹן לָמַד מֵהָרַ"י מִיגָאשׁ חָכְמָה עֲמֻקָּה וְדַרְכֵי עִיּוּן נִפְלָאוֹת בַּתַּלְמוּד הַקָּדוֹשׁ, וְשָׁמַר בְּלִבּוֹ אֶת כָּל הַפֵּרוּשִׁים וְהַהַסְבָּרוֹת שֶׁשָּׁמַע מִפִּי רַבּוֹ.

לְרַבִּי מַיְמוֹן נוֹלַד בֵּן, וּשְׁמוֹ רַבֵּנוּ מֹשֶׁה בֶּן מַיְמוֹן, הַמֻּכְנָה הָרַמְבַּ"ם. הָרַמְבַּ"ם נוֹלַד שָׁלֹשׁ שָׁנִים בִּלְבַד לִפְנֵי שֶׁהָרַ"י מִיגָאשׁ נִפְטַר מִן הָעוֹלָם. מִכֵּיוָן שֶׁהָרַמְבַּ"ם הָיָה יֶלֶד קָטָן מְאֹד כְּשֶׁהָרַ"י מִיגָאשׁ נִפְטַר, הוּא לֹא זָכָה לִלְמֹד מִמֶּנּוּ בְּפָנִים אֶל פָּנִים. אַךְ הָרַמְבַּ"ם לָמַד אֶת כָּל תּוֹרָתוֹ הַנִּפְלָאָה שֶׁל הָרַ"י מִיגָאשׁ מִתּוֹךְ מַה שֶׁקִּבֵּל אָבִיו רַבִּי מַיְמוֹן מֵרַבּוֹ, וְכֵן מִתּוֹךְ הַסְּפָרִים וְהַפֵּרוּשִׁים הָרַבִּים שֶׁכָּתַב הָרַ"י מִיגָאשׁ בְּעַצְמוֹ.

כַּאֲשֶׁר גָּדַל הָרַמְבַּ"ם וְכָתַב אֶת סִפְרוֹ הַמְפֻרְסָם, הַפֵּרוּשׁ לְמִשְׁנָיוֹת, הוּא הִזְכִּיר בְּכָבוֹד עָצוּם אֶת הָרַ"י מִיגָאשׁ בַּהַקְדָּמָה לְסִפְרוֹ. הָרַמְבַּ"ם סִפֵּר כִּי הוּא אָסַף אֶת כָּל הַפֵּרוּשִׁים שֶׁל אָבִיו וְשֶׁל חַכְמֵי יִשְׂרָאֵל, וּבַמְיֻחָד אֶת דִּבְרֵי הָרַ"י מִיגָאשׁ. הָרַמְבַּ"ם כָּתַב מְלֹא הַהִתְפַּעֲלוּת כִּי הַלֵּב שֶׁל הָרַ"י מִיגָאשׁ בַּתַּלְמוּד מַבְהִיל וּמַפְלִיא אֶת כָּל מִי שֶׁמִּסְתַּכֵּל בִּדְבָרָיו. הוּא הוֹסִיף וְכָתַב כִּי עֹמֶק שִׂכְלוֹ וְהָעִיּוּן שֶׁל הָרַ"י מִיגָאשׁ הָיוּ גְּדוֹלִים כָּל כָּךְ, עַד שֶׁכִּמְעַט אֶפְשָׁר לוֹמַר עָלָיו שֶׁכָּמוֹהוּ לֹא הָיָה מֶלֶךְ לְפָנָיו בְּחָכְמַת הַתּוֹרָה.

הָרַמְבַּ"ם אָסַף אֶת הַהֲלָכוֹת וְאֶת הַפֵּרוּשִׁים שֶׁכָּתַב הָרַ"י מִיגָאשׁ, וְהִשְׁתַּמֵּשׁ בָּהֶם כְּדֵי לְחַבֵּר אֶת הַפֵּרוּשׁ שֶׁלּוֹ לְסִדְרֵי הַמִּשְׁנָה. בְּזָכוּת הַשְׁקִידָה שֶׁל הָרַ"י מִיגָאשׁ, כְּתָבָיו וְתַלְמִידָיו, עָבְרָה הַתּוֹרָה הַקְּדוֹשָׁה מִדּוֹר לְדוֹר.

💡 כְּשֶׁלּוֹמְדִים תּוֹרָה בִּשְׁקִידָה וּבְעֹמֶק, זוֹכִים לְהָפִיץ אוֹר גָּדוֹל שֶׁמַּשְׁפִּיעַ עַל הַדּוֹרוֹת הַבָּאִים.
