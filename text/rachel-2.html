הָרַב גּוֹרֵן וְשַׁחְרוּר קֶבֶר רָחֵל

זֶהוּ סִפּוּר מְרַגֵּשׁ עַל עֵדוּתוֹ שֶׁל הָרַב שְׁלֹמֹה גּוֹרֵן זֵכֶר צַדִּיק לִבְרָכָה, שֶׁהָיָה הָרַב הַצְּבָאִי הָרָאשִׁי שֶׁל צְבָא הֲגַנָּה לְיִשְׂרָאֵל, עַל שַׁחְרוּר קֶבֶר רָחֵל אִמֵּנוּ בְּמִלְחֶמֶת שֵׁשֶׁת הַיָּמִים.

בִּזְמַן הַמִּלְחָמָה, לְאַחַר שֶׁחַיָּלֵי צַהַ"ל שִׁחְרְרוּ אֶת יְרוּשָׁלַיִם הָעַתִּיקָה, הִמְשִׁיכוּ כֹּחוֹת הַצָּבָא בִּנְסִיעָה כְּלַפֵּי דָּרוֹם, לְכִוּוּן הָעִיר בֵּית לֶחֶם, כְּדֵי לְשַׁחְרֵר אֶת קֶבֶר רָחֵל וְאֶת מְעָרַת הַמַּכְפֵּלָה.

הָרַב שְׁלֹמֹה גּוֹרֵן יָצָא אֶל הַדֶּרֶךְ בְּאוֹתוֹ לַיְלָה מַמָּשׁ יַחַד עִם הַנַּהָג שֶׁלּוֹ, כְּדֵי לְהַגִּיעַ אֶל מְקוֹם קֶבֶר רָחֵל. הָרַב גּוֹרֵן הִכִּיר אֶת מְקוֹמוֹ הַמְּדֻיָּק שֶׁל הַקֶּבֶר עוֹד מֵהַתְּקוּפָה שֶׁלִּפְנֵי מִלְחֶמֶת הָעַצְמָאוּת.

בִּזְמַן הַנְּסִיעָה נִזְכַּר הָרַב גּוֹרֵן בְּמַה שֶּׁכָּתוּב בְּסֵפֶר הַזֹּהַר הַקָּדוֹשׁ. בַּזֹּהַר כָּתוּב כִּי בַּזְּמַן שֶׁבְּנֵי יִשְׂרָאֵל יַחְזְרוּ מֵהַגָּלוּת, הֵם יָנִיחוּ אֶת יְדֵיהֶם עַל צִיּוֹן רָחֵל אִמֵּנוּ וְיֹאמְרוּ לָהּ: 'הַבָּנִים שֶׁלָּךְ שָׁבוּ מֵאֶרֶץ אוֹיֵב, הַבָּנִים שֶׁלָּךְ שָׁבוּ לִגְבוּלָם!'.

כַּאֲשֶׁר מָצָא הָרַב גּוֹרֵן אֶת מְקוֹם קֶבֶר רָחֵל, הָיְתָה הַשָּׁעָה אַחַת עֶשְׂרֵה וָרֶבַע בַּלַּיְלָה (23:15). כְּשֶׁהִגִּיעוּ הָרַב וְהַנַּהָג, הֵם רָאוּ כִּי הַשַּׁעַר הַגָּדוֹל שֶׁל הַמִּבְנֶה הָיָה נָעוּל בְּשַׁרְשְׁרוֹת חֲזָקוֹת וּבְמַנְעוּלִים. הֵם הִתְמַאֲמְצוּ וְעָבְדוּ חֲצִי שָׁעָה שְׁלֵמָה עַד שֶׁהִצְלִיחוּ לִפְתֹּחַ אֶת הַשַּׁעַר הַגָּדוֹל.

לְאַחַר שֶׁנִּכְנְסוּ פְּנִימָה, רָאוּ הָרַב וְהַנַּהָג כִּי מְקוֹם הַקֶּבֶר עַצְמוֹ נִמְצָא בְּתוֹךְ חֶדֶר פְּנִימִי, שֶׁגַּם הוּא הָיָה נָעוּל. הֵם נִסּוּ לִפְתֹּחַ אֶת הַחֶדֶר הַפְּנִימִי, אַךְ לֹא הִצְלִיחוּ לַעֲשׂוֹת זֹאת.

הָרַב גּוֹרֵן יָצָא חֲזָרָה הַחוּצָה, כְּשֶׁהוּא חוֹשֵׁב מָה עָלָיו לַעֲשׂוֹת כְּדֵי לְהִכָּנֵס. וְהִנֵּה, לְפֶתַע הִשְׁלִיךְ מִישֶׁהוּ אֶת הַמַּפְתֵּחַ שֶׁל הַחֶדֶר הַפְּנִימִי.

הָרַב גּוֹרֵן וְהַנַּהָג לָקְחוּ אֶת הַמַּפְתֵּחַ וְנִכְנְסוּ אֶל הַחֶדֶר הַפְּנִימִי. כְּשֶׁנִּכְנְסוּ פְּנִימָה אֶל מְקוֹם הַקֶּבֶר, הָיְתָה הַשָּׁעָה חֲצוֹת הַלַּיְלָה בְּדִיּוֹק.

כְּשֶׁנִּכְנַס הָרַב גּוֹרֵן, הוּא אָמַר בְּהִתְרַגְּשׁוּת אֶת הַפְּסוּקִים מִסֵּפֶר יִרְמְיָהוּ עַל שִׁיבַת הַבָּנִים לִגְבוּלָם: 'כֹּה אָמַר ה' קוֹל בְּרָמָה נִשְׁמָע נְהִי בְּכִי תַמְרוּרִים רָחֵל מְבַכָּה עַל בָּנֶיהָ מֵאֲנָה לְהִנָּחֵם עַל בָּנֶיהָ כִּי אֵינֶנּוּ. כֹּה אָמַר ה' מִנְעִי קוֹלֵךְ מִבֶּכִי וְעֵינַיִךְ מִדִּמְעָה כִּי יֵשׁ שָׂכָר לִפְעֻלָּתֵךְ נְאֻם ה' וְשָׁבוּ מֵאֶרֶץ אוֹיֵב. וְיֵשׁ תִּקְוָה לְאַחֲרִיתֵךְ נְאֻם ה' וְשָׁבוּ בָנִים לִגְבוּלָם'. כְּמוֹ כֵן קָרָא הָרַב גּוֹרֵן בַּמָּקוֹם הַקָּדוֹשׁ גַּם פִּרְקֵי תְּהִלִּים.

💡 נְבוּאוֹת הַתּוֹרָה וְהַבְטָחוֹת ה' תָּמִיד מִתְקַיְּמוֹת, וּבְנֵי יִשְׂרָאֵל חוֹזְרִים אֶל אֶרֶץ יִשְׂרָאֵל וְאֶל הַמְּקוֹמוֹת הַקְּדוֹשִׁים שֶׁלָּהֶם.
