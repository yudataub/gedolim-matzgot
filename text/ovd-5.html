הַבְּרָכָה שֶׁרִגְּשָׁה אֶת שׁוֹטְרֵי פָּרִיז

הַדָּבָר אֵרַע בְּעֵת בִּקּוּרוֹ שֶׁל הָרַב עוֹבַדְיָה יוֹסֵף בָּעִיר פָּרִיז שֶׁבְּצָרְפַת. בְּנֵי הַקְּהִלָּה הַיְּהוּדִית, שֶׁחִכּוּ לְבוֹאוֹ בְּהִתְרַגְּשׁוּת רַבָּה, קִבְּלוּ אֶת פָּנָיו בִּכְבוֹד מְלָכִים. נִכְבְּדֵי הַקְּהִלָּה הִמְתִּינוּ לוֹ בִּשְׂדֵה הַתְּעוּפָה נִרְגָּשִׁים וְנִפְעָמִים.

אַךְ לֹא רַק בְּנֵי הַקְּהִלָּה הִמְתִּינוּ לָרַב. מִשְׁטֶרֶת פָּרִיז הֶעֱמִידָה לִכְבוֹדוֹ מִשְׁמַר כָּבוֹד מְיֻחָד, וְקִבְּלָה אֶת פָּנָיו בְּכָבוֹד שֶׁשָּׁמוּר לְרָאשֵׁי מֶמְשָׁלוֹת. הַשּׁוֹטְרִים אִבְטְחוּ אֶת הַבִּקּוּר בִּקְפִידָה, וְלִוּוּ אֶת רִכְבּוֹ שֶׁל הָרַב בְּנַיָּדוֹת מִשְׁטָרָה וּבְאוֹפַנּוֹעִים עַד לְבֵיתוֹ שֶׁל רַב הַקְּהִלָּה.

כַּאֲשֶׁר הִגִּיעַ הָרֶכֶב לְבֵית הָרַב, יָצָא הָרַב עוֹבַדְיָה מִן הָרֶכֶב וּבִקֵּשׁ מֵרַבָּהּ שֶׁל הַקְּהִלָּה: "אָנָּא, קְרָא לְכָאן אֶת הַשּׁוֹטְרִים וְאַנְשֵׁי הַבִּטָּחוֹן שֶׁלִּוּוּנוּ". רַב הַקְּהִלָּה הִתְעַנְיֵן וְשָׁאַל: "לְשֵׁם מָה?". הָרַב עוֹבַדְיָה הִסְבִּיר: "אֲנִי רוֹצֶה לְבָרֵךְ אוֹתָם וּלְהוֹדוֹת לָהֶם עַל שֶׁטָּרְחוּ לְמַעֲנִי!".

רַב הַקְּהִלָּה נִסָּה לְהַסְבִּיר: "אֲבָל רַבֵּנוּ, הֲלֹא הֵם אֵינָם יְהוּדִים, וְהֵם אֵינָם יוֹדְעִים מָה זֶה 'מִי שֶׁבֵּרַךְ'". אַךְ הָרַב עוֹבַדְיָה הוֹרָה: "בְּכָל זֹאת, קְרָא לָהֶם!". אַנְשֵׁי הַבִּטָּחוֹן נִקְרְאוּ לְהִתְאַסֵּף לְיַד רִכְבּוֹ שֶׁל הָרַב.

הָרַב עוֹבַדְיָה הִבִּיט בָּהֶם בְּעֵינָיו הַטּוֹבוֹת, וּבֵרֵךְ אוֹתָם בְּחֹם מִקִּיר הַלֵּב בְּמֶשֶׁךְ דַּקּוֹת אֲרֻכּוֹת. הַשּׁוֹטְרִים הַקְּשׁוּחִים, שֶׁלָּבְשׁוּ קְסָדוֹת וְאֲפוֹדִים, עָמְדוּ וְהִקְשִׁיבוּ בִּתְשׂוּמַת לֵב. הֵם לֹא כָּל כָּךְ הֵבִינוּ אֶת שְׂפַת הַקֹּדֶשׁ שֶׁבְּפִיו שֶׁל הָרַב, אֲבָל לִבּוֹתֵיהֶם הִרְגִּישׁוּ אֶת הַחֹם שֶׁקָּרַן מִמֶּנּוּ, וּדְמָעוֹת הוֹפִיעוּ בְּזָוִיּוֹת עֵינֵיהֶם. הַמַּעֲמָד הַמְּרַגֵּשׁ הִסְתַּיֵּם.

תְּקוּפָה מְסֻיֶּמֶת לְאַחַר שֶׁהָרַב עוֹבַדְיָה שָׁב לְאֶרֶץ יִשְׂרָאֵל, הִגִּיעַ רַב הַקְּהִלָּה לְבִקּוּר בְּאַרְמוֹנוֹ שֶׁל נְשִׂיא צָרְפַת, זַ'אק שִׁירָק. רַב הַקְּהִלָּה הֻפְתַּע מְאֹד כַּאֲשֶׁר הַנָּשִׂיא פָּנָה אֵלָיו וְשָׁאַל: "מָה שְׁלוֹמוֹ שֶׁל הָרַב עוֹבַדְיָה יוֹסֵף מִיִּשְׂרָאֵל?".

רַב הַקְּהִלָּה עָנָה: "שְׁלוֹמוֹ טוֹב, בָּרוּךְ הַשֵּׁם, אֲבָל מַדּוּעַ כְּבוֹד הַנָּשִׂיא מִתְעַנְיֵן בָּזֶה?". הַנָּשִׂיא הַצָּרְפָתִי סִפֵּר בְּהִתְרַגְּשׁוּת: "דַּע לְךָ כִּי קִבַּלְתִּי מִכְתָּב מְרַגֵּשׁ מְאֹד מֵאַנְשֵׁי הַמִּשְׁטָרָה הַצָּרְפָתִית. הֵם כָּתְבוּ לִי כִּי מַנְהִיג הַיְּהוּדִים הוּא אָדָם מְיֻחָד בְּמִינוֹ, בַּעַל לֵב רַחוּם וְרָגִישׁ מֵאֵין כָּמוֹהוּ, הַחוֹלֵק כָּבוֹד עָצוּם לְכָל נִבְרָא וּמַכִּיר טוֹבָה לְכָל אֶחָד!".

רַק אָז הֵבִין רַב הַקְּהִלָּה כִּי עֵינָיו הַקְּדוֹשׁוֹת שֶׁל הָרַב צָפוּ אֶת קִדּוּשׁ הַשֵּׁם הָעָצוּם שֶׁיִּגָּרֵם כְּתוֹצָאָה מֵאוֹתוֹ 'מִי שֶׁבֵּרַךְ'.

💡 תָּמִיד כְּדַאי לְהַכִּיר טוֹבָה וּלְכַבֵּד כָּל אָדָם שֶׁעוֹזֵר לָנוּ, כִּי דֶּרֶךְ אֶרֶץ וְכָבוֹד מְבִיאִים לְקִדּוּשׁ הַשֵּׁם גָּדוֹל.
