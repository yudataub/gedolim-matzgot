הָרַב קוּק וְהַמִּפְגָּשׁ בְּפוֹרִיָּה

בִּשְׁנַת תרע"ד, שֶׁהִיא שְׁנַת 1913 לַמִּנְיָן, יָצָא הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק, הַמּוֹרָה וְהַמַּנְהִיג הָרוּחָנִי הַדָּגוּל שֶׁנִּקְרָא גַּם הָרַאיָ"ה קוּק, בְּרֹאשׁ מִשְׁלַחַת רַבָּנִים חֲשׁוּבִים מְאֹד. הֵם יָצְאוּ לְמַסָּע מְיֻחָד בְּמִינוֹ, שֶׁעָבַר בֵּין יִשּׁוּבֵי הַחֲלוּצִים הַשּׁוֹנִים בְּאֶרֶץ יִשְׂרָאֵל. מַסָּע מְפֻרְסָם זֶה נִקְרָא בְּשֵׁם "מַסַּע הַמּוֹשָׁבוֹת". יַחַד עִם הָרַב קוּק הָיוּ מִשְׁתַּתְּפִים בַּמַּסָּע רַבָּנִים נוֹסָפִים וּגְדוֹלִים, וּבֵינֵיהֶם גַּם הָרַב הַגָּאוֹן יוֹסֵף חַיִּים זֹנֶנְפֶלְד.

מַטְרָתוֹ הָעִקָּרִית וְהַחֲשׁוּבָה שֶׁל מַסַּע הַמּוֹשָׁבוֹת הָיְתָה לְקָרֵב אֶת הַלְּבָבוֹת בֵּין מַנְהִיגֵי הַדּוֹר וְהָרַבָּנִים לְבֵין הַחֲלוּצִים הַחִילוֹנִיִּים, אֲשֶׁר עָבְדוּ בִּמְסִירוּת בַּשָּׂדוֹת וּבַכְּרָמִים. הָרַבָּנִים בִּקְשׁוּ לְחַזֵּק אֶת הַזִּיקָה, אֶת הָאַהֲבָה וְאֶת הַקֶּשֶׁר הֶעָמֹק לְמִצְוֹת הַתּוֹרָה בְּקֶרֶב כָּל אַנְשֵׁי הַיִּשּׁוּבִים.

בַּמַּהֲלַךְ הַמַּסָּע הַמְּרַגֵּשׁ, הִגִּיעָה מִשְׁלַחַת הָרַבָּנִים הַחֲשׁוּבָה גַּם אֶל הַיִּשּׁוּב הַחֲלוּצִי פוֹרִיָּה. כַּאֲשֶׁר הִגִּיעוּ הָרַבָּנִים לַמָּקוֹם, יָצָא לִקְרָאתָם אֶחָד מֵחַבְרֵי הַיִּשּׁוּב, אָדָם שֶׁשְּׁמוֹ מוֹשֶׁה, אֲשֶׁר נִקְרָא גַּם מֹשֶׁה. מוֹשֶׁה מָחָה בְּתֹקֶף וּבְקוֹל רָם נֶגֶד בּוֹאָם שֶׁל הָרַבָּנִים אֶל הַיִּשּׁוּב. הוּא יָצָא לִקְרָאתָם וְקָרָא לַעֲבָרָם בְּתַרְעוֹמֶת וּבְכַעַס: "הַי, מָה אַתֶּם עוֹשִׂים פֹּה? לְכוּ מִכָּאן! אֲנַחְנוּ לֹא רוֹצִים שֶׁתַּטִּיפוּ עָלֵינוּ אֶת כָּל הַמִּצְוֹת וְהַתּוֹרוֹת שֶׁלָּכֶם." מוֹשֶׁה וְחַבְרָיו חָשְׁשׁוּ מְאֹד שֶׁהָרַבָּנִים בָּאוּ כְּדֵי לְהַטִּיף לָהֶם מוּסָר וּלְבַקֵּר אֶת אֹרַח חַיֵּיהֶם.

אַךְ הָרַב קוּק לֹא נִפְגַּע, לֹא כָּעַס וְלֹא הִתְרַגֵּז עַל הַדְּבָרִים הַקָּשִׁים שֶׁשָּׁמַע. הוּא נִגַּשׁ אֶל מוֹשֶׁה בְּסַבְלָנוּת רַבָּה, בְּאַהֲבָה וּבִמְאוֹר פָּנִים גָּדוֹל. הָרַב קוּק פָּנָה אֵלָיו בְּנַחַת וְאָמַר לוֹ בְּחִיּוּךְ: "אֲנַחְנוּ בָּאנוּ לִלְמֹד מִכֶּם. אַתֶּם אֵלֶּה שֶׁעוֹבְדִים אֶת אֶרֶץ יִשְׂרָאֵל יוֹמָם וָלַיְלָה... אֲנַחְנוּ רוֹצִים לִלְמֹד מִכֶּם, וְנִשְׂמַח לַעֲזֹר לָכֶם בְּכָל דָּבָר שֶׁתִּצְטָרְכוּ." הַדְּבָרִים הַחַמִּים וְהָאוֹהֲבִים שֶׁל הָרַב קוּק נִכְנְסוּ מִיָּד אֶל הַלֵּב וְהוֹתִירוּ רוֹשֵׁם עָמֹק וּמְרַגֵּשׁ עַל הַחֲלוּצִים.

לֹא פַּעַם, בִּמְקוֹמוֹת רַבִּים וּשְׁוֹנִים שֶׁבָּהֶם בִּקְרָה מִשְׁלַחַת הָרַבָּנִים בַּמַּהֲלַךְ מַסַּע הַמּוֹשָׁבוֹת, הִסְתַּיְמוּ מִפְגָּשִׁים שֶׁכָּאֵלֶּה בְּחַוָּיָה מְרַגֶּשֶׁת וּמַפְתִיעָה בְּיוֹתֵר. מִפְגָּשִׁים שֶׁהִתְחִילוּ לִפְעָמִים בַּחֲשַׁדְנוּת אוֹ בְּחִלּוּקֵי דֵּעוֹת הָפְכוּ לְרִקּוּדִים נִלְהָבִים וּמְשֻׁתָּפִים שֶׁל אִכָּרִים, פּוֹעֲלִים וְרַבָּנִים, הָעוֹמְדִים וְרוֹקְדִים שְׁכֶם אֶל שְׁכֶם. כָּךְ נוֹלְדָה אַחֲוָה גְּדוֹלָה, שִׂמְחָה רַבָּה וְאַהֲבַת יִשְׂרָאֵל אֲמִתִּית בֵּין כָּל הַמִּשְׁתַּתְּפִים.

💡 כַּאֲשֶׁר אָנוּ פוֹנִים אֶל הַזּוּלַת בְּאַהֲבָה, בְּכָבוֹד וּבִרְצוֹן לִלְמֹד, אָנוּ יְכוֹלִים לְקָרֵב לְבָבוֹת וּלְאַחֵד בֵּין אֲנָשִׁים.
