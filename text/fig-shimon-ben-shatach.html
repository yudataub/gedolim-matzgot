הַמַּרְגָּלִית שֶׁהֻחְזְרָה

פַּעַם אַחַת הָיוּ לְשִׁמְעוֹן בֶּן שָׁטַח תַּלְמִידִים מְסוּרִים. שִׁמְעוֹן בֶּן שָׁטַח נָתַן לַתַּלְמִידִים שֶׁלּוֹ כֶּסֶף, וּבִקֵּשׁ מֵהֶם שֶׁיִּהְיוּ הַשְּׁלִיחִים שֶׁלּוֹ וְיֵלְכוּ לִקְנוֹת עֲבוּרוֹ חֲמוֹר. הַתַּלְמִידִים הָלְכוּ כְּשְׁלִיחָיו שֶׁל שִׁמְעוֹן בֶּן שָׁטַח, הִגִּיעוּ אֶל הַגּוֹי שֶׁמָּכַר אֶת הַחֲמוֹר, וְשִׁלְמוּ לוֹ עַל הַחֲמוֹר בַּכֶּסֶף שֶׁל שִׁמְעוֹן בֶּן שָׁטַח. לְאַחַר שֶׁשִּׁלְמוּ עַל הַחֲמוֹר, לָקְחוּ הַתַּלְמִידִים אֶת הַחֲמוֹר וְהֵבִיאוּ אוֹתוֹ אֶל שִׁמְעוֹן בֶּן שָׁטַח.

כַּאֲשֶׁר הֵבִיאוּ הַתַּלְמִידִים אֶת הַחֲמוֹר אֶל שִׁמְעוֹן בֶּן שָׁטַח, הִסְתַּכֵּל שִׁמְעוֹן בֶּן שָׁטַח עַל הַחֲמוֹר. פִּתְאוֹם, רָאָה שִׁמְעוֹן בֶּן שָׁטַח שֶׁיֵּשׁ עַל הַחֲמוֹר שַׁרְשֶׁרֶת עִם אֶבֶן יְקָרָה. הַתַּלְמִידִים וְשִׁמְעוֹן בֶּן שָׁטַח רָאוּ אֶת הַשַּׁרְשֶׁרֶת הַזֹּאת שֶׁהָיְתָה תְּלוּיָה עַל הַחֲמוֹר.

כְּשֶׁרָאָה שִׁמְעוֹן בֶּן שָׁטַח אֶת הָאֶבֶן הַיְּקָרָה, הוּא שָׁאַל אֶת הַתַּלְמִידִים שֶׁלּוֹ שְׁאֵלָה חֲשׁוּבָה מְאֹד. אָמַר לָהֶם שִׁמְעוֹן בֶּן שָׁטַח: 'הַאִם הַגּוֹי שֶׁמָּכַר אֶת הַחֲמוֹר רָאָה שֶׁיֵּשׁ אֶת הָאֶבֶן הַיְּקָרָה? אוּלַי הוּא שָׁכַח אֶת זֶה?!'. שִׁמְעוֹן בֶּן שָׁטַח רָצָה לָדַעַת אִם הַמּוֹכֵר יָדַע שֶׁהָאֶבֶן הַיְּקָרָה נִמְצֵאת עַל הַחֲמוֹר אוֹ שֶׁהוּא שָׁכַח אוֹתָהּ שָׁם בְּטָעוּת. הַתַּלְמִידִים עָנוּ שֶׁהֵם אֵינָם בְּטוּחִים אִם הַגּוֹי יָדַע עַל הָאֶבֶן הַיְּקָרָה אוֹ שֶׁשָּׁכַח אוֹתָהּ.

כַּאֲשֶׁר שָׁמַע שִׁמְעוֹן בֶּן שָׁטַח שֶׁהַתַּלְמִידִים אֵינָם בְּטוּחִים, אָמַר לָהֶם מִיָּד: 'אִם אַתֶּם לֹא בְּטוּחִים, אֲנַחְנוּ עַכְשָׁיו הוֹלְכִים לְהַחְזִיר אֶת הָאֶבֶן הַיְּקָרָה'. שִׁמְעוֹן בֶּן שָׁטַח וְהַתַּלְמִידִים שֶׁלּוֹ לֹא חִכּוּ, וְהֵם הָלְכוּ מִיָּד יַחְדָּו כְּדֵי לְהַחְזִיר אֶת הָאֶבֶן הַיְּקָרָה אֶל הַגּוֹי שֶׁמָּכַר אֶת הַחֲמוֹר.

כַּאֲשֶׁר הִגִּיעוּ אֶל הַגּוֹי שֶׁמָּכַר אֶת הַחֲמוֹר, פָּנוּ אֵלָיו וְהֶחְזִירוּ לוֹ אֶת הָאֶבֶן הַיְּקָרָה. הֵם אָמְרוּ לוֹ בְּיוֹשֶׁר: 'אֲנַחְנוּ קָנִינוּ חֲמוֹר, אֲבָל לֹא קָנִינוּ אֶבֶן יְקָרָה שֶׁעָלָיו'.

הַגּוֹי שֶׁמָּכַר אֶת הַחֲמוֹר הוּפְתַּע מְאֹד. פִּתְאוֹם מַחְזִירִים לוֹ אֶת הָאֶבֶן הַיְּקָרָה שֶׁלּוֹ, וְאוֹמְרִים לוֹ שֶׁהֵם קָנוּ מִמֶּנּוּ רַק אֶת הַחֲמוֹר וְלֹא אֶת הָאֶבֶן הַיְּקָרָה. הַגּוֹי כָּל כָּךְ הִתְפַּעֵל מֵהַיֹּשֶׁר הַגָּדוֹל וּמֵהַמַּעֲשֶׂה הַיָּפֶה שֶׁל שִׁמְעוֹן בֶּן שָׁטַח וְתַלְמִידָיו. הוּא קָרָא בְּהִתְרַגְּלוּת וְאָמַר: 'אֵיזוֹ תּוֹרָה יֵשׁ לַיְּהוּדִים, אֵיזֶה חֲכָמִים גְּדוֹלִים יֵשׁ לַיְּהוּדִים!'. הַגּוֹי שָׂמַח מְאֹד מְאֹד מְאֹד שֶׁהֶחְזִירוּ לוֹ אֶת הָאֶבֶן הַיְּקָרָה, כִּי הוּא כְּבָר חָשַׁב שֶׁזֶּהוּ, הָאֶבֶן הַיְּקָרָה שֶׁלּוֹ הָלְכָה לְאִבּוּד וְהוּא לֹא יִרְאֶה אוֹתָהּ עוֹד.

💡 כַּאֲשֶׁר אֲנַחְנוּ נוֹהֲגִים בְּיוֹשֶׁר וּמַחְזִירִים מַשֶּׁהוּ שֶׁלֹּא שַׁיָּךְ לָנוּ, אֲנַחְנוּ מְקַדְּשִׁים שֵׁם שָׁמַיִם וּמַרְאִים לְכֻלָּם אֶת יְפִי הַתּוֹרָה.
