הַתְּפִלָּה שֶׁפָּתְחָה אֶת הַשֶּׁפַע

פַּעַם אַחַת בָּאוּ הַבַּעַל שֵׁם טוֹב הַקָּדוֹשׁ וְתַלְמִידָיו לְהִתְאָרֵחַ אֵצֶל יְהוּדִי מַמָּשׁ מַמָּשׁ עָנִי. לָאִישׁ הַזֶּה כִּמְעַט לֹא הָיָה כְּלוּם בַּבַּיִת. כְּשֶׁרָאָה הָעָנִי אֶת הַבַּעַל שֵׁם טוֹב, הוּא רָץ לִקְרָאתוֹ בְּשִׂמְחָה גְּדוֹלָה, חָשַׁב אֵיזוֹ זְכוּת אַדִּירָה זוֹ עֲבוּרוֹ, דָּחַף אֶת מִשְׁפַּחְתּוֹ לַצַּד וְנָתַן לָאוֹרְחִים אֶת הַמִּטָּה שֶׁלָּהֶם וְאֶת כָּל מַה שֶּׁהָיָה לָהֶם. הוּא רָץ מִיָּד לִמְכֹּר אֶת הַפְּמוֹטוֹת שֶׁל אִשְׁתּוֹ כְּדֵי לְהָבִיא כִּבּוּד לַצַּדִּיקִים הָאֵלֶּה.

הַתַּלְמִידִים לֹא הֵבִינוּ לָמָּה הַבַּעַל שֵׁם טוֹב בָּא לְהִתְאָרֵחַ דַּוְקָא אֵצֶל מִישֶׁהוּ כָּל כָּךְ מִסְכֵּן וְכָל כָּךְ עָנִי, אֲבָל הֵם לֹא שָׁאֲלוּ. כָּךְ עָבַר עוֹד יוֹם וְעוֹד יוֹם, וּבְכָל פַּעַם רָץ הָאִישׁ הַזֶּה וּמָכַר עוֹד אֵיזֶה חֵפֶץ מֵהַבַּיִת כְּדֵי לְהָבִיא עוֹד כִּבּוּד לָאוֹרְחִים הַצַּדִּיקִים הָאֵלֶּה שֶׁלּוֹ. כָּךְ נִמְשַׁךְ הַדָּבָר עַד שֶׁעָבַר שָׁבוּעַ וְנִגְמַר לוֹ הַשֶּׁקֶל הָאַחֲרוֹן, וּכְבָר לֹא נִשְׁאַר לוֹ מַה לִמְכֹּר.

הָאִישׁ הַזֶּה נִגְמַר לוֹ הַכֹּל. הוּא יָצָא מֵהַבַּיִת וְאָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם, תּוֹדָה עַל הַזְּכוּת הָאַדִּירָה שֶׁנָּתַתָּ לִי לְאָרֵחַ אֶת הַצַּדִּיקִים הָאֵלֶּה אֶצְלִי בַּבַּיִת, אֵיזֶה מַדְהִים! אֲבָל רִבּוֹנוֹ שֶׁל עוֹלָם, אֵין לִי כֶּסֶף, אָז בְּבַקָּשָׁה הַשֵּׁם, שְׁלַח לִי קְצָת פַּרְנָסָה, שְׁלַח לִי קְצָת כֶּסֶף בְּבַקָּשָׁה".

תּוֹךְ שָׁבוּעַ נִהְיָה הָאִישׁ הַזֶּה מִילְיוֹנֵר. כְּשֶׁהַחֲסִידִים שָׁמְעוּ אֶת זֶה הֵם אָמְרוּ: "וָואוֹ, רַבֵּנוּ, מַה הָיָה פֹּה? מַה קָרָה? מַה הַסִּפּוּר?".

הַבַּעַל שֵׁם טוֹב אָמַר: "אֲנִי רָאִיתִי שֶׁבַּשָּׁמַיִם נִגְזַר שֶׁהָאִישׁ הַזֶּה צָרִיךְ לִהְיוֹת מִילְיוֹנֵר. כָּךְ נִקְבַּע מֵרֹאשׁ הַשָּׁנָה, וְיֵשׁ מִילְיוֹנִים שֶׁמְּחַכִּים לוֹ בַּשָּׁמַיִם לְהַגִּיעַ עַד אֵלָיו. אֲבָל יֵשׁ בְּעָיָה, יֵשׁ חֲסִימָה. הַחֲסִימָה הִיא שֶׁהוּא אַף פַּעַם לֹא הִתְפַּלֵּל לְכֶסֶף. הוּא אַף פַּעַם לֹא חָשַׁב שֶׁיָּכוֹל לִהְיוֹת לוֹ יוֹתֵר טוֹב מִמַּה שֶׁיֵּשׁ לוֹ. אוּלַי הַשֵּׁם רוֹצֶה לָתֵת לוֹ יוֹתֵר? הוּא אַף פַּעַם לֹא חָשַׁב אֶת זֶה. הוּא הָיָה שָׂמֵחַ בְּחֶלְקוֹ, שֶׁזֶּה דָּבָר טוֹב, אֲבָל הוּא אַף פַּעַם לֹא הִתְפַּלֵּל. וְכָל עוֹד הוּא לֹא הִתְפַּלֵּל, מַה שֶּׁמְּחַכֶּה לוֹ וּמַה שֶּׁמַּגִּיעַ לוֹ – הוּא לֹא יָכֹל לְקַבֵּל אֶת זֶה".

אָמַר הַבַּעַל שֵׁם טוֹב: "הָיִיתִי חַיָּב לָבוֹא, לָקַחַת לוֹ אֶת הַשֶּׁקֶל הָאַחֲרוֹן, לְדַחֹף אוֹתוֹ לַפִּנָּה שֶׁלֹּא תִּהְיֶה לוֹ בְּרֵרָה וְהוּא יִהְיֶה חַיָּב לִהְיוֹת מִתְפַּלֵּל. וּבָרֶגַע שֶׁהוּא הִתְפַּלֵּל, הוּא פָּתַח אֶת הַחֲסִימָה, וְכָל מַה שֶּׁחִכָּה לוֹ בַּשָּׁמַיִם, כָּל הַשֶּׁפַע שֶׁהָיָה מַגִּיעַ לוֹ, יָכֹל הָיָה לְהַגִּיעַ עַד אֵלָיו".

💡 כְּשֶׁאָנוּ מִתְפַּלְלִים אֶל הַשֵּׁם מִכָּל הַלֵּב, אָנוּ פּוֹתְחִים אֶת הַשַּׁעַר לְכָל הַשֶּׁפַע וְהַטּוֹב שֶׁמְּחַכִּים לָנוּ בַּשָּׁמַיִם.
