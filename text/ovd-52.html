הַמַּלְאָכִים שֶׁשָּׁמְרוּ עַל הַתּוֹרָה

כְּשֶׁמָּלְאוּ לַנַּעַר עֹובַדְיָה יֹוסֵף אַרְבַּע שָׁנִים בִּלְבַד, עָלְתָה מִשְׁפַּחְתֹּו מֵעִירָאק לְאֶרֶץ יִשְׂרָאֵל וְהִשְׁתַּקְעָה בִּשְׁכוּנַת "בֵּית יִשְׂרָאֵל" בְּעִיר הַקֹּדֶשׁ יְרוּשָׁלַיִם. לַמְרֹות שֶׁבְּעִירָאק שִׁמֵּשׁ הָאָב כְּצֹורֵף זָהָב וָכֶסֶף מֻמְחֶה, בְּאֶרֶץ יִשְׂרָאֵל הוּא לֹא הִצְלִיחַ לִמְצֹא עֲבֹודָה. בְּמֶשֶׁךְ שָׁלֹושׁ שָׁנִים תְּמִימֹות הָיָה הָאָב מְבֻטָּל מִמְּלָאכָה, וּמַצָּבָהּ הַכַּלְכָּלִי שֶׁל הַמִּשְׁפָּחָה הָפַךְ לְקָשֶׁה וּלְמֻרְכָּב מְאֹד.

בְּסֹופֹו שֶׁל דָּבָר, הִצְלִיחַ הָאָב לִפְתֹּחַ חֲנוּת מַכֹּלֶת קְטַנָּה בִּירוּשָׁלַיִם. חֲנוּת מַכֹּלֶת זוֹ הָפְכָה לְאֶבֶן דֶּרֶךְ חֲשׁוּבָה וּמַשְׁמַעוֹתִית מְאֹד בְּחַיָּיו וּבְפָעֳלֹו שֶׁל עֹובַדְיָה יֹוסֵף. בִּגְלַל שֶׁעֹובַדְיָה הָיָה הַבֵּן הַבְּכֹור בַּמִּשְׁפָּחָה, הוּא נֶאֱלַץ לָצֵאת לַעֲבֹוד כְּבָר בְּגִיל צָעִיר מְאֹד כְּדֵי לַעֲזֹר לְפַרְנָסַת בְּנֵי מִשְׁפַּחְתֹּו. אֻלָּם נַפְשֹׁו שֶׁל הַנַּעַר עֹובַדְיָה חָשְׁקָה בְּכָל לִבָּהּ רַק בְּלִמּוּד הַתּוֹרָה הַקְּדֹושָׁה. הוּא בִּקֵּשׁ מֵאָבִיו לְנַהֵל אֶת חֲנוּת הַמַּכֹּלֶת בְּעַצְמֹו, מִתֹּוךְ תִּקְוָה שֶׁכָּךְ יַצְלִיחַ לִלְמֹד בְּסִפְרֵי הַתּוֹרָה בַּזְּמַן שֶׁבֹּו אֵין לָקֹוחֹות בַּחֲנוּת.

אַחִיו שֶׁל עֹובַדְיָה, נָעִים זִכְרֹונֹו לִבְרָכָה, סִפֵּר כִּי רֹאשׁ יְשִׁיבַת "פֹורַת יֹוסֵף", הָרַב עֶזְרָא עַטִיָּה, הִגִּיעַ בְּעַצְמֹו אֶל אֲבִיהֶם. רֹאשׁ הַיְּשִׁיבָה הִבְטִיחַ לַעֲזֹר לָאָב בְּעַצְמֹו בְּנִהוּל הַחֲנוּת, וְזֹאת בִּתְנַאי שֶׁיַּרְשֶׁה לִבְנֹו לָלֶכֶת וְלִלְמֹד תּוֹרָה בַּיְּשִׁיבָה. הָרַב עֶזְרָא עַטִיָּה הִדְגִּישׁ וְאָמַר כִּי מְדֻבָּר בְּתַלְמִיד חָכָם מְיֻחָד שֶׁיָּכֹול לִהְיֹות עִלּוּי גָּדֹול מְאֹד בַּתּוֹרָה. אֻלָּם, לַמְרֹות הַדְּבָרִים הַחֲשׁוּבִים הָאֵלֶּה מִפִּיו שֶׁל רֹאשׁ הַיְּשִׁיבָה, אֲבִי הַמִּשְׁפָּחָה לֹא הִשְׁתַּכְנַע מֵהַהַצָּעָה וְלֹא הִסְכִּים לְוַתֵּר עַל עֶזְרָתוֹ שֶׁל בְּנֹו הַבְּכֹור בַּחֲנוּת.

לְאַחַר זְמַן מָה, אֵרַע מִקְרֶה יוֹצֵא דֹּפֶן וּמַפְתִּיעַ בַּחֲנוּת הַמַּכֹּלֶת. גַּנָּבִים פָּרְצוּ אֶל הַחֲנוּת, וְהַנַּעַר עֹובַדְיָה יֹוסֵף לֹא הִצְלִיחַ לַעֲצֹר בַּעֲדָם. בַּפְּרִיצָה הַזֹּאת נִגְנְבָה סְחֹורָה יְקָרָה מְאֹד מִן הַחֲנוּת. לְאַחַר הַתַּקְרִית הַקָּשָׁה הַזֹּאת, הֶחְלִיט הָאָב כִּי עָדִיף שֶׁבְּנֹו לֹא יְנַהֵל עוֹד אֶת חֲנוּת הַמַּכֹּלֶת, וְאִפְשֵׁר לֹו לָלֶכֶת וְלִלְמֹד בַּיְּשִׁיבָה בְּלִי הַפְרָעָה. בְּנֵי הַמִּשְׁפָּחָה וּמְקֹורָבָיו שֶׁל הָרַב מְסַפְּרִים אֶת הַסִּפּוּר הַזֶּה כְּסִפּוּר נִסִּי מַמָּשׁ, וְהֵם אַף מְכַנִּים אֶת אֹותָם הַגַּנָּבִים בַּשֵּׁם "מַלְאָכִים", מִפְּנֵי שֶׁבִּזְכוּתָם זָכָה הַנַּעַר לַחְזֹר לִלְמֹד תּוֹרָה בַּיְּשִׁיבָה.
