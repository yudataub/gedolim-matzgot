הַכָּבוֹד לַתּוֹרָה וְהַמְּשִׂימָה הַסּוֹדִית

בְּפָרָשַׁת 'וַיִּגַּשׁ' אָנוּ לוֹמְדִים עַל הַגְּבוּרָה שֶׁל יְהוּדָה, שֶׁנֶּעֱמַד בְּאֹמֶץ לִפְנֵי הַמִּשְׁנֶה לְמֶלֶךְ מִצְרַיִם, יוֹסֵף, כְּדֵי לְהַצִּיל אֶת בִּנְיָמִין. כַּיּוֹצֵא בָּזֶה, מְסֻפָּר עַל גְּבוּרָתוֹ שֶׁל הָרַב מָרְדְּכַי אֵלִיָּהוּ. יוֹם אֶחָד נִגַּשׁ אֶל הָרַב 'אִישׁ מוֹסָד' וְאָמַר לוֹ שֶׁיֵּשׁ מְשִׂימָה חֲשׁוּבָה מְאֹד: יֵשׁ חֲטוּפִים שֶׁנִּמְצָאִים בְּסוּרְיָה, וְאֵין שׁוּם דֶּרֶךְ לְקַבֵּל עֲלֵיהֶם מֵידָע. אִישׁ הַמּוֹסָד בִּקֵּשׁ מֵהָרַב לְדַבֵּר עִם נְשִׂיא בְּרָזִיל, שֶׁהָיָה בִּקְשָׁרִים טוֹבִים עִם נְשִׂיא סוּרְיָה, וּלְבַקֵּשׁ מִמֶּנּוּ שֶׁיִּדַּבֵּר עִם נְשִׂיא סוּרְיָה כְּדֵי לְקַבֵּל מֵידָע עַל הַחֲטוּפִים.

הַמְּשִׂימָה הָיְתָה קָשָׁה מְאֹד. נְשִׂיא בְּרָזִיל הָיָה אָדָם נוֹקְשֶׁה וְעַצְבָּנִי מְאֹד, שֶׁכָּל דָּבָר קָטָן הִכְעִיס אוֹתוֹ, וְכָל אֲנָשָׁיו פָּחֲדוּ מִמֶּנּוּ. כַּאֲשֶׁר הִתְקַיְּמָה פְּגִישָׁה שֶׁל אֲנָשִׁים חֲשׁוּבִים מֵהַקְּהִלּוֹת הַיְּהוּדִיּוֹת בְּבְרָזִיל עִם הַנַּשִׂיא, הִצְטָרֵף אֲלֵיהֶם הָרַב מָרְדְּכַי אֵלִיָּהוּ, וְהִתְפַּלֵּל אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיַּעֲזֹר לוֹ בַּמְּשִׂימָה.

בַּחֶדֶר הַגָּדוֹל שֶׁל נְשִׂיא בְּרָזִיל, הִגִּישׁ לוֹ הָרַב בִּתְחִלַּת הַפְּגִישָׁה סֵפֶר תַּנָּ"ךְ מְתֻרְגָּם לְלָשׁוֹן לוֹעֲזִית כְּמַתָּנָה. הַנַּשִׂיא אָמַר בִּשְׂפָתוֹ "תּוֹדָה רַבָּה", וְהַמְּתוּרְגְּמָן תִּרְגֵּם אֶת דְּבָרָיו לָרַב. כַּאֲשֶׁר אֶחָד הָאֲנָשִׁים הַיְּהוּדִים קָם לְדַבֵּר, הוֹצִיא נְשִׂיא בְּרָזִיל מִכִּיסוֹ שְׁתֵּי גֻּלּוֹת מְיֻחָדוֹת, שִׂחֵק בָּהֵן עַל הַשֻּׁלְחָן וְהִנִּיחַ אוֹתָן עַל סֵפֶר הַתַּנָּ"ךְ.

הָרַב מָרְדְּכַי אֵלִיָּהוּ רָאָה זֹאת וְלֹא יָכֹל לִשְׁתֹּק. הוּא בִּקֵּשׁ מֵהַמְּתוּרְגְּמָן לְהַגִּיד לַנַּשִׂיא שֶׁיּוֹרִיד אֶת הַגֻּלּוֹת מֵהַתַּנָּ"ךְ, כִּי זֶהוּ סֵפֶר קָדוֹשׁ. הַמְּתוּרְגְּמָן נִהְיָה אָדוֹם מֵרֹב פַּחַד וְאָמַר שֶׁאָסוּר לְהַגִּיד לַנַּשִׂיא דָּבָר כָּזֶה כִּי הוּא יָכוֹל לְהִתְעַצְבֵּן. אַךְ הָרַב לֹא וִתֵּר, וְדָרַשׁ שׁוּב מֵהַמְּתוּרְגְּמָן לְהַגִּיד לַנַּשִׂיא מִיָּד שֶׁיּוֹרִיד אֶת הַגֻּלּוֹת מֵהַתַּנָּ"ךְ הַקָּדוֹשׁ.

הַנַּשִׂיא רָאָה שֶׁהָרַב מְדַבֵּר עִם הַמְּתוּרְגְּמָן, צָעַק עַל הַמְּתוּרְגְּמָן וְדָרַשׁ לָדַעַת מַה הָרַב אָמַר לוֹ לְהַגִּיד. הַמְּתוּרְגְּמָן הָרוֹעֵד סִפֵּר לַנַּשִׂיא שֶׁהָרַב בִּקֵּשׁ שֶׁיּוֹרִיד אֶת הַגֻּלּוֹת מֵהַתַּנָּ"ךְ כִּי זֶה סֵפֶר קָדוֹשׁ. הַנַּשִׂיא עָצַר מִיָּד אֶת הַפְּגִישָׁה וְקָרָא לָרַב לְהִכָּנֵס אִתּוֹ לְחַדְרוֹ הָאִישִׁי.

בַּחֶדֶר הַפְּרָטִי אָמַר הַנַּשִׂיא לָרַב שֶׁאַף אָדָם מֵעוֹלָם לֹא הֵעֵז לְהָעִיר לוֹ אוֹ לְבַקֵּר אוֹתוֹ בַּפָּנִים. כֵּיוָן שֶׁהָרַב הָיָה הָרִאשׁוֹן שֶׁהֵעֵז לַעֲשׂוֹת זֹאת, הַנַּשִׂיא אָמַר שֶׁהוּא רוֹצֶה לִשְׁמֹעַ אֶת כָּל מַה שֶׁיֵּשׁ לָרַב לוֹמַר. הָרַב נִצֵּל אֶת הַהִזְדַּמְנוּת, סִפֵּר לוֹ עַל הַחֲטוּפִים שֶׁנִּמְצָאִים בְּסוּרְיָה וּבִקֵּשׁ שֶׁיְּדַבֵּר עִם נְשִׂיא סוּרְיָה כְּדֵי לְהַשִּׂיג עֲלֵיהֶם מֵידָע. נְשִׂיא בְּרָזִיל הִבְטִיחַ וְהִתְחַיֵּב לַעֲשׂוֹת זֹאת.

כְּשֶׁנָּסַע הָרַב בַּחֲזָרָה עִם עוֹזְרוֹ, שָׁאַל אוֹתוֹ הָעוֹזֵר כֵּיצַד הִצְלִיחַ לַעֲשׂוֹת זֹאת. הָרַב הֱשִׁיב לוֹ עַל פִּי הַפָּסוּק מִסֵּפֶר מִשְׁלֵי: "סַלְסְלֶהָ וּתְרוֹמְמֶךָּ" – כְּשֶׁאָדָם מְכַבֵּד אֶת הַתּוֹרָה, הַקָּדוֹשׁ בָּרוּךְ הוּא גּוֹרֵם לְכָךְ שֶׁהַתּוֹרָה תִּתְרוֹמֵם עַל יָדוֹ.

💡 כְּשֶׁאָנוּ מְכַבְּדִים אֶת הַתּוֹרָה הַקְּדוֹשָׁה וְלוֹחֲמִים עַל כְּבוֹדָהּ, הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹזֵר לָנוּ וּמְרוֹמֵם אוֹתָנוּ.
