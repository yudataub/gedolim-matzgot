הַבְּרָכָה הַמְיֻחֶדֶת שֶׁל הָעֵץ

פַּעַם אַחַת יָשְׁבוּ יַחְדָּו שְׁנֵי חֲכָמִים גְּדוֹלִים, רַב נַחְמָן וְרַבִּי יִצְחָק, וְאָכְלוּ אֲרוּחָה מְשֻׁתָּפֶת. תּוֹךְ כְּדֵי הַסְּעוּדָה, פָּנָה רַב נַחְמָן אֶל רַבִּי יִצְחָק וּבִקֵּשׁ מִמֶּנּוּ: בְּבַקָּשָׁה מִמְּךָ, אֱמֹר לָנוּ דִּבְרֵי תוֹרָה מְתוּקִים שֶׁיְּשַׂמְחוּ אֶת לִבֵּנוּ. אָמַר לוֹ רַבִּי יִצְחָק: כָּךְ לִמֵּד אוֹתָנוּ רַבִּי יוֹחָנָן, שֶׁאֵין מְדַבְּרִים בִּזְמַן הָאֲכִילָה. עָלֵינוּ לְהִזָּהֵר מְאֹד כְּשֶׁאָנוּ אוֹכְלִים, שֶׁמָּא הָאֹכֶל יִכָּנֵס לְקָנֵה הַנְּשִׁימָה בִּמְקוֹם לַוֶּשֶׁט, וְהַדָּבָר יָבוֹא לִידֵי סַכָּנָה. לָכֵן, בִּזְמַן הָאֲכִילָה אָנוּ שׁוֹתְקִים וְאוֹכְלִים בְּנַחַת.

כַּאֲשֶׁר סִיְּמוּ אֶת הַסְּעוּדָה וְהִגִּיעָה הָעֵת לְהִפָּרֵד זֶה מִזֶּה, פָּנָה שׁוּב רַב נַחְמָן אֶל רַבִּי יִצְחָק וְאָמַר לוֹ: עַכְשָׁו, כְּשֶׁסִיַּמְנוּ לֶאֱכֹל, יְבָרֵךְ אוֹתָנוּ מַר בִּבְרָכָה טוֹבָה לַדֶּרֶךְ!

עָנָה לוֹ רַבִּי יִצְחָק: אֲסַפֵּר לְךָ מָשָׁל יָפֶה. לְמָה הַדָּבָר דּוֹמֶה? לְאָדָם שֶׁהָיָה הוֹלֵךְ בַּמִּדְבָּר הַחַם. הוּא הָיָה רָעֵב מְאֹד, עָיֵף מִן הַדֶּרֶךְ וְצָמֵא לְמַיִם. פִּתְאֹם, מָצָא בְּלֵב הַמִּדְבָּר אִילָן גָּבוֹהַּ וּמְפֹאָר. הָאִילָן הַזֶּה הָיָה מְיֻחָד מְאֹד: פֵּרוֹתָיו הָיוּ מְתוּקִים וּטְעִימִים, הַצֵּל שֶׁלּוֹ הָיָה נָאֶה וּנְעִים, וְאַמַּת מַיִם זַכָּה וּקְרִירָה עָבְרָה מַמָּשׁ תַּחְתָּיו.

שָׂמַח הָאִישׁ מְאֹד. הוּא אָכַל מִן הַפֵּרוֹת הַמְּתוּקִים שֶׁל הָעֵץ, שָׁתָה מִן הַמַּיִם הַזַּכִּים וְיָשַׁב לָנוּחַ בְּצִילּוֹ הַנָּעִים. כַּאֲשֶׁר הִגִּיעָה הָעֵת שֶׁל הָאִישׁ לְהַמְשִׁיךְ בְּדַרְכּוֹ, הִבִּיט עַל הָעֵץ וּבִקֵּשׁ לְבָרֵךְ אוֹתוֹ עַל כָּל הַטּוֹב שֶׁקִּבֵּל מִמֶּנּוּ.

אָמַר הָאִישׁ לָאִילָן: אִילָן, אִילָן, בַּמָּה אֲבָרֶכְךָ? אִם אוֹמַר לְךָ שֶׁיִּהְיוּ פֵּרוֹתֶיךָ מְתוּקִים – הֲרֵי פֵּרוֹתֶיךָ כְּבָר מְתוּקִים מְאֹד! אִם אוֹמַר לְךָ שֶׁיִּהְיֶה צִילְּךָ נָאֶה וּנְעִים – הֲרֵי צִילְּךָ כְּבָר נָאֶה מְאֹד! וְאִם אוֹמַר לְךָ שֶׁתִּהְיֶה אַמַּת מַיִם עוֹבֶרֶת תַּחְתֶּיךָ – הֲרֵי אַמַּת מַיִם כְּבָר עוֹבֶרֶת תַּחְתֶּיךָ! אֶלָּא, הַבְּרָכָה הֲכִי טוֹבָה שֶׁאֲנִי יָכוֹל לְבָרֵךְ אוֹתְךָ הִיא זוֹ: יְהִי רָצוֹן שֶׁכָּל הַנְּטִיעוֹת שֶׁיִּטְעוּ מִמְּךָ, כָּל הָעֵצִים הַחֲדָשִׁים שֶׁיִּגְדְּלוּ מֵהַזְּרָעִים שֶׁלְּךָ, יִהְיוּ טוֹבִים וּמְבֹרָכִים בְּדִיּוּק כָּמוֹךָ!

סִיֵּם רַבִּי יִצְחָק וְאָמַר לְרַב נַחְמָן: אַף אַתָּה, רַב נַחְמָן, בַּמָּה אֲבָרֶכְךָ? אִם אֲבָרֶכְךָ בַּתּוֹרָה – הֲרֵי יֵשׁ לְךָ תּוֹרָה רַבָּה! אִם אֲבָרֶכְךָ בְּעֹשֶׁר – הֲרֵי יֵשׁ לְךָ עֹשֶׁר! אִם אֲבָרֶכְךָ בְּבָנִים – הֲרֵי יֵשׁ לְךָ בָּנִים! אֶלָּא, יְהִי רָצוֹן מִלִּפְנֵי הַשֵּׁם שֶׁיִּהְיוּ צֶאֱצָאֶיךָ, כָּל הַיְּלָדִים וְהַדּוֹרוֹת שֶׁיֵּצְאוּ מִמְּךָ, צַדִּיקִים, חֲכָמִים וּמְבֹרָכִים בְּדִיּוּק כָּמוֹךָ!

💡 הַבְּרָכָה הַטּוֹבָה בְּיוֹתֵר לְאָדָם שֶׁיֵּשׁ לוֹ הַכֹּל הִיא שֶׁיְּלָדָיו וְצֶאֱצָאָיו יֵלְכוּ בִּדְרָכָיו הַטּוֹבוֹת.
