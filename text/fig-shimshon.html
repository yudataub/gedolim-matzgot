שִׁמְשׁוֹן הַשּׁוֹפֵט וְהַתְּפִלָּה עַל הַמַּיִם

זֹאת תְּפִלָּה מֻפְלָאָה הַמּוֹפִיעָה בַּתַּנָּ״ךְ, תְּפִלָּה שֶׁל אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל הַחֲשׁוּבִים, הֲלֹא הוּא שִׁמְשׁוֹן הַשּׁוֹפֵט. שִׁמְשׁוֹן נוֹלַד מֵעֵת שֶׁהָיָה יֶלֶד צָעִיר מְאֹד, מִגִּיל אֶפֶס, עִם כֹּחוֹת מְדַהֲמִים וִיכוֹלוֹת מֻפְלָאוֹת. אַךְ כְּמוֹ שֶׁהָיוּ לוֹ כֹּחוֹת פִיזִיִּים מְדַהֲמִים וְכֹחַ רַב בְּגוֹפוֹ, הָיוּ לוֹ גַּם כֹּחוֹת גְּדוֹלִים מְאֹד שֶׁל צִדְקוּת וְשֶׁל טָהֳרָה. הוּא הָיָה אִישׁ צַדִּיק צַדִּיק צַדִּיק, שֶׁבָּטַח בַּהַשֵּׁם בְּכָל לִבּוֹ וְהָלַךְ תָּמִיד בְּדַרְכֵי הַשֵּׁם. אֶחָד הַדְּבָרִים הֲכִי יָפִים וּמְרַגְּשִׁים בַּסִּפּוּר שֶׁל שִׁמְשׁוֹן הוּא שֶׁרֹואִים שֶׁפַּעֲמַיִם בְּחַיָּיו הִתְפַּלֵּל שִׁמְשׁוֹן מֵעֹמֶק הַלֵּב לְרִבּוֹנוֹ שֶׁל עוֹלָם שֶׁיַּעֲזֹר לוֹ, וּפַעֲמַיִם הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַע אֶל תְּפִלָּתוֹ וְעָנָה לוֹ.

פַּעַם אַחַת שִׁמְשׁוֹן יָצָא לְהִלָּחֵם מוּל הַפְּלִשְׁתִּים. בְּאוֹתָהּ תְּקוּפָה, עַם יִשְׂרָאֵל נוֹרָא נוֹרָא פָּחַד מֵהַפְּלִשְׁתִּים הָרְשָׁעִים שֶׁהִצִּיקוּ לַיְּהוּדִים. אַךְ שִׁמְשׁוֹן הַשּׁוֹפֵט לֹא נִפְחַד כְּלָל, וְיָצָא לְהִלָּחֵם מוּל הַפְּלִשְׁתִּים לְבַדּוֹ. לְשִׁמְשׁוֹן לֹא הָיוּ כְּלֵי נֶשֶׁק רְגִילִים, כְּמוֹ חֲרָבוֹת אוֹ חֲנִיתוֹת, אֶלָּא הוּא לָקַח עֶצֶם יְשָׁנָה שֶׁל חֲמוֹר, הַנִּקְרֵאת לְחִי שֶׁל חֲמוֹר. וְעִם הָעֶצֶם הַזֹּאת, עִם לְחִי הַחֲמוֹר הַזֹּאת בְּיָדוֹ, שִׁמְשׁוֹן נִלְחַם לְבַדּוֹ מוּל הֲמוֹן אֲנָשִׁים וְהָרַג כְּאַלֵּף אִישׁ פְּלִשְׁתִּים. הוּא הָרַג אֶת אַלֵּף הַפְּלִשְׁתִּים הָרְשָׁעִים הַלָּלוּ בְּכֹחוֹ הַגָּדוֹל שֶׁנָּתַן לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא, וְהוֹשִׁיעַ אֶת עַם יִשְׂרָאֵל.

אַחֲרֵי שֶׁהוּא הָרַג אֶת אַלֵּף הָאֲנָשִׁים, אֶת אַלֵּף הַפְּלִשְׁתִּים הָרְשָׁעִים הַלָּלוּ, שִׁמְשׁוֹן הָיָה עָיֵף מְאֹד מֵהַמַּאֲמָץ וְנוֹרָא נוֹרָא צָמֵא. זֶה הָיָה צִמָּאוֹן קָשֶׁה מְאֹד, צָמֵא בְּרָמָה כָּזֹאת שֶׁעוֹד מְעַט הוּא חָשַׁשׁ שֶׁהוּא עַלּוּל לָמוּת מִצָּמָא. כְּשֶׁהוּא הִרְגִּישׁ צָמֵא כָּל כָּךְ וְרָאָה שֶׁאֵין לוֹ שׁוּם מַיִם לִשְׁתּוֹת, שִׁמְשׁוֹן לֹא הִתְיָאֵשׁ, אֶלָּא פָּנָה מִיָּד אֶל הַשֵּׁם יִתְבָּרַךְ בִּתְפִלָּה זַכָּה מִכָּל הַלֵּב.

שִׁמְשׁוֹן אָמַר לַקָּדוֹשׁ בָּרוּךְ הוּא בִּתְפִלָּתוֹ הַמְּרַגֶּשֶׁת: "הַשֵּׁם, יִהְיֶה חִלּוּל הַשֵּׁם גָּדוֹל אִם חַס וְחָלִילָה אֲנִי אָמוּת כָּעֵת מִצָּמָא! אַחֲרֵי שֶׁהָרַגְתִּי אֶת כָּל הַפְּלִשְׁתִּים הָאֵלֶּה, אִם אָמוּת מִצָּמָא, הַפְּלִשְׁתִּים יָבֹאוּ וְיַגִּידוּ: מָה, הָאֱלֹהִים שֶׁל הַיְּהוּדִים יוֹדֵעַ לִשְׁלֹחַ כֹּחַ לְשִׁמְשׁוֹן, אֲבָל הוּא לֹא יָכוֹל לִשְׁלֹחַ לוֹ מַיִם?! בְּבַקָּשָׁה מִמְּךָ הַשֵּׁם, תַּעֲשֵׂה שֶׁלֹּא יִהְיֶה חִלּוּל הַשֵּׁם בָּעוֹלָם!".

וְכָךְ בְּדִיּוּק הָיָה! הַקָּדוֹשׁ בָּרוּךְ הוּא הִקְשִׁיב לִתְפִלָּתוֹ הַכֵּנָה שֶׁל שִׁמְשׁוֹן הַצַּדִּיק, עָשָׂה לוֹ נֵס מֻפְלָא וְהֵבִיא לוֹ מַיִם. שִׁמְשׁוֹן שָׁתָה מֵהַמַּיִם, הִתְאוֹשֵׁשׁ וְקִבֵּל כֹּחַ חָדָשׁ, וְנִצַּל בִּזְכוּת הַתְּפִלָּה הַמְּדַהֲמָה שֶׁלּוֹ שֶׁיָּצְאָה מֵעֹמֶק הַלֵּב.

💡 כְּשֶׁאָנוּ פוֹנִים אֶל הַשֵּׁם בִּתְפִלָּה כֵּנָה מֵעֹמֶק הַלֵּב, הַקָּדוֹשׁ בָּרוּךְ הוּא תָּמִיד שׁוֹמֵעַ אֶת תְּפִלָּתֵנוּ וְעוֹזֵר לָנוּ.
