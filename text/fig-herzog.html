הַפְּסוּק שֶׁלֹּא נִשְׁכַּח בַּמִּנְזָר

אַחֲרֵי הַשּׁוֹאָה, לֹא עָלֵינוּ, מֵאוֹת יְלָדִים מֵעַם יִשְׂרָאֵל הָיוּ מְפֹזָרִים בְּרַחֲבֵי אֵירוֹפָה, בְּתוֹךְ מִנְזָרִים. מִנְזָרִים הֵם בָּתִּים מְיֻחָדִים שֶׁבָּהֶם גָּרִים אַנְשֵׁי הַדָּת הַנּוֹצְרִית, וְשָׁם הֵם חַיִּים. בִּזְמַן הַשּׁוֹאָה, הוֹרִים רַבִּים שֶׁפָּחֲדוּ מְאֹד עַל גּוֹרַל יַלְדֵיהֶם, שָׁלְחוּ אֶת הַיְּלָדִים שֶׁלָּהֶם, הַיְּהוּדִים הַקְּטַנִּים, אֶל הַמִּנְזָרִים, תַּחַת הַשְׁגָּחָתָם שֶׁל אוֹתָם כְּמָרִים וּנְזָרוֹת. הַהוֹרִים עָשׂוּ זֹאת כִּי הֵם יָדְעוּ שֶׁהַצּוֹרֵר הַנַּאצִי, הָאוֹיֵב הַנּוֹרָא, לֹא יָעֵז לִשְׁלֹחַ אֶת יָדוֹ לְתוֹךְ בָּתֵּי הַנַּצְרוּת, וְכָךְ בְּדִיּוּק הֵם עָשׂוּ. הַיְּלָדִים הָאֵלֶּה נִשְׁלְחוּ לַמִּנְזָרִים כְּשֶׁהָיוּ קְטַנִּים וּצְעִירִים מְאֹד, וּבְמֶשֶׁךְ הַזְּמַן, לְאַט לְאַט, הֵם הִתְחִילוּ לִשְׁכֹּחַ אֶת זְהוּתָם הַיְּהוּדִית.

אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל שֶׁבְּכָל הַדּוֹרוֹת, הָרַב הָרָאשִׁי שֶׁלָּנוּ, הָרַב הַרְצוֹג, לִפְנֵי שָׁנִים רַבּוֹת, זֵכֶר צַדִּיק לִבְרָכָה, לָקַח עַל עַצְמוֹ שְׁלִיחוּת מְיֻחֶדֶת לְהַחְזִיר אֶת הַיְּלָדִים הָאֵלֶּה לְחֵיק עַם יִשְׂרָאֵל. הָרַב הַרְצוֹג הִסְתּוֹבֵב בְּסַבְלָנוּת וּבִמְסִירוּת בְּרַחֲבֵי אֵירוֹפָה, חָקַר וְדָרַשׁ, נִפְגַּשׁ וְשָׁאַל, עַד שֶׁהִגִּיעַ אֲפִילּוּ עַד לָאַפִיפְיוֹר, רֹאשׁ הַנַּצְרוּת. אֲבָל הַמְּלָאכָה הָיְתָה קָשָׁה מְאֹד, וְאַף אֶחָד מֵאַנְשֵׁי הַדָּת הַנּוֹצְרִית לֹא הִסְכִּים לַעֲזֹר לוֹ. לַמְרֹאת זֹאת, לְאַט לְאַט הוּא לִקֵּט וְאָסַף יֶלֶד לְיֶלֶד, עַד שֶׁהִצְלִיחַ לְהַעֲמִיס רַכֶּבֶת שְׁלֵמָה עִם חֲמֵשׁ מֵאוֹת יְלָדִים שֶׁנָּסְעוּ לְכִוּוּן אֶרֶץ יִשְׂרָאֵל.

אֲבָל הָרַב הַרְצוֹג לֹא הִפְסִיק אֶת מַסָּעוֹ. בְּאֶחָד מִגִּלְגּוּלָיו הוּא הִגִּיעַ לְמִנְזָר אֶחָד, שֶׁהוּא שָׁמַע וְיָדַע שֶׁיֵּשׁ בּוֹ קְבוּצָה לֹא קְטַנָּה שֶׁל יְלָדִים וִילָדוֹת יְהוּדִים. הוּא נִכְנַס אֶל הַמִּנְזָר וְדִבֵּר עִם רֹאשׁ הַמִּנְזָר, וְכַמּוּבָן נַעֲנָה בִּשְׁלִילָה. וְאָז הוּא בִּקֵּשׁ בַּקָּשָׁה אַחַת בִּלְבַד, וְאוֹתָהּ נָתְנוּ לוֹ. הוּא נִגַּשׁ אֶל הָאוּלָם הַמֶּרְכָּזִי שֶׁל הַמִּנְזָר, עָמַד שָׁם בָּאֶמְצַע, צָעַק וְקָרָא בְּקוֹל רָם: "שְׁמַע יִשְׂרָאֵל ה' אֱלֹהֵינוּ ה' אֶחָד!".

וּפִתְאוֹם, מֵהַפִּנָּה הָרְחוֹקָה שֶׁל הָאוּלָם, הוּא שׁוֹמֵעַ קוֹל שֶׁל יַלְדָּה קְטַנָּה עֹנָה לוֹ: "שְׁמַע יִשְׂרָאֵל!", וּמֵהַצַּד הַשֵּׁנִי יֶלֶד קָטָן רָץ אֵלָיו בְּעֵינַיִם דּוֹמְעוֹת, וּמֵאַחוֹר כַּמָּה יְלָדִים בָּאִים וּמְחַבְּקִים אוֹתוֹ. כֵּן כֵּן, הַיְּלָדִים הָאֵלֶּה כִּמְעַט וְשָׁכְחוּ שֶׁהֵם יְהוּדִים, אֲבָל אֶת הַפָּסוּק הַמֻּפְלָא, שֶׁיָּנְקוּ מֵאִמּוֹתֵיהֶם, וְשֶׁשָּׁמְעוּ בַּבַּיִת בֹּקֶר וָעֶרֶב – אֶת הַפָּסוּק הַמֻּפְלָא "שְׁמַע יִשְׂרָאֵל", אוֹתוֹ הֵם לֹא שָׁכְחוּ לְעוֹלָם.

💡 פְּסוּק "שְׁמַע יִשְׂרָאֵל" וְהַקֶּשֶׁר הַקָּדוֹשׁ לַה' וּלְעַם יִשְׂרָאֵל חֲקוּקִים בְּלֵב כָּל יֶלֶד יְהוּדִי וְאֵינָם נִשְׁכָּחִים לְעוֹלָם.
