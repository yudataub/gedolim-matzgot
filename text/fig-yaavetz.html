הָרַב שֶׁהִדְפִּיס אֶת הַסִּדּוּר בְּבֵיתוֹ

רַבִּי יַעֲקֹב עֶמְדִין, הַמּُכָּר גַּם בַּכִּנּוּי הַקָּדוֹשׁ הַיַּעֲבָעַ"ץ, נוֹלַד בָּעִיר אַלְטוֹנָה שֶׁבְּגֶרְמַנְיָה. הוּא הָיָה בְּנוֹ שֶׁל הַחָכָם צְבִי, וְלָמַד תּוֹרָה מֵאָבִיו הַדָּגוּל. מִנְּעוּרָיו שָׁקַד עַל הַתּוֹרָה, וְצָמַח לִהְיוֹת אֶחָד מִגְּדוֹלֵי חַכְמֵי יִשְׂרָאֵל בְּאַשְׁכְּנַז שֶׁל דּוֹרוֹ. בְּמַהֲלַךְ חַיָּיו כָּתַב יוֹתֵר מִשִּׁשִּׁים סְפָרִים חֲשׁוּבִים בְּכָל מִקְצוֹעוֹת הַתּוֹרָה הַקְּדוֹשָׁה, וְנִלְחַם בִּתְקִיפוּת רַבָּה בִּתְנוּעַת הַשֶּׁקֶר שֶׁל שַׁבְּתַי צְבִי כְּדֵי לְהָגֵן עַל אֱמֶת הַתּוֹרָה.

בִּשְׁנַת הַתפ"ח הֻזְמַן רַבִּי יַעֲקֹב לְכַהֵן כְּרַב הָעִיר עֶמְדִין, וּמִשֵּׁם זָכָה לְקַבֵּל אֶת שְׁמוֹ "עֶמְדִין". אֻלָּם לְאַחַר אַרְבַּע שָׁנִים בִּלְבַד שֶׁבָּהֶן כִּהֵן כְּרַב, הֶחְלִיט לַעֲזֹב אֶת מִשְׂרַת הָרַבָּנוּת. הוּא רָצָה מְאֹד לִהְיוֹת חָפְשִׁי לְהַגִּיד אֶת הָאֱמֶת הַיְשָׁרָה בְּכָל עֵת, בְּלִי לִהְיוֹת תָּלוּי בְּאַף אָדָם אוֹ בְּמִשְׂרָה. לָכֵן חָזַר רַבִּי יַעֲקֹב לְעִירוֹ אַלְטוֹנָה, וְהֶחְלִיט לְפַרְנֵס אֶת עַצְמוֹ בְּכָבוֹד מִמִּסְחָר. בְּדֶרֶךְ שֶׁל הֲלָצָה הוּא הָיָה מְסַפֵּר כִּי הוּא מְבָרֵךְ בְּכָל יוֹם "בָּרוּךְ שֶׁלֹּא עָשַׂנִי אַב בֵּית דִּין", בְּדוֹמֶה לַבְּרָכָה שֶׁמְּבָרְכִים "שֶׁלֹּא עָשַׂנִי עֶבֶד".

אֲבָל הָעֲבוֹדָה בַּמִּסְחָר לֹא עָצְרָה אֶת לִמּוּד הַתּוֹרָה הַמַּתְמִיד שֶׁלּוֹ. אֶת הַשָּׁעוֹת הַטּוֹבוֹת וְהַיְקָרוֹת בְּיוֹתֵר שֶׁל יוֹמוֹ הוּא הִקְדִּישׁ לִכְתִיבַת סְפָרִים וּלְעִיּוּן בַּתּוֹרָה. אָז עָשָׂה רַבִּי יַעֲקֹב דָּבָר מְיֻחָד וּפִלְאִי מְאֹד: הוּא בִּקֵּשׁ וְקִבֵּל רְשׁוּת מֵהַשִּׁלְטוֹן הַמְּקוֹמִי, וּפָתַח בֵּית דְּפוּס עִבְרִי בְּתוֹךְ בֵּיתוֹ הַפְּרָטִי! הֲקָמַת בֵּית הַדְּפוּס עָלְתָה לוֹ הוֹן רַב וְכֶסֶף רַב, אַךְ עַל יְדֵי כָּךְ הוּא יָכֹל הָיָה לְהַדְפִּיס אֶת סְפָרָיו הַקְּדוֹשִׁים בְּעַצְמוֹ וּבְדַיְקָנוּת.

אֶחָד מֵהַסְּפָרִים הַחֲשׁוּבִים וְהַמְּפֻרְסָמִים בְּיוֹתֵר שֶׁהִדְפִּיס בְּבֵיתוֹ הוּא הַסִּדּוּר "עַמּוּדֵי שָׁמַיִם". זֶהוּ סִדּוּר מְפֹאָר עִם פֵּרוּשִׁים לַתְּפִלּוֹת שֶׁל כָּל יְמוֹת הַשָּׁנָה, וְכֵן עִם כָּל הַדִּינִים וְהַמִּנְהָגִים שֶׁל הַתְּפִלָּה. הַסִּדּוּר הַזֶּה מוּכָר עַד הַיּוֹם בַּשֵּׁם "סִדּוּר הַיַּעֲבָעַ"ץ" אוֹ "סִדּוּר בֵּית יַעֲקֹב", וִיהוּדִים רַבִּים בְּכָל הָעוֹלָם מִתְפַּלְלִים בּוֹ בְּכַוָּנָה. כַּאֲשֶׁר נִפְטַר אָבִיו, הַחָכָם צְבִי, בָּעִיר לְבּוֹב, נָסַע רַבִּי יַעֲקֹב לְשָׁם כְּדֵי לְהַסְפִּיד אוֹתוֹ בְּכָבוֹד רַב, וְהִדְפִּיס אֶת הַהֶסְפֵּד שֶׁכָּתַב לְזִכְרוֹ.

מִסִּפּוּר חַיָּיו הַמֻּפְלָא שֶׁל רַבִּי יַעֲקֹב אָנוּ לוֹמְדִים כִּי אָדָם יָכוֹל לַעֲבֹד לְפַרְנָסָתוֹ וּבְכָל זֹאת לִהְיוֹת גָּדוֹל וְעָצוּם בַּתּוֹרָה, כַּאֲשֶׁר הוּא מַקְדִּישׁ לָהּ אֶת הַשָּׁעוֹת הַטּוֹבוֹת בְּיוֹתֵר שֶׁל יוֹמוֹ. כְּמוֹ כֵן אָנוּ לוֹמְדִים שֶׁהָאֱמֶת חֲשׁוּבָה יוֹתֵר מִכָּל כָּבוֹד, מַעֲמָד אוֹ מִשְׂרָה.

💡 כְּשֶׁמַּקְדִּישִׁים אֶת הַשָּׁעוֹת הַטּוֹבוֹת בְּיוֹתֵר לַתּוֹרָה וְשׁוֹמְרִים עַל הָאֱמֶת, יְכוֹלִים לִהְיוֹת גְּדוֹלִים בַּתּוֹרָה וְאוֹהֲבֵי אֱמֶת.
