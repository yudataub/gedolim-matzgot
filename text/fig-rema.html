הַמַּתָּנָה שֶׁל שׁוֹמֵר הַשַּׁבָּת

בָּעִיר קְרָקֹוב חַי אָדָם יָקָר וְצַדִּיק שֶׁנִּקְרָא רַבִּי יִשְׂרָאֵל אִיסֶר. רַבִּי יִשְׂרָאֵל אִיסֶר הָיָה אֶחָד מֵעֲשִׁירֵי הָעִיר וּמֵהַנְּדִיבִים הַגְּדוֹלִים שֶׁלָּהּ. הָיְתָה לוֹ חֲנוּת גְּדוֹלָה וּמְפֹאָרֶת שֶׁל בַּדִּים מְשֻׁבָּחִים וִיקָרִים מְאוֹד, וְקֹנִים רַבִּים וּמְכֻבָּדִים הָיוּ מַגִּיעִים לִקְנוֹת בָּהּ.

לְרַבִּי יִשְׂרָאֵל אִיסֶר הָיָה מִנְהָג חָשׁוּב מְאוֹד: בְּכָל יוֹם שִׁשִּׁי, מֵחֲצוֹת הַיּוֹם וְהָלְאָה, הוּא הָיָה סוֹגֵר אֶת הַחֲנוּת וְעוֹסֵק אַךְ וְרַק בַּהֲכָנָה לִקְרַאת שַׁבַּת קֹדֶשׁ.

פַּעַם אַחַת, בְּיוֹם שִׁשִּׁי, הִגִּיעַ לַחֲנוּת קוֹנֶה מְאֹד מְכֻבָּד וְעָשִׁיר. הַקּוֹנֶה בִּקֵּשׁ מֵרַבִּי יִשְׂרָאֵל אִיסֶר לִרְאוֹת בַּדִּים מְשֻׁבָּחִים וִיקָרִים. רַבִּי יִשְׂרָאֵל אִיסֶר הוֹצִיא אֶת הַבַּדִּים וְהֶרְאָה לוֹ אוֹתָם. יַחַד הֵם מִיְּנוּ אֶת הַבַּדִּים שֶׁהַקּוֹנֶה רָצָה, וּמָדְדוּ אוֹתָם.

תּוֹךְ כְּדֵי שֶׁהֵם עוֹסְקִים בִּבְחִירַת הַבַּדִּים וּבִמְדִידָתָם, הִבִּיט רַבִּי יִשְׂרָאֵל אִיסֶר וְרָאָה שֶׁכְּבָר עוֹד מְעַט מַגִּיעַ חֲצוֹת הַיּוֹם. הוּא זָכַר אֶת מִנְהָגוֹ לִסְגֹּר אֶת הַחֲנוּת בַּחֲצוֹת וּלְהִתְכּוֹנֵן לַשַּׁבָּת, וְלָכֵן אָמַר לַקּוֹנֶה בְּכָבוֹד וּבְדֶרֶךְ אֶרֶץ: "אֲנִי מִצְטַעֵר מְאוֹד, אַךְ נִצְטָרֵךְ לְהַמְשִׁיךְ בָּזֶה בְּיוֹם רִאשׁוֹן".

הַקּוֹנֶה הַמְּכֻבָּד הִתְפַּלֵּא מְאוֹד וְאָמַר: "מָה זֹאת אוֹמֶרֶת? מָה פִּתְאוֹם! אֲנִי רוֹצֶה לְסַיֵּם אֶת הַקְּנִיָּה הַיּוֹם. אֲנִי לֹא יָכוֹל לְחַכּוֹת עַד יוֹם רִאשׁוֹן".

רַבִּי יִשְׂרָאֵל אִיסֶר הֵשִׁיב לוֹ שׁוּב בְּכָבוֹד וּבְדֶרֶךְ אֶרֶץ: "אֲנִי מִצְטַעֵר, אֲנִי מֵחֲצוֹת וְהָלְאָה עוֹסֵק אַךְ וְרַק בַּהֲכָנָה לִקְרַאת שַׁבַּת קֹדֶשׁ. כָּךְ מִנְהָגִי לַעֲשׂוֹת בְּעֶרֶב שַׁבָּת".

הַקּוֹנֶה נִסָּה לְשַׁכְנֵעַ אוֹתוֹ, אַךְ כְּשֶׁרָאָה שֶׁרַבִּי יִשְׂרָאֵל אִיסֶר עוֹמֵד עַל דַּעְתּוֹ, יָצָא מֵהַחֲנוּת וְאָמַר: "בְּסֵדֶר גָּמוּר, אֲנִי אֶעֱרֹךְ אֶת הַקְּנִיָּה הַגְּדוֹלָה וְהַיְּקָרָה הַזֹּאת בַּחֲנוּת אַחֶרֶת".

רַבִּי יִשְׂרָאֵל אִיסֶר יָדַע שֶׁהוּא הִפְסִיד אֶפְשָׁרוּת לְהַרְוִיחַ כֶּסֶף רַב, אַךְ הוּא הָיָה שָׂמֵחַ שֶׁשָּׁמַר עַל מִנְהָגוֹ לְהָכִין לִקְרַאת שַׁבַּת הַמַּלְכָּה.

וּבִזְכוּת זֶה שֶׁשָּׁמַר עַל מִנְהָגוֹ לְהִתְכּוֹנֵן מֵחֲצוֹת לִקְרַאת שַׁבַּת הַמַּלְכָּה, נָתַן לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא מַתָּנָה גְּדוֹלָה. כַּעֲבֹר זְמַן לֹא רַב נוֹלַד לוֹ בֵּן צַדִּיק, שֶׁקָּרְאוּ לוֹ רַבִּי מֹשֶׁה בֵּן אִיסֶר, הֲלֹא הוּא רַבִּי מֹשֶׁה אִיסֶרְלִישׁ, הַמְּכֻנֶּה הָרַמָּ"א. רַבִּי מֹשֶׁה הָיָה הַפּוֹסֵק הַגָּדוֹל שֶׁל אַרְצוֹת אַשְׁכְּנַז וְהֵאִיר אֶת עֵינֵי יִשְׂרָאֵל בְּתוֹרָתוֹ.
