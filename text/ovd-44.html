הַזִּכָּרוֹן הַמֻּפְלָא שֶׁל הָרַב עוֹבַדְיָה

הַגָּאוֹן הָרַב עוֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק וְקָדוֹשׁ לִבְרָכָה הָיָה יָדוּעַ בְּאַהֲבָתוֹ הָעֲצוּמָה וְהַגְּדוֹלָה לְתוֹרַת יִשְׂרָאֵל. כָּל חַיָּיו, מִנְּעוּרָיו וְעַד יְמֵי זִקְנָתוֹ, עָסַק בַּתּוֹרָה יוֹמָם וָלַיְלָה בְּשִׁקְדָנוּת רַבָּה, בְּשִׂמְחָה וּבִדְבֵקוּת נִפְלָאָה. הַתּוֹרָה הָיְתָה מַמָּשׁ מַרְכַּז חַיָּיו, וְכָל מַחְשְׁבוֹתָיו הָיוּ נְתוּנוֹת לְלִמּוּד הַתּוֹרָה וּלְהַבָּנַת דִּבְרֵי חֲכָמֵינוּ.

כַּאֲשֶׁר הִגִּיעַ הָרַב עוֹבַדְיָה יוֹסֵף לְגִיל שְׁמוֹנִים וָשֶׁבַע, הוּא חָשׁ בְּרַע בְּגוּפוֹ וְנִזְקַק לַעֲבֹר נִתּוּחַ מֻרְכָּב וְקָשֶׁה בְּבֵית הַחוֹלִים. הַנִּתּוּחַ הָיָה אָרֹךְ, וְהָרוֹפְאִים הָיוּ צְרִיכִים לְהַרְדִּים אֶת הָרַב כְּדֵי לְבַצֵּעַ אֶת הַטִּפּוּל הָרְפוּאִי הַנָּחוּץ. בְּנֵי מִשְׁפַּחְתּוֹ הָאוֹהֲבִים וְתַלְמִידָיו הָרַבִּים עָמְדוּ מִחוּץ לַחֲדַר הַנִּתּוּחַ, הִתְפַּלְּלוּ מִכָּל הַלֵּב לַקָּדוֹשׁ בָּרוּךְ הוּא, וְחִכּוּ בְּדַאֲגָה רַבָּה שֶׁהַנִּתּוּחַ יַעֲבֹר בְּשָׁלוֹם וְשֶׁהָרַב יִתְעוֹרֵר מֵהַהַרְדָּמָה.

לְאַחַר שָׁעוֹת מְתוּחוֹת שֶׁל הַמְתָנָה, הַנִּתּוּחַ הִסְתַּיֵּם בְּהַצְלָחָה רַבָּה בְּסַיַּעְתָּא דִּשְׁמַיָּא. הָרַב עוֹבַדְיָה יָצָא מֵחֲדַר הַנִּתּוּחַ וְהוּעֲבַר לַחֲדַר הַהִתְאוֹשְׁשׁוּת כְּשֶׁהוּא עֲדַיִן יָשֵׁן עֲמֻקּוֹת. מְעַט לְאַחַר מִכֵּן, הֵחֵל הָרַב לְהִתְעוֹרֵר מֵהַהַרְדָּמָה הַקָּשָׁה. כָּל הַסּוֹבְבִים אוֹתוֹ הִבִּיטוּ בּוֹ בְּרִגְשׁוֹת שִׂמְחָה וְצִפִיָּה כְּדֵי לִרְאוֹת מָה יִהְיוּ מִלּוֹתָיו הָרִאשׁוֹנוֹת שֶׁל הָרַב לְאַחַר הַנִּתּוּחַ הַמְּעַיֵּף.

וְהִנֵּה, בַּרֶגַע שֶׁפָּקַח אֶת עֵינָיו, הַדָּבָר הָרִאשׁוֹן שֶׁבִּקֵּשׁ הָרַב עוֹבַדְיָה לֹא הָיָה מַשְׁקֶה מְרַעֲנָן וְלֹא מְנוּחָה. הָרַב בִּקֵּשׁ מִיָּד מַיִם לִנְטִילַת יָדַיִם! הוּא רָצָה לִנְטֹל אֶת יָדָיו כַּהֲלָכָה כְּדֵי שֶׁיּוּכַל לַחֲזֹר וְלִלְמֹד תּוֹרָה מִיָּד, בְּלִי לְבַזְבֵּז אַף רֶגַע אֶחָד שֶׁל לִמּוּד. הַסּוֹבְבִים אֶת הָרַב הִתְרַגְּשׁוּ מְאֹד לִרְאוֹת אֶת מְסִירוּת נַפְשׁוֹ וְאֶת אֲהַבְתּוֹ הַגְּדוֹלָה לַתּוֹרָה הַקְּדוֹשָׁה גַּם בְּרְגָעִים כָּאֵלֶּה.

לְאַחַר מִכֵּן, הָרוֹפְאִים וּבְנֵי הַמִּשְׁפָּחָה רָצוּ לִבְדֹק אֶת מַצַּב זִכְרוֹנוֹ שֶׁל הָרַב כְּדֵי לִרְאוֹת אִם הַנִּתּוּחַ וְהַהַרְדָּמָה לֹא הִשְׁפִּיעוּ עַל חֲדוּת מַחְשַׁבְתּוֹ. הֵם הֶחְלִיטוּ לִשְׁאֹל אֶת הָרַב שְׁאֵלוֹת שוֹנוֹת כְּדֵי לִבְחֹן אֶת זִכְרוֹנוֹ.

כַּאֲשֶׁר שָׁאֲלוּ אֶת הָרַב עַל נוֹשְׂאִים תּוֹרָנִיִּים מְתוּקִים, הָרַב עָנָה מִיָּד בְּבְקִיעוּת מַפְתִיעָה וּמַדְהִימָה. הוּא צִטֵּט לָהֶם בִּמְדֻיָּק מָקוֹר הִלְכָתִי עָמֹק מִסִּפְרוֹ שֶׁל הַחִידָ"א (רַבִּי חַיִּים יוֹסֵף דָּוִד אֲזוּלַאי), מָקוֹר שֶׁלָּמַד לִפְנֵי שִׁשִּׁים וְשָׁלֹשׁ שָׁנִים! הָרַב זָכַר כָּל מִלָּה וּכְלַל הִלְכָתִי שֶׁלָּמַד עַשְׂרוֹת שָׁנִים קֹדֶם לָכֵן בִּבְהִירוּת גְּמוּרָה, לְלֹא שׁוּם טָעוּת.

אַךְ כַּאֲשֶׁר שָׁאֲלוּ אֶת הָרַב שְׁאֵלָה פְּשׁוּטָה מֵחַיֵּי הַיּוֹם-הַיּוֹם, כְּגוֹן מָה אָכַל בַּצָּהֳרַיִם בְּאוֹתוֹ יוֹם, הָרַב לֹא זָכַר אֶת הַתְּשׁוּבָה. עִנְיְנֵי הָעוֹלָם הַזֶּה לֹא תָּפְסוּ מָקוֹר אוֹ חֲשִׁיבוּת בְּלִבּוֹ, וְהוּא לֹא הִקְדִּישׁ לָהֶם תְּשׂוּמַת לֵב. לְעֻמַּת זֹאת, כָּל דִּבְרֵי הַתּוֹרָה שֶׁלָּמַד הָיוּ חֲקוּקִים בְּלִבּוֹ וּבְנַפְשׁוֹ לְתָמִיד.

הָרוֹפְאִים וְכָל הַנֹּכָחִים עָמְדוּ נִדְהָמִים וְהִתְרַגְּשׁוּ מְאֹד. הֵם הֵבִינוּ כִּי הַתּוֹרָה הַקְּדוֹשָׁה הִיא חַיָּיו שֶׁל הָרַב עוֹבַדְיָה יוֹסֵף, וּמַה שֶׁאָדָם אוֹהֵב וּמַעֲרִיךְ מִכָּל הַלֵּב נִשְׁאָר שָׁמוּר אֶצְלוֹ לְעוֹלָם.
