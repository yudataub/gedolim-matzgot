רַבִּי עֲקִיבָא בְּבֵית הַכֶּלֶא

מְסַפְּרִים חֲכָמֵינוּ, שֶׁכַּאֲשֶׁר רַבִּי עֲקִיבָא הַגָּדוֹל יָשַׁב בְּבֵית הָאֲסוּרִים, הָיָה מַגִּיעַ אֵלָיו בְּכָל יוֹם וָיוֹם תַּלְמִידוֹ הַמָּסוּר, רַבִּי יְהוֹשֻׁעַ הַגַּרְסִי. רַבִּי יְהוֹשֻׁעַ הָיָה מֵבִיא לוֹ אֹכֶל וּמַיִם לִנְטִילַת יָדַיִם, בְּכָל יוֹם וָיוֹם, כִּי הָיָה מָסוּר אֵלָיו מְאֹד.

יוֹם אֶחָד הִגִּיעַ רַבִּי יְהוֹשֻׁעַ לְבֵית הַכֶּלֶא. עָצַר אוֹתוֹ הַסּוֹהֵר, הַשּׁוֹמֵר שֶׁבַּפֶּתַח, וְאָמַר לוֹ: "תַּרְאֶה לִי מָה הֵבֵאתָ". רַבִּי יְהוֹשֻׁעַ, כְּהֶרְגֵּלוֹ, פָּתַח וְהֶרְאָה לוֹ: מְעַט אֹכֶל וּמְעַט מַיִם, וְרָצָה לְהִכָּנֵס פְּנִימָה.

אָמַר לוֹ הַשּׁוֹמֵר: "יוֹתֵר מִדַּי מַיִם הֵבֵאתָ הַיּוֹם, זֶה לֹא טוֹב. אוּלַי אַתָּה רוֹצֶה לְהִשְׁתַּמֵּשׁ בָּהֶם כְּדֵי לְרַכֵּךְ אֶת הָאֲדָמָה וְלַחְפּוֹר מְחִלָּה, כְּדֵי לַעֲזֹר לְאֶחָד מֵהָאֲסִירִים לִבְרֹחַ, לְרַבִּי עֲקִיבָא שֶׁאַתָּה מְבַקֵּר אוֹתוֹ?".

רַבִּי יְהוֹשֻׁעַ נִסָּה לְהַסְבִּיר וְלוֹמַר לַשּׁוֹמֵר שֶׁזֶּה מָה שֶׁרַבִּי עֲקִיבָא צָרִיךְ. אַךְ הַשּׁוֹמֵר לֹא הִסְכִּים לִשְׁמֹעַ. אָמַר לוֹ הַשּׁוֹמֵר: "שְׁפֹךְ חֲצִי מֵהַכַּמּוּת, הַחֲצִי הַנּוֹתָר יַסְפִּיק לוֹ". כָּךְ נֶאֱלַץ רַבִּי יְהוֹשֻׁעַ לַעֲשׂוֹת: הוּא שָׁפַךְ חֲצִי מֵהַכַּמּוּת וְנִכְנַס אֶל תּוֹךְ הַכֶּלֶא.

כְּהֶרְגֵּלוֹ, נִכְנַס רַבִּי יְהוֹשֻׁעַ אֶל רַבִּי עֲקִיבָא וְנָתַן לוֹ אֶת הָאֹכֶל וְאֶת הַמַּיִם. רָצָה רַבִּי עֲקִיבָא לִנְטֹל אֶת יָדָיו, אַךְ פִּתְאוֹם הוּא רָאָה שֶׁכַּמּוּת הַמַּיִם מְאֹד קְטַנָּה. אָמַר רַבִּי עֲקִיבָא לְתַלְמִידוֹ רַבִּי יְהוֹשֻׁעַ: "תַּלְמִידִי הַיָּקָר, מָה אַתָּה עוֹשֶׂה? אַתָּה יוֹדֵעַ שֶׁאֲנִי זָקֵן וַאֲנִי צָרִיךְ גַּם מַיִם לִשְׁתִיָּה בְּכַמּוּת שֶׁתַּחֲזִיק אוֹתִי בַּחַיִּים, מַדּוּעַ הֵבֵאתָ לִי כָּל כָּךְ מְעַט מַיִם?".

אָמַר לוֹ רַבִּי יְהוֹשֻׁעַ: "אֲנִי מִצְטַעֵר רַבִּי, זֶה לֹא בְּאַשְׁמָתִי", וְסִפֵּר לוֹ מָה הָיָה עִם הַשּׁוֹמֵר בַּפֶּתַח.

אָמַר רַבִּי עֲקִיבָא: "טוֹב, אִם כָּךְ אָז אֲנִי אֶטּוֹל יָדַיִם וַאֲנִי לֹא אֶשְׁתֶּה הַיּוֹם". אָמַר לוֹ תַּלְמִידוֹ: "אֲבָל רַבִּי, אַתָּה צָרִיךְ לִשְׁתּוֹת, חַם, אַתָּה לֹא אָדָם צָעִיר, אַתָּה חַיָּב לִשְׁתּוֹת, אַתָּה תְּסַכֵּן אֶת חַיֶּיךָ!".

כְּשֶׁשָׁמַע אֶת זֶה רַבִּי עֲקִיבָא, זֶה מָה שֶׁהוּא עָשָׂה: בְּאוֹתוֹ הַיּוֹם הוּא לֹא אָכַל לֶחֶם, וְזֶה הָיָה כִּמְעַט כָּל מָה שֶׁהָיָה לוֹ לֶאֱכֹל. הוּא הִשְׁתַּמֵּשׁ בַּמַּיִם בִּשְׁבִיל לִשְׁתּוֹת, אַךְ בִּגְלָל שֶׁלֹּא הָיָה מַסְפִּיק מַיִם לִנְטִילַת יָדַיִם, הוּא פָּשׁוּט לֹא אָכַל. אָמַר רַבִּי עֲקִיבָא: "רְאוּיָה מִצְוַת נְטִילַת יָדַיִם שֶׁאֲנִי אֶתְיַסֵּר בִּשְׁבִילָהּ, וְלֹא אֹכַל לֶחֶם הַיּוֹם".
