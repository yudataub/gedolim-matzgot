יָאִיר הַגִּלְעָדִי וּשְׁלֹשִׁים בָּנָיו

בִּימֵי קֶדֶם, בִּתְקוּפַת הַשּׁוֹפְטִים בְּאֶרֶץ יִשְׂרָאֵל, לְאַחַר יְמֵי שְׁפִיטָתוֹ שֶׁל הַשּׁוֹפֵט תּוֹלָע בֶּן פּוּאָה, קָם מַנְהִיג חָדָשׁ וְחָשׁוּב לְעַם יִשְׂרָאֵל. שְׁמוֹ שֶׁל הַמַּנְהִיג הַזֶּה הָיָה יָאִיר הַגִּלְעָדִי. יָאִיר הַגִּלְעָדִי עָמַד בְּרֹאשׁ הָעָם וְשָׁפַט אֶת יִשְׂרָאֵל בְּמֶשֶׁךְ עֶשְׂרִים וּשְׁתַּיִם שָׁנִים תְּמִימוֹת. בְּכָל הַשָּׁנִים הָאֵלֶּה הִנְהִיג יָאִיר אֶת עַם יִשְׂרָאֵל בְּחָכְמָה, בְּצֶדֶק וּבְמֵישָׁרִים, וְדָאַג לְכָל בְּנֵי יִשְׂרָאֵל.

לַשּׁוֹפֵט יָאִיר הַגִּלְעָדִי הָיוּ שְׁלֹשִׁים בָּנִים. בָּנָיו שֶׁל יָאִיר הָיוּ אֲנָשִׁים חֲשׁוּבִים, נֶחְבָּבִים וּמְכֻבָּדִים מְאֹד בְּקֶרֶב כָּל עַם יִשְׂרָאֵל. לְכָל אֶחָד מִשְּׁלֹשִׁים הַבָּנִים הָיָה עַיִר אֶחָד, שֶׁהוּא חֲמוֹר צָעִיר. בְּאֹתָם יָמִים קְדוּמִים, רְכִיבָה עַל עֲיָרִים הָיְתָה דְּבַר חֲשִׁיבוּת גְּדוֹלָה מְאֹד, וְסִימָן מֻבְהָק שֶׁל כָּבוֹד, מַנְהִיגוּת וּמַעֲלָה הָרְאוּיָה לִבְנֵי מַנְהִיגִים. כָּל אֶחָד מִשְּׁלֹשִׁים הַבָּנִים רָכַב עַל עַיִר מִשֶּׁלּוֹ, וְכָךְ יָדְעוּ כָּל אֲנָשֵׁי הָעָם כִּי בְּנֵי הַשּׁוֹפֵט הֵם אֲנָשִׁים נַעֲלִים, חֲשׁוּבִים וּמְכֻבָּדִים.

נוֹסָף עַל הָעֲיָרִים הַצְּעִירִים, לִשְׁלֹשִׁים הַבָּנִים הָיוּ שְׁלֹשִׁים עָרִים בְּאֶרֶץ הַגִּלְעָד. כָּל בֵּן מִבְּנֵי יָאִיר הָיָה אַחֲרַאי עַל עִיר אַחַת מִבֵּין שְׁלֹשִׁים הֶעָרִים שֶׁבְּאֶרֶץ הַגִּלְעָד. אֶת הֶעָרִים הָאֵלֶּה קָרְאוּ כָּל אֲנָשֵׁי הָאָרֶץ בְּשֵׁם "חַוֺּת יָאִיר".

מַדּוּעַ נִקְרְאוּ הֶעָרִים שֶׁל הַבָּנִים בְּשֵׁם "חַוֺּת יָאִיר"? הַסִּבָּה לְכָךְ קְשׁוּרָה לַהִיסְטוֹרְיָה שֶׁל עַם יִשְׂרָאֵל. עוֹד בִּימֵי מֹשֶׁה רַבֵּנוּ, מַנְהִיג מִשֵּׁבֶט מְנַשֶּׁה שֶׁשְּׁמוֹ הָיָה יָאִיר בֶּן מְנַשֶּׁה, הָלַךְ וְלָכַד כְּפָרִים וְעָרִים בְּאֹתוֹ אֵזוֹר בְּאֶרֶץ הַגִּלְעָד, וְקָרָא לָהֶן בְּשֵׁם "חַוֺּת יָאִיר". מִכֵּיוָן שֶׁשְּׁמוֹ שֶׁל הַשּׁוֹפֵט הַחָדָשׁ הָיָה גַּם כֵּן יָאִיר, כְּמוֹ שְׁמוֹ שֶׁל יָאִיר בֶּן מְנַשֶּׁה, וּמִכֵּיוָן שֶׁעָרֵי בָּנָיו הָיוּ מְרוּבּוֹת וְנִמְצְאוּ בְּאֹתוֹ חֶבֶל אֶרֶץ בַּגִּלְעָד, קָרְאוּ גַּם לֶעָרִים שֶׁל בְּנֵי יָאִיר הַגִּלְעָדִי בְּשֵׁם "חַוֺּת יָאִיר".

כָּל עַם יִשְׂרָאֵל וְכָל תּוֹשְׁבֵי אֶרֶץ הַגִּלְעָד הִכִּירוּ אֶת חַוֺּת יָאִיר וְאֶת שְׁלֹשִׁים הַבָּנִים הָרוֹכְבִים עַל שְׁלֹשִׁים הָעֲיָרִים. הַשֵּׁם "חַוֺּת יָאִיר" נִשְׁאַר שְׁמָן שֶׁל הֶעָרִים הָאֵלֶּה בְּאֶרֶץ הַגִּלְעָד עַד הַיּוֹם הַזֶּה, כְּפִי שֶׁכָּתוּב בְּסֵפֶר שׁוֹפְטִים בַּתָּנָ"ךְ.

לְאַחַר עֶשְׂרִים וּשְׁתַּיִם שְׁנוֹת שְׁפִיטָה וְהַנְהָגָה טוֹבָה וּמְפֹאָרָה, נִפְטַר יָאִיר הַגִּלְעָדִי. עַם יִשְׂרָאֵל זָכַר אֶת יָאִיר הַשּׁוֹפֵט אֲשֶׁר הִנְהִיג אֶת הָעָם בֶּאֱמוּנָה. יָאִיר הַגִּלְעָדִי הוּבָא לִקְבוּרָה בִּמְקוֹם קְבוּרָתוֹ בָּעִיר קָמוֹן.

💡 מַנְהִיגוּת טוֹבָה וּמַעֲשִׂים טוֹבִים מְבִיאִים כָּבוֹד וּבְרָכָה לַמִּשְׁפָּחָה וְלַדּוֹרוֹת הַבָּאִים.
