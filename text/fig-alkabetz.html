הַקּוֹל הַמֻּפְלָא בְּלֵיל שָׁבוּעוֹת

רַבִּי שְׁלֹמֹה הַלֵּוִי אַלְקָבֵץ הָיָה חָכָם גָּדוֹל, מְקֻבָּל וּפַיְטָן מִמְּגֹרָשֵׁי סְפָרַד. הוּא לָמַד תּוֹרָה בָּעִיר סָלוֹנִיקִי, וּלְאַחַר מִכֵּן עָבַר לָגוּר בָּעִיר אַדְרִיאַנוֹפּוֹל לְיַד יְדִידוֹ הַטּוֹב, רַבִּי יוֹסֵף קָארוֹ, בַּעַל הַ"שֻׁלְחָן עָרוּךְ". רַבִּי שְׁלֹמֹה אָהַב מְאֹד אֶת תּוֹרַת ה' וְהֶאֱמִין כִּי דַּוְקָא בְּאֶרֶץ יִשְׂרָאֵל מִתְגַּלִּים סוֹדוֹת הַתּוֹרָה הָעֲמֻקִּים בְּיוֹתֵר. לָכֵן רָצָה מְאֹד לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל.

לִפְנֵי שֶׁעָלוּ לָאָרֶץ, כַּאֲשֶׁר הָיוּ עֲדַיִן בַּגּוֹלָה, הֶחְלִיטוּ רַבִּי שְׁלֹמֹה אַלְקָבֵץ וְרַבִּי יוֹסֵף קָארוֹ, יַחַד עִם קְבוּצָה שֶׁל תַּלְמִידֵי חֲכָמִים, לַעֲשׂוֹת מַעֲשֶׂה מְיֻחָד בְּלֵיל חַג הַשָּׁבוּעוֹת. זֶהוּ הַלַּיְלָה שֶׁלִּפְנֵי הַיּוֹם שֶׁבּוֹ קִבְּלוּ בְּנֵי יִשְׂרָאֵל אֶת הַתּוֹרָה בְּהַר סִינַי. הֵם הֶחְלִיטוּ שֶׁבַּלַּיְלָה הַזֶּה הֵם לֹא יִישְׁנוּ כְּלָל, אֶלָּא יֵשְׁבוּ וְיִלְמְדוּ תּוֹרָה בְּמֶשֶׁךְ כָּל הַלַּיְלָה.

בְּלֵיל הַחַג נֶאֶסְפוּ כֻּלָּם יַחַד בְּשִׂמְחָה גְּדוֹלָה וּבִקְדֻשָּׁה. הֵם יָשְׁבוּ שָׁעוֹת אֲרֻכּוֹת וְלָמְדוּ פְּסוּקִים מִתּוֹרַת מֹשֶׁה, מֵהַנְּבִיאִים, מֵהַכְּתוּבִים וּמִמִּשְׁנָיוֹת. הָאֲוִירָה הָיְתָה מְרוֹמֶמֶת וּמְלֵאַת אַהֲבַת ה'.

וְהִנֵּה, בְּאֶמְצַע הַלַּיְלָה, בְּדִיּוּק בַּחֲצוֹת, קָרָה דָּבָר מֻפְלָא וּמַרְהִיב. פִּתְאֹם נִשְׁמַע קוֹל חָזָק וְנָעִים מְדַבֵּר מִתּוֹךְ פִּיו שֶׁל רַבִּי יוֹסֵף קָארוֹ. כָּל הַנּוֹכָחִים שָׁמְעוּ אֶת הַדְּבָרִים בְּבֵרוּר רַב. הַקּוֹל הַקָּדוֹשׁ שִׁבַּח אֶת תַּלְמִידֵי הַחֲכָמִים עַל כָּךְ שֶׁהֵם עֵרִים בַּלַּיְלָה וְלוֹמְדִים תּוֹרָה בִּמְסִירוּת. הַקּוֹל סִפֵּר לָהֶם כִּי הַלִּמּוּד הַזַּךְ שֶׁלָּהֶם עָלָה לַשָּׁמַיִם וְשִׂמַּח מְאֹד אֶת הַשָּׁמַיִם.

הַקּוֹל הִמְשִׁיךְ וְאָמַר לָהֶם: "אַל תַּפְסִיקוּ מִלִּמּוּדְכֶם, וְהִתְחַזְּקוּ! מַהֲרוּ לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, כִּי לֹא כָּל הָעִתִּים שָׁווֹת!". כְּשֶׁשָּׁמְעוּ אֶת הַדְּבָרִים הַמְּרַגְּשִׁים הָאֵלֶּה, בָּכוּ כָּל הַחֲכָמִים מֵרֹב הִתְרַגְּשׁוּת וּדְמָעוֹת שֶׁל שִׂמְחָה. הֵם הִמְשִׁיכוּ לִלְמֹד תּוֹרָה בְּאֹרֶךְ רוּחַ וּבְהַתְמָדָה עַד שֶׁעָלָה אוֹר הַבֹּקֶר.

בַּלַּיְלָה שֶׁלְּאַחֲרָיו, שֶׁהָיָה הַיּוֹם הַשֵּׁנִי שֶׁל הַחַג בְּחוּץ לָאָרֶץ, נֶאֶסְפוּ הַחֲכָמִים שׁוּב. הַפַּעַם הִצְטַרְפוּ אֲלֵיהֶם חֲבֵרִים נוֹסָפִים שֶׁשָּׁמְעוּ עַל הַנֵּס. שׁוּב יָשְׁבוּ וְלָמְדוּ תּוֹרָה כָּל הַלַּיְלָה, וְשׁוּב נִשְׁמַע הַקּוֹל הַמֻּפְלָא וְעוֹדֵד אוֹתָם לְהַמְשִׁיךְ בְּדַרְכָּם הַקְּדוֹשָׁה.

בְּעִקְבוֹת הַלַּיְלָה הַמְּיֻחָד הַזֶּה, עָלוּ רַבִּי שְׁלֹמֹה אַלְקָבֵץ וְרַבִּי יוֹסֵף קָארוֹ לְאֶרֶץ יִשְׂרָאֵל וְהִתְיַשְּׁבוּ בָּעִיר הַקְּדוֹשָׁה צְפַת. בִּצְפַת חִבֵּר רַבִּי שְׁלֹמֹה אֶת הַפִּיּוּט הַמְּפֻרְסָם "לְכָה דּוֹדִי" לִכְבוֹד קַבָּלַת שַׁבָּת, שֶׁבּוֹ חָתַם אֶת שְׁמוֹ שׁ-ל-מ-ה ה-ל-ו-י בְּרָאשֵׁי הַבָּתִּים. עַד הַיּוֹם שָׁרִים אֶת הַשִּׁיר הַזֶּה בְּכָל בָּתֵּי הַכְּנֶסֶת בְּלֵיל שַׁבָּת.

מֵאָז וְעַד הַיּוֹם, בְּכָל קְהִילּוֹת יִשְׂרָאֵל בְּכָל הָעוֹלָם, נִשְׁאָרִים עֵרִים בְּלֵיל שָׁבוּעוֹת וְלוֹמְדִים תּוֹרָה – מִנְהָג קָדוֹשׁ הַנִּקְרָא "תִּקּוּן לֵיל שָׁבוּעוֹת". כָּךְ הָפַךְ לַיְלָה אֶחָד שֶׁל לִמּוּד תּוֹרָה שֶׁל כַּמָּה תַלְמִידֵי חֲכָמִים לְמִנְהָג מְפֹאָר שֶׁל כָּל עַם יִשְׂרָאֵל לְדוֹרוֹתָיו.

💡 לִמּוּד תּוֹרָה מִתּוֹךְ שִׂמְחָה וּמְסִירוּת מְשַׂמֵּחַ אֶת ה' וּמֵבִיא בְּרָכָה גְּדוֹלָה לְכָל עַם יִשְׂרָאֵל.
