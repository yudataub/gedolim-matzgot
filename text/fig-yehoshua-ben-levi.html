רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי וּמַלְאַךְ הַמָּוֶת

חַכְמֵינוּ זִכְרוֹנָם לִבְרָכָה מְסַפְּרִים כִּי הָיוּ אֲנָשִׁים בּוֹדְדִים בִּלְבַד שֶׁזָּכוּ לְהִכָּנֵס עוֹד בְּחַיֵּיהֶם אֶל גַּן עֵדֶן. אֶחָד מֵאוֹתָם אֲנָשִׁים יְחִידֵי סְגֻלָּה הָיָה הַתַּנָּא הַקָּדוֹשׁ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. הַשָּׂטָן, שֶׁהוּא גַּם מַלְאַךְ הַמָּוֶת, הָיָה חָבֵר שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי.

יוֹם אֶחָד פָּנָה רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי אֶל מַלְאַךְ הַמָּוֶת וְאָמַר לוֹ: "בּוֹא וְתַרְאֶה לִי אֶת הַמָּקוֹם שֶׁלִּי בְּגַן עֵדֶן אוֹ בְּגֵיהִנֹּם, אֶחָד מֵהַמְּקוֹמוֹת". עָנָה לוֹ מַלְאַךְ הַמָּוֶת: "הַמָּקוֹם שֶׁלְּךָ הוּא בְּגַן עֵדֶן!" אָמַר לוֹ רַבִּי יְהוֹשֻׁעַ: "טוֹב, תַּרְאֶה לִי". אָמַר מַלְאַךְ הַמָּוֶת: "אֲנִי אַרְאֶה לְךָ, טוֹב".

אָמַר לוֹ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לְמַלְאַךְ הַמָּוֶת: "אֲנִי אָבוֹא אִתְּךָ, אֲבָל אֲנִי פּוֹחֵד מִמְּךָ שֶׁמָּא בְּאֶמְצַע הַדֶּרֶךְ תַּהֲרֹג אוֹתִי. תֵּן לִי אֶת הַחֶרֶב שֶׁלְּךָ, הַחֶרֶב שֶׁבָּהּ אַתָּה הוֹרֵג אֲנָשִׁים". מַלְאַךְ הַמָּוֶת הִסְכִּים וְאָמַר לוֹ: "קַח, בְּסֵדֶר", וּמָסַר לְרַבִּי יְהוֹשֻׁעַ אֶת הַחֶרֶב שֶׁלּוֹ.

הָלְכוּ שְׁנֵיהֶם יַחְדָּו, וּכְשֶׁהִגִּיעוּ מֵרָחוֹק אֶל גַּן עֵדֶן, הֶרְאָה לוֹ מַלְאַךְ הַמָּוֶת אֶת הַמָּקוֹם וְאָמַר: "אַתָּה רוֹאֶה אֶת הַמָּקוֹם הַזֶּה? זֶה הַמָּקוֹם שֶׁלְּךָ". מִיָּד קָפַץ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי אֶל תּוֹךְ הַמָּקוֹם הַקָּדוֹשׁ בְּגַן עֵדֶן, יָשַׁב שָׁם וְנִשְׁבַּע: "אֲנִי לֹא קָם מֵהַמָּקוֹם הַזֶּה!"

כְּשֶׁרָאָה זאת מַלְאַךְ הַמָּוֶת, נִבְהַל מְאֹד, הִתְחִיל לִצְעֹק וְאָמַר לוֹ: "מָה אַתָּה עוֹשֶׂה? תַּגִּיד לִי, מָה אַתָּה עוֹשֶׂה?" מַלְאַךְ הַמָּוֶת לֹא יָדַע מָה לַעֲשׂוֹת, וְגַם בֵּית דִּין שֶׁל מַעְלָה אָמְרוּ: "אֲנַחְנוּ לֹא יוֹדְעִים מָה לַעֲשׂוֹת". עָלוּ וְהָלְכוּ אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא וְשָׁאֲלוּ אוֹתוֹ: "מָה עוֹשִׂים עִם הָאָדָם הַזֶּה?"

אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא לְמַלְאֲכֵי הַשָּׁרֵת: "תִּפְתְּחוּ אֶת הַפִּנְקָס שֶׁלּוֹ. תִּרְאוּ אִם הוּא פַּעַם נִשְׁבַּע וְעָשָׂה הַתָּרַת נְדָרִים, אָז גַּם כָּאן נַעֲשֶׂה לוֹ הַתָּרַת נְדָרִים. וְאִם פַּעַם הוּא הִחֵל אֶת דְּבָרוֹ וְלֹא עָמַד בְּמִילָתוֹ, אֲנַחְנוּ גַּם כָּאן נַחֵל אֶת דְּבָרוֹ".

חִפְּשׂוּ בְּכָל הַפִּנְקָס שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי, בָּדְקוּ שׁוּב וָשׁוּב, וְלֹא מָצְאוּ אֲפִילוּ פַּעַם אַחַת שֶׁהוּא הִחֵל אֶת דְּבָרוֹ אוֹ בִּטֵּל אֶת הַבְטָחָתוֹ!

אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא: "כָּכָה יִשָּׁאֵר!" כְּפִי שֶׁנֶּאֱמַר בַּתּוֹרָה: "לֹא יַחֵל דְּבָרוֹ", אִם אָדָם אֵינוֹ מְחַלֵּל אֶת דְּבָרוֹ, אָז "כְּכָל הַיֹּצֵא מִפִּיו יַעֲשֶׂה". מֵאַחַר שֶׁרַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי מְעוֹלָם לֹא חִלֵּל אֶת דְּבָרוֹ, אֲפִילוּ אִם יֵשֵׁב בְּגַן עֵדֶן, כָּכָה יֵעָשֶׂה הַדָּבָר הַזֶּה כֻּלּוֹ, וְהוּא נִשְׁאַר לָשֶׁבֶת בְּגַן עֵדֶן.

💡 כְּשֶׁאָדָם שׁוֹמֵר עַל דְּבָרוֹ וְאֵינוֹ מְחַלֵּל אֶת הַבְטָחוֹתָיו, הַקָּדוֹשׁ בָּרוּךְ הוּא מְקַיֵּם אֶת כָּל מוֹצָא פִּיו.
