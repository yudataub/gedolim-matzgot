פְּסָקִים שֶׁל אֲהָבָה וּמִסִּירוּת נֶפֶשׁ

בִּתְקוּפַת מִלְחֶמֶת הַשִּׁחְרוּר וַהֲקָמַת מְדִינַת יִשְׂרָאֵל, הִנְהִיגוּ אֶת הָעָם הָרַבָּנִים הָרָאשִׁיִּים, הָרַב יִצְחָק הֶרְצוֹג וְהָרַב עֻזִּיאֵל. הֵם חִבְּרוּ אֶת הַתְּפִלָּה לִשְׁלוֹם הַמְּדִינָה, שֶׁבָּהּ מְבָרְכִים אֶת מְדִינַת יִשְׂרָאֵל וְקוֹרְאִים לָהּ "רֵאשִׁית צְמִיחַת גְּאֻלָּתֵנוּ". בְּיָמִים קָשִׁים אֵלֶּה שֶׁל מִלְחָמָה וּסְכָנָה, נִדְרְשׁוּ הָרַבָּנִים לַעֲנוֹת עַל שְׁאֵלוֹת הֲלָכָתִיּוֹת כִּבְדַת מַשְׁקָל הַנּוֹגְעוֹת לְפִקּוּחַ נֶפֶשׁ וּלְמִלְחֶמֶת מִצְוָה, כְּדֵי לְהַצִּיל חַיִּים וּלְחַזֵּק אֶת הַלּוֹחֲמִים.

שְׁאֵלָה אַחַת הִגִּיעָה מִכְּפַר עֶצְיוֹן. הַלּוֹחֲמִים בְּגּוּשׁ עֶצְיוֹן שָׁלְחוּ מִכְתָּב וְשָׁאֲלוּ הַאִם מֻתָּר לִפְרֹק בְּשַׁבָּת שַׁיָּרוֹת שֶׁמְּבִיאוֹת לָהֶם אֹכֶל, נֶשֶׁק וְדֶלֶק. הָרַב הֶרְצוֹג עָנָה לָהֶם מִיָּד: "זֶה לֹא רַק פִּקּוּחַ נֶפֶשׁ, זֶה מִלְחֶמֶת מִצְוָה, וּבְמִלְחֶמֶת מִצְוָה עוֹשִׂים מִיָּד הַכֹּל, בְּלִי לִשְׁאֹל שְׁאֵלוֹת, בָּרוּר שֶׁזֶּה מֻתָּר". שְׁאֵלָה נוֹסֶפֶת הִגִּיעָה מֵהָרַב שֶׁל טְבֶרְיָה. תּוֹשָׁבֵי טְבֶרְיָה בִּקְשׁוּ לְהָקִים גָּדֵר סְבִיב הַשְּׁכוּנָה בְּשַׁבָּת מִחֲשָׁשׁ שֶׁתִּהְיֶה הַתְקָפָה. הָרַב שֶׁל טְבֶרְיָה שָׁאַל הַאִם מֻתָּר לְהַמְשִׁיךְ בָּעֲבוֹדָה בְּשַׁבָּת, וְהָרַב הֶרְצוֹג הֵשִׁיב: "בְּוַדַּאי שֶׁמֻּתָּר, לֹא רַק שֶׁמֻּתָּר - זוֹ מִלְחֶמֶת מִצְוָה, וְצָרִיךְ לַעֲשׂוֹת אֶת זֶה בְּמֶרֶץ, אֲפִילוּ בְּשַׁבָּת!".

נְשִׂיא הַמְּדִינָה כַּיּוֹם, יִצְחָק הֶרְצוֹג, נֶכְדּוֹ שֶׁל הָרַב הֶרְצוֹג, סִפֵּר עַל מִכְתָּב מְעַנְיֵן שֶׁמָּצָא מֵחַיָּל בְּרִיטִי. הַחַיָּל הַזֶּה הָיָה מִתְנַדֵּב שֶׁהֶחְלִיט לְהִשָּׁאֵר בָּאָרֶץ לְאַחַר הֲקָמַת הַמְּדִינָה כְּדֵי לַעֲזֹר לַיְּהוּדִים. הַחַיָּל מָצָא מַחְסְנֵי מָזוֹן עֲנָקִיִּים בְּאֶחָד הַבְּסִיסִים הַבְּרִיטִיִּים וְרָצָה לְחַלֵּק אֶת הַמָּזוֹן לַיְּהוּדִים, אַךְ לְצַעֲרוֹ גִּלָּה שֶׁהָאֹכֶל אֵינוֹ כָּשֵׁר. הַחַיָּל הִגִּיעַ לְבֵיתוֹ שֶׁל הָרַב הֶרְצוֹג וְשָׁאַל כֵּיצַד לִנְהֹג. הָרַב הֶרְצוֹג פָּסַק לוֹ: "זֶה פִּקּוּחַ נֶפֶשׁ, תֵּן אֶת זֶה לְבָתֵי חֹלִים, מֻתָּר לֶאֱכֹל אֹכֶל לֹא כָּשֵׁר עַכְשָׁו". אַךְ הָרַב הֶרְצוֹג הוֹסִיף וְאָמַר: "אֲנִי לֹא רוֹצֶה לְהַכְרִיעַ לְבַדִּי, תֵּלֵךְ בְּבַקָּשָׁה לָרַב עֻזִּיאֵל, תִּשְׁאַל אוֹתוֹ אֶת אוֹתָהּ שְׁאֵלָה, וְאַל תַּגִּיד לוֹ מָה אָמַרְתִּי". הַחַיָּל הָלַךְ אֶל הָרַב עֻזִּיאֵל, וְהָרַב עֻזִּיאֵל עָנָה לוֹ בְּדִיּוּק אֶת אוֹתָם הַדְּבָרִים: "תִּתֵּן אֶת זֶה לְבָתֵי חֹלִים, זֶה פִּקּוּחַ נֶפֶשׁ".

בַּקְּרָב הָאַחֲרוֹן בִּכְפַר עֶצְיוֹן, בְּדָלֶת בְּאִיָּר תַּשִּׁ"ח, נָפְלוּ מֵאָה עֶשְׂרִים וְשִׁבְעָה לוֹחֲמִים וְלוֹחֲמוֹת. הַהוֹדָעָה הָאַחֲרוֹנָה מִכִּנּוּי הָאֵזוֹר בְּמַכְשִׁירֵי הַקֶּשֶׁר הָיְתָה "מַלְכָּה נָפְלָה". לְמָחֳרָת, בְּהֵא בְּאִיָּר, הִכְרִיז דָּוִד בֶּן גּוּרִיּוֹן עַל הֲקָמַת מְדִינַת יִשְׂרָאֵל. כְּשֶׁשָּׁמַע הָרַב הֶרְצוֹג עַל שְׁנֵי הָאֵרוּעִים הָאֵלֶּה שֶׁקָּרוּ בִּסְמִיכוּת, אָמַר: "מַלְכָּה נָפְלָה, מַלְכוּת קָמָה".

💡 פִּקּוּחַ נֶפֶשׁ וְאַהֲבַת יִשְׂרָאֵל דּוֹחִים הַכֹּל, וּמִתּוֹךְ הַקְּשָׁיִים צֹמַחַת הַגְּאֻלָּה.
