הָרַב שֶׁצָּמָא לַתּוֹרָה

יֵשׁ יָמִים מְיֻחָדִים בַּשָּׁנָה שֶׁבָּהֶם אֲנַחְנוּ מִתְכַּנְסִים וּמִזְכִּירִים אֶת הַהִיסְטוֹרְיָה שֶׁל עַם יִשְׂרָאֵל. אֶחָד מֵהַיָּמִים הָאֵלֶּה הוּא יוֹם תַּשְׁעָה בְּאָב. יוֹם תַּשְׁעָה בְּאָב הוּא יוֹם צוֹם מְיֻחָד, שֶׁבּוֹ אֲנַחְנוּ לֹא אוֹכְלִים וְלֹא שׁוֹתִים בְּמֶשֶׁךְ יְמָמָה שְׁלֵמָה. אַךְ בְּיוֹם זֶה יֵשׁ גַּם הַנְהָגָה מְיֻחֶדֶת מְאֹד וְשׁוֹנָה מִכָּל יְמוֹת הַשָּׁנָה: זֶהוּ הַיּוֹם הַיְּחִידִי בַּשָּׁנָה שֶׁבּוֹ אָסוּר לִלְמֹד תּוֹרָה. בְּכָל יְמוֹת הַשָּׁנָה הַיְּהוּדִים אוֹהֲבִים מְאֹד לִלְמֹד אֶת תּוֹרַת הַשֵּׁם, לִלְמֹד גְּמָרָא, מִשְׁנָה וּפְסוּקִים, וְלַעֲסֹק בַּתּוֹרָה בְּשִׂמְחָה רַבָּה. אֲבָל בְּתַשְׁעָה בְּאָב, בִּגְלַל הָאֵבֶל וְהַצַּעַר, אָסוּר לִלְמֹד תּוֹרָה בְּמֶשֶׁךְ יְמָמָה תְּמִימָה.

וְהִנֵּה, כַּאֲשֶׁר מַגִּיעַ מוֹצָאֵי תַּשְׁעָה בְּאָב, הַצּוֹם הָאָרֹךְ מִסְתַּיֵּם. בְּרֶגַע זֶה, כָּל הָאֲנָשִׁים מַרְגִּישִׁים רְעֵבִים וּצְמֵאִים מְאֹד לְמַיִם וּלְמַאֲכָל. כֻּלָּם רָצִים מְהֵרָה הַבַּיְתָה כְּדֵי לֶאֱכֹל, לִשְׁתּוֹת וְלִשְׁבֹּר אֶת הַצּוֹם בְּאֹכֶל טָעִים. כָּל אָדָם רוֹצֶה לְהַכְנִיס מַשֶּׁהוּ לַפֶּה אַחֲרֵי יְמָמָה שְׁלֵמָה שֶׁל צוֹם.

גַּם הָרַב הַגָּדוֹל, הָרַב אַהֲרֹן לִיכְטֶנְשְׁטַיְן זֵכֶר צַדִּיק לִבְרָכָה, הָיָה רָץ הַבַּיְתָה בְּמוֹצָאֵי תַּשְׁעָה בְּאָב יַחַד עִם כָּל הָאֲנָשִׁים. אַךְ הָרַב אַהֲרֹן לִיכְטֶנְשְׁטַיְן לֹא רָץ הַבַּיְתָה כְּדֵי לֶאֱכֹל אֲרוּחָה. הוּא רָץ הַבַּיְתָה מִסִּבָּה אַחֶרֶת לַגְמָרֵי! הָרַב רָץ הַבַּיְתָה כְּדֵי לִפְתֹּחַ אֶת סֵפֶר הַגְּמָרָא וְלִלְמֹד דַּף גְּמָרָא.

כֵּיצַד זֶה יִתָּכֵן? מַדּוּעַ הָרַב אַהֲרֹן לִיכְטֶנְשְׁטַיְן פָּנָה קֹדֶם כָּל לְלִמּוּד הַגְּמָרָא בִּמְקוֹם לָלֶכֶת לֶאֱכֹל? הַתְּשׁוּבָה הִיא שֶׁאַחֲרֵי יְמָמָה שְׁלֵמָה שֶׁבָּהּ הָיָה אָסוּר לִלְמֹד תּוֹרָה, הָרַב הָיָה כָּל כָּךְ צָמָא לַתּוֹרָה! הַצִּמָּאוֹן שֶׁל הָרַב לְדִבְרֵי הַתּוֹרָה הַקְּדוֹשָׁה הָיָה גָּדוֹל וְחָזָק יוֹתֵר מִכָּל רָעָב אוֹ צָמָא לְמַאֲכָל וּלְמַשְׁקֶה. בְּמֶשֶׁךְ יְמָמָה שְׁלֵמָה נִמְנַע מִמֶּנּוּ הַדָּבָר הַחָשׁוּב לוֹ מִכֹּל – לִמּוּד הַתּוֹרָה, וְעַכְשָׁיו הוּא לֹא יָכֹל לַחֲכּוֹת אַף רֶגַע אֶחָד נוֹסָף.

לָכֵן, קֹדֶם כָּל הָיָה הָרַב הוֹלֵךְ וְלוֹמֵד תּוֹרָה, וּבְכָךְ הָיָה מַרְוֶה אֶת הַצִּמָּאוֹן הָרוּחָנִי שֶׁל נַפְשׁוֹ. הוּא הֶעֱדִיף לִשְׁבֹּר קֹדֶם כָּל אֶת הַצּוֹם שֶׁל הַתּוֹרָה, וְרַק אַחַר כָּךְ הָיָה הוֹלֵךְ, כְּמוֹ כָּל שְׁאָר הָאֲנָשִׁים, לִשְׁבֹּר אֶת הַצּוֹם בְּאֹכֶל וּבְמַשְׁקֶה. הַהַנְהָגָה הַמְּפֹאֶרֶת הַזֹּאת הֶרְאֲתָה לְכֻלָּם עַד כַּמָּה הָיְתָה הַתּוֹרָה יְקָרָה וַאֲהוּבָה עַל הָרַב אַהֲרֹן לִיכְטֶנְשְׁטַיְן.

💡 כַּאֲשֶׁר אֲנַחְנוּ אוֹהֲבִים אֶת הַתּוֹרָה בְּכָל לִבֵּנוּ, לִמּוּד הַתּוֹרָה הוֹפֵךְ לַדָּבָר הַחָשׁוּב וְהַמְּשַׂמֵּחַ בְּיוֹתֵר בְּחַיֵּינוּ.
