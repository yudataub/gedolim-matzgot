הַסִּימָנִים הַסּוֹדִיִּים שֶׁל רָחֵל אִמֵּנוּ

יַעֲקֹב אָבִינוּ אָהַב אֶת רָחֵל מְאֹד, וְעָבַד בִּשְׁבִילָהּ אֵצֶל אָבִיהָ לָבָן שֶׁבַע שָׁנִים תְּמִימֹות. רָחֵל יָדְעָה כִּי אָבִיהָ לָבָן הוּא אָדָם עַרְמוּמִי, וְחָשְׁשָׁה שֶׁהוּא יְנַסֶּה לְהַחְלִיף אוֹתָהּ בַּאֲחוֹתָהּ הַגְּדוֹלָה לֵאָה. לָכֵן, קָבְעוּ יַעֲקֹב וְרָחֵל מְרָאשׁ מִסְפַּר סִימָנִים סוֹדִיִּים מְיֻחָדִים, כְּדֵי שֶׁיַּעֲקֹב יוּכַל לְהַכִּיר אוֹתָהּ בְּלֵיל הַחֲתוּנָה וְלָדַעַת בְּבֵרוּר כִּי זוֹהִי רָחֵל.

כַּאֲשֶׁר הִגִּיעַ לֵיל הַחֲתוּנָה, פָּעַל לָבָן כְּפִי שֶׁחָשְׁשׁוּ, וְהֶחְלִיט לְהַכְנִיס אֶת לֵאָה תַּחַת רָחֵל. בְּאוֹתוֹ רֶגַע קָשֶׁה, מָלְאָה רָחֵל רַחֲמִים גְּדוֹלִים עַל אֲחוֹתָהּ. הִיא חָשְׁבָה עַל הַבּוּשָׁה וְהַחֶרְפָּה הַנּוֹרָאָה שֶׁתִּהְיֶה לְלֵאָה כַּאֲשֶׁר יַעֲקֹב יְגַלֶּה שֶׁזּוֹ אֵינָהּ רָחֵל, אֶלָּא אֲחוֹתָהּ. רָחֵל הֶחְלִיטָה לְוַתֵּר עַל רְצוֹנָהּ הַפְּרָטִי וְעַל שִׂמְחָתָהּ הָאִישִׁית, כְּדֵי לִמְנֹעַ מֵאֲחוֹתָהּ צַעַר וּבֹשֶׁת פָּנִים.

רָחֵל הָלְכָה אֶל לֵאָה וּמָסְרָה לָהּ אֶת כָּל הַסִּימָנִים הַסּוֹדִיִּים שֶׁקָּבְעָה עִם יַעֲקֹב, כְּדֵי שֶׁיַּעֲקֹב יַחְשֹׁב שֶׁהוּא מְדַבֵּר עִם רָחֵל. וְלֹא עוֹד, אֶלָּא שֶׁרָחֵל נִכְנְסָה וְהִתְחַבְּאָה תַּחַת הַמִּטָּה בְּאוֹתוֹ לַיְלָה. כַּאֲשֶׁר דִּבֵּר יַעֲקֹב עִם לֵאָה, שָׁתְקָה לֵאָה, וְרָחֵל עָנְתָה לוֹ בְּקוֹלָהּ מִתַּחַת לַמִּטָּה, כְּדֵי שֶׁיַּעֲקֹב לֹא יַכִּיר אֶת קוֹלָהּ שֶׁל לֵאָה וְלֵאָה לֹא תִּתְבַּיֵּשׁ. בַּבֹּקֶר, כַּאֲשֶׁר הֵאִיר הַיּוֹם, הִתְבָּרֵר לְיַעֲקֹב כִּי זוֹהִי לֵאָה, אַךְ לֵאָה נִצְּלָה מִבּוּשָׁה וּמִחֶרְפָּה בִּזְכוּת וַתְּרָנוּתָהּ הַמֻּפְלָאָה שֶׁל רָחֵל.

עָבְרוּ שָׁנִים רַבּוֹת. כַּאֲשֶׁר חָרַב בֵּית הַמִּקְדָּשׁ וּבְנֵי יִשְׂרָאֵל נִלְקְחוּ לַגָּלוּת, הָיוּ אֲבוֹת הָעוֹלָם – אַבְרָהָם, יִצְחָק, יַעֲקֹב וּמֹשֶׁה רַבֵּנוּ – בּוֹכִים וּמִתְפַּלְלִים לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא, וּמְבַקְּשִׁים שֶׁיְּרַחֵם עַל עַמּוֹ. אַךְ הַגְּזֵרָה הָיְתָה קָשָׁה. אָז קָמָה רָחֵל אִמֵּנוּ וְעָמְדָה לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא. הִיא סִפְּרָה אֵיךְ עָשְׂתָה חֶסֶד עִם אֲחוֹתָהּ, אֵיךְ לֹא קִנְּאָה בָּהּ, וְאֵיךְ וִתְּרָה וּמָסְרָה לָהּ אֶת הַסִּימָנִים כְּדֵי שֶׁלֹּא תֵּצֵא לְחֶרְפָּה. רָחֵל אָמְרָה: "אִם אֲנִי, שֶׁאֲנִי בָּשָׂר וָדָם, לֹא קִנֵּאתִי וְרִחַמְתִּי עַל אֲחוֹתִי, אֵיךְ אַתָּה מֶלֶךְ רַחֲמָן אֵינְךָ מְרַחֵם עַל בָּנֶיךָ?"

מִיָּד נִתְגַּלְגְּלוּ רַחֲמָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, וְהוּא אָמַר לָהּ: "בִּשְׁבִילֵךְ, רָחֵל, אֲנִי מַחֲזִיר אֶת יִשְׂרָאֵל לִמְקוֹמָן." הַקָּדוֹשׁ בָּרוּךְ הוּא הִבְטִיחַ לָהּ שֶׁהַדְּמָעוֹת שֶׁלָּהּ לֹא יֵרְדוּ לְרִיק, וְכִי יֵשׁ שָׂכָר לִפְעֻלָּתָהּ, וּבְנֵי יִשְׂרָאֵל יַחְזְרוּ מִן הַגָּלוּת אֶל אַרְצָם.

💡 כַּאֲשֶׁר אָנוּ מְוַתְּרִים מֵהַלֵּב וְגוֹמְלִים חֶסֶד עִם אֲחֵרִים, הַקָּדוֹשׁ בָּרוּךְ הוּא מִתְמַלֵּא רַחֲמִים וּמְבִיא לָנוּ שָׂכָר וּבְרָכָה גְּדוֹלָה.
