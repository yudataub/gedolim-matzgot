תַּעֲלוּמַת הָעַמּוּד הָרִאשׁוֹן

כַּאֲשֶׁר תַּלְמִידִים פּוֹתְחִים אֶת סֵפֶר הַגְּמָרָא הַקָּדוֹשׁ, הֵם פּוֹגְשִׁים מַעֲשֶׂה מֻפְלָא שֶׁל מְסִירוּת נֶפֶשׁ. אָנוּ יוֹדְעִים כִּי חַכְמֵי הַתַּנָּאִים שָׁנוּ אֶת הַמִּשְׁנָה, וְאַחֲרֵיהֶם בֵּאֲרוּ חַכְמֵי הָאֲמוֹרָאִים אֶת כָּל דִּבְרֵיהֶם. לְאַחַר שָׁנִים רַבּוֹת שֶׁל עָמָל וְלִמּוּד, סִדְּרוּ גְּדוֹלֵי הַדּוֹר רַב אַשִׁי וְרָבִינָא אֶת הַתַּלְמוּד הַבַּבְלִי וְחָתְמוּ אוֹתוֹ. חֲכָמֵינוּ לִמְּדוּ כִּי בִּפְטִירָתָם נִפְסְקָה הַהוֹרָאָה, וּמֵאוֹתוֹ זְמַן אֵין אִישׁ רַשַּׁאי לְהוֹסִיף דִּינִים חֲדָשִׁים אוֹ לַחֲלֹק עֲלֵיהֶם. אֲבָל כָּאן מַתְחִילָה תַּעֲלוּמָה מְרַתֶּקֶת: אִם רַב אַשִׁי וְרָבִינָא חָתְמוּ אֶת כָּל הַתַּלְמוּד, מִי כָּתַב אֶת הָעַמּוּד הָרִאשׁוֹן שֶׁל מַסֶּכֶת קִדּוּשִׁין?

כַּאֲשֶׁר פּוֹתְחִים אֶת מַסֶּכֶת קִדּוּשִׁין בַּדַּף הָרִאשׁוֹן, דַּף ב' עַמּוּד א', רוֹאִים אֶת הַמִּשְׁנָה שֶׁפּוֹתַחַת בַּמִּלִּים 'הָאִשָּׁה נִקְנֵית'. מִיָּד לְאַחַר מִכֵּן שׁוֹאֶלֶת הַגְּמָרָא קֻשְׁיוֹת חֲכָמוֹת, כְּגוֹן מַדּוּעַ נֶאֱמַר 'נִקְנֵית' וְלֹא 'הָאִישׁ קוֹנֶה', וּמְבָאֶרֶת אֶת כָּל הַסְּפֵקוֹת עַד דַּף ג' שָׁם נֶאֱמַר 'בְּכֶסֶף מְנָא לַן'. רַבִּים חָשְׁבוּ שֶׁרַב אַשִׁי וְרָבִינָא כָּתְבוּ אֶת הַדְּבָרִים הַלָּלוּ, אַךְ הָאֱמֶת שׁוֹנָה. רַב שְׁרִירָא גָּאוֹן מְגַלֶּה בְּאִגַּרְתּוֹ שֶׁקַּיֶּמֶת קַבָּלָה עַתִּיקָה מֵהַדּוֹרוֹת הָרִאשׁוֹנִים, וּמְסֻפָּר שֶׁכָּל הַסּוּגְיָה הַזּוֹ, עַד הַמִּלִּים 'בְּכֶסֶף מְנָא הָנֵי מִילֵּי', נִקְבְּעָה וְסֻדְּרָה בִּידֵי חֲכָמִים שֶׁבָּאוּ אַחֲרֵיהֶם, וְהֵם נִקְרָאִים 'רַבָּנַן סָבוֹרָאֵי'.

מִי הָיוּ חַכְמֵי הַסָּבוֹרָאִים? אֵלּוּ הָיוּ גְּדוֹלֵי יִשְׂרָאֵל שֶׁחָיוּ בְּבָבֶל לְאַחַר הָאֲמוֹרָאִים, בִּתְקוּפָה שֶׁל גְּזֵרוֹת קָשׁוֹת מִצַּד מַלְכֵי פָּרַס. הַשֵּׁם 'סָבוֹרָאִים' בָּא מֵהַמִּלָּה 'סְבָרָא', שֶׁפֵּרוּשָׁהּ הַסְבָּרָה, הֲבָנָה עֲמֻקָּה וְדִיּוּק. הַסָּבוֹרָאִים יָדְעוּ שֶׁאֵין לָהֶם רְשׁוּת לְחַדֵּשׁ דִּינִים מִשֶּׁלָּהֶם, אַךְ עֲמָלָם הַקָּדוֹשׁ הָיָה לְפָרֵשׁ וּלְהַכְרִיעַ בְּכָל מַה שֶׁנִּשְׁאַר תָּלוּי וְעוֹמֵד בְּסָפֵק. בְּאַהֲבַת תּוֹרָה גְּדוֹלָה, הֵם בֵּאֲרוּ אֶת הַתַּלְמוּד, הוֹסִיפוּ קֻשְׁיוֹת וְתֵרוּצִים, וְקָבְעוּ אֶת הַנֹּסַח הַסּוֹפִי שֶׁל הַגְּמָרָא.

הַדָּבָר הַמַּרְשִׁים בְּיוֹתֵר בְּמַעֲשֵׂיהֶם הוּא מִדַּת הָעֲנָוָה שֶׁלָּהֶם. חֲכָמִים נוֹדָעִים כְּמוֹ רַב עֵינָא, רַב סִימוֹנָא, רַב יוֹסֵף וַחֲבֵרֵיהֶם לָמְדוּ וְטָרְחוּ בְּלֹא הֶרְפֵּף, אֲבָל הֵם בָּחֲרוּ שֶׁלֹּא לַחְתֹּם אֶת שְׁמָם עַל דַּף הַגְּמָרָא. הֵם לֹא בִּקְשׁוּ כָּבוֹד אוֹ תְּהִלָּה, אֶלָּא עָשׂוּ הַכֹּל מֵאַהֲבַת הַתּוֹרָה וְלִשְׁמָהּ, כְּדֵי שֶׁכָּל לוֹמֵד וְכָל יֶלֶד יוּכַל לְהָבִין אֶת הַלִּמּוּד בְּבֵאוּר בָּהִיר. כָּךְ נִפְתְּרָה הַתַּעֲלוּמָה: הָעַמּוּד הָרִאשׁוֹן נִכְתַּב בִּידֵי חֲכָמִים צַדִּיקִים שֶׁוִּתְּרוּ עַל פִּרְסוּם שְׁמָם כְּדֵי לְהָאִיר אֶת הַתּוֹרָה לְכָל הַדּוֹרוֹת.

💡 כַּאֲשֶׁר אָנוּ עוֹשִׂים מַעֲשֶׂה טוֹב אוֹ לוֹמְדִים בַּעֲנָוָה מִבְּלִי לְבַקֵּשׁ שֶׁיְּשַׁבְּחוּ אוֹתָנוּ, אָנוּ מְאִירִים אֶת הָעוֹלָם וּמַמְשִׁיכִים אֶת דַּרְכָּם שֶׁל חַכְמֵי יִשְׂרָאֵל.
