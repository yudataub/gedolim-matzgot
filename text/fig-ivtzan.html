אִבְצָן הוּא בֹּעַז וּבִרְכַּת הַשָּׁלוֹם בַּשָּׂדֶה

בְּסֵפֶר שׁוֹפְטִים אָנוּ לוֹמְדִים עַל מַנְהִיג וְשׁוֹפֵט חָשׁוּב שֶׁשְּׁמוֹ אִבְצָן, אֲשֶׁר גָּר בָּעִיר בֵּית לֶחֶם. אִבְצָן שָׁפַט אֶת עַם יִשְׂרָאֵל מֶשֶׁךְ שֶׁבַע שָׁנִים תְּמִימוֹת, אַחֲרֵי יִפְתָּח הַשּׁוֹפֵט. הַפְּסוּקִים מְסַפְּרִים לָנוּ כִּי לְאִבְצָן הָיוּ שְׁלֹשִׁים בָּנִים וְגַם שְׁלֹשִׁים בָּנוֹת. אִבְצָן חִתֵּן אֶת שְׁלֹשִׁים בְּנֹתָיו וְשִׁלַּח אוֹתָן לְהִתְחַתֵּן מִחוּץ לַמִּשְׁפָּחָה, וְכֵן הֵבִיא שְׁלֹשִׁים בָּנוֹת מִן הַחוּץ עֲבוּר בָּנָיו. כַּאֲשֶׁר נִפְטַר אִבְצָן הַשּׁוֹפֵט, הוּא הוּבַא לִקְבוּרָה בְּעִירוֹ בֵּית לֶחֶם.

חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה לִמְּדוּ אוֹתָנוּ בַּתַּלְמוּד הַבַּבְלִי, בְּמַסֶּכֶת בָּבָא בַּתְרָא, גִּלּוּי מְפֹרָשׁ: אִבְצָן הַשּׁוֹפֵט הוּא לְמַעֲשֶׂה בֹּעַז הַמֻּכָּר לָנוּ מִמְּגִלַּת רוּת. רַבָּה בַּר רַב הוּנָא אָמַר בְּשֵׁם רַב, שֶׁאִבְצָן הוּא בֹּעַז, וְסִפֵּר שֶׁבֹּעַז עָשָׂה מֵאָה וְעֶשְׂרִים מִשְׁתָּאוֹת וַחֲגִיגוֹת גְּדוֹלוֹת עֲבוּר בָּנָיו וּבְנֹתָיו לִכְבוֹד נִשּׂוּאֵיהֶם.

בִּמְגִלַּת רוּת מְסֻפָּר עַל מַעֲשֵׂהוּ הַנָּאֶה שֶׁל בֹּעַז בַּשָּׂדֶה. בֹּעַז יָצָא מֵעִירוֹ בֵּית לֶחֶם וּבָא אֶל הַשָּׂדֶה שֶׁלּוֹ, שָׁם עָבְדוּ פּוֹעֲלִים קוֹצְרִים שֶׁאָסְפוּ אֶת הַתְּבוּאָה. כַּאֲשֶׁר בָּא בֹּעַז אֶל הַקּוֹצְרִים, הוּא בֵּרֵךְ אוֹתָם בְּאַהֲבָה וְאָמַר לָהֶם: "הַשֵּׁם עִמָּכֶם!". הַפּוֹעֲלִים הַקּוֹצְרִים שָׂמְחוּ מְאֹד לִשְׁמֹעַ אֶת בִּרְכָתוֹ שֶׁל בֹּעַז, וְהֵם עָנוּ לוֹ בְּכָבוֹד רָב: "יְבָרֶכְךָ הַשֵּׁם!".

מִמַּעֲשֵׂהוּ שֶׁל בֹּעַז בַּשָּׂדֶה לָמְדוּ חֲכָמֵינוּ לֶקַח חָשׁוּב מְאֹד עַל אֹפֶן הַבְּרָכָה וְהַשָּׁלוֹם. בַּמִּשְׁנָה בְּמַסֶּכֶת בְּרָכוֹת מְבֹאָר כִּי חֲכָמִים הִתְקִינוּ תַּקָּנָה מְיֻחֶדֶת עַל פִּי מַעֲשֵׂה בֹּעַז. הֵם לִמְּדוּ שֶׁיְּהֵא אָדָם שׁוֹאֵל בִּשְׁלוֹם חֲבֵרוֹ תוֹךְ הַזְכָּרַת שֵׁם הַשֵּׁם, כְּפִי שֶׁבֹּעַז בֵּרֵךְ אֶת הַקּוֹצְרִים. חֲכָמִים הֵבִיאוּ לְכָךְ מְקוֹרוֹת נוֹסָפִים מִן הַתָּנָ"ךְ, וּבָהֶם דִּבְרֵי הַמַּלְאָךְ לְגִדְעוֹן: "הַשֵּׁם עִמְּךָ גִּבּוֹר הֶחָיִל", הַפָּסוּק מִסֵּפֶר מִשְׁלֵי: "וְאַל תָּבוּז כִּי זָקְנָה אִמֶּךָ", וְהַפָּסוּק מִסֵּפֶר תְּהִלִּים: "עֵת לַעֲשׂוֹת לַהַ' הֵפֵרוּ תּוֹרָתֶךָ". רַבִּי נָתָן מַסְבִּיר גַּם הוּא אֶת הַפָּסוּק הַזֶּה מִתְּהִלִּים עַל חֲשִׁיבוּת הָעֲשִׂיָּה לְמַעַן הַשֵּׁם.

מִכָּל הַמְּקוֹרוֹת הַלָּלוּ אָנוּ רוֹאִים אֶת גְּדֻלָּתוֹ שֶׁל אִבְצָן, הוּא בֹּעַז הַשּׁוֹפֵט. הוּא הָיָה מַנְהִיג מְבֹרָךְ שֶׁדָּאַג לְמִשְׁפַּחְתּוֹ, שָׁפַט אֶת הָעָם בְּצֶדֶק מֶשֶׁךְ שֶׁבַע שָׁנִים, וּבֵרֵךְ אֶת הָעוֹבְדִים בַּשָּׂדֶה בְּשֵׁם הַשֵּׁם כְּדֵי לְהַרְבּוֹת אַהֲבָה, יִרְאַת שָׁמַיִם וְשָׁלוֹם בְּכָל מָקוֹם.

💡 נִלְמַד מִבֹּעַז לְבָרֵךְ אֶת חֲבֵרֵינוּ בְּסֵבֶר פָּנִים יָפוֹת וּלְהַכְנִיס אֶת שֵׁם הַשֵּׁם לְשִׂיחַת הַשָּׁלוֹם שֶׁלָּנוּ.
