הַלָּשׁוֹן הַטּוֹבָה וְהַלָּשׁוֹן הָרָעָה

לִפְנֵי שָׁנִים רַבּוֹת חַי חָכָם גָּדוֹל שֶׁנִּקְרָא רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל. רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל הָיָה נְשִׂיאָם שֶׁל כָּל יִשְׂרָאֵל, וְכָל הָעָם כִּבֵּד אוֹתוֹ מְאֹד וְלָמַד מִמֶּנּוּ תּוֹרָה וְחָכְמָה. בְּבֵיתוֹ שֶׁל רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל חַי גַּם טַבִּי, שֶׁהָיָה הָעֶבֶד הַטּוֹב שֶׁלּוֹ. טַבִּי הָיָה עֶבֶד נֶאֱמָן, מָסוּר וְחָכָם מְאֹד.

יוֹם אֶחָד קָרָא רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל לְטַבִּי עַבְדּוֹ, כְּדֵי לָתֵת לוֹ מְשִׂימָה. הוּא אָמַר לוֹ: טַבִּי, צֵא אֶל הַשּׁוּק וּתְבִיא לִי דְּבַר מַאֲכָל מַמָּשׁ מַמָּשׁ טוֹב, מְשֻׁבָּח. טַבִּי שָׁמַע אֶת דִּבְרֵי רַבּוֹ וּמִיָּד יָצָא לַדֶּרֶךְ. הוּא הָלַךְ אֶל הַשּׁוּק, נִכְנַס אֶל חֲנוּת הַבָּשָׂר אֵצֶל הַקַּצָּב, קָנָה שָׁם לָשׁוֹן שֶׁל פָּרָה, וְחָזַר וְהֵבִיא אוֹתָהּ אֶל רַבּוֹ. כַּאֲשֶׁר רָאָה רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אֶת מַה שֶּׁהֵבִיא טַבִּי, הוּא שִׁבַּח אוֹתוֹ מְאֹד וְאָמַר לוֹ: יָפֶה, כָּל הַכָּבוֹד, בֶּאֱמָת לָשׁוֹן הִיא בָּשָׂר מַמָּשׁ מַמָּשׁ מְשֻׁבָּח.

לְאַחַר מִכֵּן קָרָא רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל שׁוּב לְטַבִּי עַבְדּוֹ וְאָמַר לוֹ: אֲבָל עַכְשָׁיו, טַבִּי, מְשִׂימָה שְׁנִיָּה. צֵא שׁוּב אֶל הַשּׁוּק וּתְבִיא לִי מִשָּׁם דְּבַר מַאֲכָל גָּרוּעַ. טַבִּי הוּפְתַע מֵהַבַּקָּשָׁה הַחֲדָשָׁה, אֲבָל הוּא הֵבִין שֶׁרַבּוֹ רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל רוֹצֶה לְלַמֵּד אֶת תַּלְמִידָיו שִׁעוּר חָשׁוּב מְאֹד. לָכֵן יָצָא טַבִּי שׁוּב מִיָּד וְרָץ אֶל הַשּׁוּק, אֶל חֲנוּת הַבָּשָׂר שֶׁל הַקַּצָּב.

כַּאֲשֶׁר חָזַר טַבִּי מֵהַשּׁוּק אֶל רַבּוֹ, הוּא הֵבִיא לוֹ שׁוּב לָשׁוֹן שֶׁל פָּרָה. רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל תָּמַהּ מְאֹד עַל כָּךְ וְשָׁאַל אוֹתוֹ: הֵבֵאתָ לָשׁוֹן שֶׁל פָּרָה שׁוּב? קוֹדֵם שָׁלַחְתִּי אוֹתְךָ לְהָבִיא דְּבַר מַאֲכָל מְשֻׁבָּח וְהֵבֵאתָ לָשׁוֹן, וְעַכְשָׁיו אָמַרְתִּי לְךָ לְהָבִיא דְּבַר מַאֲכָל גָּרוּעַ וְהֵבֵאתָ שׁוּב לָשׁוֹן? אֵיךְ זֶה יָכוֹל לִהְיוֹת?

עָנָה לוֹ טַבִּי עַבְדּוֹ בְּחָכְמָה: רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל רַבִּי, אַתָּה לִמַּדְתָּ אוֹתִי שֶׁאִם הַלָּשׁוֹן הִיא טוֹבָה – אֵין טוֹבָה מִמֶּנָּה, וְאִם הִיא גָּרוּעָה – אֵין גָּרוּעָה מִמֶּנָּה. אִם אֲנַחְנוּ מִשְׁתַּמְשִׁים בַּלָּשׁוֹן לְטוֹבָה, אָז אֵין כָּמוֹהָ, הִיא טוֹבָה מְאֹד, אֲבָל אִם אֲנַחְנוּ מִשְׁתַּמְשִׁים בַּלָּשׁוֹן שֶׁלָּנוּ לְרָעָה – אֵין גָּרוּעָה מִמֶּנָּה.

רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל שָׂמַח מְאֹד עַל חָכְמָתוֹ הַגְּדוֹלָה שֶׁל טַבִּי עַבְדּוֹ. הוּא פָּנָה אֶל תַּלְמִידָיו שֶׁעָמְדוּ מִסָּבִיב וְאָמַר לָהֶם: רְאִיתֶם, תַּלְמִידַי, כַּמָּה חָכָם טַבִּי עַבְדִּי? גַּם אֲנַחְנוּ צְרִיכִים לִלְמֹד, שֶׁאִם הַלָּשׁוֹן שֶׁלָּנוּ טוֹבָה – אֵין טוֹבָה מִמֶּנָּה, וְחַס וְחָלִילָה שֶׁלֹּא נִשְׁתַּמֵּשׁ בַּלָּשׁוֹן לְרָעָה.
