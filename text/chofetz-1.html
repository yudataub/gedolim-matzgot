הַחָפֵץ חַיִּים וְהַתַּלְמִיד הַיְּרֵא שָׁמַיִם

מַעֲשֶׂה בְּתַלְמִיד שֶׁל רַבִּי יִשְׂרָאֵל מֵאִיר הַכֹּהֵן מֵרָאדִין, הַמְּכֻנֶּה עַל שֵׁם סִפְרוֹ "הֶחָפֵץ חַיִּים". תַּלְמִיד זֶה בָּא אֶל רַבּוֹ הַחָפֵץ חַיִּים וּבִקֵּשׁ לְהִתְיַעֵץ עִמּוֹ בִּשְׁאֵלָה חֲשׁוּבָה מְאֹד. הוּא סִפֵּר לֶחָפֵץ חַיִּים כִּי בְּאֵיזוֹ עִיר לוֹחֲצִים עָלָיו מְאֹד שֶׁהוּא יִהְיֶה הָרַב שֶׁל הָעִיר, וִיכַהֵן בְּתַפְקִיד הַהַנְהָגָה הָרוּחָנִית שֶׁל הַקְּהִילָה. אַךְ הַתַּלְמִיד אָמַר לָרַב כִּי הוּא אֵינוֹ רוֹצֶה כָּל כָּךְ לִהְיוֹת הָרַב, מִפְּנֵי שֶׁהוּא יָרֵא מְאֹד וְיָרֵא מֵהַשֵּׁם יִתְבָּרַךְ. הוּא הִסְבִּיר כִּי הוּא חוֹשֵׁשׁ מְאֹד שֶׁמָּא יִפְסֹק לְאַנְשֵׁי הָעִיר אֵיזֶה דָּבָר שֶׁאֵינוֹ נָכוֹן, אוֹ יַגִּיד לָהֶם אֵיזוֹ הֲלָכָה לֹא נְכוֹנָה, וְכֻלָּם יִטְעוּ בִּגְלָלוֹ. מִתּוֹךְ יִרְאַת הַשָּׁמַיִם הַגְּדוֹלָה שֶׁלּוֹ, הוּא חָשַׁשׁ לָקַחַת עַל עַצְמוֹ אֶת הָאַחֲרָיוּת הָרַבָּה הַזֹּאת, וְשָׁאַל אֶת רַבּוֹ: "מָה הָרַב אוֹמֵר?".

כְּשֶׁשָּׁמַע הֶחָפֵץ חַיִּים אֶת דִּבְרֵי הַתַּלְמִיד, הוּא עָנָה לוֹ בְּבֵרוּר וְאָמַר כִּי וַדַּאי שֶׁעָלָיו לָקַחַת אֶת רַבָּנוּת הָעִיר, וְוַדַּאי שֶׁהוּא צָרִיךְ לְהוֹבִיל וּלְהַנְהִיג אֶת הַקְּהִילָה הַזֹּאת בָּעִיר. הֶחָפֵץ חַיִּים הִסְבִּיר לַתַּלְמִיד אֶת הַסִּבָּה לְכָךְ מִתּוֹךְ לִמּוּד בְּפָרָשַׁת שְׁמוֹת. הוּא הִזְכִּיר כִּי בְּפָרָשַׁת שְׁמוֹת מְסֻפָּר עַל הַמַּיַּלְּדוֹת הָעִבְרִיּוֹת. כַּאֲשֶׁר פַּרְעֹה הַמֶּלֶךְ הָרָשָׁע, מֶלֶךְ מִצְרַיִם, צִוָּה עֲלֵיהֶן שֶׁאִם יִוָּלֵד בֵּן הֵן צְרִיכוֹת לְהָמִית אוֹתוֹ, הַמַּיַּלְּדוֹת יָכְלוּ לְהַגִּיד כִּי הֵן בְּדִיּוֹק חָשְׁבוּ לְהַפְסִיק לִהְיוֹת מַיַּלְּדוֹת, וּלְעַזֹב אֶת תַּפְקִידָן כְּדֵי לֹא לְהִסְתַּבֵּךְ. אַךְ בִּגְלַל שֶׁהֵן הָיוּ יְרֵאוֹת אֱלֹהִים, וּבִגְלַל שֶׁהֵן כָּל כָּךְ רָצוּ אֶת כְּבוֹד הַשֵּׁם וְרָצוּ לַעֲשׂוֹת לִכְבוֹדוֹ, דַּוְקָא בִּגְלַל זֶה הֵן הִמְשִׁיכוּ לִהְיוֹת הַמַּיַּלְּדוֹת. הֵן חָשְׁשׁוּ וְאָמְרוּ: מִי יוֹדֵעַ אֵילוּ מַיַּלְּדוֹת יִהְיוּ בִּמְקוֹמֵנוּ, וְאוּלַי הֵן כֵּן תִּשְׁמַעְנָה בְּקוֹל פַּרְעֹה וִימִיתוּ אֶת הַיְּלָדִים וְאֶת הַבָּנִים. לָכֵן, בִּגְלַל יִרְאַת הָאֱלֹהִים שֶׁלָּהֶן, הֵן הֶחְלִיטוּ לְהִשָּׁאֵר בְּתַפְקִידָן וּלְהוֹסִיף אוֹר וְחַיִּים.

אָמַר הֶחָפֵץ חַיִּים לַתַּלְמִיד שֶׁלּוֹ כִּי אוֹתוֹ הַדָּבָר נָכוֹן גַּם גַּבָּיו. הוּא אָמַר לוֹ כִּי דַּוְקָא בִּגְלַל שֶׁהוּא יְרֵא אֱלֹהִים, וְדַוְקָא בִּגְלַל שֶׁהוּא כָּל כָּךְ חוֹשֵׁשׁ שֶׁמָּא יַגִּיד אֵיזֶה דָּבָר אֵינוֹ נָכוֹן, בְּדִיּוֹק בִּגְלַל זֶה הוּא צָרִיךְ לִהְיוֹת הָרַב שֶׁל הָעִיר, לָקַחַת אֶת הַתַּפְקִיד הַזֶּה וּלְהַנְהִיג אֶת הַקְּהִילָה. הֶחָפֵץ חַיִּים הִסְבִּיר כִּי אִם הוּא לֹא יִהְיֶה הָרַב, מִי יוֹדֵעַ מִי יִהְיֶה בִּמְקוֹמוֹ, וְהַאִם אוֹתוֹ אָדָם אַחֵר יִהְיֶה יְרֵא הַשֵּׁם וְיַחְשֹׁשׁ לִפְסֹק וּלְהַגִּיד דָּבָר שֶׁאֵינוֹ מַסְפִּיק נָכוֹן. בְּכָךְ לִמֵּד הֶחָפֵץ חַיִּים אֶת תַּלְמִידוֹ כִּי דַּוְקָא מִי שֶׁחוֹשֵׁשׁ וְיָרֵא שָׁמַיִם הוּא הָרָאוּי בְּיוֹתֵר לְהַנְהִיג.

💡 דַּוְקָא מִי שֶׁיָּרֵא מֵהַשֵּׁם וְחוֹשֵׁשׁ לִטְעוֹת הוּא הַמַּתְאִים בְּיוֹתֵר לְהוֹבִיל וְלַעֲשׂוֹת טוֹב.
