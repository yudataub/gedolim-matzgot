הַמַּתָּנָה הַמְּיֻחֶדֶת שֶׁל הָרַבָּנִית מַרְגָּלִית

בִּשְׁנוֹת הַנִּשּׂוּאִין הָרִאשׁוֹנוֹת שֶׁל מָרָן הָרַב עוֹבַדְיָה יוֹסֵף וְרַעְיָתוֹ הָרַבָּנִית מַרְגָּלִית, הָיָה בֵּיתָם צָנוֹעַ מְאֹד. הָרַבָּנִית וִתְּרָה עַל כָּל מַנְעַמֵּי הָעוֹלָם וְלֹא בִּקְשָׁה לְעַצְמָהּ בֶּגֶד, תַּכְשִׁיט, רְהִיט יָקָר אוֹ חֵפֶץ נוֹי. אוּלָם מִשְׁאָלָה צְנוּעָה אַחַת הָיְתָה בְּלִבָּהּ: לִרְכֹּשׁ עֲבוּר בְּנֵי הַבַּיִת אֲרוֹן בְּגָדִים. בְּאוֹתָם יָמִים לֹא הָיָה בְּבֵיתָם אֲרוֹן בְּגָדִים כְּלָל, וּבִגְדֵי הַיְּלָדִים הַקְּטַנִּים הָיוּ צְרוּרִים בְּתוֹךְ שַׂקֵּי בַּד. בְּכָל פַּעַם שֶׁהָרַבָּנִית בִּקְשָׁה לְהִשְׁתַּמֵּשׁ בְּבֶגֶד כָּלְשְׁהוּ, הָיָה עָלֶיהָ לְהוֹצִיא כַּמָּה וְכַמָּה בְּגָדִים מִתּוֹךְ הַשַּׂק.

בְּאוֹתָם יְמֵי צְנַע, קִבְּלוּ תּוֹשַׁבֵּי יְרוּשָׁלַיִם אֶת מְזוֹנָם בְּאֶמְצָעוּת תְּלוּשִׁים מֻקְצָבִים. הָרַבָּנִית הֶחְלִיטָה לְצַמְצֵם מִמְּזוֹנָהּ כְּדֵי לַחְסֹךְ פְּרוּטָה לִפְרוּטָה. בְּמֶשֶׁךְ חֳדָשִׁים אֲרֻכִּים סָחֲרָה הָרַבָּנִית בְּחֵלֶק מִן הַתְּלוּשִׁים שֶׁקִּבְּלָה, וּבְתוּשִיָּה רַבָּה צֵרְפָה גְּרוּשׁ לִגְרוּשׁ וְלִירָה לְלִירָה, עַד שֶׁהִצְטַבֵּר בְּיָדָהּ סְכוּם נָאֶה שֶׁל מָאתַיִם לִירוֹת תְּמִימוֹת. הָרַבָּנִית הָיְתָה מְלֵאַת תִּקְוָה וְהִרְהֲרָה בְּשִׂמְחָה כֵּיצַד תִּקְנֶה אֶת הָאֲרוֹן, תְּסַדֵּר אֶת הַבְּגָדִים עַל הַמַּדָּפִים הָרְחָבִים וְתִתְלֶה אֶת הַחֲלִיפוֹת עַל הַקּוֹלָבִים. אֶת צְרוֹר הַבַּד וּבוֹ מָאתַיִם הַלִּירוֹת הִצְנִיעָה בִּמְקוֹם סֵתֶר, וְהִמְתִּינָה לְהִזְדַּמְנוּת הַמַּתְאִימָה לִרְכֹּשׁ אֶת הָאֲרוֹן הַמְּיֻחָל.

וְהִנֵּה, בְּאֶחָד מֵהַיָּמִים, הִבְחִינָה הָרַבָּנִית כִּי מָרָן טָרוּד, וְתַחוּשַׁת מַתַּח נִכְרָה עַל פָּנָיו וְהוּא אָחוּז מַחְשָׁבוֹת. כְּשֶׁשָּׁאֲלָה אוֹתוֹ מָה אֵרַע, סִפֵּר לָהּ מָרָן בְּשֶׁקֶט וּבְכְאֵב עַל חִבּוּרוֹ הָרִאשׁוֹן, סֵפֶר הַשּׁוּאֵל וּמֵשִׁיב "יַבִּיעַ אוֹמֵר", שֶׁכְּבָר מוּכָן לִדְפוּס, אַךְ אֵין בְּיָדוֹ כֶּסֶף כְּדֵי לְמַמֵּן אֶת הַדְפָּסָתוֹ. הָרַב אָמַר כִּי זֶהוּ חִבּוּרוֹ הָרִאשׁוֹן וְהוּא רוֹצֶה כָּל כָּךְ לִרְאוֹתוֹ יוֹצֵא לָאוֹר.

הָרַבָּנִית הִתְלַבְּטָה לְשְׁנִיָּה אַחַת בִּלְבַד. מִצַּד אֶחָד עָמַד הַחֲלוֹם הַיָּשָׁן עַל אֲרוֹן הַבְּגָדִים הַחִיּוּנִי כָּל כָּךְ, וּמִצַּד שֵׁנִי עָמְדָה הַתּוֹרָה וְהַחִבּוּר הֶחָשׁוּב שֶׁל הָרַב. הִיא הֶחְלִיטָה מִיָּד. בִּצְעָדִים מְהִירִים נִגְּשָׁה אֶל מְקוֹם הַמַּחְבּוֹא, שָׁלְפָה אֶת הַצְּרוֹר הַסָּמוּי, וְהִגִּישָׁה אוֹתוֹ לְמָרָן כְּשֶׁדִּמְעוֹת הִתְרַגְּשׁוּת בְּעֵינֶיהָ. הִיא אָמְרָה לוֹ: "קַח אֶת הַכֶּסֶף וְהַדְפֵס אֶת הַסֵּפֶר! הָאֲרוֹן יָכוֹל לְהַמְתִּין". מָרָן הִשְׁתָּאָה מִמַּסִירוּתָהּ, וְהָרַבָּנִית הוֹסִיפָה וְאָמְרָה: "הַתּוֹרָה חֲשׁוּבָה מִכֹּל! עַל הַכֹּל אֲוַתֵּר עֲבוּר הַתּוֹרָה!". וּבָאֱמֶת, הַסֵּפֶר "יַבִּיעַ אוֹמֵר" הֻדְפַּס וְהִתְקַבֵּל בִּבְרָכָה רַבָּה, וְהַכֶּסֶף שֶׁהִתְקַבֵּל מִמְּכִירַת כַּמָּה מֵהַכְּרָכִים הָרִאשׁוֹנִים הִסְפִּיק לְהַדְפָּסַת הַסִּדְרָה כֻּלָּהּ.

💡 כַּאֲשֶׁר אָנוּ מְוַתְּרִים עַל רְצוֹנוֹתֵינוּ הַפְּרָטִיִּים לְמַעַן הַתּוֹרָה וְהַזּוּלָת, אָנוּ זוֹכִים לְהָבִיא בְּרָכָה וְאוֹר גָּדוֹל לָעוֹלָם.
