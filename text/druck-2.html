הָרַב חַיִּים דְּרוּקְמַן – הָרוֹפֵא וְהַמְּסִירוּת לְשַׂמֵּחַ חָתָן וְכַלָּה

רַבִּי אַרְיֵה לִיפּוֹ מְסַפֵּר עַל אָבִיו, דּוֹקְטוֹר לִיפּוֹ. הָאָב הוּא רוֹפֵא מְסוּר שֶׁטִּפֵּל בַּרַב צְבִי יְהוּדָה הַכֹּהֵן קוּק, וְגַם טִפֵּל בַּרַב חַיִּים דְּרוּקְמַן. הָרוֹפֵא הֶחָכָם פִּתֵּחַ שִׁיטָה מְיֻחֶדֶת שֶׁנִּקְרֵאת "כִּנּוֹר לֵב". שִׁיטָה זוֹ מְתַרְגֶּמֶת אֶת קֶצֶב דֹּפֶק הַלֵּב לְנִגּוּן נָעִים, שֶׁמַּרְגִּיעַ אֶת הָרֵאָה וְעוֹזֵר לְחוֹלֵי קוֹרוֹנָה לִנְשֹׁם בְּקַלּוּת.

כַּמָּה חֳדָשִׁים לִפְנֵי שֶׁהָרַב חַיִּים דְּרוּקְמַן נִפְטַר, כְּשֶׁהוּא חָלָה בְּקוֹרוֹנָה, בִּקֵּשׁ דּוֹקְטוֹר לִיפּוֹ מִבְּנוֹ אַרְיֵה לִנְסֹעַ אֶל הָרַב דְּרוּקְמַן וּלְלַמֵּד אוֹתוֹ לַעֲשׂוֹת אִמּוּן בְּשִׁיטַת "כִּנּוֹר לֵב". אַרְיֵה הִתְקַשֵּׁר אֶל הָרַב דְּרוּקְמַן וְהִצִּיעַ לָבוֹא אֵלָיו. הָרַב הֱשִׁיב בִּפְשָׁטוּת: "אֵין צֹרֶךְ לִטְרֹחַ, אֲנִי בְּסֵדֶר". אַרְיֵה אָמַר לוֹ: "הָרַב, זֶה לֹא טֹרַח בִּשְׁבִילִי, זֶה זְכוּת בִּשְׁבִילִי". כְּשֶׁשָּׁמַע זֹאת הָרַב, נַעֲנָה וְאָמַר: "טוֹב, אִם אַתָּה רוֹצֶה, אַתָּה מוּזְמָן".

אַרְיֵה, שֶׁהָיָה בִּירוּשָׁלַיִם, נָסַע כְּשָׁעָה עַד לְבֵיתוֹ שֶׁל הָרַב דְּרוּקְמַן. כְּשֶׁנִּכְנַס לַבַּיִת, רָאָה אֶת הָרַב יוֹשֵׁב בְּאֵזוֹר מְבוֹדָד עִם נַיְלוֹנִים בִּגְלַל הַמַּחֲלָה. אַרְיֵה הִבְחִין שֶׁבַּסָּלוֹן שֶׁל הָרַב יֵשׁ שׁוּרוֹת רַבּוֹת שֶׁל סַפְסָלִים וְשׁוּלְחָנוֹת. הַסַּפְסָלִים הַלָּלוֹ מְיוֹעָדִים לְתַלְמִידִים רַבִּים שֶׁבָּאִים לִלְמֹד אֵצֶל הָרַב שִׁעוּר אַחֲרֵי שִׁעוּר אַחֲרֵי שִׁעוּר, עַד הַשָּׁעָה שְׁתַּיִם בַּלַּיְלָה.

אַרְיֵה הִתְקִין לָרַב אֶת הַמַּעֲרֶכֶת וְאֶת הַחַיְשָׁן שֶׁמִּתְחַבֵּר לַטֶּלֶפֿוֹן, וְהָרַב דְּרוּקְמַן בִּצֵּעַ נְשִׁימוֹת יַחַד עִם הַנִּגּוּן וְהִרְגִּישׁ טוֹב יוֹתֵר. כְּשֶׁסִיְּמוּ אֶת הַטִּפּוּל, בִּקֵּשׁ אַרְיֵה בְּרָכָה בִּשְׁבִיל אָבִיו הָרוֹפֵא. הָרַב דְּרוּקְמַן בֵּרֵךְ אוֹתוֹ בְּחֹם וְאָמַר: "בְּוַדַּאי, אַבָּא שֶׁלְּךָ חַיָּב לְהַצְלִיחַ, עִם הַ'כִּנּוֹר לֵב', עִם כָּל הַמִּפְעָל שֶׁלּוֹ, כִּי הוּא עוֹשֶׂה לְשֵׁם שָׁמַיִם. וְגַם הוּא עוֹשֶׂה דְּבָרִים חֲשׁוּבִים לְעַם יִשְׂרָאֵל וּדְבָרִים מְיֻחָדִים לְעַם יִשְׂרָאֵל, וְלָכֵן הוּא וַדַּאי חַיָּב לְהַצְלִיחַ".

כְּדֵי לְהַרְאוֹת אֶת מְסִירוּת הַנֶּפֶשׁ הַגְּדוֹלָה שֶׁל הָרַב דְּרוּקְמַן וְאֶת הָעֲשִׂיָּה שֶׁלּוֹ לְשֵׁם שָׁמַיִם, סִפֵּר אַרְיֵה מַעֲשֶׂה נוֹסָף: בַּיּוֹם שֶׁבּוֹ אֲחוֹתוֹ הִתְחַתְּנָה, בְּסוֹף הַחֲתֻנָּה בַּשָּׁעָה שְׁתֵּים עֶשְׂרֵה בַּלַּיְלָה, הִגִּיעַ הָרַב דְּרוּקְמַן בְּעַצְמוֹ לָאוּלָם כְּדֵי לְבָרֵךְ אֶת הַחָתָן וְאֶת הַכַּלָּה. הָרַב נָסַע שַׁעֲתַיִם תְּמִימוֹת מִבֵּיתוֹ עַד לָאוּלָם, וְאַחַר כָּךְ נָסַע עוֹד שַׁעֲתַיִם בַּחֲזָרָה לְבֵיתוֹ, וְכָל זֶה בְּסוֹף יוֹם עָמוּס מְאֹד. הָרַב עָשָׂה אֶת כָּל הַדֶּרֶךְ הָאֲרֻכָּה הַזֹּאת רַק כְּדֵי לְהַגִּיד מַזָּל טוֹב וּלְשַׂמֵּחַ חָתָן וְכַלָּה.

💡 כְּשֶׁאָנוּ עוֹשִׂים מַעֲשֵׂי חֶסֶד וּמִצְוֹת בְּאַהֲבָה וּלְשֵׁם שָׁמַיִם, אָנוּ מְבִיאִים אוֹר וְשִׂמְחָה גְּדוֹלָה לְעַם יִשְׂרָאֵל.
