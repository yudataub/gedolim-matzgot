הַמִּשְׁנָיוֹת שֶׁבַּכִּיס

רַבִּי יְהוּדָה צְדָקָה זֵכֶר צַדִּיק לִבְרָכָה הָיָה חָכָם גָּדוֹל, צַדִּיק וְרֹאשׁ יְשִׁיבָה מֻפְלָא, שֶׁהִקְפִּיד מְאֹד מְאֹד עַל שְׁמִירַת הַזְּמָן שֶׁלּוֹ. הוּא יָדַע שֶׁכָּל רֶגַע וְרֶגַע בַּחַיִּים הוּא מַתָּנָה יְקָרָה מֵהַשֵּׁם יִתְבָּרַךְ, וְעַל כֵּן הוּא לֹא רָצָה לְבַזְבֵּז אַף רֶגַע אֶחָד לַשָּׁוְא. רַבֵּנוּ נִצֵּל אֶת כָּל הַזְּמָן שֶׁלּוֹ לְלִמּוּד תּוֹרָה, לִתְפִלָּה וּלְמַעֲשִׂים טוֹבִים. גַּם כַּאֲשֶׁר הָיָה צָרִיךְ לָצֵאת מֵהַבַּיִת וְלִנְסֹעַ מִמָּקוֹם אֶחָד לְמָקוֹם אַחֵר בַּמְּכוֹנִית, הַזְּמָן שֶׁל הַנְּסִיעָה לֹא עָבַר סְתָם כָּךְ בַּבַּטָּלָה אוֹ בְּשִׂיחוֹת בְּטֵלוֹת, אֶלָּא הָפַךְ לִזְמָן יָקָר שֶׁל תּוֹרָה וּתְפִלָּה.

בְּכָל פַּעַם שֶׁרַבֵּנוּ רַבִּי יְהוּדָה צְדָקָה נָסַע בִּמְכוֹנִית מִמָּקוֹם אֶחָד לַמִּשְׁנֵהוּ, מִיָּד וּבְלִי שׁוּם שְׁהִיּוֹת, עִכּוּבִים אוֹ הַמְתָנָה, הוּא הוֹצִיא מֵחֵיקוֹ סֵפֶר תְּהִלִּים קָטָן וּמְיֻחָד. סֵפֶר הַתְּהִלִּים הַזֶּה הָיָה שָׁמוּר אֶצְלוֹ תָּמִיד. בִּתְחִלַּת הַסֵּפֶר, בָּעַמּוֹדִים הָרִאשׁוֹנִים שֶׁבְּרֹאשׁוֹ, הָיוּ כַּמָּה דַּפִּים שֶׁרַבֵּנוּ רָשַׁם בְּעַצְמוֹ בִּכְתַב יָדוֹ. עַל הַדַּפִּים הָאֵלֶּה הָיוּ רְשׁוּמִים רָאשֵׁי דְבָרִים שֶׁל פִּרְקֵי מִשְׁנָה רַבִּים. הָרְשִׁימוֹת הָאֵלֶּה סִיְּעוּ לוֹ לִזְכֹּר וּלְשַׁנֵּן אֶת הַמִּשְׁנָיוֹת בְּקַלּוּת וּבִמְהִירוּת.

בַּתְּחִלָּה, בְּמַהֲלַךְ הַנְּסִיעָה בַּמְּכוֹנִית, הָיָה רַבִּי יְהוּדָה צְדָקָה קוֹרֵא וְאֹמֵר מִסְפָּר מִזְמוֹרֵי תְּהִלִּים מִתּוֹךְ הַסֵּפֶר הַקָּטָן. אַחַר כָּכָה, מִיָּד כְּשֶׁסִּיֵּם אֶת אֲמִירַת מִזְמוֹרֵי הַתְּהִלִּים, הוּא עָבַר לִלְמֹד וּלְשַׁנֵּן אֶת הַמִּשְׁנָיוֹת עַל פִּי הָרְשִׁימוֹת הַמְּיֻחָדוֹת שֶׁכָּתַב בַּדַּפִּים. הַדֶּרֶךְ שֶׁלּוֹ לִלְמֹד אֶת הַמִּשְׁנָיוֹת בַּנְּסִיעָה הָיְתָה מַפְלִיאָה וּמְעוֹרֶרֶת הַשְׁרָאָה. הוּא הָיָה מֵצִיץ בַּדַּפִּים הָרְשׁוּמִים בִּכְתַב יָדוֹ, רוֹאֶה אֶת הַמִּלִּים הָרִאשׁוֹנִים שֶׁל כָּל פֶּרֶק וּפֶרֶק מִשְׁנָה, וּמִיָּד הָיָה מַמְשִׁיךְ לִלְמֹד, לַחֲזֹר וּלְשַׁנֵּן אֶת כָּל הַפֶּרֶק מִזִּכְרוֹנוֹ הַנִּפְלָא בְּעַל פֶּה.

מִנְהָג זֶה הָיָה קָבוּעַ וַחֲזָק אֶצְלוֹ תָּמִיד, וְהוּא לֹא שִׁנָּה מִמֶּנּוּ אַף פַּעַם. אֲפִילּוּ כַּאֲשֶׁר נָסַע בַּמְּכוֹנִית נְסִיעָה קְצָרָה בְּיוֹתֵר, כְּגוֹן בַּדֶּרֶךְ הַקְּצָרָה מִבֵּיתוֹ אֶל הַיְשִׁיבָה, הוּא לֹא שִׁנָּה מִמִּנְהָגוֹ הַטּוֹב וְקָרָא מִתּוֹךְ הַסֵּפֶר. הוּא לֹא אָמַר שֶׁהַנְּסִיעָה קְצָרָה מִדַּי וְשֶׁאֵין טַעַם לִפְתֹּחַ אֶת הַסֵּפֶר בִּשְׁבִיל דַּקּוֹת סְפוּרוֹת, אֶלָּא מִיָּד פָּתַח אֶת סֵפֶר הַתְּהִלִּים הַקָּטָן, אָמַר תְּהִלִּים וְשִׁנֵּן אֶת הַמִּשְׁנָיוֹת. דֻּגְמָה זוֹ שֶׁל הַצַּדִּיק מְלַמֶּדֶת אוֹתָנוּ שֶׁכָּל רֶגַע קָטָן בַּחַיִּים הוּא מַתָּנָה יְקָרָה, וְשֶׁגַּם בִּנְסִיעוֹת קְצָרוֹת אֶפְשָׁר לְנַצֵּל אֶת הַזְּמָן לְלִמּוּד תּוֹרָה.
