הָרַב שֶׁאָהַב אֶת הָעֲנִיִּים וְאֶת יְרוּשָׁלַיִם

בְּעִיר בַּגְדָּאד שֶׁבְּעִירָאק נוֹלַד הַצַּדִּיק הָרַב יְהוּדָה מֹשֶׁה פְּתַיָּה. כְּשֶׁהָיָה יֶלֶד קָטָן, לָמַד בְּתַלְמוּד תּוֹרָה וּבִישִׁיבַת "מִדְרַשׁ בֵּית זִילְכָּה", וְשָׁמַע בְּשִׂמְחָה אֶת הַדְּרָשׁוֹת שֶׁל הַחָכָם הַגָּדוֹל הַבַּעַל "בֶּן אִישׁ חַי". יְהוּדָה הַקָּטָן שָׁקַד עַל הַתּוֹרָה בִּרְצִינוּת רַבָּה, וּבְגִיל שְׁבַע עֶשְׂרֵה בִּלְבַד כְּבָר הֻסְמַךְ לִהְיוֹת רַב וּמוֹרֶה הוֹרָאָה, וְאַף לָמַד אֶת חָכְמַת הַקַּבָּלָה אֶצְל הַבַּעַל "בֶּן אִישׁ חַי". בִּרְבוֹת הַשָּׁנִים הָפַךְ הָרַב יְהוּדָה לַמַּנְהִיג הָרוּחָנִי שֶׁל כָּל יְהוּדֵי בָּבֶל, לְאֶחָד מִגְּדוֹלֵי הַמְּקֻבָּלִים, וְחִבֵּר סְפָרִים חֲשׁוּבִים רַבִּים, וּבָהֶם הַסְּפָרִים "מִנְחַת יְהוּדָה" וְ"בֵית לֶחֶם יְהוּדָה". בְּבַגְדָּאד, כְּשֶׁאָמְרוּ הָאֲנָשִׁים אֶת הַמִּלִּים "חָכָם יְהוּדָה", כֻּלָּם יָדְעוּ מִיָּד בְּדִיּוּק לְמִי הַכַּוָּנָה.

הָרַב יְהוּדָה פְּתַיָּה הָיָה מְפֻרְסָם מְאֹד בְּכָל רְחוֹבוֹת בַּגְדָּאד, וְכָל הָאֲנָשִׁים כִּבְדוּ אוֹתוֹ מְאֹד. אַךְ בְּבֵית הַכְּנֶסֶת "יִצְחָק חַיִּים", שֶׁבּוֹ הָיָה מִתְפַּלֵּל, הוּא לֹא הִסְכִּים לָשֶׁבֶת בַּמָּקוֹם הַמְּכֻבָּד בְּיוֹתֵר שֶׁלְּיַד אֲרוֹן הַקֹּדֶשׁ. אֵיפֹה הוּא יָשַׁב? הָרַב קָבַע אֶת מְקוֹם תְּפִלָּתוֹ דַּוְקָא בַּפִּנָּה הַצְּנוּעָה שֶׁבָּהּ יָשְׁבוּ הָעֲנִיִּים וּפְשׁוּטֵי הָעָם, וְשָׁם הִתְפַּלֵּל עִמָּהֶם. כָּל אָדָם שֶׁבָּא אֵלָיו – עָשִׁיר אוֹ עָנִי, חָכָם אוֹ פָּשׁוּט – הָרַב קִבֵּל אוֹתוֹ בְּסֵבֶר פָּנִים יָפוֹת, וּבְסַבְלָנוֹת גְּדוֹלָה הִקְשִׁיב לוֹ. אֲנָשִׁים רַבִּים מְאֹד בָּאוּ אֵלָיו כְּדֵי לְבַקֵּשׁ אֶת בִּרְכָתוֹ הַקְּדוֹשָׁה.

וְעוֹד דָּבָר חָשׁוּב מְאֹד הָיָה בְּלִבּוֹ שֶׁל הָרַב כָּל יָמָיו: גַּעְגּוּעִים עֲמֻקִּים לִירוּשָׁלַיִם, עִיר הַקֹּדֶשׁ. בִּשְׁנַת הַ'תַּרְסָ"ה הוּא עָלָה עִם בְּנֵי בֵּיתוֹ לִירוּשָׁלַיִם, אַךְ כַּעֲבֹר שְׁנָתַיִם נֶאֱלַץ לַחֲזֹר לְבַגְדָּאד. בִּשְׁנַת הַ'תַּרְפָּ"ב נִסָּה שׁוּב לְהִתְיַשֵּׁב בִּירוּשָׁלַיִם, וְשׁוּב נֶאֱלַץ לַחֲזֹר. אַךְ הָרַב יְהוּדָה לֹא וִתֵּר בְּשׁוּם אֹפֶן! בַּפַּעַם הַשְּׁלִישִׁית, בִּשְׁנַת הַ'תַּרְצָ"ה, הוּא עָלָה שׁוּב לִירוּשָׁלַיִם, וְהַפַּעַם זָכָה לְהִשָּׁאֵר בָּהּ עַד יוֹמוֹ הָאַחֲרוֹן.

בִּירוּשָׁלַיִם הֶעֱמִיד תַּלְמִידִים רַבִּים וְלִמֵּד אוֹתָם תּוֹרָה. בִּשְׁנוֹת מִלְחֶמֶת הָעוֹלָם הַשְּׁנִיָּה, בַּיָּמִים הַקָּשִׁים שֶׁל עַם יִשְׂרָאֵל, הוּא קָם בַּחֲצוֹת הַלַּיְלָה וְהִתְפַּלֵּל עַל עַם יִשְׂרָאֵל. הָרַב נִפְטַר בִּירוּשָׁלַיִם בְּכָ"ז בְּאָב הַ'תַּשָּׁ"ב, וְנִקְבַּר בְּהַר הַזֵּיתִים. עַד הַיּוֹם רְחוֹבוֹת בִּירוּשָׁלַיִם, בְּרָמַת גַּן וּבִבְנֵי בְּרָק נִקְרָאִים עַל שְׁמוֹ.
