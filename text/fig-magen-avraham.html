הַמְּלַמֵּד הַצָּנוּעַ מִקָּאלִישׁ

רַבִּי אַבְרָהָם אַבְלִי הַלֵּוִי גּוֹמְבִּינֶר נוֹלַד בָּעִיר גּוֹמְבִּין שֶׁבְּפוֹלִין. כְּשֶׁהָיָה בֶּן שְׁמוֹנֶה עֶשְׂרֵה שָׁנִים, רָצְחוּ הַקּוֹזַקִים אֶת אָבִיו, הַדַּיָּן רַבִּי חַיִּים הַלֵּוִי. לַנַּעַר הַצָּעִיר לֹא הָיוּ חַיִּים קַלִּים, אַךְ הוּא הִמְשִׁיךְ לִלְמֹד תּוֹרָה בִּשְׁקִידָה וּבֶאֱמוּנָה.

בִּשְׁנַת הַ'תּ"כּ הִגִּיעַ רַבִּי אַבְרָהָם לָעִיר קָאלִישׁ. הוּא הָיָה תַּלְמִיד חָכָם עָצוּם וְגָאוֹן גָּדוֹל, אַךְ הוּא לֹא סִפֵּר עַל כָּךְ לְאַף אֶחָד. הוּא בָּחַר לִחְיוֹת בִּצְנִיעוּת וּבְעַנְוָה רַבָּה, וְהִתְחִיל לְלַמֵּד תּוֹרָה בַּחוּרִים צְעִירִים. רַבִּי אַבְרָהָם לִמֵּד אֶת הַיְלָדִים בְּסַבְלָנוּת וּבְאַהֲבָה גְּדוֹלָה, וְלֹא בִּקֵּשׁ לְעַצְמוֹ שׁוּם כָּבוֹד אוֹ פִּרְסוּם.

רַק לַאט לַאט גִּלּוּ הַתַּלְמִידִים כַּמָּה גָּדוֹל הוּא בַּתּוֹרָה. הַשֵּׁמַע עַל הַמְּלַמֵּד הַחָכָם הִתְפַּשֵּׁט, וּבָאוּ אֵלָיו גַּם תַּלְמִידִים מְבֻגָּרִים יוֹתֵר שֶׁבִּקְשׁוּ לִלְמֹד מִמֶּנּוּ. כָּכָה נַעֲשָׂה רַבִּי אַבְרָהָם לְרֹאשׁ יְשִׁיבָה, מִבְּלִי שֶׁחִפֵּשׂ זֹאת כְּלָל.

פַּעַם אַחַת הִגִּיעַ לְבִקּוּר בָּעִיר קָאלִישׁ גְּדוֹל הַדּוֹר, הַגָּאוֹן רַבִּי שַׁבְתַּאי הַכֹּהֵן, הַמְּכֻנֶּה הַשָּׁ"ךְ. בְּעֵת בִּקּוּרוֹ, שָׁאַל הַשָּׁ"ךְ שְׁאֵלָה קָשָׁה וּמֻרְכֶּבֶת מְאֹד בַּתּוֹרָה. אֶחָד מִתַּלְמִידָיו שֶׁל רַבִּי אַבְרָהָם עָנָה לוֹ מִיָּד תְּשׁוּבָה חַדָּה, עֲמֻקָּה וְנִפְלָאָה.

הַשָּׁ"ךְ הִתְפַּלֵּא מְאֹד מֵהַתְּשׁוּבָה הַמַּבְרִיקָה. כְּשֶׁבֵּרֵר וְהֵבִין מִי הוּא הַמּוֹרֶה שֶׁלִּמֵּד אֶת הַבַּחוּרִים הָאֵלֶּה, פָּנָה מִיָּד אֶל בְּנֵי הָעִיר וְדָרַשׁ מֵהֶם: "אָדָם כָּזֶה גָּדוֹל צָרִיךְ לְקַבֵּל מִשְׂרָה שֶׁל רַב!"

אַךְ רַבִּי אַבְרָהָם הָיָה עָנָיו אֲמִתִּי. הוּא סֵרֵב בְּתֹקֶף לְקַבֵּל מִשְׂרָה שֶׁתִּגְרֹם לְרַב אַחֵר לְאַבֵּד אֶת מְקוֹמוֹ. הוּא לֹא רָצָה לִפְגֹּעַ בְּאַף אָדָם. הוּא הִסְכִּים לְקַבֵּל רַק מִשְׂרָה שֶׁלֹּא פּוֹגַעַת בְּאַף אֶחָד – תַּפְקִיד שֶׁל דַּיָּן וּמוֹרֶה הוֹרָאָה בְּבֵית הַדִּין שֶׁל הָעִיר.

בְּמֶשֶׁךְ חַיָּיו, כִּמְעַט וְלֹא יָדְעוּ אֲנָשִׁים עַל גְּדֻלָּתוֹ הָעֲצוּמָה. הוּא יָשַׁב בִּצְנִיעוּת בְּמֶשֶׁךְ שֵׁשׁ שָׁנִים וְכָתַב אֶת פֵּרוּשׁוֹ הַקָּדוֹשׁ "מָגֵן אַבְרָהָם" עַל חֵלֶק "אוֹרַח חַיִּים" שֶׁל הַשֻּׁלְחָן עָרוּךְ. סֵפֶר זֶה כּוֹלֵל אֶת כָּל הַהֲלָכוֹת שֶׁל חַיֵּי הַיּוֹם-יוֹם: הִלְכוֹת תְּפִלָּה, בְּרָכוֹת, שַׁבָּת וְחַגִּים.

רַק כַּמָּה שָׁנִים אַחֲרֵי פְּטִירָתוֹ, כְּשֶׁהַסֵּפֶר נִדְפַּס, הִכִּיר כָּל עַם יִשְׂרָאֵל אֶת גְּדֻלָּתוֹ. כִּמְעַט בְּכָל מַחֲלֹקֶת נִפְסְקָה הַהֲלָכָה כְּדַעְתּוֹ, וְגַם סְפָרִים מְפֻרְסָמִים כְּמוֹ הַמִּשְׁנָה בְּרוּרָה וְשֻׁלְחָן עָרוּךְ הָרַב נִשְׁעָנִים עַל דְּבָרָיו. עַל שֵׁם סִפְרוֹ הַקָּדוֹשׁ זֶה הוּא נִקְרָא עַד הַיּוֹם "הַמָּגֵן אַבְרָהָם".

💡 עַנְוָה אֲמִתִּית הִיא לִלְמֹד וְלַעֲשׂוֹת טוֹב בִּצְנִיעוּת מִבְּלִי לְחַפֵּשׂ כָּבוֹד, וּלְעוֹלָם לֹא לִפְגֹּעַ בִּכְבוֹדוֹ אוֹ בְּפַרְנָסָתוֹ שֶׁל אָדָם אַחֵר.
