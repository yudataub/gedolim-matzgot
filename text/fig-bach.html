הַשֻּׁלְחָן הַפָּתוּחַ שֶׁל הַבַּ"ח

לִפְנֵי שָׁנִים רַבּוֹת נוֹלַד בָּעִיר לוּבְּלִין שֶׁבְּפוֹלִין יֶלֶד מְיוּחָד בְּשֵׁם יוֹאֵל סִירְקִישׁ. הוּא נוֹלַד לְמִשְׁפַּחַת יָפֶה הַמְּיוּחֶסֶת וְהַמְּפֹאָרָה. כְּשֶׁהָיָה צָעִיר, לָמַד תּוֹרָה בְּשַׁקְדָנוּת רַבָּה מִפִּי אָבִיו, וְגַם מִפִּי גְּדוֹלֵי הַחֲכָמִים וְהָרַבָּנִים שֶׁחָיוּ בְּפוֹלִין. כַּאֲשֶׁר גָּדַל רַבִּי יוֹאֵל וְהָפַךְ לְתַלְמִיד חָכָם מֻפְלָג, הוּא נַעֲשָׂה לְרַב וְשִׁמֵּשׁ בְּרַבָּנוּת בְּעָרִים רַבּוֹת, וּבָהֵן בֶּעָרִים מֶזְ'בּוּז' וּבֶעֶלְזָא. בִּשְׁנַת ה'שְׁעָ"ה הוּא נַעֲשָׂה רַבָּהּ שֶׁל הָעִיר בְּרִיסְק, וּבִשְׁנַת ה'שְׁעָ"ט הוּא מוֹנָה לִהְיוֹת רַבָּהּ שֶׁל הָעִיר הַגְּדוֹלָה קְרָקוֹב וְגַם לְרֹאשׁ הַיְּשִׁיבָה שָׁם, תַּפְקִיד חָשׁוּב שֶׁבּוֹ שִׁמֵּשׁ עַד יוֹם פְּטִירָתוֹ.

רַבִּי יוֹאֵל הָיָה מוֹכָר וּמְפֻרְסָם בְּפִי כָּל עַם יִשְׂרָאֵל בַּכִּנּוּי "הַבַּ"ח". כִּנּוּי מְיֻחָד זֶה נִתַּן לוֹ עַל שֵׁם סִפְרוֹ הַגָּדוֹל וְהַחָשׁוּב שֶׁנִּקְרָא "בַּיִת חָדָשׁ". סֵפֶר זֶה הוּא פֵּרוּשׁ מַקִּיף וִיסוֹדִי עַל הַסֵּפֶר "אַרְבָּעָה טוּרִים", שֶׁהוּא הַסֵּפֶר הַמֶּרְכָּזִי שֶׁעָלָיו בָּנוּי וּמְבֻסָּס סֵפֶר הַ"שֻּׁלְחָן עָרוּךְ".

הַבַּ"ח הָיָה אָדָם עָשִׁיר גָּדוֹל מְאֹד וְזָכָה לִנְכָסִים וּלְמָמוֹן רַב. אֲבָל אֶת עָשְׁרוֹ הוּא לֹא שָׁמַר לְעַצְמוֹ בִּלְבַד. הַבַּ"ח פָּתַח אֶת בֵּיתוֹ וְהֶאֱכִיל תַּלְמִידִים רַבִּים עַל שֻׁלְחָנוֹ הַפָּתוּחַ בְּכָל יוֹם וָיוֹם. הוּא עָשָׂה זֹאת כְּדֵי שֶׁהַתַּלְמִידִים יוּכְלוּ לָשֶׁבֶת בִּמְנוּחָה, לִלְמֹד תּוֹרָה וּלְהַשְׁקִיעַ בְּלִמּוּדָם בְּלִי לִדְאֹג כְּלָל לְאֹכֶל וּלְמָזוֹן.

בַּיְּשִׁיבָה שֶׁל הַבַּ"ח בָּעִיר קְרָקוֹב גָּדְלוּ תַּלְמִידִים חֲכָמִים שֶׁהָפְכוּ לִגְדוֹלֵי תּוֹרָה מְפֻרְסָמִים. אֶחָד מֵהַתַּלְמִידִים הַלַּלּוּ הָיָה רַבִּי דָּוִד הַלֵּוִי סֶגַל, הַמּוֹכָר לְכֻלָּנוּ בַּכִּנּוּי "הַטַּ"ז". רַבִּי דָּוִד זָכָה לְהֵעָשׂוֹת חֲתָנוֹ שֶׁל הַבַּ"ח, וּבְרַבּוֹת הַיָּמִים נַעֲשָׂה גַּם הוּא לְאֶחָד מִגְּדוֹלֵי הַמְּפָרְשִׁים שֶׁל סֵפֶר הַ"שֻּׁלְחָן עָרוּךְ".

בְּסִפְרוֹ "בַּיִת חָדָשׁ" לִמֵּד הַבַּ"ח דֶּרֶךְ חֲשׁוּבָה וִיסוֹדִית בְּלִמּוּד הַהֲלָכָה. הוּא הִסְבִּיר כִּי לִפְנֵי שֶׁפּוֹסְקִים הֲלָכָה, צָרִיךְ תָּמִיד לַחְזֹר אֶל הַשֹּׁרֶשׁ – לִלְמֹד בַּגְּמָרָא, לְעַיֵּן בְּפֵרוּשׁ רַשַׁ"י, בְּתוֹסָפוֹת וּבְדִבְרֵי הָרִאשׁוֹנִים – וּלְהָבִין הֵיטֵב מֵאֵיפֹה כָּל הֲלָכָה בָּאָה. הַבַּ"ח עָבַר בְּשַׁקְדָנוּת עַל כָּל אַרְבַּעַת הַטּוּרִים, וּלְכָל הֲלָכָה וַהֲלָכָה הֶרְאָה אֶת הַמָּקוֹר שֶׁלָּהּ וְאֶת הַטַּעַם הָעָמֹק שֶׁלָּהּ.

בְּאֹתָן שָׁנִים כִּבְדוּ אוֹתוֹ מְאֹד כָּל קְהִלּוֹת פּוֹלִין. הַבַּ"ח הָיָה גַּם מֵרָאשֵׁי "וַעַד אַרְבַּע אֲרָצוֹת" – הַוַּעַד הַמֶּרְכָּזִי שֶׁהִנְהִיג אֶת כָּל קְהִלּוֹת יִשְׂרָאֵל בְּפוֹלִין. הוּא הִנְהִיג אֶת הַצִּבּוּר בְּחָכְמָה וּבְמַנְהִיגוּת, וְנִפְטַר בָּעִיר קְרָקוֹב בְּתַאֲרִיךְ כ' בְּאַדָּר בִּשְׁנַת ה'ת'.

💡 עֹשֶׁר וְכֹחַ הֵם מַתָּנָה מֵהַשֵּׁם כְּדֵי לַעֲזֹר לַאֲחֵרִים וְלִתְמֹךְ בְּלוֹמְדֵי תּוֹרָה, וּמִי שֶׁרוֹצֶה לָדַעַת הֲלָכָה בֶּאֱמֶת לוֹמֵד אוֹתָהּ מִן הַשֹּׁרֶשׁ.
