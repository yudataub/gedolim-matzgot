מַקֵּל הַהֲלִיכָה שֶׁמַּזְכִּיר לְהוֹדוֹת

בְּאֶחָד מִיְּמֵי הַבֹּקֶר יָרְדוּ הָרַב שָׁלוֹם מַשָּׁאשׁ וְהָעוֹזֵר שֶׁלּוֹ בִּזְרִיזוּת בַּמַּדְרֵגוֹת בְּדַרְכָּם לִתְפִלַּת שַׁחֲרִית בְּבֵית הַמִּדְרָשׁ. כַּאֲשֶׁר הִגִּיעוּ לְתַחְתִּית הַמַּדְרֵגוֹת, נֶעֱצַר הָרַב פִּתְאֹם וְאָמַר לְעוֹזְרוֹ כִּי הוּא שָׁכַח אֶת מַקֵּל הַהֲלִיכָה שֶׁלּוֹ לְמַעְלָה. הָעוֹזֵר הַמָּסוּר רָץ מִיָּד לְמַעְלָה, הֵבִיא לָרַב אֶת הַמַּקֵּל, אַךְ תָּמַהּ וְשָׁאַל אֶת הָרַב לְפֵשֶׁר הַדָּבָר. הָעוֹזֵר שָׁאַל כֵּיצַד זֶה שֶׁהָרַב יָרַד אֶת כָּל הַמַּדְרֵגוֹת בְּלֹא כָּל קֹשִׁי, וְגַם בִּפְעָמִים רַבּוֹת כְּלָל אֵינוֹ מִשְׁתַּמֵּשׁ בַּמַּקֵּל, וּבְכָל זֹאת חָשׁוּב לוֹ כָּל כָּךְ לָקַחַת אוֹתוֹ.

הָרַב שָׁלוֹם מַשָּׁאשׁ הִבִּיט בְּעוֹזְרוֹ וְהֶחְלִיט לְסַפֵּר לוֹ אֵרוּעַ מֵהָעָבָר. הוּא סִפֵּר שֶׁלִּפְנֵי שָׁנִים רַבּוֹת, בָּעִיר שֶׁבָּהּ חַי, הִתְפַּשְׁטָה מַגֵּפָה קָשָׁה שֶׁל חַיְדָּק מְסֻכָּן שֶׁגָּרְמָה לַמַּחֲלָה קָשָׁה וּלְמוֹתָם שֶׁל אֲנָשִׁים רַבִּים. גַּם הָרַב עַצְמוֹ חָלָה בְּאוֹתָהּ מַחֲלָה קָשָׁה וְסָבַל מֵחֹם גּוּף גָּבוֹהַּ מְאֹד. הָרַב סִפֵּר שֶׁאִלְמָלֵא הַמִּשְׁפָּחָה שֶׁלּוֹ וְכָל הַקְּרוֹבִים שֶׁלּוֹ הָיוּ מִתְפַּלְּלִים לַה' שֶׁהוּא יַבְרִיא, וְאִלְמָלֵא הָרוֹפְאִים הַטּוֹבִים הָיוּ בָּאִים וּמַרְפְּאִים אוֹתוֹ, הוּא לֹא הָיָה בֵּין הַחַיִּים כַּיּוֹם. מִתּוֹךְ הַחֹם הַגָּבוֹהַּ שֶׁל הַמַּחֲלָה חָלַם הָרַב חֲלוֹמוֹת רַבִּים, וּבֵין הַיְתֶר חָלַם שֶׁהוּא מַבְרִיא וְרָץ בְּשִׂמְחָה לְהִתְפַּלֵּל תְּפִלַּת שַׁחֲרִית בְּבֵית הַמִּדְרָשׁ.

לְאַחַר שֶׁהָרוֹפְאִים נִסּוּ תְּרוּפוֹת שׁוֹנוֹת, הַחֹם הַגָּבוֹהַּ יָרַד לְאַט לְאַט וְהָרַב הִתְחִיל לְהַרְגִּישׁ טוֹב יוֹתֵר. בַּתְּחִלָּה, בְּמַהֲלַךְ תַּהֲלִיךְ הַשִּׁקּוּם, הָרַב נִזְקַק לְמַקֵּל הֲלִיכָה מִכֵּיוָן שֶׁלֹּא יָכֹל לָלֶכֶת בְּכֹחוֹת עַצְמוֹ. אַךְ עִם הַזְּמַן מַצָּבוֹ הַבְּרִיאוּתִי הִשְׁתַּפֵּר עוֹד וָעוֹד, עַד שֶׁהָיָה יָכוֹל לָלֶכֶת כְּבָר לְלֹא שׁוּם צֹרֶךְ בַּמַּקֵּל.

כַּאֲשֶׁר שָׁאַל הַתַּלְמִיד שׁוּב לָמָּה הָרַב מַמְשִׁיךְ לָקַחַת אִתּוֹ אֶת הַמַּקֵּל, הִסְבִּיר לוֹ הָרַב שָׁלוֹם מַשָּׁאשׁ אֶת הַתַּפְקִיד הֶחָשׁוּב שֶׁל הַמַּקֵּל. הָרַב הִסְבִּיר כִּי הַמַּקֵּל מַזְכִּיר לוֹ תָּמִיד אֶת הַמַּגֵּפָה הַקָּשָׁה שֶׁעָבְרָה עָלָיו. הוּא מַזְכִּיר לוֹ שֶׁלּוּלֵא הַתְּפִלּוֹת שֶׁל קְרוֹבָיו וְלוּלֵא הָרוֹפְאִים הַטּוֹבִים, הוּא לֹא הָיָה חַי כַּיּוֹם. הַמַּקֵּל מַזְכִּיר לוֹ לְהוֹדוֹת תָּמִיד לַקָּדוֹשׁ בָּרוּךְ הוּא עַל כָּל הַטּוֹבוֹת שֶׁהוּא עוֹשֶׂה עִמָּנוּ. הוּא גַּם מַזְכִּיר לוֹ אֶת הָאֲנָשִׁים הָרַבִּים שֶׁנִּפְטְרוּ בְּאוֹתָהּ מַגֵּפָה קָשָׁה. בְּנוֹסָף, אָמַר הָרַב, הַמַּקֵּל מַזְכִּיר גַּם לְכָל הָאֲנָשִׁים שֶׁסּוֹבְבִים אוֹתוֹ לִזְכֹּר תָּמִיד לְהוֹדוֹת לַה' עַל כָּל הַדְּבָרִים הַטּוֹבִים שֶׁהוּא עוֹשֶׂה בִּשְׁבִילָם.
