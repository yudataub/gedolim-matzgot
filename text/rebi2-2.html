הַיְלָדִים הָאִלְמִים שֶׁלֹּא וִתְּרוּ

בִּשְׁכוּנָתוֹ שֶׁל הָרַב הַגָּדוֹל, רַבִּי יְהוּדָה הַנָּשִׂיא, חָיוּ שְׁנֵי יְלָדִים מְתוּקִים. לִשְׁנֵי הַיְלָדִים הָאֵלֶּה הָיָה דּוֹד שֶׁהָיָה תַּלְמִיד חָכָם גָּדוֹל מְאֹד. הַיְלָדִים הָאָחִים הָאֵלֶּה רָצוּ גַּם הֵם מְאֹד מְאֹד לִהְיוֹת תַּלְמִידֵי חֲכָמִים גְּדוֹלִים כְּמוֹ דּוֹדָם, וְלִלְמֹד תּוֹרָה בְּשִׂמְחָה וּבִשְׁקֵדָנוּת. אַךְ לִשְׁנֵי הַיְלָדִים הַלָּלּוּ הָיְתָה בְּעָיָה קָשָׁה וּמַעֲצִיבָה – הֵם הָיוּ אִלְמִים. אִלְמִים אֵלֶּה אֲנָשִׁים שֶׁאֵינָם יְכוֹלִים לְדַבֵּר. הֵם מְנַסִּים וּמְנַסִּים לְדַבֵּר, לְהַגִּיד מִלִּים וּלְהוֹצִיא קוֹל, אַךְ הֵם פְּשׁוּט לֹא מַצְלִיחִים לְדַבֵּר כְּלָל.

וְלַמְרוֹת שֶׁהֵם הָיוּ אִלְמִים וְלֹא יָכְלוּ לְדַבֵּר, שְׁנֵי הַיְלָדִים הַמְּתוּקִים הָאֵלֶּה לֹא הִתְיָאֲשׁוּ לְרֶגַע. הֵם לֹא הִתְיָאֲשׁוּ לְרֶגַע אֶחָד! מָה שֶׁהֵם עָשׂוּ כָּל יוֹם – אֵיךְ שֶׁהֵם הָיוּ רוֹאִים אֶת הָרַב הַגָּדוֹל, אֶת רַבִּי יְהוּדָה הַנָּשִׂיא, מַגִּיעַ לְבֵית הַמִּדְרָשׁ לְלַמֵּד תּוֹרָה, הֵם הָיוּ רָצִים, רָצִים, רָצִים בְּשִׂמְחָה וּמִתְיַשְּׁבִים יָשָׁר בַּשּׁוּרָה הָרִאשׁוֹנָה. הֵם הָיוּ יוֹשְׁבִים שָׁם בַּשּׁוּרָה הָרִאשׁוֹנָה בְּבֵית הַמִּדְרָשׁ, מַבִּיטִים בָּרַב וּמַקְשִׁיבִים בְּצָמָא גָּדוֹל לְכָל מָה שֶׁרַבִּי יְהוּדָה הַנָּשִׂיא לִמֵּד אֶת הַתַּלְמִידִים.

וְלַמְרוֹת שֶׁהֵם נוֹרָא נוֹרָא רָצוּ לִשְׁאֹל שְׁאֵלוֹת בַּלִּמּוּד כְּמוֹ שְׁאָר הַלּוֹמְדִים בְּבֵית הַמִּדְרָשׁ – הֵם לֹא יָכְלוּ לְדַבֵּר. וְזֶה הָיָה נוֹרָא נוֹרָא מְתַסְכֵּל לִפְעָמִים. זֶה מְתַסְכֵּל מְאֹד כְּשֶׁרוֹצִים כָּל כָּךְ לִשְׁאֹל שְׁאֵלָה וּלְהָבִין, וְאֵין יְכוֹלִים לְהוֹצִיא אַף מִלָּה מֵהַפֶּה. אַךְ גַּם אָז הֵם לֹא הִתְיָאֲשׁוּ כְּלָל. הֵם כָּל כָּךְ רָצוּ לִלְמֹד תּוֹרָה, שֶׁהֵם הִתְמִידוּ וְהִמְשִׁיכוּ לָבוֹא מִדֵּי יוֹם לְבֵית הַמִּדְרָשׁ וּלְהַקְשִׁיב בְּקֶשֶׁב רַב, לַמְרוֹת שֶׁלֹּא הִצְלִיחוּ לְדַבֵּר.

כְּשֶׁרָאָה רַבִּי יְהוּדָה הַנָּשִׂיא כַּמָּה אוֹתָם יְלָדִים מְתוּקִים מִתְאַמְּצִים בְּכָל כֹּחָם לִלְמֹד תּוֹרָה לַמְרוֹת שֶׁקָּשֶׁה לָהֶם כָּל כָּךְ, הוּא הֶחְלִיט לְהִתְפַּלֵּל עֲלֵיהֶם אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא. רַבִּי יְהוּדָה הַנָּשִׂיא הִתְפַּלֵּל מֵעֹמֶק הַלֵּב לַה' וְאָמַר: "ה', תַּעֲשֶׂה שֶׁאוֹתָם יְלָדִים מְתוּקִים יַבְרִיאוּ וְיוּכְלוּ לְדַבֵּר."

וְכָךְ הָיָה. הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַע אֶת תְּפִלָּתוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. יוֹם אֶחָד הֵם הִגִּיעוּ כְּהֶרְגֵּלָם לְבֵית הַמִּדְרָשׁ, וּפִתְאֹם הֵם הִצְלִיחוּ לְדַבֵּר! כָּל הָאֲנָשִׁים בְּבֵית הַמִּדְרָשׁ הִתְפַּלְאוּ מְאֹד, וְהִתְחִילוּ לִשְׁאֹל אוֹתָם שְׁאֵלוֹת רַבּוֹת בַּלִּמּוּד. אָז רָאוּ כֻּלָּם שֶׁהַיְלָדִים עָנִים מְאֹד מְאֹד יָפֶה עַל כָּל הַשְּׁאֵלוֹת, וְהֵבִינוּ שֶׁהֵם בֶּאֱמַת הִקְשִׁיבוּ בְּלִבָּם וְלָמְדוּ בִּשְׁקֵדָנוּת כָּל אוֹתָן הַשָּׁנִים.

💡 כְּשֶׁמִּתְאַמְּצִים וְלֹא מִתְיָאֲשִׁים מִקְּשָׁאִים, מַצְלִיחִים לִלְמֹד תּוֹרָה וּלְהָבִין אוֹתָהּ בְּצוּרָה נִפְלָאָה.
