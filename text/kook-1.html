הָרַב קוּק וְהַשְּׁתִיל בְּטָ"ו בִּשְׁבָט

בְּיוֹם חַג טַ"ו בִּשְׁבָט הַשָּׂמֵחַ וְהַמְּיֻחָד, הִגִּיעַ הָרַב הָרָאשִׁי שֶׁל אֶרֶץ יִשְׂרָאֵל, מָרָן הָרַב אַבְרָהָם יִצְחָק הַכֹּהֵן קוּק זֵכֶר צַדִּיק לִבְרָכָה, לַחֲגִיגָה גְּדוֹלָה וּמְרַגֶּשֶׁת בַּמּוֹשָׁבָה הַחֲדָשָׁה מַגְדִּיאֵל, שֶׁכַּיּוֹם אָנוּ מַכִּירִים אוֹתָהּ בְּשֵׁם הוֹד הַשָּׁרוֹן. אַנְשֵׁי הַמּוֹשָׁבָה וְהָאוֹרְחִים הָרַבִּים שֶׁהִתְכַּנְסוּ בַּמָּקוֹם שָׂמְחוּ מְאֹד לִרְאוֹת אֶת הָרַב הֶחָשׁוּב מַגִּיעַ לַחֹג עִמָּהֶם אֶת חַג הָאִילָנוֹת בְּאַהֲבָה רַבָּה. כְּחֵלֶק מֵהַחֲגִיגָה הַחֲגִיגִית, רָצוּ אַנְשֵׁי הַמּוֹשָׁבָה לְכַבֵּד אֶת הָרַב קוּק בִּכְבוֹד גָּדוֹל, וְהִצִּעוּ לוֹ לִנְטֹעַ שְׁתִיל צָעִיר בְּאַדְמַת הַמּוֹשָׁבָה.

לִכְבוֹד מַעֲשֵׂה הַנְּטִיעָה הֶחָשׁוּב, הִגִּישׁוּ לָרַב מַעְדֵּר כְּדֵי שֶׁיּוּכַל לַחְפֹּר בּוֹ אֶת הַחֹר בָּאֲדָמָה. אַךְ לְהַפְתָּעָתָם הָרַבָּה שֶׁל כָּל הָאֲנָשִׁים שֶׁעָמְדוּ מִסָּבִיב, הָרַב קוּק סֵרֵב לְקַחַת אֶת הַמַּעְדֵּר מִיְּדֵיהֶם. בִּמְקוֹם לְהִשְׁתַּמֵּשׁ בַּכְּלִי הָעֲבוֹדָה, הָרַב הִתְכּוֹפֵף לְמַטָּה לְעֵבֶר הָעָפָר, וּבְיָדָיו שֶׁלּוֹ מַמָּשׁ הֵחֵל לְטַפֵּל בָּאֲדָמָה בְּעַדְנוּת וּבְמַסִּירוּת. כָּל גּוּפוֹ שֶׁל הָרַב רָעַד מֵרֹב הִתְרַגְּשׁוּת וּקְדֻשָּׁה כְּשֶׁנָּגַע בְּיָדָיו בְּאַדְמַת אֶרֶץ יִשְׂרָאֵל הַקְּדוֹשָׁה.

כַּאֲשֶׁר סִיֵּם הָרַב לִנְטֹעַ אֶת הַשְּׁתִיל הָרַךְ בָּאֲדָמָה, הוּא נִשְׁאַר כָּפוּף עַל הָאָרֶץ וְלֹא קָם עַל רַגְלָיו. הַקָּהָל הַמֻּפְתָּע שֶׁעָמַד סְבִיבוֹ לֹא יָדַע וְלֹא הֵבִין מָה קָרָה לָרַב, וּמַדּוּעַ הוּא מַמְשִׁיךְ לִרְכֹּן כָּכָה עַל הָעָפָר. הָאֲנָשִׁים הִבִּיטוּ בּוֹ בְּתִמָּהוֹן רַב וְחִכּוּ לִרְאוֹת מָה יִקְרֶה. אָז הִתְקָרֵב אֵלָיו הָרַב זְאֵב גּוֹלְד זֵכֶר צַדִּיק לִבְרָכָה, הוֹשִׁיט לוֹ יָד וְהֵרִים אוֹתוֹ מֵהָאֲדָמָה. כְּשֶׁהֵרִים אוֹתוֹ, רָאָה הָרַב גּוֹלְד שֶׁפָּנָיו שֶׁל הָרַב קוּק הָיוּ מְשׁוּלְהָבוֹת בְּאוֹר מְיֻחָד וּבְעֵינָיו הָיוּ דְּמָעוֹת שֶׁל הִתְרַגְּשׁוּת גְּדוֹלָה.

אֶחָד מִמְּקֹרָבָיו שֶׁל הָרַב, שֶׁרָאָה אֶת הַדְּמָעוֹת וְאֶת הַהִתְרַגְּשׁוּת הָרַבָּה, הִתְפַּלֵּא מְאֹד וְשָׁאַל אוֹתוֹ: "מַדּוּעַ הָרַב כָּל כָּךְ נִרְגָּשׁ?". הָרַב קוּק עָנָה לוֹ וְהִסְבִּיר שֶׁנְּטִיעַת עֵץ פְּרִי בְּאֶרֶץ יִשְׂרָאֵל הִיא לֹא סְתָם מַעֲשֶׂה חַקְלָאִי רָגִיל. הוּא אָמַר שֶׁזֶּהוּ מַעֲשֶׂה מְיֻחָד שֶׁל דְּבֵקוּת בַּמִּדּוֹת שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁהֲרֵי הַפְּעֻלָּה הָרִאשׁוֹנָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה בְּגַן עֵדֶן הָיְתָה נְטִיעַת עֵצִים. הָרַב הוֹסִיף וְסִפֵּר שֶׁמִּרֹב הִתְרַגְּשׁוּת עֲצוּמָה מֵהַמִּצְוָה הַגְּדוֹלָה הָזֹאת, הָיָה לוֹ קָשֶׁה מְאֹד לַעֲזֹב אֶת הָאֲדָמָה וְלַעֲמֹד עַל רַגְלָיו.

💡 כַּאֲשֶׁר אָנוּ נוֹטְעִים עֵצִים בְּאֶרֶץ יִשְׂרָאֵל, אָנוּ מִתְחַבְּרִים לַה' וְהוֹלְכִים בִּדְרָכָיו.
