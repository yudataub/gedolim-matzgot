שָׁלוֹם לַמְּלָכִים

בְּמֶשֶׁךְ שָׁנִים רַבּוֹת, בִּתְקוּפַת רַבָּנַן סָבוֹרָאֵי, עָבְרוּ עַל עַם יִשְׂרָאֵל בְּבָבֶל יָמִים קָשִׁים מְאֹד. מַלְכֵי פָּרַס גָּזְרוּ גְּזֵרוֹת נוֹרָאוֹת לְבַטֵּל אֶת הַשַּׁבָּת, סָגְרוּ אֶת בָּתֵּי הַכְּנֶסֶת וְאֶת הַיְּשִׁיבוֹת לְעַשְׂרוֹת שָׁנִים, וְרָדְפוּ אֶת מַנְהִיגֵי הָעָם. אַךְ חַכְמֵי יִשְׂרָאֵל שָׁמְרוּ עַל גַּחֶלֶת הַתּוֹרָה וְעַל הַגְּמָרָא בִּמְסִירוּת נֶפֶשׁ שְׁקֵטָה, עַד שֶׁנִּפְתְּחוּ הַיְּשִׁיבוֹת מֵחָדָשׁ.

לִקְרַאת סוֹף תְּקוּפַת הַסָּבוֹרָאִים הִתְרַחֵשׁ מְאֹרָע גָּדוֹל. צְבָאוֹת הָעֲרָבִים בְּרָאשׁוּת הַחָ'לִיף עָלִי בֶּן אֲבִי טָאלֶב כָּבְשׁוּ אֶת אֶרֶץ בָּבֶל מִידֵי הַפַּרְסִים, וְהִגִּיעוּ עַד לָעִיר פִּירוֹז שָׁבוּר. הַיְּהוּדִים בְּבָבֶל חָשְׁשׁוּ מֵאֵימַת הַמִּלְחָמָה וּמֵחִלּוּפֵי הַשִּׁלְטוֹן, אַךְ מַנְהִיגֵי הַדּוֹר יָדְעוּ שֶׁיֵּשׁ לִנְהֹג בְּאֹמֶץ, בִּתְבוּנָה וּבִכְבוֹד מַלְכוּת.

בְּאוֹתָהּ עֵת עָמַד בְּרֹאשׁ הַיְּשִׁיבָה הֶחָכָם הַגָּדוֹל רַב יִצְחָק, וּלְצִדּוֹ עָמַד מַר בּוּסְתְּנַאי, שֶׁהָיָה צֶאֱצָא מְיֻחָס מִזֶּרַע דָּוִד הַמֶּלֶךְ. כַּאֲשֶׁר קָרַב הַחָ'לִיף עָלִי אֶל הָעִיר, יָצָא רַב יִצְחָק רֹאשׁ הַיְּשִׁיבָה לִקְרָאתוֹ, וְהִקְבִּיל אֶת פָּנָיו בְּסֵבֶר פָּנִים יָפוֹת וּבִכְבוֹד רַב. יַחַד עִמּוֹ יָצָא גַּם מַר בּוּסְתְּנַאי בְּרֹאשׁ קָהָל גָּדוֹל מִיִּשְׂרָאֵל כְּדֵי לְכַבֵּד אֶת הַשַּׁלִּיט הֶחָדָשׁ. הַחָ'לִיף עָלִי רָאָה אֶת מַנְהִיגֵי הַיְּהוּדִים, קִבֵּל אוֹתָם בְּהַעֲרָכָה רַבָּה, וְהֶעֱנִיק לָהֶם מַתָּנוֹת יְקָרוֹת.

מְסֻפָּר בַּמָּסוֹרֶת שֶׁבְּעֵת שֶׁעָמַד מַר בּוּסְתְּנַאי לִפְנֵי הַשַּׁלִּיט, בָּא זְבוּב טָרְדָנִי וְעָקַץ אוֹתוֹ בְּמִצְחוֹ עַד שֶׁזָּב דָּם. מַר בּוּסְתְּנַאי שָׁמַר עַל אִפּוּק מֻפְלָא: הוּא לֹא הֵזִיז אֶת רַגְלוֹ, לֹא נִבְהַל וְלֹא הֵרִים אֶת יָדוֹ לְגָרֵשׁ אֶת הַזְּבוּב, אֶלָּא הִמְשִׁיךְ לַעֲמוֹד בְּיִרְאַת כָּבוֹד. כַּאֲשֶׁר רָאָה הַשַּׁלִּיט אֶת הַדָּם הַיּוֹרֵד עַל פָּנָיו, תָּמַהּ וְשָׁאַל מַדּוּעַ אֵינוֹ מַבְרִיחַ אֶת הַזְּבוּב מֵעַל מִצְחוֹ.

הֵשִׁיב לוֹ מַר בּוּסְתְּנַאי בַּעֲנָוָה כִּי כָּךְ יָרְשׁוּ מֵאֲבוֹתֵיהֶם לְבֵית דָּוִד הַמֶּלֶךְ, שֶׁכַּאֲשֶׁר עוֹמְדִים לִפְנֵי מֶלֶךְ, עוֹמְדִים בְּדֶרֶךְ אֶרֶץ גְּמוּרָה, בְּלִי לְשַׂחֵק וּבְלִי לְהָרִים יָד שֶׁלֹּא לְצֹרֶךְ הַמֶּלֶךְ. הַחָ'לִיף עָלִי הִתְפַּעֵל עֲמֻקּוֹת מִמִּדַּת הָאִפּוּק וּמִדֶּרֶךְ הָאֶרֶץ שֶׁלּוֹ. הוּא אִשֵּׁר מִיָּד אֶת מַעֲמָדוֹ שֶׁל מַר בּוּסְתְּנַאי כְּרֹאשׁ הַגּוֹלָה לְכָל יְהוּדֵי בָּבֶל. מְסֻפָּר גַּם כִּי לְאוֹת כָּבוֹד נָתַן לוֹ הַשַּׁלִּיט לְאִשָּׁה אֶת בַּת מֶלֶךְ פָּרַס שֶׁנִּשְׁבְּתָה, וְהִיא הִתְגַּיְּרָה וְנִשְּׂאָה לוֹ כְּדַת מֹשֶׁה וְיִשְׂרָאֵל. מֵאָז נִקְבַּע צִיּוּר הַזְּבוּב עַל חוֹתָמָם הָרִשְׁמִי שֶׁל רָאשֵׁי הַגּוֹלָה.

הַמִּפְגָּשׁ הַמְּכֻבָּד שֶׁל רַב יִצְחָק וּמַר בּוּסְתְּנַאי עִם הַחָ'לִיף עָלִי הֵבִיא בְּשׂוֹרָה טוֹבָה לְכָל הַיְּהוּדִים. הַשַּׁלִּיט הֶעֱנִיק חֵרוּת דָּתִית מְלֵאָה, וּבָתֵּי הַמִּדְרָשׁ וְהַיְּשִׁיבוֹת בְּסוּרָא, בִּפְוּמְבְּדִיתָא וּבִנְהַרְדְּעָא חָזְרוּ לִפְרוֹחַ בְּשַׁלְוָה, כִּתְחִלָּתָהּ שֶׁל תְּקוּפָה חֲדָשָׁה וּמְאִירָה בְּתוֹלְדוֹת עַם יִשְׂרָאֵל.

💡 כַּאֲשֶׁר הוֹלְכִים בְּדֶרֶךְ הַתּוֹרָה, מִתְנַהֲגִים בְּדֶרֶךְ אֶרֶץ וְשׁוֹלְטִים בְּעַצְמֵנוּ בְּאִפּוּק וּבְנִמּוּס, זוֹכִים לְקִדּוּשׁ שֵׁם שָׁמַיִם וּלְכָבוֹד בְּעֵינֵי כָּל רוֹאֵינוּ.
