הַבְּרָכָה וְהַדְּמָעוֹת שֶׁל מָרָן

בְּבֵיתוֹ שֶׁל אַחַד הָאַבְרֵכִים הַצְּעִירִים בִּירוּשָׁלַיִם צִלְצֵל מַכְשִׁיר הַטֶּלֶפוֹן. הָאַבְרֵךְ הֵרִים אֶת הַשְּׁפוֹפֶרֶת וְשָׁמַע קוֹל מִן הָעֵבֶר הַשֵּׁנִי. הַדּוֹבֵר שָׁאַל אוֹתוֹ אִם הוּא הָרַב פְּלוֹנִי, וְהָאַבְרֵךְ עָנָה כֵּן וְשָׁאַל מִי מְדַבֵּר. הָאִישׁ בַּטֶּלֶפוֹן סִפֵּר כִּי הוּא מְדַבֵּר מִבֵּיתוֹ שֶׁל הָרַב הַגָּדוֹל עוֹבַדְיָה יוֹסֵף, וְהוֹסִיף כִּי הָרַב מְבַקֵּשׁ מִמֶּנּוּ לָבוֹא וְלִרְאוֹת אוֹתוֹ. הָאַבְרֵךְ הַצָּעִיר נִבְהַל מְאֹד וְלֹא הֶאֱמִין לְמַה שֶּׁשָּׁמַע. הוּא תָּמַהּ בְּלִבּוֹ אֵיךְ גְּדוֹל הַדּוֹר מַכִּיר אוֹתוֹ וּמַדּוּעַ הוּא קוֹרֵא לוֹ לְבֵיתוֹ. לַמְרוֹת הַתְּמִיהָה הַגְּדוֹלָה, הוּא לֹא שָׁאַל שְׁאֵלוֹת מִיּוּתָרוֹת. הוּא לָבַשׁ מִיָּד אֶת הַחֲלִיפָה שֶׁלּוֹ, חָבַשׁ אֶת הַכּוֹבַע, וּמִהֵר בְּרַגְלַיִם נִרְגָּשׁוֹת אֶל בֵּיתוֹ שֶׁל מָרָן.

כַּאֲשֶׁר הִגִּיעַ הָאַבְרֵךְ אֶל הַבַּיִת, הִכְנִיסוּ אוֹתוֹ בְּכָבוֹד וְהִפְנוּ אוֹתוֹ אֶל חֲדַר הַלִּמּוּד שֶׁל הָרַב עוֹבַדְיָה. הָאַבְרֵךְ נִכְנַס פְּנִימָה וְהִרְגִּישׁ כִּי לִבּוֹ פּוֹעֵם בִּפְרָאוּת מֵרֹב הִתְרַגְּשׁוּת. הוּא הִבִּיט עַל הַשֻּׁלְחָן הַגָּדוֹל וְרָאָה שָׁם קוּנְטְרֵס קָטָן בְּעִנְיְנֵי תְּרוּמוֹת וּמַעַשְׂרוֹת. קוּנְטְרֵס זֶה הָיָה חוֹבֶרֶת תּוֹרָנִית שֶׁהָאַבְרֵךְ עַצְמוֹ חִבֵּר וְכָתַב זְמַן קָצָר קֹדֶם לָכֵן. מָרָן הָרַב עוֹבַדְיָה הֵרִים אֶת עֵינָיו מִן הַסְּפָרִים, הִבִּיט בּוֹ בְּחִיּוּךְ וְאָמַר לוֹ בְּחֹם: "בְּנִי, חֲבִיבִי, בּוֹא וְגַשׁ הֵנָּה". הָאַבְרֵךְ הִתְקָרֵב אֶל הַשֻּׁלְחָן, וּמָרָן אָמַר לוֹ כִּי הוּא עוֹסֵק עַכְשָׁו בְּלִמּוּד הִלְכוֹת תְּרוּמוֹת וּמַעַשְׂרוֹת, וְהוּא הִסְתַּיֵּעַ מְאֹד בַּקּוּנְטְרֵס הֶחָשׁוּב שֶׁכָּתַב הָאַבְרֵךְ וְשֶׁהִגִּיעַ לְיָדָיו.

הָאַבְרֵךְ הָיָה נִדְהָם וְלֹא יָדַע אֶת נַפְשׁוֹ מֵרֹב הִשְׁתּוֹמְמוּת. הוּא לֹא הֶאֱמִין שֶׁהַחוֹבֶרֶת הַקְּטַנָּה שֶׁהוּא חִבֵּר עָזְרָה לְמָרָן גְּדוֹל הַדּוֹר, שֶׁכָּל הַתּוֹרָה כֻּלָּהּ גְּלוּיָה וּפְרוּשָׂה לְפָנָיו כְּכַף יָד. הוּא הֵבִין כִּי זוֹהִי מִדַּת הָעֲנָוָה הָעֲצוּמָה שֶׁל הָרַב, שֶׁבִּגְלָלָהּ הוּא טָרַח לִקְרֹא לוֹ כְּדֵי לְעוֹדֵד אֶת רוּחוֹ בְּחֹם וּבְאַהֲבָה. הָאַבְרֵךְ הִרְכִּין אֶת רֹאשׁוֹ וְנִצֵּל אֶת הָרֶגַע הַמְּיֻחָד. הוּא סִפֵּר לָרַב כִּי עָבְרוּ כַּמָּה שָׁנִים מֵאָז נִשּׂוּאָיו וַעֲדַיִן לֹא זָכָה לִילָדִים, וּבִקֵּשׁ מֵהָרַב שֶׁיַּבְטִיחַ לוֹ שֶׁיִּזְכֶּה לְכָךְ. מָרָן הֵשִׁיב לוֹ בְּעַנְוְתָנוּת כִּי הוּא אֵינוֹ יָכוֹל לְהַבְטִיחַ דָּבָר, אַךְ הוּא מְבָרֵךְ אוֹתוֹ וְיִתְפַּלֵּל עָלָיו מִכָּל הַלֵּב.

הָאַבְרֵךְ הַנִּרְגָּשׁ לֹא וִתֵּר, הוּא בָּכָה וְהִתְחַנֵּן מְאֹד לְהַבְטָחָה. אָז הֵרִים מָרָן אֶת עֵינָיו לַשָּׁמַיִם, וּדְמָעוֹת חַמּוֹת הֵחֵלּוּ לִזְלֹג עַל לְחָיָיו. הָרַב אָמַר שׁוּב כִּי לְהַבְטִיחַ אֵינוֹ יָכוֹל, אַךְ הוּא יִתְפַּלֵּל עָלָיו, וּבֵרַךְ אוֹתוֹ בְּחֹם עָצוֹם. וְכָךְ אָכֵן הָיָה: כַּעֲבֹר שָׁנָה אֶחָת, חָבְקוּ הָאַבְרֵךְ וְאִשְׁתּוֹ בֵּן זָכָר בְּמַזָּל טוֹב וּבְשִׂמְחָה גְּדוֹלָה.

💡 מִמָּרָן אָנוּ לוֹמְדִים עֲנָוָה רַבָּה וְכֹחַ עָצוֹם שֶׁל תְּפִלָּה הַיּוֹצֵאת מֵעֹמֶק הַלֵּב.
