הַהֲלִיכָה בַּשֶּׁלֶג לִכְבוֹד אַבָּא

הָרַב אֱלִישָׁע וִישְׁלִיצְקִי הָיָה אָדָם גָּדוֹל שֶׁהִקְפִּיד מְאֹד עַל מִצְוַת כִּבּוּד הוֹרִים. הוּא יָדַע שֶׁמִּצְוָה זוֹ הִיא מִצְוָה חֲשׁוּבָה מְאֹד שֶׁל הַכָּרַת טוֹבָה לַהוֹרִים עַל כָּל מָה שֶׁעָשׂוּ לְמַעֲנֵנוּ. הָרַב אֱלִישָׁע הֶרְאָה לְכָל סְבִיבָתוֹ אֵיךְ מְכַבְּדִים הוֹרִים בְּאַהֲבָה רַבָּה, בְּכָל מַצָּב וּבְכָל עֵת.

כַּאֲשֶׁר סַבָּא, אָבִיו שֶׁל הָרַב אֱלִישָׁע, הָיָה מַגִּיעַ לְבֵיתוֹ לִסְעוּדַת שַׁבָּת, הָרַב אֱלִישָׁע לֹא הָיָה עוֹשֶׂה אֶת הַקִּדּוּשׁ בְּעַצְמוֹ. הוּא אָמַר לְאָבִיו בְּכָבוֹד רַב שֶׁהוּא הַמֶּלֶךְ בַּבַּיִת, וְעַל כֵּן מַגִּיעַ לוֹ לְבָרֵךְ וְלַעֲשׂוֹת אֶת הַקִּדּוּשׁ עֲבוּר כָּל בְּנֵי הַמִּשְׁפָּחָה. הָרַב אֱלִישָׁע הִרְגִּישׁ שֶׁזּוֹ זְכוּת גְּדוֹלָה לָתֵת לְאָבִיו אֶת הַכָּבוֹד הָרָאוּי לוֹ בְּשֻׁלְחַן הַשַּׁבָּת.

כַּאֲשֶׁר אָבִיו שֶׁל הָרַב אֱלִישָׁע חָלָה, הוּא עָבַר לָגוּר בְּבֵית אָבוֹת. בֵּית הָאָבוֹת הָיָה מָקוֹם שֶׁבּוֹ מִתְגּוֹרְרִים וּמִתְאָרְחִים אֲנָשִׁים זְקֵנִים, וְהוּא הָיָה מְמֻקָּם בְּמֶרְחַק חֲצִי שָׁעָה שֶׁל הֲלִיכָה בְּרֶגֶל מִבֵּיתוֹ שֶׁל הָרַב אֱלִישָׁע. הָרַב אֱלִישָׁע לֹא וִתֵּר אַף פַּעַם עַל מִצְוַת כִּבּוּד אָב, וְהָיָה הוֹלֵךְ מְיֻחָד בְּרֶגֶל כָּל יוֹם וָיוֹם כְּדֵי לְבַקֵּר אֶת אָבִיו הַחֹלֶה, לִשְׁאֹל בִּשְׁלוֹמוֹ וּלְשַׂמֵּחַ אֹתוֹ.

גַּם בְּיָמִים רְגִילִים וְגַם בְּיَوْם שַׁבָּת קֹדֶשׁ, הָרַב אֱלִישָׁע הָיָה צוֹעֵד אֶת כָּל הַדֶּרֶךְ, חֲצִי שָׁעָה לְכָל כִּוּוּן, כְּדֵי לִהְיוֹת לְצַד אָבִיו. בְּיָמִים רְגִילִים עָשָׂה זֹאת בְּעַצְמוֹ, וּבְיָמִים שֶׁל שַׁבָּת הָיְתָה כָּל הַמִּשְׁפָּחָה מִצְטָרֶפֶת אֵלָיו וְהוֹלֶכֶת יַחַד לְבֵית הָאָבוֹת כְּדֵי לְבַקֵּר אֶת סַבָּא.

פַּעַם אַחַת, בְּיَوْם שַׁבָּת, יָרַד שֶׁלֶג כָּבֵד מְאֹד. הַמֶּזֶג אֲוָויר הָיָה קַר וְהַדְּרָכִים הָיוּ מְכֻסּוֹת בִּשְׁכָבָה עֲבִיָּה שֶׁל שֶׁלֶג. הַהֲלִיכָה בַּחוּץ הָיְתָה קָשָׁה מְאֹד וּמְעַיֶּפֶת. אַף עַל פִּי כֵן, הָרַב אֱלִישָׁע לֹא חָשַׁב לְרֶגַע אֶחָד לְוַתֵּר עַל הַבִּקּוּר אֵצֶל אָבִיו. בְּנֵי הַמִּשְׁפָּחָה זוֹכְרִים הֵיטֵב אֵיךְ אֲבִיהֶם, הָרַב אֱלִישָׁע, הָיָה מְבוֹסֵס בַּשֶּׁלֶג הָעָמֹק, צוֹעֵד בִּמְאַמָּץ רַב וְאֵינוֹ מִתְיָאֵשׁ. הוּא צָעַד בַּשֶּׁלֶג וְאָמַר בְּנֶחְלָטִיּוּת: "אֲנִי אֵלֵךְ לְכַבֵּד אֶת אַבָּא שֶׁלִּי!".

הָרַב אֱלִישָׁע הִמְשִׁיךְ לִצְעֹד בַּקֹּר וּבַשֶּׁלֶג עַד שֶׁהִגִּיעַ אֶל אָבִיו. מַעֲשֶׂה זֶה הִשְׁאִיר רֹשֶׁם עָמֹק עַל כָּל בְּנֵי הַמִּשְׁפָּחָה וְעַל כָּל מִי שֶׁשָּׁמַע עַל כָּךְ. זֶה הָיָה הָרַב אֱלִישָׁע וִישְׁלִיצְקִי – אָדָם שֶׁלֹּא וִתֵּר עַל שׁוּם מַאֲמָץ וְעַל שׁוּם הִזְדַּמְנוּת כְּדֵי לְקַיֵּם אֶת מִצְוַת כִּבּוּד אָב בְּשִׂמְחָה וּבִמְסִירוּת רַבָּה.

💡 מִצְוַת כִּבּוּד הוֹרִים הִיא מִצְוָה יְקָרָה שֶׁל הַכָּרַת טוֹבָה, וְשָׁוֶה לְהִתְאַמֵּץ עֲבוּרָהּ בְּכָל הַלֵּב גַּם כְּשֶׁקָּשֶׁה.
