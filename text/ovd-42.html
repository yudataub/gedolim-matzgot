הַחֲתוּלִים וְאֲרוּחַת הַצָּהֳרַיִם

בְּכָל יוֹם שִׁשִּׁי בַּבֹּקֶר הָיָה מָרָן הָרַב עוֹבַדְיָה יוֹסֵף יוֹצֵא מִבֵּיתוֹ שֶׁבִּרְחוֹב אֶלְקָנָה וְהוֹלֵךְ אֶל בֵּית הַכְּנֶסֶת הַסָּמוּךְ לְבֵיתוֹ. בְּיָדוֹ הָאַחַת הֶחֱזִיק אֶת סְפָרָיו הָרַבִּים, וּבְיָדוֹ הַשְּׁנִיָּה הֶחֱזִיק אֶת מַחְבַּרְתּוֹ. הָרַב הִנְהִיג עַצְמוֹ בְּמִנְהָג מְיֻחָד: לְהַקְדִּישׁ אֶת יְמֵי שִׁשִּׁי לִכְתִיבַת תְּשׁוּבוֹת הַלָּכָתִיּוֹת. מֵהַבֹּקֶר וְעַד אַחַר הַצָּהֳרַיִם יָשַׁב מָרָן בְּבֵית הַמִּדְרָשׁ וְעָרַךְ אֶת תְּשׁוּבוֹתָיו בָּזוֹ אַחַר זוֹ בִּשְׁקִידָה נִפְלָאָה.

וּבְכָל יוֹם שִׁשִּׁי הָיְתָה הָרַבָּנִית הַמְּסוּרָה מַגִּיעָה אֶל בֵּית הַכְּנֶסֶת בְּדִיּוּק בְּשָׁעָה שְׁתֵּים עֶשְׂרֵה בַּצָּהֳרַיִם. הִיא הִנִּיחָה עַל שֻׁלְחָנוֹ צַלַּחַת עִם תַּבְשִׁיל חַם, וְשָׁבָה לְבֵיתָהּ. גַּבַּאי בֵּית הַכְּנֶסֶת, שֶׁהָיָה מַגִּיעַ כַּעֲבֹר שָׁעָה קַלָּה כְּדֵי לְהָכִין אֶת סֵפֶר הַתּוֹרָה לִקְרִיאַת הַתּוֹרָה, הָיָה מִתְנַדֵּב לְהַחֲזִיר אֶת הַצַּלַּחַת הָרֵיקָה לְבֵיתוֹ שֶׁל מָרָן.

בְּאֶחָד מֵאוֹתָם יְמֵי שִׁשִּׁי אָמַר הַגַּבַּאי לָרַבָּנִית כִּי הוּא חוֹשֵׁב שֶׁכְּדַאי לָהּ לְהוֹסִיף מְעַט אֹכֶל בַּצַּלַּחַת. הוּא סִפֵּר לָהּ כִּי הָרַב מְסַיֵּם אֶת כָּל הַמָּנָה עַד הַפֵּרוּר הָאַחֲרוֹן, וְיָכוֹל לִהְיוֹת שֶׁהוּא זָקוּק לְעוֹד אֹכֶל. הָרַבָּנִית הִתְפַּלְּאָה מְעַט, כִּי הִיא יָדְעָה שֶׁמָּרָן מְמַעֵט לֶאֱכֹל וְלִפְעָמִים אֵינוֹ אוֹכֵל כְּלָל. אַךְ הִיא לֹא הִתְנַגְּדָה, וּבְיוֹם שִׁשִּׁי הַבָּא הִגְדִּילָה אֶת מְנַת הָאֹכֶל בַּצַּלַּחַת וְהִנִּיחָה אוֹתָהּ עַל שֻׁלְחָנוֹ שֶׁל מָרָן.

אוּלָם גַּם בַּפַּעַם הַזֹּאת שָׁבָה הַצַּלַּחַת לְבֵיתָהּ כְּשֶׁהִיא רֵיקָה לְגַמְרֵי. הַגַּבַּאי שׁוּב אָמַר כִּי מוּטָב לְהוֹסִיף מְעַט אֹכֶל לְמַעַן בְּרִיאוּתוֹ שֶׁל מָרָן. גַּם בַּפַּעַם הַבָּאָה חָזַר הַדָּבָר עַל עַצְמוֹ: הַצַּלַּחַת חָזְרָה רֵיקָה, וּפְלִיאָתָהּ שֶׁל הָרַבָּנִית גָּבְרָה עוֹד יוֹתֵר.

בַּפַּעַם הַשְּׁלִישִׁית הֶחְלִיט הַגַּבַּאי לִבְדֹּק אֶת הָעִנְיָן בְּעַצְמוֹ, שֶׁהֲרֵי גַּם הוּא יָדַע כִּי מָרָן מְמַעֵט לֶאֱכֹל. הוּא הִקְדִּים אֶת הֲגָעָתוֹ לְבֵית הַכְּנֶסֶת בְּשָׁעָה קַלָּה וְסִיֵּם אֶת הֲכָנַת סֵפֶר הַתּוֹרָה מֻקְדָּם מִן הָרָגִיל. לְאַחַר מִכֵּן נִכְנַס אֶל אַחַד הָאֲרוֹנוֹת שֶׁבְּבֵית הַכְּנֶסֶת וְסָגַר עַל עַצְמוֹ אֶת הַדֶּלֶת כִּמְעַט לְגַמְרֵי. דֶּרֶךְ הַמִּרְוָח הַצַּר שֶׁנּוֹתַר בֵּין הַדֶּלֶת לָאָרוֹן הֵצִיץ בִּדְרִיכוּת וְחִכָּה לִרְאוֹת מָה יִקְרֶה.

תּוֹךְ כַּמָּה רְגָעִים נִפְתְּרָה הַתַּעֲלוּמָה. אַךְ יָצְאָה הָרַבָּנִית מִבֵּית הַכְּנֶסֶת, זִנְּקוּ כַּמָּה חֲתוּלִים מִן הַחַלּוֹן, טִפְּסוּ עַל הַשֻּׁלְחָן וְחִסְּלוּ אֶת כָּל תְּכוּלַת הַצַּלַּחַת עַד הַפֵּרוּר הָאַחֲרוֹן. בְּכָל אוֹתָהּ עֵת יָשַׁב מָרָן כְּשֶׁהוּא שָׁקוּעַ כֻּלּוֹ בְּתַלְמוּדוֹ, וְלֹא שָׂם לִבּוֹ כְּלָל לַמִּתְרַחֵשׁ סְבִיבוֹ.

💡 כַּאֲשֶׁר אָדָם שָׁקוּעַ בְּלִמּוּד הַתּוֹרָה בְּרִכּוּז עָמֹק וּבְאַהֲבָה רַבָּה, הוּא אֵינוֹ מַרְגִּישׁ בְּשׁוּם הַפְרָעָה שֶׁמִּסָּבִיב.
