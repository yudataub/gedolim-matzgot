הָרַב גּוֹרֵן וְהַשּׁוֹפָר בַּכּוֹתֶל הַמַּעֲרָבִי

הָרַב הָרָאשִׁי לְצַהַ"ל שֶׁל אָז, הָרַב שְׁלֹמֹה גּוֹרֵן זֵכֶר צַדִּיק לִבְרָכָה, הָיָה מֵרִאשׁוֹנֵי הַלּוֹחֲמִים שֶׁצָּעֲדוּ לְכִוּוּן יְרוּשָׁלַיִם. הָרַב שְׁלֹמֹה גּוֹרֵן עוֹדֵד אֶת הַחַיָּלִים לִפְנֵי שֶׁיָּצְאוּ לַקְּרָב, הוּא עוֹדֵד אֶת הַחַיָּלִים בְּמַהֲלַךְ הָרִיצָה, וְגַם בְּמַהֲלַךְ הַקְּרָב הַמָּתוּחַ בַּדֶּרֶךְ אֶל הַכּוֹתֶל הַמַּעֲרָבִי. בְּכָל הַדֶּרֶךְ הֶעֱלָה אֶת רוּחָם שֶׁל הַחַיָּלִים וְחִזֵּק אֶת לִבֹּתֵיהֶם, עַד שֶׁהִגִּיעַ בְּעַצְמוֹ בֵּין רִאשׁוֹנֵי הַחַיָּלִים אֶל הַכּוֹתֶל הַמַּעֲרָבִי.

הָרַב שְׁלֹמֹה גּוֹרֵן הִגִּיעַ אֶל הַכּוֹתֶל הַמַּעֲרָבִי כַּאֲשֶׁר הוּא אוֹחֵז סֵפֶר תּוֹרָה בְּיָדוֹ וְעִמּוֹ גַּם שׁוֹפָר. בְּהִתְרַגְּשׁוּת רַבָּה וּבִקְדֻשָּׁה, הָרַב שָׂם אֶת סֵפֶר הַתּוֹרָה בֵּין אַבְנֵי הַכּוֹתֶל, נָטַל אֶת הַשּׁוֹפָר וְתָקַע בּוֹ תְּקִיעָה גְּדוֹלָה. הָרַב וְהַחַיָּלִים הַנִּרְגָּשִׁים נָשְׂאוּ יַחְדָּו תְּפִלָּה זַכָּה וְאָמְרוּ פִּרְקֵי הוֹדָיָה. הֵם הוֹדוּ לַהַשֵּׁם מִכָּל הַלֵּב עַל הַנֵּס הַגָּדוֹל וְעַל הַחֶסֶד הַגָּדוֹל שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ לַחֲזֹר וּלְהִתְקָרֵב אֵלָיו, וְשֶׁנּוּכַל לְהִתְפַּלֵּל בַּמָּקוֹם הַקָּדוֹשׁ שֶׁמִּמֶּנּוּ הַשְּׁכִינָה מֵעוֹלָם לֹא זָזָה, מֵהַכּוֹתֶל הַמַּעֲרָבִי.

מְפַקֵּד חֲטִיבַת הַצַּנְחָנִים, אֲלוּף מִשְׁנֶה מוּטָה גּוּר, כָּתַב בְּסִפְרוֹ "הַר הַבַּיִת בְּיָדֵינוּ" כִּי הָרַב גּוֹרֵן וַחֲבוּרָתוֹ הָיוּ הָרִאשׁוֹנִים בִּרְחֲבַת הַכּוֹתֶל. מוּטָה גּוּר תִּאֵר כִּי מֵרֹב הִתְרַגְּשׁוּת, הָרַב נָפַל עַל רִצְפַת הַכּוֹתֶל, נִשַּׁק אוֹתָהּ וּבֵרֵךְ בְּשִׂמְחָה בִּרְכַּת "שֶׁהֶחֱיָנוּ". לְאַחַר מִכֵּן, הָרַב גּוֹרֵן שָׁלַח אֶת הָרֶכֶב שֶׁלּוֹ כְּדֵי לְהָבִיא אֶל הַכּוֹתֶל אֶת חָמִיו, הַנָּזִיר הַיְרוּשַׁלְמִי – הָרַב דָּוִד כֹּהֵן, שֶׁהָיָה אִישׁ צַדִּיק, קָדוֹשׁ וּמְיֻחָד בְּמִינוֹ. מוּטָה גּוּר תִּאֵר אֶת הַנָּזִיר הַיְרוּשַׁלְמִי עוֹמֵד דָּבוּק מַמָּשׁ לְאַבְנֵי הַכּוֹתֶל. הוּא כָּתַב כִּי מִבַּעַד לְגּוּפוֹ הַכְּמוֹ מְשׁוּתָק שֶׁל הַנָּזִיר, הִרְגִּישׁ אֶת דְּפִיקוֹת הַלֵּב הַיְהוּדִי בּוֹקְעוֹת מִתּוֹךְ הָאֶבֶן. הֵם עָמְדוּ יַחְדָּו דְּבוּקִים אֶל הַכּוֹתֶל מִסְפַּר דַּקּוֹת מְרוֹמָמוֹת – הַנָּזִיר הָרַב דָּוִד כֹּהֵן, הָרַב שְׁלֹמֹה גּוֹרֵן, מוּטָה גּוּר וְהַכּוֹתֶל הַקָּדוֹשׁ.

וְעוֹד בְּאוֹתוֹ הַיּוֹם, הִגִּיעוּ הָרַב וְהַחַיָּלִים אֶל מָקוֹם מְיֻחָד נוֹסָף שֶׁנִּקְרָא "קֶבֶר רָחֵל", סָמוּךְ לִירוּשָׁלַיִם, בָּעִיר בֵּית לֶחֶם. כַּאֲשֶׁר הִגִּיעוּ לְקִבְרָהּ שֶׁל רָחֵל אִמֵּנוּ, הָרַב שְׁלֹמֹה גּוֹרֵן אָמַר אֶת הַמִּלִּים הַמְּרַגְּשׁוֹת: "מִנְעִי קוֹלֵךְ מִבֶּכִי וְעֵינַיִךְ מִדִּמְעָה כִּי יֵשׁ שָׂכָר לִפְעֻלָּתֵךְ... וְשָׁבוּ בָנִים לִגְבוּלָם". וּלְמָחֳרָת, הָרַב שְׁלֹמֹה גּוֹרֵן וְחַיָּלֵי צְבָא הֲגָנָה לְיִשְׂרָאֵל הִגִּיעוּ לְמָקוֹם מְיֻחָד נוֹסָף, וְהוּא "מְעָרַת הַמַּכְפֵּלָה".

💡 תָּמִיד נִזְכֹּר לְהוֹדוֹת לַהַשֵּׁם עַל הַנִּסִּים וּלְאֱהֹב אֶת יְרוּשָׁלַיִם וְאֶת הַמְּקוֹמוֹת הַקְּדוֹשִׁים.
