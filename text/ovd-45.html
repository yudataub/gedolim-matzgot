הַתֻּכִּי וְהַשַּׁקְדָנוּת שֶׁל מָרָן

גְּדֻלָּתוֹ וּשְׁקִיעָתוֹ שֶׁל מָרָן הָרַב עוֹבַדְיָה יוֹסֵף זַצַ"ל בְּלִמּוּד הַתּוֹרָה הָיוּ עֲצוּמוֹת מְאֹד. פְּעָמִים רַבּוֹת הָיָה הָרַב שָׁקוּעַ בַּסְּפָרִים בְּכָל לִבּוֹ, עַד שֶׁכְּלָל לֹא שָׂם לֵב לְמָה שֶׁמִּתְרַחֵשׁ סְבִיבוֹ. גַּם כַּאֲשֶׁר קָרָה מַשֶּׁהוּ שֶׁעוֹרֵר תְּשׂוּמַת לֵב בְּאוֹפֶן בָּרוּר, הָרַב הִמְשִׁיךְ לִלְמֹד בְּרִכּוּז רַב בְּלִי לְהַפְסִיק.

פַּעַם אַחַת, נֶאֱלַץ הָרַב עוֹבַדְיָה יוֹסֵף לַעֲשׂוֹת אֶת שַׁבַּת הַקֹּדֶשׁ בָּעִיר אֶלְעָד. הָרַב הִגִּיעַ לָעִיר לְרֶגֶל שִׂמְחָה מִשְׁפַּחְתִּית שֶׁל אַחַת מִבְּנוֹתָיו. תּוֹשָׁבֵי הָעִיר אֶלְעָד הִתְרַגְּשׁוּ מְאֹד וְחָרְדוּ לִכְבוֹדוֹ שֶׁל הַגָּאוֹן הַגָּדוֹל. אֶחָד מִמְּקֹרָבָיו שֶׁל הָרַב זָכָה בְּכָבוֹד עָצוּם: הוּא קִבֵּל אֶת הַזְּכוּת לְאָרֵחַ אֶת מָרָן הָרַב עוֹבַדְיָה וְאֶת הָרַבָּנִית בְּבֵיתוֹ בְּמַהֲלַךְ כָּל הַשַּׁבָּת.

לִפְנֵי שֶׁהִתְחִילָה הַשַּׁבָּת, פָּנָה מָרָן אֶל בַּעַל הַבַּיִת וּבִקֵּשׁ מִמֶּנּוּ בַּבַּקָּשָׁה הַבָּאָה: "אָנָּא מִמְּךָ, הַשְׁאֵר עֲבוּרִי מְנוֹרָה קְטַנָּה דּוֹלֶקֶת בַּחֲדַר הַלִּמּוּד הַסָּמוּךְ לַחֲדַר הַשֵּׁנָה. אֲנִי רוֹצֶה שֶׁאוּכַל לָשֶׁבֶת וְלִלְמֹד בַּסְּפָרִים בְּמֶשֶׁךְ כָּל הַלַּיְלָה". בַּעַל הַבַּיִת הַמְּאָרֵחַ נַעֲנָה בְּשִׂמְחָה וּבְרָצוֹן רַב. הוּא הֵכִין עֲבוּר הָרַב אֶת הַחֶדֶר הַסָּמוּךְ, וְהִשְׁאִיר שָׁם מְנוֹרָה דּוֹלֶקֶת לְמֶשֶׁךְ כָּל הַלַּיְלָה.

לְאַחַר שֶׁסִיְּמוּ אֶת סְעֻדַּת הַשַּׁבָּת הַחֲגִיגִית, פָּנָה מָרָן הָרַב עוֹבַדְיָה אֶל הַחֶדֶר שֶׁהוּכַן עֲבוּרוֹ. הוּא פָּתַח אֶת הַסֵּפֶר וְשָׁקַע בְּלִמּוּד הַתּוֹרָה הַקְּדוֹשָׁה, כְּמוֹ מִי שֶׁצּוֹלֵל בְּמַיִם אַדִּירִים. הוּא הָיָה מְרֻכָּז כֻּלּוֹ בַּמִּלִּים הַקְּדוֹשׁוֹת.

פִּתְאֹם, נִשְׁמְעָה בֶּחָלָל הַחֶדֶר קְרִיאָה חַדָּה וּבְרוּרָה: "שַׁבַּת שָׁלוֹם!". מִי אָמַר זֹאת? זֶה לֹא הָיָה אֶחָד מִבְּנֵי הַבַּיִת. בְּפִנַּת הַחֶדֶר עָמַד כְּלוּב גָּדוֹל, וּבְתוֹכוֹ הָיָה תֻּכִּי מְדַבֵּר. הַתֻּכִּי הַזֶּה יָדַע לוֹמַר מִלִּים רַבּוֹת, כְּמוֹ "שַׁבַּת שָׁלוֹם", "בֹּקֶר טוֹב", "עֶרֶב טוֹב" וְעוֹד מִלִּים אֲחָרוֹת. הַתֻּכִּי שֶׁעָמַד בַּכְּלוּב הִכְרִיז בְּקוֹל רָם: "שַׁבַּת שָׁלוֹם!". מָרָן הָרַב עוֹבַדְיָה, שֶׁהָיָה שָׁקוּעַ עָמֹק בַּסֵּפֶר, הֵשִׁיב לוֹ מִבְּלִי מֵשִׂים: "שַׁבַּת שָׁלוֹם", וְהִמְשִׁיךְ לִלְמֹד בְּלִי לַעֲצֹר.

חָלְפוּ כַּמָּה דַּקּוֹת, וְשׁוּב הִשְׁמִיעַ הַתֻּכִּי אֶת קוֹלוֹ וְקָרָא: "שַׁבַּת שָׁלוֹם!". שׁוּב הֵשִׁיב הָרַב: "שַׁבַּת שָׁלוֹם", וְהִמְשִׁיךְ לִלְמֹד בִּשְׁקִידָה. הָרַב אֲפִלּוּ לֹא הֵרִים אֶת עֵינָיו מִן הַסֵּפֶר כְּדֵי לִרְאוֹת מִי מְדַבֵּר אִתּוֹ. הַמַּחֲזֶה הַזֶּה חָזַר עַל עַצְמוֹ שׁוּב וָשׁוּב, עֶשְׂרוֹת פְּעָמִים בְּמֶשֶׁךְ כָּל הַלַּיְלָה! הַתֻּכִּי קוֹרֵא בְּקוֹל: "שַׁבַּת שָׁלוֹם!", וְהָרַב מֵשִׁיב מִיָּד: "שַׁבַּת שָׁלוֹם", וּמַמְשִׁיךְ לִלְמֹד בִּדְבֵקוּת.

בַּבֹּקֶר, כַּאֲשֶׁר נִגַּשׁ בַּעַל הַבַּיִת אֶל חַדְרוֹ שֶׁל הָרַב כְּדֵי לִדְרֹשׁ בִּשְׁלוֹמוֹ, פָּנָה אֵלָיו מָרָן וְאָמַר לוֹ: "יִשְׁמַע נָא כְּבוֹדוֹ, יֵשׁ כָּאן בַּסְּבִיבָה אֵיזֶה אָדָם שֶׁאֵין לוֹ כָּל כָּךְ דֶּרֶךְ אֶרֶץ. הוּא בָּא אֵלַי, כַּנִּרְאֶה דֶּרֶךְ הַחַלּוֹן, וְאוֹמֵר לִי 'שַׁבַּת שָׁלוֹם', פַּעַם אַחַת, וְעוֹד פַּעַם, וְעוֹד פַּעַם... מָה זֶה?".

בַּעַל הַבַּיִת עָמַד בִּמְקוֹמוֹ כַּהֲלוּם רַעַם וּבְתִמָּהוֹן עָצוּם. הוּא לֹא הֶאֱמִין לְמִשְׁמַע אָזְנָיו! הַאִם יִתָּכֵן שֶׁהָרַב כְּלָל לֹא הֵרִים אֶת עֵינָיו מִן הַסֵּפֶר בְּמֶשֶׁךְ כָּל הַלַּיְלָה, עַד כְּדֵי כָּךְ שֶׁלֹּא הִבְחִין בַּכְּלוּב הַגָּדוֹל הָעוֹמֵד בְּפִנַּת הַחֶדֶר?! אָכֵן כֵּן, עַד כְּדֵי כָּךְ הָיְתָה עֲמֻקָּה שְׁקִיעָתוֹ שֶׁל הָרַב בַּתּוֹרָה.
