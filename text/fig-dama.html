הָאֶבֶן הַיְקָרָה וְכִבּוּד הָאָב

לִפְנֵי שָׁנִים רַבּוֹת, כְּשֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם, הָיָה לַכֹּהֵן הַגָּדוֹל חֹשֶׁן מְיֻחָד עַל הֶחָזֶה. יוֹם אֶחָד, אַחַת מֵהָאֲבָנִים שֶׁל הַחֹשֶׁן נֶאֶבְדָה, וְחַכְמֵי יִשְׂרָאֵל לֹא מָצְאוּ אוֹתָהּ. הֵם הֶחְלִיטוּ לָצֵאת וּלְחַפֵּשׁ אֶבֶן כָּזֹאת בְּכָל מָקוֹם. חַכְמֵי יִשְׂרָאֵל הָלְכוּ וְחִפְּשׂוּ מְאֹד, עַד שֶׁהִגִּיעוּ לָעִיר אַשְׁקְלוֹן. שָׁם הֵם הִגִּיעוּ לְבֵיתוֹ שֶׁל גּוֹי אֶחָד שֶׁשְּׁמוֹ דָּמָא בֶּן נְתִינָה, כִּי שָׁמְעוּ שֶׁיֵּשׁ לוֹ אֶבֶן יְקָרָה וּמַתְאִימָה. חַכְמֵי יִשְׂרָאֵל דָּפְקוּ בְּדַלְתּוֹ, הֵם דָּפְקוּ וְדָפְקוּ שׁוּב וָשׁוּב, עַד שֶׁדָּמָא בֶּן נְתִינָה פָּתַח אֶת הַדֶּלֶת.

כְּשֶׁפָּתַח אֶת הַדֶּלֶת, אָמַר לָהֶם דָּמָא בְּשֶׁקֶט: "שַׁשַׁשַׁ... אַבָּא שֶׁלִּי יָשֵׁן עַכְשָׁו שְׁנַת צָהֳרַיִם. מָה אַתֶּם צְרִיכִים?" אָמְרוּ לוֹ חַכְמֵי יִשְׂרָאֵל: "שָׁמַעְנוּ שֶׁיֵּשׁ לְךָ אֶבֶן יְקָרָה שֶׁאֲנַחְנוּ בְּדִיּוּק צְרִיכִים אוֹתָהּ עֲבוּר הַחֹשֶׁן שֶׁל הַכֹּהֵן הַגָּדוֹל". עָנָה לָהֶם דָּמָא: "אַהּ... בְּשִׂמְחָה אֲנִי אָבִיא לָכֶם אֶת הָאֶבֶן, אֲבָל עַכְשָׁו אַבָּא שֶׁלִּי יָשֵׁן! אֲנִי אָבִיא לָכֶם אוֹתָהּ אַחַר כָּךְ". חַכְמֵי יִשְׂרָאֵל חָשְׁבוּ שֶׁהוּא מְהַסֵּס, וְאָמְרוּ לוֹ: "אָז נִתֵּן לְךָ מָאתַיִם מַטְבְּעוֹת, נִתֵּן לְךָ מָאתַיִם".

אָמַר לָהֶם דָּמָא: "כֵּן, אֲבָל הַמַּפְתֵּחַ שֶׁל הַכַּסֶפֶת שֶׁשָּׁם נִמְצֵאת הָאֶבֶן הַיְקָרָה נִמְצָא מִתַּחַת לַכַּר שֶׁל אַבָּא שֶׁלִּי, וְאֵינִי יָכוֹל לְהָבִיא לָכֶם אוֹתָהּ עַכְשָׁו כְּדֵי לֹא לְהָעִיר אוֹתוֹ". חַכְמֵי יִשְׂרָאֵל הִמְשִׁיכוּ לַחְשֹׁב שֶׁהוּא אֵינוֹ רוֹצֶה לָתֵת לָהֶם אֶת הָאֶבֶן, וְלָכֵן הֵם הִצִּיעוּ לוֹ סְכוּמִים גְּבוֹהִים יוֹתֵר וְיוֹתֵר. הֵם אָמְרוּ לוֹ: "אָז נָבִיא לְךָ שְׁלֹשׁ מֵאוֹת מַטְבְּעוֹת! אַרְבַּע מֵאוֹת! חֲמֵשׁ מֵאוֹת!" הֵם הִמְשִׁיכוּ לְהַעֲלוֹת אֶת הַמְּחִיר עַד שֶׁהִגִּיעוּ לְאֶלֶף מַטְבְּעוֹת זָהָב עֲבוּר הָאֶבֶן הַיְקָרָה!

אַךְ דָּמָא בֶּן נְתִינָה עָמַד בְּדַעְתּוֹ וְאָמַר לָהֶם שׁוּב: "אֲנִי אָבִיא לָכֶם אֶת הָאֶבֶן, אֲבָל לֹא עַכְשָׁו, כִּי אַבָּא יָשֵׁן!" כְּשֶׁרָאוּ חַכְמֵי יִשְׂרָאֵל שֶׁהוּא אֵינוֹ מוֹכֵר לָהֶם אֶת הָאֶבֶן כָּעֵת, הֵם הִתְיַאֲשׁוּ וְהִתְחִילוּ לָלֶכֶת מִשָּׁם.

לְפֶתַע, רָץ דָּמָא בֶּן נְתִינָה אַחֲרֵיהֶם, מִהֵר וְקָרָא לָהֶם: "הִנֵּה, הִנֵּה הָאֶבֶן הַיְקָרָה, קְחוּ, בְּשִׂמְחָה!" שָׁאֲלוּ אוֹתוֹ חַכְמֵי יִשְׂרָאֵל: "כַּמָּה אֲנַחְנוּ צְרִיכִים לְשַׁלֵּם לְךָ?" עָנָה לָהֶם דָּמָא בֶּן נְתִינָה: "רַק מֵאָה! כְּמוֹ שֶׁבִּקַּשְׁתֶּם וּרְצִיתֶם לְהָבִיא בַּהַתְחָלָה".

חַכְמֵי יִשְׂרָאֵל הִתְפַּעֲלוּ מְאֹד מֵאֹתוֹ הַגּוֹי. הֵם הֵבִינוּ שֶׁעַכְשָׁו, אַחֲרֵי שֶׁאָבִיו הִתְעוֹרֵר מִשְּׁנָתוֹ, אֶפְשָׁר הָיָה לִקַּחַת אֶת הַמַּפְתֵּחַ מִתַּחַת לַכַּר, לִפְתֹּחַ אֶת הַכַּסֶפֶת וּלְהָבִיא לָהֶם אֶת הָאֶבֶן הַיְקָרָה. דָּמָא לֹא הִסְכִּים לְהָעִיר אֶת אָבִיו וְלֹא לָקַח מַטְבֵּעַ אֶחָד יוֹתֵר מִמַּה שֶׁהִצִּיעוּ לוֹ בַּתְּחִילָה. עַל כֵּן, בֵּרְכוּ אוֹתוֹ חַכְמֵי יִשְׂרָאֵל שֶׁיִּזְכֶּה לְעֹשֶׁר גָּדוֹל.

💡 מִצְוַת כִּבּוּד הוֹרִים הִיא מִצְוָה חֲשׁוּבָה וִיקָרָה מְאֹד, וּמִי שֶׁמְּכַבֵּד אֶת הוֹרָיו וְאֵינוֹ מַפְרִיעַ לִמְנוּחָתָם זוֹכֶה לִבְרָכָה גְּדוֹלָה.
