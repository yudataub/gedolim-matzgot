הַחָכְמָה שֶׁהִפְתִיעָה אֶת הַמַּדְעָן

חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה לִמְּדוּנוּ כִּי הַתּוֹרָה הַקְּדוֹשָׁה מְכִילָה בְּתוֹכָהּ חָכְמָה עֲמֻקָּה מְאוֹד, וּמָרָן הָרַב עוֹבַדְיָה יוֹסֵף זֵכֶר צַדִּיק לִבְרָכָה חָשַׂף פְּעָמִים רַבּוֹת טֶפַח מִגְּאוֹנוּתוֹ הַגְּדוֹלָה בִּתְחוּמִים שׁוֹנִים, גְּאוֹנוּת מֻפְלָאָה שֶׁנָּבְעָה מִכֹּחַ תּוֹרָתוֹ.

מַעֲשֶׂה שֶׁהָיָה כָּךְ הָיָה: בְּיוֹם אֶחָד הִגִּיעַ לְבֵיתוֹ שֶׁל מָרָן הָרַב עוֹבַדְיָה יוֹסֵף פְּרוֹפֶסוֹר מְפֻרְסָם וְנוֹדָע, שֶׁכִּהֵן כְּרֹאשׁ הַמַּחְלָקָה לְגֶנֶטִיקָה בְּאֶחָד מִבָּתֵּי הַחוֹלִים הַמְּפֻרְסָמִים בְּאֶרֶץ יִשְׂרָאֵל. הַפְּרוֹפֶסוֹר הַחָשׁוּב בִּקֵּשׁ לִפְגֹּשׁ אֶת הָרַב עַל מְנָת לְהִתְיַעֵץ עִמּוֹ בִּשְׁאֵלָה הַלְכָתִית וּרְפוּאִית מֻרְכֶּבֶת מְאוֹד, שֶׁעָלְתָה בַּמַּעֲבָדָה בַּמַּהֲלָךְ עֲבוֹדָתוֹ הַמַּדְעִית.

שְׁאֵלָתוֹ שֶׁל הַפְּרוֹפֶסוֹר הָיְתָה: "הַאִם מֻתָּר, לְצֹרֶךְ מֶחְקָר רְפוּאִי שֶׁנַּעֲשֶׂה עַל בַּעֲלֵי חַיִּים, לְנַתֵּחַ חֻלְדָּה בְּאֹפֶן כָּזֶה שֶׁהַנִּתּוּחַ יִגְרֹם לְכָךְ שֶׁלֹּא יִהְיוּ לָהּ וְלָדוֹת בֶּעָתִיד?". הַפְּרוֹפֶסוֹר הִסְבִּיר לָרַב כִּי עַל פִּי חֻקֵּי הַהֲלָכָה, פְּעֻלָּה כָּזוֹ, כַּאֲשֶׁר הִיא נַעֲשֵׂית לְלֹא צֹרֶךְ, אֲסוּרָה הִיא מִן הַדִּין. אַךְ הוּא בִּקֵּשׁ לָדַעַת מָה הַדִּין כַּאֲשֶׁר הַדָּבָר נַעֲשֶׂה לְצֹרֶךְ מֶחְקָר רְפוּאִי, שֶׁיֵּשׁ בּוֹ תּוֹעֶלֶת רַבָּה לִבְנֵי אָדָם.

מָרָן הָרַב עוֹבַדְיָה יוֹסֵף הִקְשִׁיב לַשְּׁאֵלָה בְּכֹבֶד רֹאשׁ וּבִתְשׂוּמַת לֵב רַבָּה. תַּחַת לַעֲנוֹת מִיָּד תְּשׁוּבָה פְּשׁוּטָה שֶׁל מֻתָּר אוֹ אָסוּר, נָטַל הָרַב דַּף נְיָר וְעֵט, וְהֵחֵל לְשַׁרְטֵט עַל גַּבֵּי הַדַּף אֶת צוּרַת גּוּפָהּ שֶׁל הַחֻלְדָּה. הָרַב הִבִּיט בַּשִּׁרְטוּט וְהִתְבּוֹנֵן בּוֹ עֲמֻקּוֹת.

לְאַחַר מִכֵּן פָּנָה הָרַב אֶל הַפְּרוֹפֶסוֹר וְשָׁאַל אוֹתוֹ בְּנַחַת: "מַדּוּעַ תְּנַתְחוּ אֶת הַחֻלְדָּה דַּוְקָא בַּדֶּרֶךְ הַזֹּאת?". אָז לָקַח הָרַב עוֹבַדְיָה אֶת הָעֵט, וְהֵחֵל לְשַׁרְטֵט עַל הַדַּף דֶּרֶךְ חֲדָשָׁה וּשְׁוֹנָה לְבִצּוּעַ הַנִּתּוּחַ. הָרַב הֶרְאָה לַמַּדְעָן כֵּיצַד אֶפְשָׁר לְבַצֵּעַ אֶת הַנִּתּוּחַ הָרְפוּאִי בְּצוּרָה מְיֻחֶדֶת, כָּכָה שֶׁהַנִּתּוּחַ יִפְתֹּר אֶת הַבְּעָיָה הַמַּדְעִית וְיַגִּיעַ לַתּוֹצָאָה הַמְּבוּקֶשֶׁת, אַךְ בְּאֹפֶן שֶׁלֹּא יִפְגַּע כְּלָל בִּכְשָׁרוֹן הַהוֹלָדָה שֶׁל הַחֻלְדָּה.

הַפְּרוֹפֶסוֹר הַלָּמוּד נִשְׁאַר נִדְהָם וּמֻפְתָּע עַד מְאוֹד! בַּחֲלוֹמוֹ הָרָחוֹק בְּיוֹתֵר הוּא לֹא דִּמְיֵן, שֶׁמָּרָן הָרַב עוֹבַדְיָה יוֹסֵף, שֶׁיּוֹשֵׁב כָּל יָמָיו בְּבֵית הַמִּדְרָשׁ וְשׁוֹקֵד עַל לִמּוּד הַתּוֹרָה הַקְּדוֹשָׁה, יֵדַע לְהַדְרִיךְ אוֹתוֹ וּלְלַמֵּד אוֹתוֹ כֵּיצַד לְנַתֵּחַ חֻלְדָּה בַּמַּעֲבָדָה. בְּלָשׁוֹן מְגַמְגֶמֶת מֵרֹב הִתְרַגְּשׁוּת וְהַעֲרָכָה, הוֹדָה הַפְּרוֹפֶסוֹר לָרַב עוֹבַדְיָה וְאָמַר כִּי דֶּרֶךְ הַנִּתּוּחַ הַחֲכָמָה שֶׁהִצִּיעַ הָרַב, אָכֵן פּוֹתֶרֶת אֶת הַבְּעָיָה הַמַּדְעִית עַל הַצַּד הַטּוֹב בְּיוֹתֵר.

💡 כְּשֶׁאָדָם לוֹמֵד אֶת הַתּוֹרָה הַקְּדוֹשָׁה בִּשְׁקִידָה וּבְאַהֲבָה, ה׳ מַעֲנִיק לוֹ חָכְמָה עֲמֻקָּה וּתְבוּנָה רַבָּה לְהָבִין בְּכָל תְּחוּמֵי הַחַיִּים.
