תְּפִלָּתוֹ שֶׁל מֹשֶׁה הַקָּטָן

כְּשֶׁרַבִּי מֹשֶׁה בֶּן מַיְמוֹן, הָרַמְבַּ"ם הַגָּדוֹל, הָיָה יֶלֶד קָטָן, כֻּלָּם קָרְאוּ לוֹ מֹשֶׁה הַקָּטָן. אָבִיו הָיָה תַּלְמִיד חָכָם גָּדוֹל מְאֹד וְשְׁמוֹ רַבִּי מַיְמוֹן. כְּשֶׁמֹּשֶׁה הָיָה יֶלֶד קָטָן, מַמָּשׁ מַמָּשׁ מַמָּשׁ לֹא חָשְׁבוּ שֶׁיֵּצֵא מִמֶּנּוּ רַב. הַסִּבָּה לְכָךְ הָיְתָה שֶׁמֹּשֶׁה הַקָּטָן לֹא אָהַב לִקְרֹא וְלֹא אָהַב לִלְמֹד כְּלָל. בְּכָל פַּעַם שֶׁאָבִיו רַבִּי מַיְמוֹן אָמַר לוֹ: "בּוֹא, שֵׁב, נִלְמַד יַחַד", מֹשֶׁה מַמָּשׁ לֹא רָצָה לִלְמֹד. מָה שֶׁעִנְיַן אוֹתוֹ יוֹתֵר מִכֹּל הָיוּ הַפַּרְפָּרִים, הַצִּפּוֹרִים וְהַפְּרָחִים שֶׁבַּחוּץ, וְשׁוּם דָּבָר מֵהַלִּמּוּדִים לֹא עִנְיַן אוֹתוֹ.

יוֹם אֶחָד יָשַׁב מֹשֶׁה הַקָּטָן יַחַד עִם אָבִיו רַבִּי מַיְמוֹן. אָבִיו פָּנָה אֵלָיו וְאָמַר לוֹ: "תַּגִּיד אָלֶף". אַךְ מֹשֶׁה מַמָּשׁ לֹא רָצָה לְהַגִּיד אָלֶף, כִּי הוּא הָיָה עָסוּק בַּפַּרְפָּרִים שֶׁרָאָה. אָבִיו רַבִּי מַיְמוֹן הִתְאַכְזֵב מִמֶּנּוּ מְאֹד. אָבִיו אָמַר לוֹ בְּאַכְזָבָה גְּדוֹלָה: "זֶהוּ, אַתָּה כְּבָר לֹא תּוּכַל לִקְרֹא, אַתָּה לֹא מַצְלִיחַ לְהַגִּיד אֲפִלּוּ אָלֶף".

מֹשֶׁה הַקָּטָן הִתְאַכְזֵב מְאֹד מִכָּךְ שֶׁהִתְאַכְזֵב אָבִיו. מֵרֹב צַעַר, הוּא קָם וְרָץ וְרָץ עַד שֶׁהִגִּיעַ אֶל הַיַּעַר. כְּשֶׁעָמַד בַּיַּעַר, הוּא רָאָה מֵרָחוֹק אוֹר קָטָן. הָאוֹר הַזֶּה הָיָה בֵּית הַכְּנֶסֶת. מֹשֶׁה נִכְנַס אֶל בֵּית הַכְּנֶסֶת, עָמַד יָשָׁר מוּל אֲרוֹן הַקֹּדֶשׁ, וּבָכָה וּבָכָה וּבָכָה. הוּא קָרָא לַה' מֵעֹמֶק הַלֵּב וְאָמַר: "ה', תַּעֲזֹר לִי! אֲנִי רוֹצֶה שֶׁאַהֲבַת הַתּוֹרָה תִּכָּנֵס בְּלִבִּי, שֶׁאֲנִי אֶצְלִיחַ לִקְרֹא, וְשֶׁאֲנִי אוּכַל לְהִתְפַּלֵּל". מֹשֶׁה הַקָּטָן בָּכָה כָּל כָּךְ הַרְבֵּה, עַד שֶׁמֵּרֹב שֶׁבָּכָה הוּא נִרְדַּם מַמָּשׁ לְיַד אֲרוֹן הַקֹּדֶשׁ.

בַּבֹּקֶר, כְּשֶׁמֹּשֶׁה הִתְעוֹרֵר מֵאוֹר הַשֶּׁמֶשׁ שֶׁהֵאִירָה, פִּתְאֹם הוּא הִרְגִּישׁ הַרְגָּשָׁה מְיֻחֶדֶת בַּלֵּב. הוּא הִרְגִּישׁ שֶׁהוּא מַמָּשׁ רוֹצֶה לִלְמֹד, וְשֶׁמַּמָּשׁ מַתְאִים לוֹ עַכְשָׁו לְהַתְחִיל לִלְמֹד: אָלֶף, בֵּית, גִּימֶל, דָּלֶת. הוּא מַמָּשׁ מַמָּשׁ הִרְגִּישׁ אֶת זֶה בַּלֵּב שֶׁהוּא כְּבָר יָכוֹל לִלְמֹד.

בִּזְכוּת הַתְּפִלָּה הַזֹּאת שֶׁבָּאָה יָשָׁר מֵהַלֵּב, מֹשֶׁה הָלַךְ, לָמַד וְהִשְׁקִיעַ רַבּוֹת. הוּא הִמְשִׁיךְ לִלְמֹד בִּשְׁקִידָה עַד שֶׁהוּא נִהְיָה לְרַב גָּדוֹל, הֲלֹא הוּא רַבִּי מֹשֶׁה בֶּן מַיְמוֹן – הָרַמְבַּ"ם. אֶת הַסִּפּוּר הַמְּרַגֵּשׁ הַזֶּה מְסַפְּרִים מוֹיְשִׁי וְהַמּוֹרָה מֵירָב לַיְלָדִים בְּבֵית הַכְּנֶסֶת מוּל אֲרוֹן הַקֹּדֶשׁ, שֶׁעַל הַפָּרֹכֶת שֶׁלּוֹ כְּתוּבָה הַכְּתֹבֶת: "דָּרַשְׁתִּי קִרְבָתְךָ, בְּכָל לִבִּי קְרָאתִיךָ".

💡 כְּשֶׁמִּתְפַּלְלִים לַה' מֵעֹמֶק הַלֵּב, ה' שׁוֹמֵעַ אֶת הַתְּפִלָּה וְעוֹזֵר לָנוּ לְהַצְלִיחַ.
