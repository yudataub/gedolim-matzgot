תְּקִיעָה מִכָּל הַלֵּב

לְהַבַּעַל שֵׁם טוֹב הַקָּדוֹשׁ וְהַצַּדִּיק הָיָה אִישׁ אֶחָד, יְהוּדִי יָקָר וְצַדִּיק, שֶׁהָיָה הַבַּעַל תּוֹקֵעַ שֶׁתָּקַע בַּשּׁוֹפָר בְּכָל תְּפִלּוֹת רֹאשׁ הַשָּׁנָה. לִפְנֵי יְמֵי הֶחָג, פָּנָה אֵלָיו הַבַּעַל שֵׁם טוֹב וְאָמַר לוֹ: "תִּשְׁמַע, יְהוּדִי יָקָר, אֲנִי רוֹצֶה שֶׁתִּלְמַד כַּוָּנוֹת מְיֻחָדוֹת שֶׁתַּחְשֹׁב עֲלֵיהֶן תּוֹךְ כְּדֵי שֶׁאַתָּה תּוֹקֵעַ בַּתְּפִלָּה בְּרֹאשׁ הַשָּׁנָה, כְּדֵי שֶׁהַתְּקִיעוֹת יִהְיוּ הֲכִי אֲהוּבוֹת עַל ה' יִתְבָּרַךְ". הַתּוֹקֵעַ הַצַּדִּיק הִקְשִׁיב לִדְבָרָיו, וְרָשַׁם לְעַצְמוֹ עַל גַּבֵּי דַּף אֶת כָּל הַכַּוָּנוֹת הַמְּיֻחָדוֹת – מָה לְכַוֵּן בִּתְקִיעַת הַתְּקִיעָה, מָה לְכַוֵּן בַּשְׁבָרִים וּמָה לְכַוֵּן בַּתְּרוּעָה, כְּדֵי שֶׁיּוּכַל לְהַבִּיט בַּדַּף הַזֶּה בְּמַהֲלַךְ הַתְּקִיעוֹת. אוּלָם, הַבַּעַל שֵׁם טוֹב דָּאַג שֶׁהַתּוֹקֵעַ לֹא יִמְצָא אֶת הַדַּף, וְהֶחְבִּיא לוֹ אוֹתוֹ.

כְּשֶׁהִגִּיעַ יוֹם רֹאשׁ הַשָּׁנָה, בָּא הַתּוֹקֵעַ לְבֵית הַכְּנֶסֶת כְּשֶׁהוּא לָבוּשׁ בְּקִיטֶל וּמַחֲזִיק בַּשּׁוֹפָר. כְּשֶׁהִגִּיעָה שְׁעַת הַתְּקִיעוֹת וְהוּא בָּא לִתְקֹעַ, חִפֵּשׂ הַתּוֹקֵעַ אֶת הַדַּף עִם הַכַּוָּנוֹת, אַךְ לֹא מָצָא אוֹתוֹ כְּלָל. הוּא הִתְחִיל לְהַחְוִיר וְלִרְעֹד מִלַּחַץ רַב וּמִדְּאָגָה עֲמֻקָּה. הוּא שָׁאַל אֶת עַצְמוֹ בִּמְבוּכָה: מָה אֲנִי עוֹשֶׂה? מָה אֲנִי עוֹשֶׂה? אֲנִי לֹא זוֹכֵר אֶת כָּל הַכַּוָּנוֹת שֶׁתִּכְנַנְתִּי לְכַוֵּן, אֵיךְ אֲנִי יָכוֹל לִתְקֹעַ טוֹב? אֵיךְ הַתְּקִיעוֹת שֶׁלִּי יַעֲלוּ כָּךְ לַשָּׁמַיִם? בְּאֵין בְּרֵרָה, תָּקַע הַתּוֹקֵעַ בַּשּׁוֹפָר מִתּוֹךְ בְּכִי גָּדוֹל וּמִכָּל הַלֵּב, בִּדְמָעוֹת שֶׁזָּלְגוּ מֵעֵינָיו, כְּשֶׁהוּא יוֹדֵעַ שֶׁאֵינוֹ יוֹדֵעַ וְאֵינוֹ זוֹכֵר אֶת הַכַּוָּנוֹת.

לְאַחַר סִיּוּם הַתְּקִיעוֹת, נִגַּשׁ אֵלָיו הַבַּעַל שֵׁם טוֹב וְאָמַר לוֹ: "תֵּדַע לְךָ, שֶׁהַתְּקִיעוֹת הָאֵלֶּה הָיוּ הֲכִי טוֹבוֹת שֶׁיֵּשׁ. זֶה בְּדִיּוּק מָה שֶׁה' אוֹהֵב, כִּי תָּקַעְתָּ בְּלֵב שָׁבוּר. הִרְגַּשְׁתָּ שֶׁאַתָּה כָּל כָּךְ מִסְכֵּן, כָּל כָּךְ בּוֹכֶה וְכָל כָּךְ צָרִיךְ אֶת ה'. זֶה בְּדִיּוּק מָה שֶׁה' רוֹצֶה, וְזֶה מָה שֶׁהִתְכַּוַּנְתִּי. זֶה חָשׁוּב יוֹתֵר מִכָּל הַכַּוָּנוֹת שֶׁל הַתְּקִיעוֹת".

הַבַּעַל שֵׁם טוֹב הִסְבִּיר אֶת הַדָּבָר עַל יְדֵי מָשָׁל: לְמֶלֶךְ אֶחָד יֵשׁ בְּאַרְמוֹנוֹ חֲדָרִים רַבִּים. הַכַּוָּנוֹת הֵן כְּמוֹ מַפְתְּחוֹת לַחֲדָרִים הָאֵלֶּה בָּאַרְמוֹן, אֲבָל יֵשׁ דָּבָר אֶחָד שֶׁשּׁוֹבֵר אֶת כָּל הַמַּנְעוּלִים. זֶה לֹא מַפְתֵּחַ, אֶלָּא גַּרְזֶן. מִי שֶׁיֵּשׁ לוֹ גַּרְזֶן, הוּא יָכוֹל לִשְׁבֹּר אֶת כָּל הַדְּלָתוֹת וּלְהִכָּנֵס לְכָל הַחֲדָרִים. הַגַּרְזֶן הַזֶּה הוּא לֵב שָׁבוּר. מִי שֶׁמַּרְגִּישׁ שֶׁהוּא חַיָּב אֶת אָבָא שֶׁלָּנוּ בַּשָּׁמַיִם וְרוֹצֶה מְאֹד לַחְזֹר בִּתְשׁוּבָה – זֶה שָׁוֶה יוֹתֵר מִכָּל הַכַּוָּנוֹת, וְלָכֵן הַתְּקִיעָה הַזֹּאת עָלְתָה יָשָׁר לַשָּׁמַיִם.

💡 תְּפִלָּה וּתְקִיעָה בְּלֵב שָׁבוּר וּמִתּוֹךְ רָצוֹן אֲמִתִּי לְהִתְקָרֵב לַה' יְקָרוֹת וַאֲהוּבוֹת לְפָנָיו יוֹתֵר מִכָּל דָּבָר אַחֵר.
