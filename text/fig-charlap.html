לִשְׁמֹעַ אֶת בִּנְיַן יְרוּשָׁלַיִם

הָרַב יַעֲקֹב מֹשֶׁה חַרְלַ"פ כָּל חַיָּיו כָּל כָּךְ אָהַב לִרְאוֹת אֶת יְרוּשָׁלַיִם בְּבִנְיָנָהּ, וְאֶת כָּל אֶרֶץ יִשְׂרָאֵל. הָרַב חַרְלַ"פ הָיָה הָרַב שֶׁל שְׁכוּנַת רְחַבְיָה, וְאַחֲרֵי זֶה הָיָה הָרַב שֶׁל שְׁכוּנַת שַׁעֲרֵי חֶסֶד בִּירוּשָׁלַיִם. הוּא הָיָה אוֹהֵב מְאֹד לְהִסְתּוֹבֵב בִּרְחוֹבוֹת יְרוּשָׁלַיִם וְלִרְאוֹת כֵּיצַד יְרוּשָׁלַיִם נִבְנֵית: עוֹד בַּיִת חָדָשׁ, עוֹד שְׁכוּנָה חָדָשָׁה, וְעוֹד רְחוֹב חָדָשׁ. כָּל בַּיִת וְכָל רְחוֹב שֶׁנִּבְנוּ בָּעִיר הָיוּ מְמַלְּאִים אֶת לִבּוֹ בְּשִׂמְחָה גְּדוֹלָה מְאֹד.

בְּסוֹף יָמָיו הָרַב חַרְלַ"פ כְּבָר נָפַל לְמִשְׁכָּב, הָיָה חוֹלֶה מְאֹד, וְהוּא לֹא הִרְגִּישׁ טוֹב. הוּא שָׁכַב עַל מִטָּתוֹ בְּחַדְרוֹ וְנִסָּה קְצָת לְהַבְרִיא וּלְהִתְחַזֵּק. דּוֹקְטוֹר שְׁטֶרְנְבַּךְ, שֶׁהָיָה הָרוֹפֵא שֶׁלּוֹ, הָיָה מְטַפֵּל בּוֹ בִּמְסִירוּת רַבָּה יוֹם-יוֹם כְּדֵי לַעֲזֹר לוֹ.

יוֹם אֶחָד, כְּשֶׁהַדּוֹקְטוֹר טִפֵּל בָּרַב חַרְלַ"פ בְּחַדְרוֹ, הַדּוֹקְטוֹר שׁוֹמֵעַ פִּתְאֹם רַעַשׁ גָּדוֹל שֶׁל טְרַקְטוֹרִים, קְדִיחוֹת, מַכּוֹת פַּטִּישִׁים, וְקוֹלוֹת שֶׁל פּוֹעֲלִים שֶׁקּוֹרְאִים מִבַּחוּץ: "הֵיי, תָּבִיא אֶת זֶה לְפֹה!", "תֵּלֵךְ לְשָׁם!". הוּא שׁוֹמֵעַ אֶת כָּל הָרַעַשׁ הַחָזָק הַזֶּה, וְהָרַעַשׁ הַזֶּה מַגִּיעַ מֵהַחַלּוֹן הַפָּתוּחַ שֶׁל הָרַב חַרְלַ"פ. מִחוּץ לְחַדְרוֹ שֶׁל הָרַב חַרְלַ"פ, בְּאוֹתוֹ רְחוֹב, עָשׂוּ עֲבוֹדוֹת, שִׁפְרוּ תַּשְׁתִּיּוֹת, הִרְחִיבוּ אֶת הַכְּבִישׁ, וּבָנוּ מִדְרָכוֹת חֲדָשׁוֹת, וְכָל הָרַעַשׁ הַזֶּה הִגִּיעַ יָשָׁר לְחַדְרוֹ שֶׁל הָרַב חַרְלַ"פ.

חָשַׁב לְעַצְמוֹ הַדּוֹקְטוֹר: "אִם לִי הָרַעַשׁ הַזֶּה מַפְרִיעַ, בְּוַדַּאי זֶה מַפְרִיעַ לָרַב חַרְלַ"פ שֶׁשּׁוֹכֵב בַּמִּטָּה וּמְנַסֶּה לָנוּחַ וּלְהִתְחַזֵּק". הַדּוֹקְטוֹר רָצָה לָלֶכֶת מִיָּד לָעִירִיָּה וּלְדַבֵּר עִם הָאֲנָשִׁים שֶׁאַחֲרָאִים עַל עֲבוֹדוֹת הַבְּנִיָּה, וּלְהַגִּיד לָהֶם: "תִּשְׁמְעוּ, יֵשׁ פֹּה אָדָם חוֹלֶה, אָדָם צַדִּיק שֶׁצָּרִיךְ אֶת הַשֶּׁקֶט הַזֶּה כְּדֵי לָנוּחַ. אוּלַי יֵשׁ מָקוֹם אַחֵר שֶׁצָּרִיךְ עַכְשָׁו לַעֲבֹד בּוֹ, וְתָבוֹאוּ כְּשֶׁהָרַב יִתְחַזֵּק, תָּבוֹאוּ לַעֲבֹד בָּרְחוֹב שֶׁבּוֹ הָרַב חַרְלַ"פ נִמְצָא?".

אֲבָל כְּשֶׁשָּׁמַע הָרַב חַרְלַ"פ אֶת הָרַעְיוֹן שֶׁל הַדּוֹקְטוֹר – לָלֶכֶת וְלַעֲצֹר אֶת עֲבוֹדוֹת הַבְּנִיָּה בְּאוֹתוֹ רְחוֹב, הָרַב חַרְלַ"פ עָצַר אוֹתוֹ מִיָּד, וְאָמַר לוֹ: "עֲצֹר, עֲצֹר, עֲצֹר, עֲצֹר. כָּל יָמַי הָיִיתִי נֶהֱנֶה וְהָיִיתִי אוֹהֵב לְהִסְתּוֹבֵב בִּרְחוֹבוֹת יְרוּשָׁלַיִם, לִרְאוֹת אוֹתָהּ בְּבִנְיָנָהּ. הַיּוֹם כְּשֶׁאֲנִי לֹא יָכוֹל כְּבָר לְהִסְתּוֹבֵב, תִּרְאֶה אֲנִי מְרֻתָּק לַמִּטָּה, אֲנִי חוֹלֶה, אֲנִי כְּבָר חַלָּשׁ וְלֹא מַרְגִּישׁ טוֹב, הַיּוֹם אֲנִי לֹא יָכוֹל לְהִסְתּוֹבֵב וְלִרְאוֹת בְּבִנְיָנָהּ שֶׁל יְרוּשָׁלַיִם. לְפָחוֹת תַּשְׁאִיר לִי אֶת הַיְּכֹלֶת לִשְׁמֹעַ בְּבִנְיָנָהּ שֶׁל יְרוּשָׁלַיִם. אַל תַּגִּיד לָהֶם שֶׁיַּעַצְרוּ. אֲנִי פֹּה עַל הַמִּטָּה, וַאֲנִי מְבַקֵּשׁ לִשְׁמֹעַ אֶת בִּנְיָנָהּ שֶׁל יְרוּשָׁלַיִם".

💡 אֲהָבָה אֲמִתִּית לִירוּשָׁלַיִם מְמַלְּאָה אֶת הַלֵּב בְּשִׂמְחָה לְשֵׁמַע כָּל בְּנִיָּה וְהִתְחַדְּשׁוּת שֶׁלָּהּ.
