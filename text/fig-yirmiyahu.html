יִרְמְיָהוּ הַנָּבִיא וְהַהַצָּלָה מִבּוֹר הַטִּיט

יִרְמְיָהוּ בֶּן חִלְקִיָּהוּ הָיָה נָבִיא מִן הַכֹּהֲנִים שֶׁבָּעִיר עֲנָתוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא בָּחַר בּוֹ לִהְיוֹת נָבִיא עוֹד לִפְנֵי שֶׁנּוֹצַר בְּבֶטֶן אִמּוֹ, וּמִנָּה אוֹתוֹ לִהְיוֹת נָבִיא לְכָל הַגּוֹיִם. כַּאֲשֶׁר אָמַר ה' לְיִרְמְיָהוּ לְדַבֵּר אֶל הָעָם, חָשַׁשׁ יִרְמְיָהוּ וְאָמַר כִּי הוּא רַק נַעַר וְאֵינוֹ יוֹדֵעַ לְדַבֵּר. אַךְ ה' אָמַר לוֹ שֶׁלֹּא יִירָא וְשֶׁלֹּא יִפְחַד מִפְּנֵי אִישׁ, כִּי ה' יִהְיֶה אִתּוֹ תָּמִיד כְּדֵי לְהַצִּיל אוֹתוֹ. ה' שָׁלַח אֶת יָדוֹ, נָגַע בְּפִיו שֶׁל יִרְמְיָהוּ וְשָׂם אֶת דְּבָרָיו בְּפִיו, וְיִרְמְיָהוּ קָרָא לְכָל הָעָם לַעֲשׂוֹת תְּשׁוּבָה.

יִרְמְיָהוּ הַנָּבִיא דִּבֵּר אֶל כָּל הָעָם וְאָמַר לָהֶם אֶת דִּבְרֵי ה'. הוּא הִזְהִיר שֶׁמִּי שֶׁיִּשָּׁאֵר בָּעִיר יְרוּשָׁלַיִם יָמוּת בַּחֶרֶב, בָּרָעָב וּבַדֶּבֶר, וּמִי שֶׁיֵּצֵא אֶל מֶלֶךְ בָּבֶל וְאֶל הַכַּשְׂדִּים יִחְיֶה. כַּאֲשֶׁר שָׁמְעוּ הַשָּׂרִים – שְׁפַטְיָה בֶּן מַתָּן, גְּדַלְיָהוּ בֶּן פַּשְׁחוּר, יוּכַל בֶּן שֶׁלֶמְיָהוּ וּפַשְׁחוּר בֶּן מַלְכִּיָּה – אֶת דִּבְרֵי יִרְמְיָהוּ, הֵם כָּעֲסוּ עָלָיו מְאֹד. הַשָּׂרִים הָלְכוּ אֶל הַמֶּלֶךְ צִדְקִיָּהוּ וְאָמְרוּ לוֹ שֶׁצָּרִיךְ לְהָמִית אֶת יִרְמְיָהוּ, כִּי בִּדְבָרָיו הוּא מַחֲלִישׁ אֶת יְדֵי אַנְשֵׁי הַמִּלְחָמָה וְאֶת יְדֵי כָּל הָעָם וְאֵינוֹ דּוֹרֵשׁ שָׁלוֹם. הַמֶּלֶךְ צִדְקִיָּהוּ עָנָה לָהֶם שֶׁיִּרְמְיָהוּ נִמְצָא בִּידֵיהֶם, כִּי הַמֶּלֶךְ אֵינוֹ יָכוֹל לַעֲמֹד כְּנֶגְדָם.

אָז לָקְחוּ הַשָּׂרִים אֶת יִרְמְיָהוּ הַנָּבִיא וְהִשְׁלִיכוּ אוֹתוֹ אֶל בּוֹר הַמַּיִם שֶׁל מַלְכִיָּהוּ בֶן הַמֶּלֶךְ, אֲשֶׁר הָיָה בַּחֲצַר הַמַּטָּרָה. הֵם הוֹרִידוּ אֶת יִרְמְיָהוּ אֶל תּוֹךְ הַבּוֹר בְּחֲבָלִים. בַּבּוֹר הַזֶּה לֹא הָיוּ מַיִם כְּלָל, אֶלָּא רַק טִיט עָמֹק, וְיִרְמְיָהוּ הַנָּבִיא הֵחֵל לִטְבֹּעַ בַּטִּיט.

בְּאוֹתוֹ זְמַן, עֶבֶד מֶלֶךְ הַכּוּשִׁי, אִישׁ חָשׁוּב שֶׁעָבַד בְּבֵית הַמֶּלֶךְ, שָׁמַע שֶׁהִשְׁלִיכוּ אֶת יִרְמְיָהוּ אֶל הַבּוֹר. הוּא יָצָא מִיָּד מִבֵּית הַמֶּלֶךְ וְהָלַךְ אֶל הַמֶּלֶךְ צִדְקִיָּהוּ, שֶׁיָּשַׁב בְּאוֹתָהּ שָׁעָה בְּשַׁעַר בִּנְיָמִין. עֶבֶד מֶלֶךְ אָמַר לַמֶּלֶךְ בְּאֹמֶץ לֵב כִּי הָאֲנָשִׁים הָאֵלֶּה עָשׂוּ רָעָה גְּדוֹלָה לְיִרְמְיָהוּ הַנָּבִיא, וְכִי יִרְמְיָהוּ עָלוּל לָמוּת בַּבּוֹר מִפְּנֵי הָרָעָב, כִּי נִגְמַר הַלֶּחֶם בָּעִיר. הַמֶּלֶךְ צִדְקִיָּהוּ שָׁמַע אֶת דְּבָרָיו וְצִוָּה עַל עֶבֶד מֶלֶךְ לָקַחַת אִתּוֹ שְׁלֹשִׁים אֲנָשִׁים, וּלְהַעֲלוֹת אֶת יִרְמְיָהוּ הַנָּבִיא מִן הַבּוֹר מַהֵר בְּטֶרֶם יָמוּת.

עֶבֶד מֶלֶךְ לָקַח אֶת שְׁלֹשִׁים הָאֲנָשִׁים וְהָלַךְ אֶל מִתַּחַת לְאוֹצַר בֵּית הַמֶּלֶךְ. מִשָּׁם הוּא לָקַח בְּלוֹיֵי סְחָבוֹת וּבְגָדִים יְשָׁנִים וְרַכִּים, וְהוֹרִיד אוֹתָם אֶל יִרְמְיָהוּ אֶל תּוֹךְ הַבּוֹר בְּחֲבָלִים. עֶבֶד מֶלֶךְ קָרָא אֶל יִרְמְיָהוּ וְאָמַר לוֹ לָשִׂים אֶת הַסְּחָבוֹת הָרַכּוֹת תַּחַת בֵּית הַשְּׁחִי שֶׁלּוֹ, מִתַּחַת לַחֲבָלִים, כְּדֵי שֶׁהַחֲבָלִים לֹא יִכְאֲבוּ לוֹ כְּשֶׁיִּמְשְׁכוּ אוֹתוֹ. יִרְמְיָהוּ עָשָׂה כֵּן. הָאֲנָשִׁים מָשְׁכוּ אֶת יִרְמְיָהוּ בַּחֲבָלִים בִּזְהִירוּת, וְהֶעֱלוּ אוֹתוֹ מִן הַבּוֹר הֶעָמֹק. יִרְמְיָהוּ הַנָּבִיא נִצַּל, וְהוּא נִשְׁאַר לָשֶׁבֶת בַּחֲצַר הַמַּטָּרָה.

💡 תָּמִיד כְּדַאי לַעֲזֹר לַזּוּלָת בְּאֹמֶץ לֵב וּבִרְגִישׁוּת, וְהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר עַל מִי שֶׁעוֹשֶׂה אֶת הַטּוֹב.
