אֱלָהָא דְרַבִּי מֵאִיר עֲנֵנִי

פַּעַם אַחַת, בִּימֵי קֶדֶם, שָׁלְטוּ הָרוֹמָאִים הָרְשָׁעִים בְּאֶרֶץ יִשְׂרָאֵל. בְּאוֹתָם יָמִים קְשִׁים, תָּפְסוּ הָרוֹמָאִים בַּת אַחַת שֶׁהָיְתָה אֲחוֹתָהּ שֶׁל בְּרוּרְיָה, אִשְׁתּוֹ שֶׁל רַבִּי מֵאִיר. הָרוֹמָאִים הִכְנִיסוּ אֶת הַבַּת הַמִּסְכֵּנָה לְבֵית הַכֶּלֶא וְסָגְרוּ אוֹתָהּ בְּתוֹךְ תָּא קָטָן. בְּנֵי הַמִּשְׁפָּחָה וְהַיְּהוּדִים הַדּוֹאֲגִים רָצוּ מְאֹד לִפְדּוֹת אוֹתָהּ וּלְהַחֲזִיר אוֹתָהּ לְבֵיתָהּ. הֵם אָסְפוּ כֶּסֶף רַב וּפָנוּ אֶל הָרוֹמָאִים בְּבַקָּשָׁה: קְחוּ אֶת הַכֶּסֶף הַזֶּה וְתַחְזִירוּ לָנוּ אֶת הַבַּת. אַךְ הָרוֹמָאִים הָרְשָׁעִים סֵרְבוּ בְּתֹקֶף וְלֹא הִסְכִּימוּ לְשַׁחְרֵר אוֹתָהּ כְּלָל. הַבַּת הַמִּסְכֵּנָה בָּכְתָה כָּל כָּךְ הַרְבֵּה בְּתוֹךְ בֵּית הַכֶּלֶא, וְכָל הָאֲנָשִׁים הִתְפַּלְּלוּ וְהִתְפַּלְּלוּ לַהַקָּדוֹשׁ בָּרוּךְ הוּא, וְלֹא יָדְעוּ מָה עוֹד אֶפְשָׁר לַעֲשׂוֹת כְּדֵי לְהַצִיל אוֹתָהּ.

כַּאֲשֶׁר רָאָה רַבִּי מֵאִיר אֶת הַצַּעַר הַגָּדוֹל, הוּא אָמַר לָאֲנָשִׁים: תְּנוּ לִי אֶת הַכֶּסֶף שֶׁאָסַפְתֶּם, וַאֲנִי אֵלֵךְ וַאֲנִי אֲשַׁחְרֵר אוֹתָהּ. רַבִּי מֵאִיר לָקַח אֶת הַכֶּסֶף וְהָלַךְ אֶל בֵּית הַכֶּלֶא. הוּא הִגִּיעַ עַד שׁוֹמֵר הַתָּא שֶׁל הַבַּת הַמִּסְכֵּנָה וְאָמַר לוֹ: תִּרְאֶה, אֲנִי מְדַבֵּר אִתְּךָ וְאַף אֶחָד לֹא רוֹאֶה אוֹתָנוּ. אֲנִי נוֹתֵן לְךָ פֹּה הַרְבֵּה כֶּסֶף. תֵּן לִי אוֹתָהּ, הִיא אֲחוֹת שֶׁל אִשְׁתִּי, הִיא בַּת יִשְׂרָאֵל, תֵּן לִי אוֹתָהּ! מָה אַתָּה מַרְוִיחַ מִזֶּה שֶׁהִיא פֹּה? כְּלוּם. אֲנִי מֵבִיא לְךָ פֹּה הַרְבֵּה כֶּסֶף, קַח לְךָ וְתַגִּיד שֶׁהִיא בָּרְחָה.

הַשּׁוֹמֵר שֶׁל בֵּית הַכֶּלֶא פָּחַד מְאֹד וְאָמַר לְרַבִּי מֵאִיר: מָה פִּתְאוֹם! הֵם יָבֹאוּ לִבְדֹּק וְיַגִּידוּ לִי, מָה זֹאת אוֹמֶרֶת הִיא בָּרְחָה? אִם הִיא בָּרְחָה אָז אֲנַחְנוּ נָשִׂים אוֹתְךָ בַּתָּא פֹּה בִּמְקוֹמָהּ! עָנָה לוֹ רַבִּי מֵאִיר וְאָמַר: תִּקְשִׁיב, תְּנַסֶּה לְשַׁחֵד אוֹתָם. אֲנִי מֵבִיא לְךָ הַרְבֵּה כֶּסֶף, חֵלֶק תִּתֵּן לָהֶם וְחֵלֶק תִּשְׁמֹר לְעַצְמְךָ וְתַחְבִּיא. וְאִם זֶה לֹא יַעֲזֹר וְהֵם עֲדַיִן יִכְעֲסוּ עָלֶיךָ וְיִרְצוּ לְהַעֲנִישׁ אוֹתְךָ, תִּתְפַּלֵּל לַה' בִּשְׁמִי וְתַגִּיד אַרְבַּע מִלִּים: אֱלָהָא דְרַבִּי מֵאִיר עֲנֵנִי! שָׁאַל הַשּׁוֹמֵר: וְזֶה יַעֲזֹר לִי? עָנָה רַבִּי מֵאִיר: כֵּן, ה' יִשְׁמַע אוֹתְךָ! רַבִּי מֵאִיר שִׁכְנַע מְאֹד אֶת הַשּׁוֹמֵר, נָתַן לוֹ אֶת כָּל סְכוּם הַכֶּסֶף, שִׁחְרֵר אֶת אֲחוֹת אִשְׁתּוֹ וְהֶחֱזִיר אוֹתָהּ לְבֵיתָהּ.

וּבֶאֱמֶת, אַחֲרֵי כַּמָּה זְמַן, בָּאוּ הָרוֹמָאִים לִבְדֹּק שֶׁכָּל הַכְּלוּאִים נִמְצָאִים בְּבֵית הַכֶּלֶא. הֵם רָאוּ שֶׁמִּישֶׁהִי אַחַת חֲסֵרָה, וְאָמְרוּ לַשּׁוֹמֵר: סְלִיחָה, סְלִיחָה, הָיְתָה כָּאן עוֹד בַּת, אֵיפֹה הִיא נִמְצֵאת? עָנָה הַשּׁוֹמֵר: הִיא בָּרְחָה, אֲנִי לֹא יוֹדֵעַ! כָּעֲסוּ הָרוֹמָאִים וְאָמְרוּ: מָה זֶה לֹא יוֹדֵעַ, אַתָּה הַשּׁוֹמֵר! בּוֹא אִתָּנוּ אֶל הַמֶּלֶךְ! הַשּׁוֹמֵר נִבְהַל וְאָמַר: אוֹי וַאֲבוֹי, אוֹי וַאֲבוֹי! אֲנִי אֶתֵּן לָכֶם קְצַת כֶּסֶף, תַּעַזְבוּ אוֹתִי, תְּשַׁחְרְרוּ אוֹתִי! אַךְ הָרוֹמָאִים עָנוּ: לֹא, בּוֹא אִתָּנוּ אֶל הַמֶּלֶךְ!

כַּאֲשֶׁר בָּא הַשּׁוֹמֵר לִפְנֵי הַמֶּלֶךְ, אָמַר הַמֶּלֶךְ בְּכַעַס: מָה זֶה הַדָּבָר הַזֶּה, אַתָּה לֹא שׁוֹמֵר כְּמוֹ שֶׁצָּרִיךְ עַל הַכְּלוּאִים בְּבֵית הַכֶּלֶא? אֲנַחְנוּ נוֹצִיא אוֹתְךָ לְהוֹרֵג עִם חֶרֶב! הַחַיָּלִים לָקְחוּ אֶת הַחֶרֶב וְעָמְדוּ לְהַעֲנִישׁ אוֹתוֹ. וְאָז נִזְכַּר הַשּׁוֹמֵר שֶׁל בֵּית הַכֶּלֶא וְאָמַר לְעַצְמוֹ: אוֹי, מָה הוּא אָמַר לִי לְהַגִּיד? הוּא נִזְכַּר בְּאַרְבַּע הַמִּלִּים שֶׁל רַבִּי מֵאִיר, וְצָעַק: אֱלָהָא דְרַבִּי מֵאִיר עֲנֵנִי! הוּא אָמַר אֶת הַמִּלִּים הָאֵלֶּה שָׁלֹשׁ פְּעָמִים. וּפִתְאֹם, בּוּם! הַחֶרֶב הִתְנַפְּצָה לִרְסִיסִים! כָּל הָאֲנָשִׁים וְהַמֶּלֶךְ רָאוּ אֶת הַנֵּס הַגָּדוֹל הַזֶּה שֶׁנַּעֲשָׂה. הֵם הִתְפַּלְאוּ מְאֹד וְאָמְרוּ לוֹ: טוֹב, בְּסֵדֶר, אֲנַחְנוּ לֹא נִפְגַּע בְּךָ, אַתָּה גַּם יָכוֹל לָצֵאת לַחָפְשִׁי.

💡 כַּאֲשֶׁר אֲנַחְנוּ נִמְצָאִים בְּצָרָה וּמִתְפַּלְּלִים לַה' בֶּאֱמוּנָה בִּזְכוּת הַצַּדִּיקִים, הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ אֶת תְּפִלָּתֵנוּ וְעוֹשֶׂה עִמָּנוּ נִסִּים.
